// === Load client data function ===async function loadClientData(userId) {
    try {
        const resp = await fetch(`${BASE_URL}/clients`);
        const data = safeParse(await resp.text());
        
        if (!data || !data.clients || data.clients.length === 0) {
            logToPanel("[WARN] No clients array found in DB payload");
            return;
        }

        // 🔍 Fixed Object Traversal: Target nested profile.name property
        const client = data.clients.find(c => c.profile && c.profile.name === userId);
        
        if (!client) {
            logToPanel(`[WARN] ${userId} nested profile records not found in DB`);
            return;
        }

        logToPanel(`[INFO] Found database record for user: ${userId}`);

        // 🪙 Extract values safely using deep structural maps + non-null fallbacks
        const walletBalance = (client.wallet && client.wallet.balance !== null) ? parseFloat(client.wallet.balance) : 5000;
        const profitTarget  = (client.settings && client.settings.profit_target) ? parseFloat(client.settings.profit_target) : 1.0;
        const stopLoss      = (client.settings && client.settings.stop_loss) ? parseFloat(client.settings.stop_loss) : 0.5;
        const profileName   = client.profile.name || userId;

        // Fill controls safely
        document.getElementById("startBalance").value = walletBalance;
        document.getElementById("investBalance").value = walletBalance;
        document.getElementById("profitTarget").value = profitTarget;
        document.getElementById("stopLoss").value = stopLoss;
        
        const userLabel = document.getElementById("currentUserLabel");
        if (userLabel) userLabel.textContent = profileName;

        // Unlock buttons instantly since balance is set
        if (walletBalance > 0) {
            setButtonStates("afterDeposit");
        } else {
            setButtonStates("init");
        }

    } catch (e) {
        logToPanel("[ERROR] loadClientData failed structural mapping: " + e);
        // Emergency fail-safe fallback so testing never blocks
        setButtonStates("afterDeposit"); 
    }
}
async function refreshBalances() {
    try {
        const resp = await fetch(`${BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ wallet_address: USER_ADDR, private_key: USER_PK, amount: 0 })
        });
        const parsed = safeParse(await resp.text());
        if (parsed?.fund_token_balance !== undefined) {
            document.getElementById("startBalance").value = parsed.fund_token_balance;
            document.getElementById("investBalance").value = parsed.fund_token_balance;
            logToPanel(`[INFO] Fund balance: ${parsed.fund_token_balance}, ETH=${parsed.eth_balance_wei}`);
        } else {
            logToPanel("[WARN] No balance data received");
        }
    } catch (err) {
        logToPanel(`[ERROR] Balance refresh failed: ${err.message}`);
    }
}
async function runDeposit(amount) {
    try {
        const approveResp = await fetch(`${BASE_URL}/runbot/approve`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ wallet_address: USER_ADDR, private_key: USER_PK, amount })
        });
        const approveParsed = safeParse(await approveResp.text());
        if (approveParsed?.tx_hash) {
            logToPanel(`[INFO] Approve executed, TX=${approveParsed.tx_hash}`);
        }

        const depositResp = await fetch(`${BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ wallet_address: USER_ADDR, private_key: USER_PK, amount })
        });
        const depositParsed = safeParse(await depositResp.text());
        if (depositParsed) {
            logToPanel(`[INFO] Deposit executed, TX=${depositParsed.tx_hash}, Fund=${depositParsed.fund_token_balance}, ETH=${depositParsed.eth_balance_wei}`);
            document.getElementById("startBalance").value = depositParsed.fund_token_balance;
            document.getElementById("investBalance").value = depositParsed.fund_token_balance;
        }
    } catch (err) {
        logToPanel(`[ERROR] Deposit failed: ${err.message}`);
    }
}
async function runWithdraw(amount) {
    try {
        const resp = await fetch(`${BASE_URL}/runbot/payout`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ wallet_address: USER_ADDR, private_key: USER_PK, amount })
        });
        const parsed = safeParse(await resp.text());
        if (parsed) {
            logToPanel(`[INFO] Payout executed, TX=${parsed.tx_hash}`);
        }
        await refreshBalances();
    } catch (err) {
        logToPanel(`[ERROR] Withdraw failed: ${err.message}`);
    }
}
// === State machine ===function setButtonStates(state) {
    const startBtn = document.getElementById("startBtn");
    const stopBtn = document.getElementById("stopBtn");
    const depositBtn = document.getElementById("depositBtn");
    const withdrawBtn = document.getElementById("withdrawBtn");
    const changeUserBtn = document.getElementById("changeUserBtn");

    const startBalanceBox = document.getElementById("startBalance");
    const depositBox = document.getElementById("depositAmount");
    const withdrawBox = document.getElementById("withdrawAmount");
    const statusLabel = document.getElementById("statusLabel");

    if (!startBtn || !stopBtn || !statusLabel) return;

    switch(state) {
        case "init":
            startBtn.disabled = true;
            stopBtn.disabled = true;
            if(depositBtn) depositBtn.disabled = false;
            if(withdrawBtn) withdrawBtn.disabled = true;
            if(changeUserBtn) changeUserBtn.disabled = true;
            if(startBalanceBox) startBalanceBox.readOnly = true;
            if(depositBox) depositBox.disabled = false;
            if(withdrawBox) withdrawBox.disabled = true;
            window.setStatus("INIT");
            statusLabel.style.color = "gray";
            break;

        case "afterDeposit":
            startBtn.disabled = false;
            stopBtn.disabled = true;
            if(depositBtn) depositBtn.disabled = false;
            if(withdrawBtn) withdrawBtn.disabled = false; 
            if(changeUserBtn) changeUserBtn.disabled = false;
            if(startBalanceBox) startBalanceBox.readOnly = true;
            if(depositBox) depositBox.disabled = false;
            if(withdrawBox) withdrawBox.disabled = false;
            window.setStatus("READY TO RUN");
            statusLabel.style.color = "blue";
            break;

        case "running":
            startBtn.disabled = true;
            stopBtn.disabled = false;
            if(depositBtn) depositBtn.disabled = true;
            if(withdrawBtn) withdrawBtn.disabled = true;
            if(changeUserBtn) changeUserBtn.disabled = true;
            if(startBalanceBox) startBalanceBox.disabled = true;
            if(depositBox) depositBox.disabled = true;
            if(withdrawBox) withdrawBox.disabled = true;
            window.setStatus("RUNNING");
            statusLabel.style.color = "red";
            break;

        case "completed":
            startBtn.disabled = false;
            stopBtn.disabled = true;
            if(depositBtn) depositBtn.disabled = false;
            if(withdrawBtn) withdrawBtn.disabled = false;
            if(changeUserBtn) changeUserBtn.disabled = false;
            if(startBalanceBox) startBalanceBox.disabled = false; 
            if(depositBox) depositBox.disabled = false;
            if(withdrawBox) withdrawBox.disabled = false;
            window.setStatus("COMPLETED");
            statusLabel.style.color = "green";
            break;
    }
}
