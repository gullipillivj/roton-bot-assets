// === Load client data function ===
async function loadClientData(username = "mockuser1") {
    try {
        const response = await fetch("/clients");
        const data = await response.json();
        
        if (!data.clients || data.clients.length === 0) {
            logToPanel("[WARN] No clients array found in DB payload");
            return;
        }

        // 🔍 Fixed Object Traversal: Target nested profile.name property
        const client = data.clients.find(c => c.profile && c.profile.name === username);
        
        if (!client) {
            logToPanel(`[WARN] ${username} nested profile records not found in DB`);
            return;
        }

        logToPanel(`[INFO] Found database record for user: ${username}`);

        // 🪙 Extract values safely using deep structural maps + non-null fallbacks
        const walletBalance = (client.wallet && client.wallet.balance !== null) ? parseFloat(client.wallet.balance) : 5000;
        const profitTarget  = (client.settings && client.settings.profit_target) ? parseFloat(client.settings.profit_target) : 1.0;
        const stopLoss      = (client.settings && client.settings.stop_loss) ? parseFloat(client.settings.stop_loss) : 0.5;
        const profileName   = client.profile.name || username;

        // Fill controls safely
        document.getElementById("startBalance").value = walletBalance;
        document.getElementById("investBalance").value = walletBalance;
        document.getElementById("profitTarget").value = profitTarget;
        document.getElementById("stopLoss").value = stopLoss;
        document.getElementById("currentUserLabel").textContent = profileName;

        // Unlock buttons instantly since balance is set
        if (walletBalance > 0) {
            setButtonStates("afterDeposit");
        } else {
            setButtonStates("init");
        }

    } catch (e) {
        logToPanel("[ERROR] loadClientData failed structural mapping: " + e);
        // Emergency fail-safe fallback so testing never blocks
        setButtonStates("afterDeposit"); 
    }
}

// === State machine ===
function setButtonStates(state) {
    const startBtn = document.getElementById("startBtn");
    const stopBtn = document.getElementById("stopBtn");
    const depositBtn = document.getElementById("depositBtn");
    const withdrawBtn = document.getElementById("withdrawBtn");
    const changeUserBtn = document.getElementById("changeUserBtn");

    const startBalanceBox = document.getElementById("startBalance");
    const depositBox = document.getElementById("depositAmount");
    const withdrawBox = document.getElementById("withdrawAmount");
    const statusLabel = document.getElementById("statusLabel");

    if (!startBtn || !stopBtn || !statusLabel) return;

    switch(state) {
        case "init":
            startBtn.disabled = true;
            stopBtn.disabled = true;
            if(depositBtn) depositBtn.disabled = false;
            if(withdrawBtn) withdrawBtn.disabled = true;
            if(changeUserBtn) changeUserBtn.disabled = true;
            if(startBalanceBox) startBalanceBox.readOnly = true;
            if(depositBox) depositBox.disabled = false;
            if(withdrawBox) withdrawBox.disabled = true;
            statusLabel.textContent = "STATE: INIT";
            statusLabel.style.color = "gray";
            break;

        case "afterDeposit":
            startBtn.disabled = false;
            stopBtn.disabled = true;
            if(depositBtn) depositBtn.disabled = false;
            if(withdrawBtn) withdrawBtn.disabled = false; 
            if(changeUserBtn) changeUserBtn.disabled = false;
            if(startBalanceBox) startBalanceBox.readOnly = true;
            if(depositBox) depositBox.disabled = false;
            if(withdrawBox) withdrawBox.disabled = false;
            statusLabel.textContent = "STATE: READY TO RUN";
            statusLabel.style.color = "blue";
            break;

        case "running":
            startBtn.disabled = true;
            stopBtn.disabled = false;
            if(depositBtn) depositBtn.disabled = true;
            if(withdrawBtn) withdrawBtn.disabled = true;
            if(changeUserBtn) changeUserBtn.disabled = true;
            if(startBalanceBox) startBalanceBox.disabled = true;
            if(depositBox) depositBox.disabled = true;
            if(withdrawBox) withdrawBox.disabled = true;
            statusLabel.textContent = "STATE: RUNNING";
            statusLabel.style.color = "red";
            break;

        case "completed":
            startBtn.disabled = false;
            stopBtn.disabled = true;
            if(depositBtn) depositBtn.disabled = false;
            if(withdrawBtn) withdrawBtn.disabled = false;
            if(changeUserBtn) changeUserBtn.disabled = false;
            if(startBalanceBox) startBalanceBox.disabled = false; 
            if(depositBox) depositBox.disabled = false;
            if(withdrawBox) withdrawBox.disabled = false;
            statusLabel.textContent = "STATE: COMPLETED";
            statusLabel.style.color = "green";
            break;
    }
}

// === Expose state callbacks globally ===
window.updateUIState = function(state) {
    setButtonStates(state);
};

// === UI glue ===
document.addEventListener("DOMContentLoaded", () => {
    logToPanel("UI ready");
    setButtonStates("init"); 
    loadClientData("mockuser1");

    // Deposit button
    const depositBtn = document.getElementById("depositBtn");
    if (depositBtn) {
        depositBtn.addEventListener("click", async () => {
            try {
                const amount = parseFloat(document.getElementById("depositAmount").value) || 0;
                const currentUser = "mockuser1";
                const commission = 5;
                const netAmount = amount - commission;

                if (amount <= commission) {
                    logToPanel("[WARN] Deposit amount must be greater than commission fee");
                    return;
                }

                await window.approveToken(window.userTokenAddress, amount, currentUser);
                await window.depositToFund(window.userTokenAddress, netAmount, currentUser);
                await window.depositToFund(window.userTokenAddress, commission, "0xa5Daf94b8b3dCE7D758bd419ffDd7aC1B83CE0c5");

                const startBalanceBox = document.getElementById("startBalance");
                const currentBalance = parseFloat(startBalanceBox.value) || 0;
                startBalanceBox.value = currentBalance + netAmount;

                logToPanel(`[INFO] ${currentUser} deposited ${amount} tokens → net credited ${netAmount}, commission ${commission} coins sent to Admin`);

                document.getElementById("depositAmount").value = 0;

                if (parseFloat(startBalanceBox.value) > 0) {
                    setButtonStates("afterDeposit");
                }
            } catch (err) {
                logToPanel("[ERROR] Deposit failed: " + err);
            }
        });
    }

    // Withdraw button
    const withdrawBtn = document.getElementById("withdrawBtn");
    if (withdrawBtn) {
        withdrawBtn.addEventListener("click", async () => {
            try {
                const currentUser = "mockuser1";
                const amount = parseFloat(document.getElementById("withdrawAmount").value) || 0;

                logToPanel(`[INFO] Checking available balance for ${currentUser} in SolidarityFund...`);
                const available = await window.fund.methods.balanceOf(currentUser).call();
                const availableAmount = parseFloat(available) || 0;

                if (amount <= 0) {
                    logToPanel("[WARN] Withdraw amount must be greater than 0");
                    return;
                }
                if (availableAmount < amount) {
                    logToPanel(`[WARN] Requested ${amount} exceeds available ${availableAmount} for ${currentUser}`);
                    return;
                }

                logToPanel(`[INFO] Ledger shows ${availableAmount} tokens available for ${currentUser}`);

                await window.withdrawFromFund(window.userTokenAddress, amount, currentUser);

                const startBalanceBox = document.getElementById("startBalance");
                const currentBalance = parseFloat(startBalanceBox.value) || 0;
                startBalanceBox.value = currentBalance - amount;

                logToPanel(`[INFO] Withdrawn ${amount} tokens for ${currentUser} from SolidarityFund`);

                document.getElementById("withdrawAmount").value = 0;
            } catch (err) {
                logToPanel("[ERROR] Withdraw failed: " + err);
            }
        });
    }

    // Start Bot button
    const startBtn = document.getElementById("startBtn");
    if (startBtn) {
        startBtn.addEventListener("click", async () => {
            const launchEngine = window.startSimulation || window.initBot;
            if (typeof launchEngine === "function") {
                setButtonStates("running");
                try {
                    await launchEngine(10);
                } catch (err) {
                    logToPanel("[ERROR] Bot execution runtime error: " + err.message);
                } finally {
                    setButtonStates("completed");
                }
            } else {
                logToPanel("[ERROR] Simulation loop entry engine not found.");
            }
        });
    }

    // Stop Bot button
    const stopBtn = document.getElementById("stopBtn");
    if (stopBtn) {
        stopBtn.addEventListener("click", () => {
            window.isBotRunning = false;
            if (typeof window.stopBot === "function") {
                window.stopBot();
            }
            setButtonStates("completed");
            logToPanel("[MANUAL] Stop signal flagged. Halting background sequence.");
        });
    }
});
