// === Helpers ===
function safeParse(text) { try { return JSON.parse(text); } catch { return null; } }

function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) panel.value += msg + "\n";
    else console.log(msg);
}

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) statusLabel.textContent = "STATE: " + text;
    else logToPanel(`[STATUS UPDATE] ${text}`);
};

// === Data loader (fetch-only, DOM-free) ===
async function fetchClientRecord(userId) {
    try {
        logToPanel(`[INFO] Fetching client record for ${userId} (DB only)...`);
        const resp = await fetch(`${window.BASE_URL}/clients`, {
            headers: { "ngrok-skip-browser-warning": "true" }
        });
        const data = safeParse(await resp.text());
        if (!data || !Array.isArray(data.clients) || data.clients.length === 0) {
            logToPanel("[WARN] No clients found in DB payload");
            return null;
        }
        const client = data.clients.find(c => c.profile && c.profile.name === userId);
        if (!client) {
            logToPanel(`[WARN] User ${userId} not found in database records.`);
            return null;
        }

        // Normalize values
        const walletBalance = client.wallet?.balance ? parseFloat(client.wallet.balance) : 0;
        const profitTarget  = client.settings?.profit_target ? parseFloat(client.settings.profit_target) : 1.0;
        const stopLoss      = client.settings?.stop_loss ? parseFloat(client.settings.stop_loss) : 0.5;
        const profileName   = client.profile?.name || userId;

        return {
            profileName,
            walletBalance,
            profitTarget,
            stopLoss
        };
    } catch (e) {
        logToPanel("[ERROR] fetchClientRecord failed: " + e.message);
        return null;
    }
}

// === On-chain & fund balances fetch (fetch-only) ===
async function fetchLiveBalances(userid, walletAddr) {
    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/balances`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ userid, wallet_address: walletAddr })
        });
        const balData = safeParse(await resp.text());
        if (!balData) {
            logToPanel("[WARN] /runbot/balances returned no JSON");
            return null;
        }
        if (balData.status === "balances fetched") {
            return balData;
        } else if (balData.status === "error") {
            logToPanel(`[ERROR] Balances endpoint error: ${balData.message}`);
            return null;
        } else {
            logToPanel("[WARN] Unexpected balances response");
            return null;
        }
    } catch (err) {
        logToPanel("[ERROR] fetchLiveBalances failed: " + err.message);
        return null;
    }
}

// === Apply client data to DOM (safe checks) ===
async function applyClientDataToUI(userId) {
    // Fetch DB record
    const client = await fetchClientRecord(userId);
    if (!client) {
        // still attempt to fetch live balances to show something
        const fallbackBal = await fetchLiveBalances(userId, window.USER_ADDR);
        if (fallbackBal) {
            logToPanel(`[INFO] Fund token balance: ${fallbackBal.fund_token_balance}, ETH balance: ${fallbackBal.eth_balance_wei}`);
        }
        return;
    }

    // Update DOM elements only if they exist
    const startEl = document.getElementById("startBalance");
    const investEl = document.getElementById("investBalance");
    const profitEl = document.getElementById("profitTarget");
    const stopEl = document.getElementById("stopLoss");
    const userLabel = document.getElementById("currentUserLabel");

    if (startEl) startEl.value = client.walletBalance;
    if (investEl) investEl.value = client.walletBalance;
    if (profitEl) profitEl.value = client.profitTarget;
    if (stopEl) stopEl.value = client.stopLoss;
    if (userLabel) userLabel.textContent = client.profileName;

    logToPanel(`[DB SUCCESS] Loaded ${client.profileName}. Balance: ${client.walletBalance} USDT.`);

    // Fetch live balances and log them
    const live = await fetchLiveBalances(userId, window.USER_ADDR);
    if (live) {
        logToPanel(`[INFO] Fund token balance: ${live.fund_token_balance}, ETH balance: ${live.eth_balance_wei}`);
    }

    // Sync clients.db with current balance (best-effort)
    try {
        await fetch(`${window.BASE_URL}/runbot/update_client`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                userid: window.USER_ID,
                current_balance: client.walletBalance
            })
        });
        logToPanel("[DB SUCCESS] Client balance synchronized on page load");
    } catch (err) {
        logToPanel("[WARN] Failed to sync client balance: " + err.message);
    }
}

// === Withdraw / Payout Handler ===
async function runWithdraw(amount) {
    if (amount <= 0) {
        logToPanel("[WARN] Withdraw amount must be greater than zero.");
        return;
    }

    logToPanel(`[INFO] Initiating withdrawal of ${amount} USDT`);

    const gasFeeUSDT = 0.05;
    const netWithdraw = amount - gasFeeUSDT;

    if (netWithdraw <= 0) {
        logToPanel("[ERROR] Withdrawal too small after gas fee.");
        return;
    }

    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/payout`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                wallet_address: window.USER_ADDR,
                private_key: window.USER_PK,
                amount: netWithdraw
            })
        });

        const parsed = safeParse(await resp.text());

        if (parsed?.status === "withdrawal executed") {
            logToPanel(`[SUCCESS] Withdrawal transaction confirmed. TxHash=${parsed.tx_hash}`);

            const investEl = document.getElementById("investBalance");
            const startEl = document.getElementById("startBalance");
            if (investEl) investEl.value = netWithdraw.toFixed(2);
            if (startEl) startEl.value = netWithdraw.toFixed(2);

            // Sync clients.db with new balance
            await fetch(`${window.BASE_URL}/runbot/update_client`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
                body: JSON.stringify({
                    userid: window.USER_ID,
                    current_balance: netWithdraw,
                    amount_withdrawn: amount
                })
            });
            logToPanel("[DB SUCCESS] Client balance updated in clients.db");
        } else if (parsed?.status === "error") {
            logToPanel(`[ERROR] Backend error: ${parsed.message}`);
        } else {
            logToPanel("[WARN] Withdrawal rejected by backend.");
            if (parsed) logToPanel(`[DEBUG] Backend response: ${JSON.stringify(parsed)}`);
        }
    } catch (err) {
        logToPanel(`[ERROR] Withdraw request failed: ${err.message}`);
    }
}

// === Deposit Handler ===
async function runDeposit(amount) {
    if (amount <= 0) {
        logToPanel("[WARN] Deposit amount must be greater than zero.");
        return;
    }

    logToPanel(`[INFO] Initiating deposit of ${amount} USDT`);

    const gasFeeUSDT = 0.05;
    const commissionCoins = 2;
    const commissionUSDT = commissionCoins * 1; // USDT stable
    const netDeposit = amount - commissionUSDT - gasFeeUSDT;

    if (netDeposit <= 0) {
        logToPanel("[ERROR] Deposit too small after fees/commission.");
        return;
    }

    logToPanel(`[INFO] Commission deducted: ${commissionCoins} coins (${commissionUSDT.toFixed(2)} USDT), Gas fee: ${gasFeeUSDT} USDT`);
    logToPanel(`[INFO] Net deposit credited: ${netDeposit.toFixed(2)} USDT`);

    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                wallet_address: window.USER_ADDR,
                private_key: window.USER_PK,
                amount: netDeposit
            })
        });

        const parsed = safeParse(await resp.text());

        if (parsed?.status === "deposit executed") {
            logToPanel(`[SUCCESS] Deposit transaction confirmed. TxHash=${parsed.tx_hash}`);

            const investEl = document.getElementById("investBalance");
            const startEl = document.getElementById("startBalance");
            if (investEl) investEl.value = netDeposit.toFixed(2);
            if (startEl) startEl.value = netDeposit.toFixed(2);

            await fetch(`${window.BASE_URL}/runbot/update_client`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
                body: JSON.stringify({
                    userid: window.USER_ID,
                    current_balance: netDeposit,
                    amount_deposited: amount
                })
            });
            logToPanel("[DB SUCCESS] Client balance updated in clients.db");
        } else if (parsed?.status === "error") {
            logToPanel(`[ERROR] Backend error: ${parsed.message}`);
        } else {
            logToPanel("[WARN] Deposit rejected by backend.");
            if (parsed) logToPanel(`[DEBUG] Backend response: ${JSON.stringify(parsed)}`);
        }
    } catch (err) {
        logToPanel(`[ERROR] Deposit request failed: ${err.message}`);
    }
}
