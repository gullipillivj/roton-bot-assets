// === Helpers ===
function safeParse(text) { try { return JSON.parse(text); } catch { return null; } }
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) panel.value += msg + "\n";
    else console.log(msg);
}
window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) statusLabel.textContent = "STATE: " + text;
    else logToPanel(`[STATUS UPDATE] ${text}`);
};

// === Data loader ===
async function fetchClientRecord(userId) {
    try {
        logToPanel(`[INFO] Fetching client record for ${userId}...`);
        const resp = await fetch(`${window.BASE_URL}/clients`, {
            headers: { "ngrok-skip-browser-warning": "true" }
        });
        const data = safeParse(await resp.text());
        if (!data || !Array.isArray(data.clients)) return null;
        return data.clients.find(c => c.profile?.name === userId) || null;
    } catch (e) {
        logToPanel("[ERROR] fetchClientRecord failed: " + e.message);
        return null;
    }
}

async function applyClientDataToUI(userId) {
    const client = await fetchClientRecord(userId);
    if (!client) return;

    const walletBalance = parseFloat(client.wallet?.balance || 0);
    const startEl = document.getElementById("startBalance");
    const investEl = document.getElementById("investBalance");
    const userLabel = document.getElementById("currentUserLabel");

    if (startEl) startEl.value = walletBalance;
    if (investEl) investEl.value = walletBalance;
    if (userLabel) userLabel.textContent = client.profile?.name || userId;

    logToPanel(`[DB SUCCESS] Loaded ${userId}. Balance: ${walletBalance} USDT.`);
}

// === Withdraw Handler ===
async function runWithdraw(amount) {
    if (amount <= 0) {
        logToPanel("[WARN] Withdraw amount must be greater than zero.");
        return;
    }
    logToPanel(`[INFO] Initiating withdrawal of ${amount} USDT`);

    const gasFeeUSDT = 0.05;
    const netWithdraw = amount - gasFeeUSDT;
    if (netWithdraw <= 0) {
        logToPanel("[ERROR] Withdrawal too small after gas fee.");
        return;
    }

    try {
        const resp = await fetch(`${window.BASE_URL}/runbot/payout`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                wallet_address: window.USER_ADDR,
                private_key: window.USER_PK,
                amount: netWithdraw
            })
        });
        const parsed = safeParse(await resp.text());

        if (parsed?.status === "withdrawal executed") {
            logToPanel(`[SUCCESS] Withdrawal confirmed. TxHash=${parsed.tx_hash}`);

            document.getElementById("investBalance").value = netWithdraw.toFixed(2);
            document.getElementById("startBalance").value = netWithdraw.toFixed(2);

            // ✅ Only send amount_withdrawn
            await fetch(`${window.BASE_URL}/runbot/update_client`, {
                method: "POST",
                headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
                body: JSON.stringify({
                    userid: window.USER_ID,
                    amount_withdrawn: amount
                })
            });
            logToPanel("[DB SUCCESS] Withdrawal recorded in clients.db");
        } else {
            logToPanel("[WARN] Withdrawal rejected.");
        }
    } catch (err) {
        logToPanel(`[ERROR] Withdraw failed: ${err.message}`);
    }
}

// === Deposit Handler ===

        logToPanel("[ERROR] Deposit too small after fees.");
        return;

            try {
                await launchEngine(10);
            } catch (err) {
                logToPanel("[ERROR] Simulation failed: " + err.message);
            } finally {
                window.setStatus("COMPLETED");
                if (typeof commitFinalPnLToDatabase === "function") {
                    await commitFinalPnLToDatabase(window.cumulativePnL);
                }
            }
        }
    });

    document.getElementById("stopBtn")?.addEventListener("click", () => {
        window.isBotRunning = false;
        window.setStatus("STOPPED");
        logToPanel("[MANUAL] Bot stopped.");
    });

    document.getElementById("logoutBtn")?.addEventListener("click", () => {
        logToPanel("[INFO] Logging out...");
        window.location.href = `${window.BASE_URL}/login`;
    });
