// === Helpers ===
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) panel.value += msg + "\n";
}
function safeParse(text) { try { return JSON.parse(text); } catch { return null; } }

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) {
        statusLabel.textContent = "STATE: " + text;
    } else {
        logToPanel(`[STATUS UPDATE] ${text}`);
    }
};
if (typeof globalThis !== 'undefined') globalThis.setStatus = window.setStatus;

// === Chart Initialization ===
function initChart() {
    const canvas = document.getElementById("balanceChart");
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    window.balanceChart = new Chart(ctx, {
        type: "line",
        data: {
            labels: [],
            datasets: [
                { label: "Running Balance", data: [], borderColor: "#00FF00", fill: false },
                { label: "Cumulative PnL", data: [], borderColor: "#FF3333", fill: false }
            ]
        },
        options: { animation: false, responsive: true }
    });
}
window.updateChart = function(runningBalance, cumulativePnL) {
    try {
        if (window.balanceChart && window.balanceChart.data) {
            const timeLabel = new Date().toLocaleTimeString();
            window.balanceChart.data.labels.push(timeLabel);
            window.balanceChart.data.datasets[0].data.push(runningBalance);
            window.balanceChart.data.datasets[1].data.push(cumulativePnL);
            window.balanceChart.update();
        }
    } catch (e) { console.error("Chart updating skipped: ", e.message); }
};

// === Client Data Loader ===
async function loadClientData(userId) {
    try {
        logToPanel(`[INFO] Fetching live database records for ${userId}...`);
        const resp = await fetch(`${BASE_URL}/clients`, {
            headers: { "ngrok-skip-browser-warning": "true" }
        });
        const data = safeParse(await resp.text());
        if (!data || !data.clients || data.clients.length === 0) {
            logToPanel("[WARN] No clients found in DB payload");
            return;
        }
        const client = data.clients.find(c => c.profile && c.profile.name === userId);
        if (!client) {
            logToPanel(`[WARN] User ${userId} not found in database records.`);
            return;
        }
        const walletBalance = (client.wallet && client.wallet.balance !== null) ? parseFloat(client.wallet.balance) : 1000;
        const profitTarget  = (client.settings && client.settings.profit_target) ? parseFloat(client.settings.profit_target) : 1.0;
        const stopLoss      = (client.settings && client.settings.stop_loss) ? parseFloat(client.settings.stop_loss) : 0.5;
        const profileName   = client.profile.name || userId;

        logToPanel(`[DB SUCCESS] Loaded ${profileName}. Balance: ${walletBalance} USDT.`);

        document.getElementById("startBalance").value = walletBalance;
        document.getElementById("investBalance").value = walletBalance;
        document.getElementById("profitTarget").value = profitTarget;
        document.getElementById("stopLoss").value = stopLoss;
        const userLabel = document.getElementById("currentUserLabel");
        if (userLabel) userLabel.textContent = profileName;
    } catch (e) {
        logToPanel("[ERROR] loadClientData query failed: " + e.message);
    }
}

// === Balance / Deposit / Withdraw ===
async function refreshBalances() {
    try {
        const resp = await fetch(`${BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ wallet_address: window.USER_ADDR, private_key: window.USER_PK, amount: 0 })
        });
        const parsed = safeParse(await resp.text());
        if (parsed?.fund_token_balance !== undefined) {
            document.getElementById("startBalance").value = parsed.fund_token_balance;
            document.getElementById("investBalance").value = parsed.fund_token_balance;
            logToPanel(`[INFO] Fund Smart Contract Balance: ${parsed.fund_token_balance} units.`);
        }
    } catch (err) {
        logToPanel(`[ERROR] Balance sync skipped: ${err.message}`);
    }
}

async function runDeposit(amount) {
    try {
        logToPanel(`[INFO] Routing deposit payload for ${amount}...`);
        await fetch(`${BASE_URL}/runbot/approve`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ wallet_address: window.USER_ADDR, private_key: window.USER_PK, amount })
        });
        const resp = await fetch(`${BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ wallet_address: window.USER_ADDR, private_key: window.USER_PK, amount })
        });
        const parsed = safeParse(await resp.text());
        if (parsed?.tx_hash) {
            logToPanel(`[SUCCESS] On-Chain Deposit TX confirmed: ${parsed.tx_hash}`);
            await loadClientData(window.USER_ID);
        }
    } catch (err) {
        logToPanel(`[ERROR] Deposit sequence aborted: ${err.message}`);
    }
}

async function runWithdraw(amount) {
    try {

        const resp = await fetch(`${BASE_URL}/runbot/payout`, {

        });
        const parsed = safeParse(await resp.text());
        if (parsed?.tx_hash) {
            logToPanel(`[SUCCESS] Payout TX confirmed: ${parsed.tx_hash}`);
            await loadClientData(window.USER_ID);
        }
    } catch (err) {
        logToPanel(`[ERROR] Withdrawal execution failed: ${err.message}`);
    }
}

        const launchEngine = window.startSimulation;
        if (typeof launchEngine === "function") {
            window.setStatus("RUNNING");
            logToPanel("[INFO] Start Trading button pressed — launching simulation loop");
            try {
                await launchEngine(10); // simulation handles DB commit internally
            } catch (err) {
                logToPanel("[ERROR] Runtime simulation engine drop: " + err.message);
            } finally {
                window.setStatus("COMPLETED");
            }
        } else {
            logToPanel("[ERROR] Target simulation script execution engine not found.");
        }
    });

    document.getElementById("stopBtn")?.addEventListener("click", () => {
        window.isBotRunning = false;
        window.setStatus("STOPPED");

    });
