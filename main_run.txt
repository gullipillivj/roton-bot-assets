function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.value += msg + "\n";
        panel.scrollTop = panel.scrollHeight; 
    }
}

function safeParse(text) {
    try { return JSON.parse(text); } catch { return null; }
}

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) {
        statusLabel.textContent = "STATE: " + text;
    } else {
        logToPanel(`[STATUS UPDATE] ${text}`);
    }
};

if (typeof globalThis !== 'undefined') {
    globalThis.setStatus = window.setStatus;
}

window.updateChart = function(runningBalance, cumulativePnL) {
    try {
        if (window.balanceChart && window.balanceChart.data) {
            const timeLabel = new Date().toLocaleTimeString();
            window.balanceChart.data.labels.push(timeLabel);
            
            if (window.balanceChart.data.datasets[0] && window.balanceChart.data.datasets[0].data) {
                window.balanceChart.data.datasets[0].data.push(runningBalance);
            }
            if (window.balanceChart.data.datasets[1] && window.balanceChart.data.datasets[1].data) {
                window.balanceChart.data.datasets[1].data.push(cumulativePnL);
            }
            window.balanceChart.update();
        }
    } catch (e) {
        console.error("Chart updating skipped: ", e.message);
    }
};

window.approveToken = async function(amount, walletAddress, privateKey) {
    try {
        const weiValueStr = (BigInt(Math.floor(parseFloat(amount) * 1000000)) * BigInt(10**12)).toString();
        const resp = await fetch(`${BASE_URL}/runbot/approve`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                amount: weiValueStr,
                wallet_address: walletAddress,
                private_key: privateKey
            })
        });
        return safeParse(await resp.text());
    } catch (err) {
        logToPanel(`[ERROR] approveToken API bridge failed: ${err.message}`);
        return null;
    }
};

async function runDeposit(amount) {
    try {

        const weiValueStr = (BigInt(Math.floor(parseFloat(amount) * 1000000)) * BigInt(10**12)).toString();
        const resp = await fetch(`${BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ wallet_address: USER_ADDR, private_key: USER_PK, amount: weiValueStr })
        });
        const parsed = safeParse(await resp.text());
        if (parsed?.tx_hash) {
            logToPanel(`[SUCCESS] On-Chain Deposit TX confirmed: ${parsed.tx_hash}`);
        }
    } catch (err) {
        logToPanel(`[ERROR] Deposit sequence aborted: ${err.message}`);
    }
}

    document.getElementById("investBalance").value = 1000;

        document.getElementById("currentUserLabel").textContent = window.USER_ID;
    }

    document.getElementById("depositBtn")?.addEventListener("click", () => {
        const amt = parseFloat(document.getElementById("depositAmount").value) || 0;

    });

    document.getElementById("withdrawBtn")?.addEventListener("click", () => {

        runWithdraw(amt);
    });

    document.getElementById("startBtn")?.addEventListener("click", async () => {
        const launchEngine = window.startSimulation || window.initBot;
        if (typeof launchEngine !== "function") {
            logToPanel("[ERROR] Simulation loop entry engine not found.");
            return;
        }

        logToPanel("[INFO] Start Trading button pressed — launching simulation loop");
        
        try {
            window.cumulativePnL = 0;
            
            // A. Execute trades from simulate_run.txt first
            await launchEngine(10); 
            
            // B. SUCCESS PATH: Complete loop cleanly without hitting any SQLite updates
            window.setStatus("COMPLETED");
            logToPanel("[FINISHED] All trading cycles processed in memory successfully.");
            
        } catch (err) {
            // C. CRASH PATH: Logs errors natively while ensuring database channels remain untouched
            window.setStatus("CRASHED");
            logToPanel("[CRITICAL ERROR] Trading loop broke mid-flight: " + err.message);
        }
    });

    document.getElementById("stopBtn")?.addEventListener("click", () => {
        window.isBotRunning = false;
        window.setStatus("STOPPED");
        logToPanel("[MANUAL] Bot process sequence flagged to stop.");
    });
});
