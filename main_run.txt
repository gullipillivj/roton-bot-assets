function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) { panel.value += msg + "\n"; panel.scrollTop = panel.scrollHeight; }
}

function safeParse(text) {
    try { return JSON.parse(text); } catch { return null; }
}

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) {
        statusLabel.textContent = "STATE: " + text;
    } else {
        logToPanel(`[STATUS UPDATE] ${text}`);
    }
};

if (typeof globalThis !== 'undefined') {
    globalThis.setStatus = window.setStatus;
}

// 🛡️ BUTTON METRICS TRACKER: Added back to context
window.setButtonStates = function(status) {
    const startBtn = document.getElementById("startBtn");
    const stopBtn = document.getElementById("stopBtn");
    
    if (!startBtn || !stopBtn) return;

    if (status === "running") {
        startBtn.disabled = true;
        startBtn.style.backgroundColor = "#2e7d32";
        startBtn.textContent = "Trading Active...";
        stopBtn.disabled = false;
        window.setStatus("RUNNING");
    } else if (status === "completed" || status === "stopped") {
        startBtn.disabled = false;
        startBtn.style.backgroundColor = "green";
        startBtn.textContent = "Start Trading";
        stopBtn.disabled = true;
        window.setStatus("COMPLETED");
    }
};

// 📈 CHART APPARATUS HOOK: Maps data streams cleanly matching chart.txt arrays
window.updateChart = function(balance, profit = null) {
    try {
        const chart = window.balanceChart;
        if (!chart || !chart.data) return;

        const cycle = chart.data.labels.length + 1;
        chart.data.labels.push(`Cycle ${cycle}`);

        if (chart.data.datasets && chart.data.datasets[0] && chart.data.datasets[0].data) {
            chart.data.datasets[0].data.push(balance);
        }
        if (profit !== null && chart.data.datasets && chart.data.datasets[1] && chart.data.datasets[1].data) {
            chart.data.datasets[1].data.push(profit);
        }

        chart.update();
        logToPanel(`[CHART] Render complete. Cycle ${cycle} -> Balance: ${balance.toFixed(2)} USDT`);
    } catch (e) {
        console.error("Chart push error handled: ", e.message);
    }
};

// 🛡️ REVERSE PROXY INTERCEPTOR
const originalFetch = window.fetch;
window.fetch = async function(url, options) {
    let urlStr = typeof url === 'string' ? url : '';
    
    if (options) {
        if (!options.headers) options.headers = {};
        options.headers["ngrok-skip-browser-warning"] = "true";
    }

    if (options && options.body && (urlStr.includes('approve') || urlStr.includes('swap') || urlStr.includes('deposit') || urlStr.includes('payout'))) {
        try {
            let parsedBody = JSON.parse(options.body);
            if (parsedBody.amountIn !== undefined && parsedBody.amount === undefined) {
                parsedBody.amount = parsedBody.amountIn;
            }
            if (parsedBody.amount !== undefined) {
                parsedBody.amount = Math.floor(parseFloat(parsedBody.amount));
            }
            options.body = JSON.stringify(parsedBody);
        } catch(e) {}
    }
    return originalFetch(url, options);
};

// Load initial balance records directly from your live Clients database endpoint
async function loadClientData(userId) {
    try {
        logToPanel(`[INFO] Fetching live database records for ${userId}...`);
        const resp = await fetch(`${BASE_URL}/clients`);
        const data = safeParse(await resp.text());
        
        if (!data || !data.clients || data.clients.length === 0) {
            logToPanel("[WARN] No clients found in DB payload");
            return;
        }

        const client = data.clients.find(c => c.profile && c.profile.name === userId);
        if (!client) {
            logToPanel(`[WARN] User ${userId} not found in database records.`);
            return;
        }

        const walletBalance = (client.wallet && client.wallet.balance !== null) ? parseFloat(client.wallet.balance) : 1000;
        const profitTarget  = (client.settings && client.settings.profit_target) ? parseFloat(client.settings.profit_target) : 1.0;
        const stopLoss      = (client.settings && client.settings.stop_loss) ? parseFloat(client.settings.stop_loss) : 0.5;
        const profileName   = client.profile.name || userId;

        logToPanel(`[DB SUCCESS] Loaded ${profileName}. Balance: ${walletBalance} USDT.`);

        document.getElementById("startBalance").value = walletBalance;
        document.getElementById("investBalance").value = walletBalance;
        document.getElementById("profitTarget").value = profitTarget;
        document.getElementById("stopLoss").value = stopLoss;
        
        const userLabel = document.getElementById("currentUserLabel");
        if (userLabel) userLabel.textContent = profileName;

    } catch (e) {
        logToPanel("[ERROR] loadClientData query failed: " + e.message);
    }
}

// Write P&L metrics back to your SQLite server storage file
async function commitFinalPnLToDatabase(finalPnL) {
    try {
        const startingAllocated = parseFloat(document.getElementById("startBalance").value) || 1000;
        const updatedNetBalance = startingAllocated + finalPnL;
        
        logToPanel(`[DB UPDATE] Simulation complete. Synchronizing updated balance (${updatedNetBalance.toFixed(2)} USDT) to SQLite...`);

        const resp = await fetch(`${BASE_URL}/runbot/update_client`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                userid: window.USER_ID,
                amount_deposited: Math.floor(startingAllocated),
                current_balance: Math.floor(updatedNetBalance)
            })
        });
        
        const parsed = safeParse(await resp.text());
        if (parsed?.status === "client updated") {
            logToPanel(`[SQLITE SUCCESS] Final performance data saved to clients.db for ${window.USER_ID}.`);
            document.getElementById("investBalance").value = updatedNetBalance.toFixed(2);
        } else {
            logToPanel("[WARN] Database adjustment rejected by server script.");
        }
    } catch (err) {
        logToPanel(`[DATABASE CRITICAL ERROR] Failed to save end balance state: ${err.message}`);
    }
}
