async function runDeposit(amount) {
    logToPanel(`[MOCK DEPOSIT] Processing local test ledger update for ${amount} tokens.`);
    const startBalanceBox = document.getElementById("startBalance");
    const currentBalance = parseFloat(startBalanceBox.value) || 0;
    startBalanceBox.value = currentBalance + amount;
}

async function runWithdraw(amount) {
    logToPanel(`[MOCK WITHDRAWAL] Processing local test ledger payout for ${amount} tokens.`);
    const startBalanceBox = document.getElementById("startBalance");
    const currentBalance = parseFloat(startBalanceBox.value) || 0;
    if (currentBalance >= amount) {
        startBalanceBox.value = currentBalance - amount;
    } else {
        logToPanel("[WARN] Insufficient balance for local withdrawal simulated block.");
    }
}
