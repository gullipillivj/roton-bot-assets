// === Helpers ===
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.value += msg + "\n";
        panel.scrollTop = panel.scrollHeight; 
    }
}

function safeParse(text) {
    try { return JSON.parse(text); } catch { return null; }
}

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) {
        statusLabel.textContent = "STATE: " + text;
    } else {
        logToPanel(`[STATUS UPDATE] ${text}`);
    }
};

if (typeof globalThis !== 'undefined') {
    globalThis.setStatus = window.setStatus;
}

// === Chart Initialization ===
function initChart() {
    const ctx = document.getElementById("balanceChart").getContext("2d");
    window.balanceChart = new Chart(ctx, {
        type: "line",
        data: {
            labels: [],
            datasets: [
                { label: "Running Balance", data: [], borderColor: "#00FF00" },
                { label: "Cumulative PnL", data: [], borderColor: "#FF3333" }
            ]
        }
    });
}

window.updateChart = function(runningBalance, cumulativePnL) {
    try {
        if (window.balanceChart && window.balanceChart.data) {
            const timeLabel = new Date().toLocaleTimeString();
            window.balanceChart.data.labels.push(timeLabel);
            window.balanceChart.data.datasets[0].data.push(runningBalance);
            window.balanceChart.data.datasets[1].data.push(cumulativePnL);
            window.balanceChart.update();
        }
    } catch (e) {
        console.error("Chart updating skipped: ", e.message);
    }
};

// === API Calls ===
window.approveToken = async function(amount, walletAddress, privateKey) {
    try {
        const resp = await fetch(`${BASE_URL}/runbot/approve`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ amount, wallet_address: walletAddress, private_key: privateKey })
        });
        return safeParse(await resp.text());
    } catch (err) {
        logToPanel(`[ERROR] approveToken failed: ${err.message}`);
        return null;
    }
};

// === Deposit Function ===
async function runDeposit(amount) {
    logToPanel(`[INFO] Routing deposit payload for ${amount}...`);
    try {
        const resp = await fetch(`${BASE_URL}/runbot/deposit`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({ wallet_address: USER_ADDR, private_key: USER_PK, amount })
        });
        const parsed = safeParse(await resp.text());
        if (parsed?.tx_hash) {
            logToPanel(`[SUCCESS] Deposit TX confirmed: ${parsed.tx_hash}`);
        }
        if (parsed?.fund_token_balance !== undefined) {
            document.getElementById("startBalance").value = parsed.fund_token_balance;
            document.getElementById("investBalance").value = parsed.fund_token_balance;
        }
    } catch (err) {
        logToPanel(`[ERROR] Deposit failed: ${err.message}`);
    }
}

// === Commit Final PnL to Database ===
async function commitFinalPnLToDatabase(finalPnL) {
    try {
        const startingAllocated = parseFloat(document.getElementById("startBalance").value) || 1000;
        const updatedNetBalance = startingAllocated + finalPnL;

        logToPanel(`[DB UPDATE] Simulation complete. Synchronizing updated balance (${updatedNetBalance.toFixed(2)} USDT) to SQLite...`);

        const resp = await fetch(`${BASE_URL}/runbot/update_client`, {
            method: "POST",
            headers: { "Content-Type": "application/json", "ngrok-skip-browser-warning": "true" },
            body: JSON.stringify({
                userid: window.USER_ID,
                current_balance: updatedNetBalance,
                wallet_address: USER_ADDR,
                email: `${window.USER_ID}@example.com`,
                status: "ACTIVE"
            })
        });

        const parsed = safeParse(await resp.text());
        if (parsed?.status === "client updated") {
            logToPanel(`[SQLITE SUCCESS] Final performance data persisted for user ${window.USER_ID}.`);
            document.getElementById("investBalance").value = updatedNetBalance.toFixed(2);
        } else {
            logToPanel("[WARN] Database adjustment rejected by server script.");
        }
    } catch (err) {
        logToPanel(`[DATABASE CRITICAL ERROR] Failed to save end balance state: ${err.message}`);
    }
}
