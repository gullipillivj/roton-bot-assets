async function categorizeCoins() {
    try {
        // Fetch pre-processed tokens directly from the multi-stage filter pipeline
        const backendUrl = "/prodbot/filtertokens";
        const res = await fetch(backendUrl);
        let responseData = await res.json();

        let shortTermList = [];
        if (responseData && responseData.short_term) { shortTermList = responseData.short_term; }
        else if (Array.isArray(responseData)) { shortTermList = responseData; }

        // 🌟 FRONTEND ISOLATION LAYER: Extract and lock token type parameters in client runtime state
        shortTermList.forEach(function(coin) {
            coin.network_assigned = coin.tokentype || "bep20";
        });

        window.logToPanel("✅ Filter Synchronized: " + shortTermList.length + " assets passed verification.");
        return { short_term: shortTermList, mid_term: [], long_term: [] };

    } catch (err) {
        window.logToPanel("⚠️ Coin Filter Layer Exception: " + err.message);
        return { short_term: [], mid_term: [], long_term: [] };
    }
}

async function getTradeCandidates() {
    try {
        var marketData = await categorizeCoins();
        var allTokens = [];
        if (marketData && marketData.short_term) {
            allTokens = marketData.short_term;
        }

        // Sort based on the updated profit key schema '%profit'
        allTokens.sort(function (a, b) {
            return Number(b["%profit"] || 0) - Number(a["%profit"] || 0);
        });
        return allTokens;
    } catch (err) {
        return [{ symbol: "MOVR", address: "0x0f93f40d6fa22e2b0de75b11eb55df1d47864cb5", "%profit": 87.57, price: 2.671, rank: 1, "usd value": 50000 }];
    }
}
window.getTradeCandidates = getTradeCandidates;

window.getBinancePairs = async function () {
    var candidates = await window.getTradeCandidates();
    return candidates.map(function (coin) {
        return coin.symbol + "USDT";
    });
};

window.evaluateCoin = async function (symbol) {
    var candidates = await window.getTradeCandidates();
    var cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
    var i;
    
    for (i = 0; i < candidates.length; i++) {
        if (String(candidates[i].symbol).toUpperCase().trim() === cleanSymbol) {
            var token = candidates[i];
            return {
                symbol: token.symbol,
                tokenAddress: token.address,
                network: "bep20",
                router: "pancakeswap",
                change24h: Number(token["%profit"] || 0), // Bridges old property variable names
                lastPrice: Number(token.price || 0)
            };
        }
    }
    return null;
};
