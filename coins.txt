(function() {
  window.logToPanel("coins.txt loaded safely.");

  // ✅ Dynamic token discoverer injecting your custom tokens directly into fallback paths
  async function categorizeCoins() {
    window.logToPanel("🔎 Querying live dynamic market trends from custom backend server proxy...");
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
        const data = await res.json();
        
        if (data && Array.isArray(data.short_term) && data.short_term.length > 0) {
            const absoluteTopCandidate = data.short_term[0];
            window.logToPanel(`📈 Live market metrics parsed. Leading velocity vector: ${absoluteTopCandidate.symbol} (+${absoluteTopCandidate.change24h}%)`);
            return {
                short_term: data.short_term,
                mid_term: data.mid_term || data.short_term,
                long_term: data.long_term || data.short_term
            };
        }
        throw new Error("Payload did not provide valid token list arrays.");
    } catch (err) {
        console.error("Backend dynamic matrix check failed, applying fallbacks:", err);
        
        const customFallbackPool = [
            { symbol: "RR-BNB", tokenAddress: "0xba67186fd454D0256a0D3744783538bE08C58feD", network: "bep20", change24h: 12.4 },
            { symbol: "RR-ETH", tokenAddress: "0x8A0cAC8E7F6305cE6EeB585005919290F78808f9", network: "bep20", change24h: 9.1 },
            { symbol: "XVS", tokenAddress: "0xcF6BB5389c92Bdda8a3747Ddb454cB7a64E86902", network: "bep20", change24h: 5.2 }
        ];
        return { long_term: customFallbackPool, mid_term: customFallbackPool, short_term: customFallbackPool };
    }
  }

  // Global bindings safely attached to window interface scope
  window.getBinancePairs = async () => ["CAKEUSDT", "XVSUSDT", "LINKUSDT", "RR-BNBUSDT", "RR-ETHUSDT"];
  window.evaluateCoin = async (s) => null;
  window.categorizeCoins = categorizeCoins;
  window.logToPanel("coins.txt initialized with dynamic custom channels.");
})();
