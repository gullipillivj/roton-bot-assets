// ==================================================================
// 🎛️ COINS FRONTEND MANAGEMENT CONTEXT INTERFACE (DYNAMIC PROD)
// ==================================================================

(function () {

    window.logToPanel("coins.txt loaded safely.");

    // ==============================================================
    // 📊 Market Analysis Loader
    // ==============================================================
    async function categorizeCoins() {
        try {
            const baseApiUrl = window.API_BASE || window.location.origin;
            const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
            const data = await res.json();

            return {
                short_term: data.short_term || [],
                mid_term: data.mid_term || [],
                long_term: data.long_term || []
            };
        } catch (err) {
            console.error("Market analysis connection failure:", err);
            return { short_term: [], mid_term: [], long_term: [] };
        }
    }

    // ==============================================================
    // 🪙 Supported Trading Pairs Mirror Loop
    // ==============================================================
    window.getBinancePairs = async () => {
        try {
            const marketData = await categorizeCoins();
            const allTokens = [
                ...(marketData.short_term || []),
                ...(marketData.mid_term || []),
                ...(marketData.long_term || [])
            ];
            
            if (allTokens.length > 0) {
                return allTokens.map(t => t.marketPair || `${t.symbol}USDT`);
            }
        } catch (e) {
            console.error(e);
        }
        return ["CAKEUSDT"];
    };

    // ==============================================================
    // 🔍 Token Resolver (Bridges Address vectors safely to simulate_prod)
    // ==============================================================
    window.evaluateCoin = async function (symbol) {
        try {
            const marketData = await categorizeCoins();
            const allTokens = [
                ...(marketData.short_term || []),
                ...(marketData.mid_term || []),
                ...(marketData.long_term || [])
            ];

            // Strip out base trading extensions to cleanly check ticker symbols
            const cleanSymbol = String(symbol)
                .replace("USDT", "")
                .toUpperCase()
                .trim();

            const token = allTokens.find(
                x => String(x.symbol).toUpperCase().trim() === cleanSymbol
            );

            if (!token) {
                window.logToPanel(`[WARN] Token not discovered inside active pools: ${cleanSymbol}`);
                return null;
            }

            // 🚀 SUCCESS: Extract the verified contract address string provided directly by the backend response
            const verifiedContractAddress = token.tokenAddress;

            if (!verifiedContractAddress) {
                window.logToPanel(`[WARN] Payload vector is missing tokenAddress string properties for token: ${cleanSymbol}`);
                return null;
            }

            return {
                symbol: token.symbol,
                tokenAddress: verifiedContractAddress, // Passed securely down to simulateCycle()
                network: "bep20",
                router: "pancakeswap",
                change24h: token.change24h
            };

        } catch (err) {
            window.logToPanel(`[ERROR] evaluateCoin resolution crashed: ${err.message}`);
            return null;
        }
    };

    window.categorizeCoins = categorizeCoins;
})();
