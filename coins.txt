/**
 * Checks a public BSC Mainnet RPC node to verify if an address 
 * hosts active smart contract bytecode on the blockchain.
 */
async function verifyBnbMainnetToken(address) {
    try {
        // 1. Structural hex address validation
        if (!address || !/^0x[a-fA-F0-9]{40}$/.test(address)) {
            return false;
        }

        // 2. Direct POST request to the public BSC RPC gateway (handles CORS safely)
        const rpcUrl = "https://binance.org";
        const response = await fetch(rpcUrl, {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                jsonrpc: "2.0",
                method: "eth_getCode",
                params: [address, "latest"],
                id: 1
            })
        });

        if (!response.ok) {
            return false;
        }

        const data = await response.json();

        // 3. True contract check: Deployed tokens return compiled bytecode.
        // Wallets or un-deployed contracts return "0x" or "0x00".
        if (data && data.result && data.result !== "0x" && data.result !== "0x00") {
            return true;
        }

        return false;

    } catch (err) {
        window.logToPanel("BSC RPC VERIFY ERROR: " + err.message);
        return false;
    }
}

/**
 * Parses raw application market JSON data, applies blacklist criteria,
 * and confirms token address validity against the active BSC Mainnet database.
 */
async function categorizeCoins() {
    try {
        const backendUrl = "/prodbot/market-analysis";
        const res = await fetch(backendUrl);
        let rawData = await res.json();

        if (rawData && rawData.data) {
            rawData = rawData.data;
        }

        if (!Array.isArray(rawData)) {
            try {
                rawData = JSON.parse(rawData);
            } catch (e) {
                rawData = [];
            }
        }

        const flatList = [];

        for (let i = 0; i < rawData.length; i++) {
            const item = rawData[i];
            if (!item) continue;

            let rawName = String(item.name || "").trim();
            let rawSymbol = item.token_symbol && String(item.token_symbol).trim().toUpperCase() !== "N/A"
                ? item.token_symbol
                : rawName;

            rawSymbol = String(rawSymbol || "").toUpperCase().trim();
            let rawAddress = String(item.address_or_geckoid || "").trim();

            // Ignore incomplete records
            if (!rawAddress || rawAddress === "N/A") {
                continue;
            }

            // Reject explicit non-BSC chain tags
            if (/^(solana|arbitrum|base|avax|polygon|optimism|ethereum|thorchain|sonic):/i.test(rawAddress)) {
                continue;
            }

            // Strip the explicit bsc prefix if present
            if (rawAddress.startsWith("bsc:")) {
                rawAddress = rawAddress.substring(4);
            }

            // Enforce custom internal structural exclusions
            if (rawName.toUpperCase().includes("THENA") || rawSymbol === "THE" || rawAddress.toLowerCase() === "0xf4c8e32eadec4bfe97e0f595add0f4450a863a11") {
                continue;
            }

            // Exclude baseline market assets from selection pool
            if (["BNB", "WBNB", "BTC", "WBTC", "ETH", "USDT", "USDC", "BUSD", "DAI"].includes(rawSymbol)) {
                continue;
            }

            // Run direct BSC Mainnet deployment check via RPC node
            const verified = await verifyBnbMainnetToken(rawAddress);

            if (!verified) {
                window.logToPanel("⛔ REJECTED NON-BSC TOKEN: " + rawSymbol + " | " + rawAddress);
                continue;
            }

            // Map required mathematical pricing models from working JSON
            const priceUsd = parseFloat(item.live_price_usd || item.lastPrice || 0) || 0;
            const change24h = parseFloat(item.change_24h_percent || item.change24h || 0) || 0;
            const volume = parseFloat(item.volume || 0) || 0;

            flatList.push({
                symbol: rawSymbol,
                address: rawAddress,
                lastPrice: priceUsd,
                change24h: change24h,
                volume: volume
            });

            window.logToPanel("✅ VERIFIED BSC TOKEN: " + rawSymbol);
        }

        // Sort descending based on performance metrics
        flatList.sort((a, b) => Number(b.change24h || 0) - Number(a.change24h || 0));

        // Restrict outputs to top 5 candidates for target evaluation
        const finalList = flatList.slice(0, 5);

        window.logToPanel("✅ Production Filter Accepted " + finalList.length + " verified BNB Mainnet assets.");

        return {
            short_term: finalList,
            mid_term: [],
            long_term: []
        };

    } catch (err) {
        window.logToPanel("⚠️ Coin Filter Error: " + err.message);
        return { short_term: [], mid_term: [], long_term: [] };
    }
}
