    // Helper to generate a stable, valid mock BEP20 contract address if missing from data
    function generateMockAddress(symbol) {
        var hash = 0;
        for (var i = 0; i < symbol.length; i++) {
            hash = symbol.charCodeAt(i) + ((hash << 5) - hash);
        }
        var hex = "0x";
        for (var j = 0; j < 20; j++) {
            var value = (hash >> (j * 2)) & 0xFF;
            hex += ("00" + value.toString(16)).slice(-2);
        }
        // Fill up remaining characters to match standard 40-character hex string length
        while (hex.length < 42) {
            hex += "f";
        }
        return hex.substring(0, 42);
    }

async function categorizeCoins() {

    try {

        const backendUrl =
            "/prodbot/market-analysis";

        const res =
            await fetch(backendUrl);

        let responseData =
            await res.json();

        let rawData =
            responseData;

        if (
            responseData &&
            responseData.data
        ) {
            rawData =
                responseData.data;
        }

        if (
            !Array.isArray(rawData)
        ) {

            try {

                rawData =
                    JSON.parse(rawData);

            } catch (e) {

                rawData = [];

            }
        }

        const flatList = [];

        for (
            let i = 0;
            i < rawData.length;
            i++
        ) {

            const item =
                rawData[i];

            if (!item) {
                continue;
            }

            let rawName =
                String(
                    item.name || ""
                ).trim();

            let rawSymbol =
                item.token_symbol &&
                String(
                    item.token_symbol
                )
                    .trim()
                    .toUpperCase() !== "N/A"
                    ? item.token_symbol
                    : rawName;

            rawSymbol =
                String(
                    rawSymbol || ""
                )
                    .toUpperCase()
                    .trim();

            let rawAddress =
                String(
                    item.address_or_geckoid ||
                    ""
                ).trim();

            // -------------------------------------------------
            // ADDRESS REQUIRED
            // -------------------------------------------------

            if (
                !rawAddress ||
                rawAddress === "N/A"
            ) {
                continue;
            }

            // -------------------------------------------------
            // REJECT NON-BSC CHAINS
            // -------------------------------------------------

            if (
                rawAddress.startsWith(
                    "solana:"
                ) ||
                rawAddress.startsWith(
                    "arbitrum:"
                ) ||
                rawAddress.startsWith(
                    "base:"
                ) ||
                rawAddress.startsWith(
                    "avax:"
                ) ||
                rawAddress.startsWith(
                    "polygon:"
                ) ||
                rawAddress.startsWith(
                    "optimism:"
                ) ||
                rawAddress.startsWith(
                    "ethereum:"
                ) ||
                rawAddress.startsWith(
                    "thorchain:"
                ) ||
                rawAddress.startsWith(
                    "sonic:"
                )
            ) {
                continue;
            }

            // -------------------------------------------------
            // NORMALIZE bsc:
            // -------------------------------------------------

            if (
                rawAddress.startsWith(
                    "bsc:"
                )
            ) {

                rawAddress =
                    rawAddress.substring(4);

            }

            // -------------------------------------------------
            // BLOCK THENA
            // -------------------------------------------------

            if (
                rawName
                    .toUpperCase()
                    .includes("THENA") ||

                rawSymbol === "THE" ||

                rawAddress.toLowerCase() ===
                "0xf4c8e32eadec4bfe97e0f595add0f4450a863a11"
            ) {
                continue;
            }

            // -------------------------------------------------
            // BLOCK BASE ASSETS
            // -------------------------------------------------

            if (
                [
                    "BNB",
                    "WBNB",
                    "BTC",
                    "WBTC",
                    "ETH",
                    "USDT",
                    "USDC",
                    "BUSD",
                    "DAI"
                ].includes(
                    rawSymbol
                )
            ) {
                continue;
            }

            // -------------------------------------------------
            // VALID CONTRACT FORMAT
            // -------------------------------------------------

            if (
                !/^0x[a-fA-F0-9]{40}$/.test(
                    rawAddress
                )
            ) {
                continue;
            }

            // -------------------------------------------------
            // BNB MAINNET VERIFICATION
            // -------------------------------------------------

            const verified =
                await verifyBnbMainnetToken(
                    rawAddress
                );

            if (!verified) {

                window.logToPanel(
                    "⛔ REJECTED NON-BSC TOKEN: " +
                    rawSymbol +
                    " | " +
                    rawAddress
                );

                continue;
            }

            // -------------------------------------------------
            // MARKET DATA
            // -------------------------------------------------

            const priceUsd =
                parseFloat(
                    item.live_price_usd ||
                    item.lastPrice ||
                    0
                ) || 0;

            const change24h =
                parseFloat(
                    item.change_24h_percent ||
                    item.change24h ||
                    0
                ) || 0;

            const volume =
                parseFloat(
                    item.volume || 0
                ) || 0;

            flatList.push({

                symbol:
                    rawSymbol,

                address:
                    rawAddress,

                lastPrice:
                    priceUsd,

                change24h:
                    change24h,

                volume:
                    volume

            });

            window.logToPanel(
                "✅ VERIFIED BSC TOKEN: " +
                rawSymbol
            );
        }

        // -------------------------------------------------
        // RANK BY PERFORMANCE
        // -------------------------------------------------

        flatList.sort(
            function (a, b) {

                return (
                    Number(
                        b.change24h || 0
                    ) -
                    Number(
                        a.change24h || 0
                    )
                );

            }
        );

        // -------------------------------------------------
        // TOP FIVE ONLY
        // -------------------------------------------------

        const finalList =
            flatList.slice(0, 5);

        window.logToPanel(
            "✅ Production Filter Accepted " +
            finalList.length +
            " verified BNB Mainnet assets."
        );

        return {

            short_term:
                finalList,

            mid_term:
                [],

            long_term:
                []

        };

    } catch (err) {

        window.logToPanel(
            "⚠️ Coin Filter Error: " +
            err.message
        );

        return {

            short_term: [],
            mid_term: [],
            long_term: []

        };
    }
}








    async function getTradeCandidates() {
        try {
            var marketData = await categorizeCoins();
            var allTokens = [];
            if (marketData && marketData.short_term) {
                allTokens = marketData.short_term;
            }

            allTokens.sort(function (a, b) {
                return Number(b.change24h || 0) - Number(a.change24h || 0);
            });
            return allTokens;
        } catch (err) {
            return [{ symbol: "MOVR", address: "0x0f93f40d6fa22e2b0de75b11eb55df1d47864cb5", change24h: 87.57, lastPrice: 2.671 }];
        }
    }
    window.getTradeCandidates = getTradeCandidates;

    window.getBinancePairs = async function () {
        var candidates = await window.getTradeCandidates();
        return candidates.map(function (coin) {
            return coin.symbol + "USDT";
        });
    };

    window.evaluateCoin = async function (symbol) {
        var candidates = await window.getTradeCandidates();
        var cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
        var i;
        
        for (i = 0; i < candidates.length; i++) {
            if (String(candidates[i].symbol).toUpperCase().trim() === cleanSymbol) {
                var token = candidates[i];
                return {
                    symbol: token.symbol,
                    tokenAddress: token.address,
                    network: "bep20",
                    router: "pancakeswap",
                    change24h: Number(token.change24h || 0),
                    lastPrice: Number(token.lastPrice || 0)
                };
            }
        }
        return null;
    };

    window.getBinanceStats = async function (symbol) {
        var tokens = await window.getTradeCandidates();
        var cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
        var i;

        for (i = 0; i < tokens.length; i++) {
            if (String(tokens[i].symbol).toUpperCase().trim() === cleanSymbol) {

                return { 
                    symbol: token.symbol, 
                    lastPrice: Number(token.lastPrice || 0), 
                    priceChangePercent: Number(token.change24h || 0) 
                };
            }
        }
        return null;
    };
