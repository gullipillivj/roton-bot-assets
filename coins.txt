// ==================================================================
// 🎛️ COST-AWARE COIN SELECTION ENGINE
// ==================================================================

(function () {

    window.logToPanel("coins.txt loaded safely.");

    const MIN_PROFIT_MOVE = 1.0;
    const MAX_PROFIT_MOVE = 20.0;
    const TOP_POOL_SIZE = 5;

    // ==============================================================
    // MARKET DATA
    // ==============================================================

    async function categorizeCoins() {

        try {

            const baseApiUrl =
                window.API_BASE ||
                window.location.origin;

            const res = await fetch(
                `${baseApiUrl}/prodbot/market-analysis`
            );

            const data =
                await res.json();

            return {
                short_term:
                    data.short_term || [],
                mid_term:
                    data.mid_term || [],
                long_term:
                    data.long_term || []
            };

        } catch (err) {

            console.error(err);

            return {
                short_term: [],
                mid_term: [],
                long_term: []
            };

        }

    }

    window.categorizeCoins =
        categorizeCoins;

    // ==============================================================
    // BUILD PROFITABLE TOKEN POOL
    // ==============================================================

    async function getTradeCandidates() {

        const marketData =
            await categorizeCoins();

        let allTokens = [

            ...(marketData.short_term || []),

            ...(marketData.mid_term || []),

            ...(marketData.long_term || [])

        ];

        allTokens = allTokens.filter(
            token => {

                const move =
                    Number(
                        token.change24h || 0
                    );

                return (
                    move >= MIN_PROFIT_MOVE &&
                    move <= MAX_PROFIT_MOVE &&
                    token.address
                );

            }
        );

        allTokens.sort(
            (a, b) =>
                Number(
                    b.change24h || 0
                ) -
                Number(
                    a.change24h || 0
                )
        );

        const finalPool =
            allTokens.slice(
                0,
                TOP_POOL_SIZE
            );

        window.logToPanel(
            `[POOL] ${finalPool.length} cost-aware candidates`
        );

        return finalPool;
    }

    // ==============================================================
    // PAIRS
    // ==============================================================

    window.getBinancePairs =
        async function () {

            try {

                const candidates =
                    await getTradeCandidates();

                if (
                    candidates &&
                    candidates.length > 0
                ) {

                    return candidates.map(
                        coin =>
                            `${coin.symbol}USDT`
                    );

                }

            } catch (err) {

                console.error(err);

            }

            return ["CAKEUSDT"];

        };

    // ==============================================================
    // TOKEN LOOKUP
    // ==============================================================

    window.evaluateCoin =
        async function(symbol) {

            try {

                const candidates =
                    await getTradeCandidates();

                const cleanSymbol =
                    String(symbol)
                        .replace("USDT", "")
                        .toUpperCase()
                        .trim();

                const token =
                    candidates.find(
                        x =>
                            String(
                                x.symbol
                            )
                            .toUpperCase()
                            .trim() ===
                            cleanSymbol
                    );

                if (!token) {

                    window.logToPanel(
                        `[SKIP] ${cleanSymbol} failed profitability filter`
                    );

                    return null;

                }

                const address =
                    token.address ||
                    token.tokenAddress;

                if (!address) {

                    window.logToPanel(
                        `[INVALID] Missing contract address for ${cleanSymbol}`
                    );

                    return null;
                }

                window.logToPanel(
                    `[TARGET]
${cleanSymbol}
MOVE=${token.change24h}%
ADDR=${address}`
                );

                return {

                    symbol:
                        token.symbol,

                    tokenAddress:
                        address,

                    network:
                        "bep20",

                    router:
                        "pancakeswap",

                    change24h:
                        Number(
                            token.change24h || 0
                        ),

                    lastPrice:
                        Number(
                            token.lastPrice || 0
                        )

                };

            } catch (err) {

                window.logToPanel(
                    `[ERROR] evaluateCoin: ${err.message}`
                );

                return null;

            }

        };

    // ==============================================================
    // MARKET STATS
    // ==============================================================

    window.getBinanceStats =
        async function(symbol) {

            try {

                const cleanSymbol =
                    String(symbol)
                        .replace("USDT", "")
                        .toUpperCase()
                        .trim();

                const tokens =
                    await getTradeCandidates();

                const token =
                    tokens.find(
                        t =>
                            String(
                                t.symbol
                            )
                            .toUpperCase()
                            .trim() ===
                            cleanSymbol
                    );

                if (!token) {
                    return null;
                }

                return {

                    symbol:
                        token.symbol,

                    lastPrice:
                        Number(
                            token.lastPrice || 0
                        ),

                    priceChangePercent:
                        Number(
                            token.change24h || 0
                        )

                };

            } catch (err) {

                window.logToPanel(
                    `[ERROR] getBinanceStats: ${err.message}`
                );

                return null;

            }

        };

    window.logToPanel(
        `coins.txt initialized.
Min Move=${MIN_PROFIT_MOVE}%
Max Move=${MAX_PROFIT_MOVE}%
Top Pool=${TOP_POOL_SIZE}`
    );

})();
