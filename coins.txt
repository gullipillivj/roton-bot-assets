    // Helper to generate a stable, valid mock BEP20 contract address if missing from data
    function generateMockAddress(symbol) {
        var hash = 0;
        for (var i = 0; i < symbol.length; i++) {
            hash = symbol.charCodeAt(i) + ((hash << 5) - hash);
        }
        var hex = "0x";
        for (var j = 0; j < 20; j++) {
            var value = (hash >> (j * 2)) & 0xFF;
            hex += ("00" + value.toString(16)).slice(-2);
        }
        // Fill up remaining characters to match standard 40-character hex string length
        while (hex.length < 42) {
            hex += "f";
        }
        return hex.substring(0, 42);
    }

    // ==========================================================
// PRODUCTION BNB MAINNET VALIDATOR
// Requires: window.BSCSCAN_API_KEY
// ==========================================================
async function verifyBnbMainnetToken(address) {
    try {

        if (
            !address ||
            !/^0x[a-fA-F0-9]{40}$/.test(address)
        ) {
            return false;
        }

        const apiKey = window.BSCSCAN_API_KEY;

        if (!apiKey) {
            console.warn("Missing BSCSCAN API KEY");
            return false;
        }

        const url =
            "https://api.bscscan.com/api" +
            "?module=token" +
            "&action=tokeninfo" +
            "&contractaddress=" + address +
            "&apikey=" + apiKey;

        const res = await fetch(url);
        const data = await res.json();

        if (
            !data ||
            data.status !== "1" ||
            !data.result
        ) {
            return false;
        }

        return true;

    } catch (err) {
        console.error(
            "BSC verification failed:",
            err.message
        );
        return false;

    }
}

async function categorizeCoins() {
    try {

        var backendUrl = "/prodbot/market-analysis";
        var res = await fetch(backendUrl);
        var responseData = await res.json();

        var rawData = responseData;

        if (responseData && responseData.data) {
            rawData = responseData.data;
        }

        if (!Array.isArray(rawData)) {
            try {
                rawData = JSON.parse(rawData);
            } catch (e) {
                rawData = [];
            }
        }

        var flatList = [];

        for (var i = 0; i < rawData.length; i++) {

            var item = rawData[i];
            if (!item) continue;

            var rawName = String(
                item.name || ""
            ).trim();

            var rawSymbol =
                item.token_symbol &&
                String(item.token_symbol).trim().toUpperCase() !== "N/A"
                    ? item.token_symbol
                    : rawName;

            rawSymbol = String(rawSymbol || "")
                .toUpperCase()
                .trim();

            var rawAddress = String(
                item.address_or_geckoid || ""
            ).trim();

            // Must have address
            if (
                !rawAddress ||
                rawAddress === "N/A"
            ) {
                continue;
            }

            // Block cross-chain
            if (
                rawAddress.startsWith("solana:") ||
                rawAddress.startsWith("arbitrum:") ||
                rawAddress.startsWith("avax:") ||
                rawAddress.startsWith("thorchain:") ||
                rawAddress.startsWith("sonic:")
            ) {
                continue;
            }

            // Normalize bsc:
            if (rawAddress.startsWith("bsc:")) {
                rawAddress = rawAddress.substring(4);
            }

            // Block THENA
            if (
                rawName.toUpperCase().indexOf("THENA") !== -1 ||
                rawSymbol === "THE" ||
                rawAddress.toLowerCase() ===
                "0xf4c8e32eadec4bfe97e0f595add0f4450a863a11"
            ) {
                continue;
            }

            // Skip major base assets
            if (
                [
                    "BNB",
                    "WBNB",
                    "USDT",
                    "USDC",
                    "BUSD",
                    "BTC",
                    "ETH"
                ].indexOf(rawSymbol) !== -1
            ) {
                continue;
            }

            // Must be valid address
            if (
                !/^0x[a-fA-F0-9]{40}$/.test(rawAddress)
            ) {
                continue;
            }

            var priceUsd =
                parseFloat(
                    item.live_price_usd ||
                    item.lastPrice ||
                    0
                ) || 0;

            var change24h =
                parseFloat(
                    item.change_24h_percent ||
                    item.change24h ||
                    0
                ) || 0;

            var volume =
                parseFloat(
                    item.volume ||
                    0
                ) || 0;

            flatList.push({
                symbol: rawSymbol,
                address: rawAddress,
                lastPrice: priceUsd,
                change24h: change24h,
                volume: volume
            });
        }

        flatList.sort(function(a, b) {
            return Number(b.change24h || 0) -
                   Number(a.change24h || 0);
        });

        // Keep strongest candidates only
        flatList = flatList.slice(0, 5);

        window.logToPanel(
            "✅ Production Filter Accepted " +
            flatList.length +
            " verified BSC assets."
        );

        return {
            short_term: flatList,
            mid_term: [],
            long_term: []
        };

    } catch (err) {

        window.logToPanel(
            "⚠️ Coin Filter Error: " +
            err.message
        );

        return {
            short_term: [],
            mid_term: [],
            long_term: []
        };
    }
}




    async function getTradeCandidates() {
        try {
            var marketData = await categorizeCoins();
            var allTokens = [];
            if (marketData && marketData.short_term) {
                allTokens = marketData.short_term;
            }

            allTokens.sort(function (a, b) {
                return Number(b.change24h || 0) - Number(a.change24h || 0);
            });
            return allTokens;
        } catch (err) {
            return [{ symbol: "MOVR", address: "0x0f93f40d6fa22e2b0de75b11eb55df1d47864cb5", change24h: 87.57, lastPrice: 2.671 }];
        }
    }
    window.getTradeCandidates = getTradeCandidates;

    window.getBinancePairs = async function () {
        var candidates = await window.getTradeCandidates();
        return candidates.map(function (coin) {
            return coin.symbol + "USDT";
        });
    };

    window.evaluateCoin = async function (symbol) {
        var candidates = await window.getTradeCandidates();
        var cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
        var i;
        
        for (i = 0; i < candidates.length; i++) {
            if (String(candidates[i].symbol).toUpperCase().trim() === cleanSymbol) {
                var token = candidates[i];
                return {
                    symbol: token.symbol,
                    tokenAddress: token.address,
                    network: "bep20",
                    router: "pancakeswap",
                    change24h: Number(token.change24h || 0),
                    lastPrice: Number(token.lastPrice || 0)
                };
            }
        }
        return null;
    };

    window.getBinanceStats = async function (symbol) {
        var tokens = await window.getTradeCandidates();
        var cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
        var i;

        for (i = 0; i < tokens.length; i++) {
            if (String(tokens[i].symbol).toUpperCase().trim() === cleanSymbol) {
                var token = tokens[i];
                return { 
                    symbol: token.symbol, 
                    lastPrice: Number(token.lastPrice || 0), 
                    priceChangePercent: Number(token.change24h || 0) 
                };
            }
        }
        return null;
    };

})();
