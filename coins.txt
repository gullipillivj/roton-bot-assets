// ==================================================================
// 🎛️ DIRECT WALLET-COMPATIBLE LIVE COIN ENGINE (DECENTRALIZED MODE)
// ==================================================================

(function () {

    window.logToPanel("coins.txt loaded safely via Open-Source DeFiLlama API Engine Layer.");

    // ==============================================================
    // DIRECT WALLET FETCH PIPELINE (POLLS DYNAMIC FLASK DEFILLAMA LAYER)
    // ==============================================================
    async function categorizeCoins() {
        try {
            // Queries your new open-source endpoint that handles proxies, retries, and TVL filters
            const backendUrl = "/prodbot/market-analysis";
            const res = await fetch(backendUrl);
            const responseData = await res.json();

            // Failover check if the response status is not flagged as ok
            if (!responseData || responseData.status !== "ok") {
                throw new Error("Backend response status failed or data malformed.");
            }

            const rawData = responseData.data || [];
            let flatList = [];

            for (let i = 0; i < rawData.length; i++) {
                let item = rawData[i];
                
                // Maps the unified backend structure straight to operational layout frames
                flatList.push({
                    symbol: String(item.symbol || "UNKNOWN").toUpperCase(),
                    address: item.address || "0x0e09fabb73bd3ade0a17ecc321fd13a19e81ce82",
                    lastPrice: parseFloat(item.lastPrice || 0),
                    change24h: parseFloat(item.change24h || 0),
                    volume: parseFloat(item.volume || 0)
                });
            }

            window.logToPanel(`🎉 [DIRECT SUCCESS] Synchronized ${flatList.length} real-time open-source tokens from DeFiLlama!`);
            
            // Maintains the exact return signature object structure
            return { short_term: flatList, mid_term: [], long_term: [] };

        } catch (err) {
            window.logToPanel(`⚠️ Open-Source Network Fetch stalled: ${err.message}. Reverting to emergency backup.`);
            // Mirroring your authoritative error layout structures exactly
            return { 
                short_term: [{ symbol: "CAKE", address: "0x0e09fabb73bd3ade0a17ecc321fd13a19e81ce82", change24h: 1.8, lastPrice: 1.95, volume: 55000.0 }], 
                mid_term: [], 
                long_term: [] 
            };
        }
    }
    window.categorizeCoins = categorizeCoins;

    // ==============================================================
    // BUILD PROFITABLE CANDIDATES POOL
    // ==============================================================
    async function getTradeCandidates() {
        try {
            const marketData = await categorizeCoins();
            let allTokens = marketData.short_term || [];

            // Filters and structures items based on the active 24h change values
            allTokens.sort((a, b) => Number(b.change24h || 0) - Number(a.change24h || 0));
            return allTokens;
        } catch (err) {
            return [{ symbol: "CAKE", address: "0x0e09fabb73bd3ade0a17ecc321fd13a19e81ce82", change24h: 1.8, lastPrice: 1.95 }];
        }
    }
    window.getTradeCandidates = getTradeCandidates;

    // Expected by pickRandomCoin() in simulate_prod.txt -> appends USDT to match legacy filter layers
    window.getBinancePairs = async function () {
        const candidates = await window.getTradeCandidates();
        return candidates.map(coin => `${coin.symbol}USDT`);
    };

    // Expected by evaluateCoin() in simulate_prod.txt -> resolves token metadata arrays
    window.evaluateCoin = async function(symbol) {
        const candidates = await window.getTradeCandidates();
        const cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
        const token = candidates.find(x => String(x.symbol).toUpperCase().trim() === cleanSymbol);
        if (!token) return null;

        return {
            symbol: token.symbol,
            tokenAddress: token.address,
            network: "bep20",
            router: "pancakeswap",
            change24h: Number(token.change24h || 0),
            lastPrice: Number(token.lastPrice || 0)
        };
    };

    // Maintained hook for statistical monitoring engines
    window.getBinanceStats = async function(symbol) {
        const tokens = await window.getTradeCandidates();
        const cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
        const token = tokens.find(t => String(t.symbol).toUpperCase().trim() === cleanSymbol);
        if (!token) return null;
        
        return { 
            symbol: token.symbol, 
            lastPrice: Number(token.lastPrice || 0), 
            priceChangePercent: Number(token.change24h || 0) 
        };
    };

})();
