// ==================================================================
// 🎛️ COINS FRONTEND MANAGEMENT CONTEXT INTERFACE (DYNAMIC PROD)
// ==================================================================

(function () {

    window.logToPanel("coins.txt loaded safely.");

    // ==============================================================
    // 📊 Market Analysis Loader
    // ==============================================================
    async function categorizeCoins() {
        try {
            const baseApiUrl = window.API_BASE || window.location.origin;
            const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
            const data = await res.json();

            // Direct mapping of the pure dynamic arrays passed from the backend
            return {
                short_term: data.short_term || [],
                mid_term: data.mid_term || [],
                long_term: data.long_term || []
            };
        } catch (err) {
            console.error("Market analysis critical failure:", err);
            return { short_term: [], mid_term: [], long_term: [] };
        }
    }

    // ==============================================================
    // 🪙 Supported Trading Pairs (Completely Dynamic Mirror)
    // ==============================================================
    window.getBinancePairs = async () => {
        try {
            const marketData = await categorizeCoins();
            const allTokens = [
                ...(marketData.short_term || []),
                ...(marketData.mid_term || []),
                ...(marketData.long_term || [])
            ];
            
            // Map the true real-time operational pairs currently tracking up
            if (allTokens.length > 0) {
                return allTokens.map(t => t.marketPair || `${t.symbol}USDT`);
            }
        } catch (e) {
            console.error(e);
        }
        // Emergency stop backup string to keep runtime execution fluid
        return ["CAKEUSDT"];
    };

    // ==============================================================
    // 🔍 Token Resolver (Registry Independent)
    // ==============================================================
    window.evaluateCoin = async function (symbol) {
        try {
            const marketData = await categorizeCoins();
            const allTokens = [
                ...(marketData.short_term || []),
                ...(marketData.mid_term || []),
                ...(marketData.long_term || [])
            ];

            const cleanSymbol = String(symbol)
                .replace("USDT", "")
                .toUpperCase()
                .trim();

            // Locate target token profile details inside the active trend dataset
            const token = allTokens.find(
                x => String(x.symbol).toUpperCase().trim() === cleanSymbol
            );

            if (!token) {
                window.logToPanel(`[WARN] Token not found in active rising pool: ${cleanSymbol}`);
                return null;
            }

            // 🚀 DYNAMIC INJECTION: Pull smart contract address parameters directly from live metrics
            // Fallback to safe defaults if asset data strings are missing
            const extractedAddress = token.tokenAddress || "0x55d398326f99059fF775485246999027B3197955";

            return {
                symbol: token.symbol,
                tokenAddress: extractedAddress,
                network: "bep20",
                router: "pancakeswap",
                change24h: token.change24h
            };

        } catch (err) {
            window.logToPanel(`[ERROR] evaluateCoin failed: ${err.message}`);
            return null;
        }
    };

    window.categorizeCoins = categorizeCoins;
})();
