document.addEventListener("DOMContentLoaded", function () {

    logToPanel("UI ready");

    const startBtn =
        document.getElementById("startBtn");

    const stopBtn =
        document.getElementById("stopBtn");

    if (!startBtn) {
        logToPanel(
            "ERROR: startBtn not found"
        );
    } else {

        logToPanel(
            "Start button attached"
        );

            logToPanel(
                "[BUTTON] Start Trading clicked"
            );

            logToPanel(
                "startSimulation type = " +
                typeof window.startSimulation
            );

            if (
                typeof window.startSimulation ===
                "function"
            ) {

                logToPanel(
                    "[BOT] Starting simulation"
                );

                window.startSimulation(10);

            } else {

                logToPanel(
                    "ERROR: startSimulation function missing"
                );

            }
        };
    }

    if (!stopBtn) {

        logToPanel(
            "ERROR: stopBtn not found"
        );

    } else {

        stopBtn.onclick = function () {

            logToPanel(
                "[BUTTON] Stop Trading clicked"
            );

            if (
                typeof window.stopBot ===
                "function"
            ) {

                window.stopBot();

            } else {

                logToPanel(
                    "ERROR: stopBot function missing"
                );

            }
        };
    }

});

// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

    const canvas = document.getElementById("balanceChart");
    if (!canvas) return;

    console.log(`[CHART UPDATE] Rendering metrics — Current Balance: ${balance}`);
}

window.updateChart = updateChart;

// Changed from const to let so they can be modified at runtime
let currentCycle = 1;
let cycleTimer = null;
let cumulativeProfit = 0;
let botRunning = false; // Fixed: Changed "bool" to "let"

// ======================================================
// STOP BUTTON SUPPORT
// ======================================================

function stopBot() {
    botRunning = false;

    if (cycleTimer) {
        clearInterval(cycleTimer);
        cycleTimer = null;
    }

    window.logToPanel(
        `[INFO] ${new Date().toLocaleTimeString()} — Simulation stopped`
    );
}

window.stopBot = stopBot;

// ======================================================
// PICK RANDOM COIN
// ======================================================

async function pickRandomCoin() {
    try {
        const pairs = await window.getBinancePairs();

        if (!pairs || !pairs.length) {
            return "CAKEUSDT";
        }

        return pairs[Math.floor(Math.random() * pairs.length)];
    } catch (err) {
        return "CAKEUSDT";
    }
}

// ======================================================
// LIVE PROFIT UI
// ======================================================

function updateLiveProfit(value) {
    const el = document.getElementById("liveProfitValue");

    if (el) {
        if (value >= 0) {
            el.style.color = "#00cc66";
            el.textContent = `+${value.toFixed(2)} USDT`;
        } else {
            el.style.color = "#ff4444";
            el.textContent = `${value.toFixed(2)} USDT`;
        }
    }
}

// ======================================================
// SINGLE CYCLE
// ======================================================

async function simulateCycle(cycle) {
    if (!botRunning) return;

    window.logToPanel(`[INFO] Cycle ${cycle} started`);

    const pair = await pickRandomCoin();
    const coin = pair.replace("USDT", "");
    const coinInfo = await window.evaluateCoin(pair);

    if (!coinInfo) {
        window.logToPanel(`[WARN] Unable to load ${coin}`);
        return;
    }

    // Simulated entry price
    const buyPrice = Number((Math.random() * 100 + 1).toFixed(4));

    window.logToPanel(`[BUY] ${coin} @ ${buyPrice} USDT`);

    await new Promise(resolve => setTimeout(resolve, 1200));

    // Changed from const to let to allow modification inside the conditional below
    let marketMove = Number((Math.random() * 12 - 6).toFixed(2));

    if (typeof coinInfo.change24h !== "undefined") {
        marketMove = Number(coinInfo.change24h);
    }

    const profit = buyPrice * (marketMove / 100);
    cumulativeProfit += profit;

    await new Promise(resolve => setTimeout(resolve, 1000));

    window.logToPanel(`[SELL] ${coin}`);

    if (profit >= 0) {
        window.logToPanel(`[PROFIT] +${profit.toFixed(2)} USDT`);
    } else {
        window.logToPanel(`[LOSS] ${profit.toFixed(2)} USDT`);
    }

    updateLiveProfit(cumulativeProfit);

    // Chart update
    if (typeof window.updateChart === "function") {
        window.updateChart(buyPrice, cumulativeProfit);
    }

    window.logToPanel(`[TOTAL PNL] ${cumulativeProfit.toFixed(2)} USDT`);
    window.logToPanel(`[INFO] Cycle ${cycle} completed`);
}

// ======================================================
// START BOT
// ======================================================

function startSimulation(totalCycles = 10) {
    stopBot();

    botRunning = true;
    currentCycle = 1;
    cumulativeProfit = 0;

    updateLiveProfit(0);

    window.logToPanel(`[BOT START] ${totalCycles} cycles`);

    // run first cycle immediately
    simulateCycle(currentCycle);
    currentCycle++;

    cycleTimer = setInterval(async () => {
        if (!botRunning) {
            clearInterval(cycleTimer);
            return;
        }

        if (currentCycle > totalCycles) {
            clearInterval(cycleTimer);
            botRunning = false;

            window.logToPanel(
                `[BOT FINISHED] Final Profit = ${cumulativeProfit.toFixed(2)} USDT`
            );
            return;
        }

        await simulateCycle(currentCycle);
        currentCycle++;
    }, 60000);
}

// ======================================================
// INIT
// ======================================================

function initBot(totalCycles = 10) {
    startSimulation(totalCycles);
}

// ======================================================
// GLOBAL EXPORTS
// ======================================================

window.simulateCycle = simulateCycle;
window.startSimulation = startSimulation;
window.initBot = initBot;
window.stopBot = stopBot;

(function () {

    window.logToPanel("coins.txt loaded safely.");

    // ==============================================================
    // 📊 Market Analysis Loader
    // ==============================================================

    async function categorizeCoins() {

        window.logToPanel(
            "🔎 Querying live dynamic market trends from backend..."
        );

        try {

            const baseApiUrl =
                window.API_BASE || window.location.origin;

            const res = await fetch(
                `${baseApiUrl}/prodbot/market-analysis`
            );

            const data = await res.json();

            if (
                data &&
                Array.isArray(data.short_term) &&
                data.short_term.length > 0
            ) {

                const top =
                    data.short_term[0];

                window.logToPanel(
                    `📈 Market scan complete. Top candidate: ${top.symbol} (+${top.change24h}%)`
                );

                return {
                    short_term: data.short_term,
                    mid_term: data.mid_term || data.short_term,
                    long_term: data.long_term || data.short_term
                };
            }

            throw new Error(
                "Invalid market analysis payload."
            );

        } catch (err) {

            console.error(
                "Market analysis failure:",
                err
            );

            // ✅ BEP20 fallback only
            const fallbackPool = [

                {
                    symbol: "CAKE",
                    tokenAddress:
                        "0x0E09FaBB73Bd3Ade0a17ECC321fD13a19e81cE82",
                    network: "bep20",
                    change24h: 3.1
                },

                {
                    symbol: "XVS",
                    tokenAddress:
                        "0xcF6BB5389c92Bdda8a3747Ddb454cB7a64E86902",
                    network: "bep20",
                    change24h: 2.4
                },

                {
                    symbol: "LINK",
                    tokenAddress:
                        "0xF8aEe1784988F6E59c1664195e67042c244514d8",
                    network: "bep20",
                    change24h: 1.8
                }

            ];

            return {
                short_term: fallbackPool,
                mid_term: fallbackPool,
                long_term: fallbackPool
            };
        }
    }

    // ==============================================================
    // 🪙 Supported Trading Pairs (BEP20 Only)
    // ==============================================================

    window.getBinancePairs = async () => [

        "CAKEUSDT",
        "XVSUSDT",
        "LINKUSDT",
        "DOTUSDT",
        "ADAUSDT",
        "DOGEUSDT",
        "MATICUSDT",
        "XRPUSDT",
        "FILUSDT",
        "LTCUSDT"

    ];

    // ==============================================================
    // 🔍 Token Resolver
    // ==============================================================

    window.evaluateCoin = async function (symbol) {

        try {

            const marketData =
                await categorizeCoins();

            const allTokens = [

                ...(marketData.short_term || []),
                ...(marketData.mid_term || []),
                ...(marketData.long_term || [])

            ];

            const cleanSymbol = String(symbol)
                .replace("USDT", "")
                .toUpperCase()
                .trim();

            const token = allTokens.find(
                x =>
                    String(x.symbol)
                        .toUpperCase()
                        .trim() === cleanSymbol
            );

            if (!token) {

                window.logToPanel(
                    `[WARN] Token not found: ${cleanSymbol}`
                );

                return null;
            }

            return {

                symbol: token.symbol,

                tokenAddress:
                    token.tokenAddress,

                network: "bep20",

                router: "pancakeswap"

            };

        } catch (err) {

            window.logToPanel(
                `[ERROR] evaluateCoin failed: ${err.message}`
            );

            return null;
        }
    };

    // ==============================================================
    // 🌐 Global Bindings
    // ==============================================================

    window.categorizeCoins =
        categorizeCoins;

    window.logToPanel(
        "coins.txt initialized successfully (BEP20 mode)."
    );

})();

window.getBinanceStats = async function(symbol) {

    try {

        const cleanSymbol = String(symbol)
            .replace("USDT", "")
            .toUpperCase()
            .trim();

        const marketData =
            await categorizeCoins();

        const allTokens = [
            ...(marketData.short_term || []),
            ...(marketData.mid_term || []),
            ...(marketData.long_term || [])
        ];

        const token = allTokens.find(
            x =>
                String(x.symbol)
                    .toUpperCase()
                    .trim() === cleanSymbol
        );

        if (!token) {
            return null;
        }

        return {
            symbol: cleanSymbol,
            lastPrice:
                (Math.random() * 100 + 1).toFixed(4),
            priceChangePercent:
                token.change24h || 0
        };

    } catch (err) {

        logToPanel(
            `[ERROR] getBinanceStats: ${err.message}`
        );

        return null;
    }
};
