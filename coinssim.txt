// 🎛️ GLOBAL LIVE CHART UPDATE HANDLER
// =====================================================
function updateChart(balance, profit = null) {
    // ✅ DYNAMIC: Directly use the existing Chart object instantiated by your HTML template
    const chartInstance = window.balanceChart;
    if (!chartInstance) {
        console.warn("[CHART] Native chart context object balanceChart not yet initialized.");
        return;
    }

    const nextIndex = chartInstance.data.labels.length + 1;
    
    // Push new coordinate values directly into your template's layout datasets
    chartInstance.data.labels.push(`Trade ${nextIndex}`);
    
    // Dataset 0: Account Equity balance track line
    chartInstance.data.datasets[0].data.push(Number(balance.toFixed(2)));
    
    // Dataset 1: Cumulative Profit metric tracking curve
    if (profit !== null) {
        chartInstance.data.datasets[1].data.push(Number(profit.toFixed(2)));
    }

    // Force the active UI canvas layout to repaint the vectors instantly
    chartInstance.update();
}
window.updateChart = updateChart;

// Helper framework mapping layer to direct string dumps right inside your HTML textarea element
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.value += `[TRADER] ${msg}\n`;
        panel.scrollTop = panel.scrollHeight; // Auto-scroll to latest event logs
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

// ==================================================================
// 🚀 DYNAMIC SIMULATION RUNTIME RUNTIME ENGINE
// ==================================================================
let currentCycle = 1;
let cycleTimer = null;
let cumulativeProfit = 0;
let botRunning = false;
let trackingBalance = 1000; 

function stopBot() {
    if (!botRunning) return;
    botRunning = false;
    if (cycleTimer) {
        clearInterval(cycleTimer);
        cycleTimer = null;
    }
    window.logToPanel("⚠️ STOP SIGNAL RECEIVED — Real-time execution halted.");
}
window.stopBot = stopBot;

function updateLiveProfit(value) {
    const el = document.getElementById("liveProfitValue");
    if (el) {
        if (value >= 0) {
            el.style.color = "#00cc66";
            el.textContent = `+${value.toFixed(2)} USDT`;
        } else {
            el.style.color = "#ff4444";
            el.textContent = `${value.toFixed(2)} USDT`;
        }
    }
}

async function simulateCycle(cycle) {
    if (!botRunning) return;

    window.logToPanel(`⏳ Initiating Cycle #${cycle} trade scanning loop...`);
    
    // 1. Live backend query via context analysis endpoint
    const marketData = await window.categorizeCoins();
    const activeGreenPool = [
        ...(marketData.short_term || []),
        ...(marketData.mid_term || []),
        ...(marketData.long_term || [])
    ];

    // Capital Shield break check if broader market has no rising assets
    if (activeGreenPool.length === 0) {
        window.logToPanel("❌ [SKIPPED] Protection Mode: 0 rising tokens found. Skipping execution to shield liquidity.");
        return; 
    }

    // 2. Extract configuration boundary limits typed directly into your HTML inputs
    const investmentAmt = Number(document.getElementById("investBalance")?.value) || 500;
    const profitTargetPct = Number(document.getElementById("profitTarget")?.value) || 1;
    const stopLossPct = Number(document.getElementById("stopLoss")?.value) || 0.5;

    // 3. Select target assets out of live climbing categories
    const selectedToken = activeGreenPool[Math.floor(Math.random() * activeGreenPool.length)];
    const coin = selectedToken.symbol;
    
    window.logToPanel(`🔍 [RADAR] Detected high-velocity token: ${coin} (+${selectedToken.change24h}%)`);
    
    // Simulated market execution calculation values
    const entryPrice = selectedToken.lastPrice || Number((Math.random() * 50 + 1).toFixed(4));
    window.logToPanel(`🛒 [BUY] Allocation initialized: Swapping ${investmentAmt} USDT into ${coin} @ ${entryPrice} USDT`);

    // UI Step Delay to cleanly follow transaction states visually
    await new Promise(resolve => setTimeout(resolve, 2000));
    if (!botRunning) return;

    // 4. Processing profit outcomes bound by configured percentages
    const marketMove = Number(selectedToken.change24h);
    let tradeProfit = investmentAmt * (marketMove / 100);

    const maxGain = investmentAmt * (profitTargetPct / 100);
    const maxLoss = -investmentAmt * (stopLossPct / 100);

    // Apply dashboard constraint limits
    if (tradeProfit > maxGain) tradeProfit = maxGain;
    if (tradeProfit < maxLoss) tradeProfit = maxLoss;

    // Re-adjust account values
    cumulativeProfit += tradeProfit;
    trackingBalance += tradeProfit;

    window.logToPanel(`📦 [SELL] Order executed: Liquidating asset holdings for ${coin}.`);
    
    if (tradeProfit >= 0) {
        window.logToPanel(`📈 [PROFIT] Realized Return: +${tradeProfit.toFixed(2)} USDT.`);
    } else {
        window.logToPanel(`📉 [LOSS] Realized Risk Correction: ${tradeProfit.toFixed(2)} USDT.`);
    }

    // 5. Force elements to match calculated adjustments instantly
    updateLiveProfit(cumulativeProfit);
    window.updateChart(trackingBalance, cumulativeProfit);

    window.logToPanel(`📊 [SUMMARY] Balance: ${trackingBalance.toFixed(2)} USDT | Net PnL: ${cumulativeProfit.toFixed(2)} USDT`);
    window.logToPanel(`✅ Cycle ${cycle} completed.\n-----------------------------------`);
}

function startSimulation() {
    stopBot();
    
    // Capture base start line value defined directly on your template input container
    const initialDeposit = Number(document.getElementById("startBalance")?.value) || 1000;
    
    // Clear and reset systemic tracking variables
    botRunning = true;
    currentCycle = 1;
    cumulativeProfit = 0;
    trackingBalance = initialDeposit;

    // Reset layout elements
    const panel = document.getElementById("logPanel");
    if (panel) panel.value = "";
    updateLiveProfit(0);

    // Clear structural vector data on the existing chart instance
    const chartInstance = window.balanceChart;
    if (chartInstance) {
        chartInstance.data.labels = ['Start'];
        chartInstance.data.datasets[0].data = [initialDeposit];
        chartInstance.data.datasets[1].data =;
        chartInstance.update();
    }

    window.logToPanel(`🚀 SYSTEM ONLINE — Processing loops at automated 15-second tracking blocks.`);

    // Start cycle 1 right away
    setTimeout(() => {
        simulateCycle(currentCycle);
        currentCycle++;
    }, 100);

    // Dynamic looping interval setup (Configured to 15 seconds so you can watch execution feed updates live)
    cycleTimer = setInterval(async () => {
        if (!botRunning) {
            clearInterval(cycleTimer);
            return;
        }
        await simulateCycle(currentCycle);
        currentCycle++;
    }, 15000); 
}

window.simulateCycle = simulateCycle;
window.startSimulation = startSimulation;
window.stopBot = stopBot;

// ==================================================================
// 🎛&nbsp;COINS FRONTEND MANAGEMENT CONTEXT INTERFACE
// ==================================================================
(function () {
    async function categorizeCoins() {
        try {
            const baseApiUrl = window.API_BASE || window.location.origin;
            const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
            const data = await res.json();
            return {
                short_term: data.short_term || [],
                mid_term: data.mid_term || [],
                long_term: data.long_term || []
            };
        } catch (err) {
            console.error("Market analysis endpoint connection error:", err);
            return { short_term: [], mid_term: [], long_term: [] };
        }
    }
    window.categorizeCoins = categorizeCoins;
})();

// =====================================================
// 💻 HTML CLICK HANDLER CAPTURE LISTENERS
// =====================================================
document.addEventListener("DOMContentLoaded", function () {
    const startBtn = document.getElementById("startBtn");
    const stopBtn = document.getElementById("stopBtn");

    if (startBtn) {
        startBtn.onclick = function () {
            window.startSimulation();
        };
    }

    if (stopBtn) {
        stopBtn.onclick = function () {
            window.stopBot();
        };
    }
});
