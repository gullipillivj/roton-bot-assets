async function getBinancePairs() {
  try {
    const baseApiUrl = window.API_BASE || window.location.origin;
    const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
    const data = await res.json();
    
    if (data && Array.isArray(data.short_term) && data.short_term.length > 0) {
        return data.short_term.map(t => `${t.symbol}USDT`);
    }
    return ["BTCUSDT", "ETHUSDT", "BNBUSDT", "XVSUSDT"];
  } catch (e) {
    console.warn("[WARN] Ticker fetch failed, using fallback list:", e);
    return ["BTCUSDT", "ETHUSDT", "BNBUSDT", "XVSUSDT"];
  }
}

async function getBinanceStats(symbol) {
  try {
    const baseApiUrl = window.API_BASE || window.location.origin;
    const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
    const data = await res.json();
    
    const plainSymbol = symbol.replace("USDT", "");
    let tokenMatch = null;
    
    if (data) {
        const mergedPool = [].concat(data.short_term || [], data.mid_term || [], data.long_term || []);
        tokenMatch = mergedPool.find(t => t.symbol === plainSymbol);
    }
    
    if (tokenMatch) {
        return {
          symbol: symbol,
          lastPrice: parseFloat(tokenMatch.lastPrice || 10.0),
          priceChangePercent: parseFloat(tokenMatch.change24h || 1.2),
          volume: parseFloat(tokenMatch.volume || 5000)
        };
    }
    return { symbol, lastPrice: 5.0, priceChangePercent: 0.8, volume: 1000 };
  } catch (e) {
    console.warn("[WARN] Stats mapping failed:", e);
    return null;
  }
}

// 🌟 EXPLICIT REGISTRATION: Expose these globally to fix the freeze
window.getBinancePairs = getBinancePairs;
window.getBinanceStats = getBinanceStats;

window.logToPanel("coins.txt methods fully initialized globally.");
