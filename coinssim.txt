// =====================================================
// 🎛️ GLOBAL CORE LOGGER & INITIALIZER
// =====================================================
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;
logToPanel("Core framework components loaded.");

function updateChart(balance, profit = null) {
    const canvas = document.getElementById("balanceChart");
    if (!canvas) return;
    console.log(`[CHART UPDATE] Rendering metrics — Current Balance: ${balance}`);
}
window.updateChart = updateChart;

// ==================================================================
// 🚀 FULL SIMULATION RUNTIME ENGINE
// ==================================================================
let currentCycle = 1;
let cycleTimer = null;
let cumulativeProfit = 0;
let botRunning = false; 

function stopBot() {
    botRunning = false;
    if (cycleTimer) {
        clearInterval(cycleTimer);
        cycleTimer = null;
    }
    window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Simulation stopped`);
}
window.stopBot = stopBot;

function updateLiveProfit(value) {
    const el = document.getElementById("liveProfitValue");
    if (el) {
        if (value >= 0) {
            el.style.color = "#00cc66";
            el.textContent = `+${value.toFixed(2)} USDT`;
        } else {
            el.style.color = "#ff4444";
            el.textContent = `${value.toFixed(2)} USDT`;
        }
    }
}

async function simulateCycle(cycle) {
    if (!botRunning) return;

    window.logToPanel(`[INFO] Cycle ${cycle} started`);
    
    // 📡 Pull active rising token arrays directly from the server endpoint
    const marketData = await window.categorizeCoins();
    
    // Combine all currently rising pools into an open actionable index list
    const activeGreenPool = [
        ...(marketData.short_term || []),
        ...(marketData.mid_term || []),
        ...(marketData.long_term || [])
    ];

    // 🛑 CYCLE PROTECTION SIGNAL: Skip loop if market data contains 0 rising metrics
    if (activeGreenPool.length === 0) {
        window.logToPanel(`[SKIPPED] Capital Protection Triggered: 0 rising assets found across market. Cycle ${cycle} skipped.`);
        window.logToPanel(`[TOTAL PNL] ${cumulativeProfit.toFixed(2)} USDT`);
        window.logToPanel(`[INFO] Cycle ${cycle} completed`);
        return; 
    }

    // 🚀 DYNAMIC RUNTIME MATCHING: Pick directly out of whatever tokens are climbing
    const selectedToken = activeGreenPool[Math.floor(Math.random() * activeGreenPool.length)];
    const coin = selectedToken.symbol;

    const buyPrice = Number((Math.random() * 100 + 1).toFixed(4));
    window.logToPanel(`[BUY] Dynamic Target: ${coin} @ ${buyPrice} USDT`);

    await new Promise(resolve => setTimeout(resolve, 1200));

    // Map true real-time metric change passed dynamically out of the python API instances
    let marketMove = Number(selectedToken.change24h);

    const profit = buyPrice * (marketMove / 100);
    cumulativeProfit += profit;

    await new Promise(resolve => setTimeout(resolve, 1000));
    window.logToPanel(`[SELL] ${coin}`);

    if (profit >= 0) {
        window.logToPanel(`[PROFIT] +${profit.toFixed(2)} USDT`);
    } else {
        window.logToPanel(`[LOSS] ${profit.toFixed(2)} USDT`);
    }

    updateLiveProfit(cumulativeProfit);

    if (typeof window.updateChart === "function") {
        window.updateChart(buyPrice, cumulativeProfit);
    }

    window.logToPanel(`[TOTAL PNL] ${cumulativeProfit.toFixed(2)} USDT`);
    window.logToPanel(`[INFO] Cycle ${cycle} completed`);
}

function startSimulation(totalCycles = 10) {
    stopBot();
    botRunning = true;
    currentCycle = 1;
    cumulativeProfit = 0;
    updateLiveProfit(0);

    window.logToPanel(`[BOT START] Requesting ${totalCycles} cycles`);

    // Asynchronously schedule execution so DOM thread isn't blocked
    setTimeout(() => {
        simulateCycle(currentCycle);
        currentCycle++;
    }, 50);

    cycleTimer = setInterval(async () => {
        if (!botRunning) {
            clearInterval(cycleTimer);
            return;
        }
        if (currentCycle > totalCycles) {
            clearInterval(cycleTimer);
            botRunning = false;
            window.logToPanel(`[BOT FINISHED] Final Profit = ${cumulativeProfit.toFixed(2)} USDT`);
            return;
        }
        await simulateCycle(currentCycle);
        currentCycle++;
    }, 60000);
}

// Global scope initialization
window.simulateCycle = simulateCycle;
window.startSimulation = startSimulation;
window.stopBot = stopBot;

// ==================================================================
// 🎛️ COINS FRONTEND MANAGEMENT CONTEXT INTERFACE
// ==================================================================
(function () {
    async function categorizeCoins() {
        try {
            const baseApiUrl = window.API_BASE || window.location.origin;
            const res = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
            const data = await res.json();

            // Return pure responsive state pools directly
            return {
                short_term: data.short_term || [],
                mid_term: data.mid_term || [],
                long_term: data.long_term || []
            };
        } catch (err) {
            console.error("Market analysis connection failure:", err);
            return { short_term: [], mid_term: [], long_term: [] };
        }
    }

    // Dynamic resolution context mapping layer
    window.evaluateCoin = async function (symbol) {
        try {
            const marketData = await categorizeCoins();
            const allTokens = [
                ...(marketData.short_term || []),
                ...(marketData.mid_term || []),
                ...(marketData.long_term || [])
            ];
            const cleanSymbol = String(symbol).replace("USDT", "").toUpperCase().trim();
            const token = allTokens.find(x => String(x.symbol).toUpperCase().trim() === cleanSymbol);

            if (!token) return null;
            return { 
                symbol: token.symbol, 
                network: "bep20", 
                router: "pancakeswap", 
                change24h: token.change24h 
            };
        } catch (err) {
            return null;
        }
    };
    
    window.categorizeCoins = categorizeCoins;
})();

// =====================================================
// 💻 UI DOM INTERACTION EVENT BINDINGS
// =====================================================
document.addEventListener("DOMContentLoaded", function () {
    logToPanel("UI ready");
    const startBtn = document.getElementById("startBtn");
    const stopBtn = document.getElementById("stopBtn");

    if (!startBtn) {
        logToPanel("ERROR: startBtn element missing in HTML");
    } else {
        logToPanel("Start button attached");
        startBtn.onclick = function () {
            logToPanel("[BUTTON] Start Trading clicked");
            
            const targetStart = window.startSimulation || startSimulation;
            if (typeof targetStart === "function") {
                logToPanel("[BOT] Executing simulation sequence...");
                targetStart(10);
            } else {
                logToPanel("ERROR: startSimulation function runtime execution failed");
            }
        };
    }

    if (stopBtn) {
        stopBtn.onclick = function () {
            logToPanel("[BUTTON] Stop Trading clicked");
            const targetStop = window.stopBot || stopBot;
            if (typeof targetStop === "function") {
                targetStop();
            } else {
                logToPanel("ERROR: stopBot function missing");
            }
        };
    }
});
