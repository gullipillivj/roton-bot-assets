// ==================================================================
// 🚀 FIXED SIMULATION RUNTIME ENGINES
// ==================================================================
window.logToPanel("simulate.txt loaded");

let currentCycle = 1;
let cycleTimer = null;

async function simulateCycle(cycle) {
    window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — simulateCycle(${cycle}) started`);

    // Pick coin from Binance pairs
    const coin = await pickRandomCoin();
    const stats = await window.getBinanceStats(coin);

    if (!stats) {
        window.logToPanel(`[WARN] Cycle ${cycle}: No stats for ${coin}, skipping`);
        return;
    }

    const buyPrice = parseFloat(stats.lastPrice);
    window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Cycle ${cycle}: Bought ${coin} at ${buyPrice} USDT`);

    // Profit/loss based on actual 24hr % change
    const result = calculatePnL(buyPrice, parseFloat(stats.priceChangePercent));

    // Update the live profit label
    updateLiveProfit(result);

    if (result >= 0) {
        window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Cycle ${cycle}: Profit +${result.toFixed(2)} USDT`);
    } else {
        window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Cycle ${cycle}: Loss ${result.toFixed(2)} USDT`);
    }

    // Chart update integration
    if (typeof window.updateChart === "function") {
        window.updateChart(buyPrice, result);
    }

    window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — simulateCycle(${cycle}) complete`);
}

function startSimulation(totalCycles) {
    window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — initBot started for ${totalCycles} cycles (1‑minute interval)`);

    currentCycle = 1;
    if (cycleTimer) clearInterval(cycleTimer);

    // 🌟 STEP 1: Execute Cycle 1 immediately on click
    simulateCycle(currentCycle);
    currentCycle++;

    // 🌟 STEP 2: Queue up subsequent intervals
    cycleTimer = setInterval(async () => {
        if (currentCycle > totalCycles) {
            clearInterval(cycleTimer);
            window.logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Simulation complete`);
            return;
        }

        await simulateCycle(currentCycle);
        currentCycle++;
    }, 60000);
}

function initBot(totalCycles = 10) {
    startSimulation(totalCycles);
}

// === Helpers ===
async function pickRandomCoin() {
    try {
        const pairs = await window.getBinancePairs();
        if (!pairs || pairs.length === 0) return "BTCUSDT";
        return pairs[Math.floor(Math.random() * pairs.length)];
    } catch (e) {
        return "BTCUSDT";
    }
}

function updateLiveProfit(value) {
    const el = document.getElementById("liveProfitValue");
    if (el) {
        el.textContent = `${value.toFixed(2)} USDT`;
    }
}

function calculatePnL(price, percentChange) {
    if (!percentChange) return 0;
    return price * (percentChange / 100);
}

// Expose globally
window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
window.initBot = initBot;
