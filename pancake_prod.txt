// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;



// ✅ runSwap (frontend delegates router choice to backend)
async function runSwap(userAddr, usdtAmount, tokenAddressOrObject, networkParam="erc20") {
    try {
        const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";
        const baseApiUrl = window.API_BASE || "";

        let targetAddress = tokenAddressOrObject && typeof tokenAddressOrObject === "object"
            ? tokenAddressOrObject.tokenAddress 
            : (tokenAddressOrObject || "0xeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeee");
            
        let finalNetwork = tokenAddressOrObject && typeof tokenAddressOrObject === "object"
            ? tokenAddressOrObject.network 
            : networkParam;

        let parsedAmount = parseFloat(usdtAmount);
        if (isNaN(parsedAmount) || parsedAmount <= 0) parsedAmount = 55.55; 
        const amountWei = BigInt(Math.floor(parsedAmount * 1e18)).toString();

        // 1. Submit Request to backend to generate the compiled data string
        const res = await fetch(`${baseApiUrl}/prodbot/swap`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                user_address: userAddr,
                from_token: USDT_ADDRESS,
                to_token: targetAddress,
                amount: amountWei,
                network: finalNetwork
            })
        });

        const data = await res.json();
        
        if (data && data.status === "ok") {
            logToPanel(`[INFO] Requesting secure signature from your browser wallet...`);
            
            // 🚀 BROADCAST DIRECTLY FROM THE CONNECTED USER WALLET VIA THE BROWSER
            const txHash = await window.ethereum.request({
                method: 'eth_sendTransaction',
                params: [{
                    from: userAddr,                            // The user wallet holding the V12 internal balance
                    to: data.to_contract,                      // V12 Trading Engine Address
                    data: data.compiled_data,                  // The low-level bytecode generated by Python
                    value: '0x0'                               // 0 Native BNB for ERC20 swaps
                }],
            });

            logToPanel(`✅ Live Transaction Sent! Hash: ${txHash}`);
            return { status: "ok", tx_hash: txHash, txHash: txHash };
        }

        return data;

    } catch(err) {
        logToPanel(`❌ Swap Error: ${err.message}`);
        return { status: "error", message: err.message };
    }
}


window.runSwap = runSwap;
