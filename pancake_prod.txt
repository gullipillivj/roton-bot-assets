// ==================================================================
// 📊 GLOBAL LOGGING UTILITY
// ==================================================================
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// 🌟 STAGE 3: LINEAR SIMULATION PASS-THROUGH PIPELINE
async function runsimulateSwapUniswap(userAddr, amountWei, fromToken, toToken, targetRouter) {
    const res = await fetch("/prodbot/swap/simulateuniswap", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
            router: targetRouter,
            tokenIn: fromToken,
            tokenOut: toToken,
            amountIn: amountWei,
            amountOutMin: "0"
        })
    });
    return await res.json();
}
window.runsimulateSwapUniswap = runsimulateSwapUniswap;
