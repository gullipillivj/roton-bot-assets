// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;



// ✅ runSwap (frontend delegates router choice to backend)
// ✅ Synchronized runSwap Execution Framework (USDT -> Selected Asset Target)
async function runSwap(userAddr, usdtAmount, tokenAddress, network="erc20") {

    try {

        const USDT_ADDRESS =
            "0x55d398326f99059fF775485246999027B3197955";

        const amountWei =
            BigInt(Math.floor(Number(usdtAmount) * 1e18)).toString();

        const res = await fetch(
            `${window.API_BASE}/prodbot/swap`,
            {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    from_token: USDT_ADDRESS,
                    to_token: tokenAddress,
                    amount: amountWei,
                    network: network
                })
            }
        );

        return await res.json();

    } catch(err) {

        return {
            status: "error",
            message: err.message
        };
    }
}

window.runSwap = runSwap;
