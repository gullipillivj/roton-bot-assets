// ==================================================================
// 📊 GLOBAL LOGGING UTILITY
// ==================================================================
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// ==================================================================
// 🥞 PIPELINE 1: DISPATCH LOCAL BSC UNISWAP / PANCAKE ROUTING
// ==================================================================
async function runbscSwapUniswap(userAddr, amountWei, fromToken, toToken) {
    try {
        const base = window.API_BASE || "";
        const cleanAmount = amountWei ? amountWei.toString().trim() : "0";
        const pancakeRouterAddress = "0x10ED43C718714eb63d5aA57B78B54704E256024E";
        const amountOutMin = "0";

        logToPanel(`[ROUTE] Dispatching Uniswap pipeline query for ${cleanAmount} units...`);
        logToPanel(`TOKEN_IN=${fromToken}\nTOKEN_OUT=${toToken}\nAMOUNT_IN=${cleanAmount}\nMIN_OUT=${amountOutMin}`);

        const res = await fetch(`${base}/prodbot/swap/bscuniswap`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                router: pancakeRouterAddress,
                tokenIn: fromToken,
                tokenOut: toToken,
                amountIn: cleanAmount,
                amountOutMin: amountOutMin
            })
        });

        const data = await res.json();

        if (data && data.status === "ok") {
            logToPanel("[INFO] Sending transaction payload directly to browser extension...");
            const txHash = await window.ethereum.request({
                method: "eth_sendTransaction",
                params: [{
                    from: userAddr,
                    to: data.to_contract,
                    data: data.compiled_data,
                    value: "0x0"
                }]
            });
            logToPanel(`✅ Swap Transaction broadcasted successfully: ${txHash}`);
            return { status: "ok", tx_hash: txHash };
        }
        return data;
    } catch (err) {
        logToPanel(`❌ Uniswap Pipeline Error: ${err.message}`);
        return { status: "error", message: err.message };
    }
}
window.runbscSwapUniswap = runbscSwapUniswap;
