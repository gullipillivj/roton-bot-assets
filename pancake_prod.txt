// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;



// ✅ runSwap (frontend delegates router choice to backend)
// ✅ runSwap (USDT -> Selected Token)
async function runSwap(userAddr, usdtAmount, tokenAddress) {
  try {

    const USDT_ADDRESS =
      "0x55d398326f99059fF775485246999027B3197955";

    const amountWei =
      Math.floor(usdtAmount * 1e18).toString();

    const res = await fetch(`${API_BASE}/prodbot/swap`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify({
        from_token: USDT_ADDRESS,
        to_token: tokenAddress,
        amount: amountWei
      })
    });

    const data = await res.json();

    if (data.status !== "ok") {
      logToPanel(`❌ Swap failed: ${data.message}`);
      return data;
    }

    logToPanel(
      `✅ Swap prepared: ${tokenAddress}`
    );

    const txHash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [{
        from: userAddr,
        to: data.contract,
        data: data.swap_call
      }]
    });

    logToPanel(`🎉 Swap submitted: ${txHash}`);

    return {
      status: "ok",
      tx_hash: txHash
    };

  } catch (err) {

    logToPanel(`❌ Swap Error: ${err.message}`);

    return {
      status: "error",
      message: err.message
    };
  }
}



window.runSwap = runSwap;
