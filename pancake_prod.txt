// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;



// ✅ runSwap (frontend delegates router choice to backend)
// ✅ Synchronized runSwap Execution Framework (USDT -> Selected Asset Target)
async function runSwap(userAddr, usdtAmount, tokenAddress) {
  try {
    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";

    if (!userAddr || userAddr === "{wallet}") {
      logToPanel("❌ Swap Aborted: Web3 execution identity is invalid or disconnected.");
      return { status: "error", message: "Invalid user wallet configuration" };
    }

    if (!tokenAddress) {
      logToPanel("❌ Swap Aborted: Target destination token address missing.");
      return { status: "error", message: "Target token undefined" };
    }

    // Convert raw simulation inputs into standard decimal notation
    const amountWei = BigInt(Math.floor(parseFloat(usdtAmount) * 1e18)).toString();

    logToPanel(`⏳ Transmitting swap data to server: ${usdtAmount} USDT -> ${tokenAddress}`);

    const res = await fetch(`${window.API_BASE}/prodbot/swap`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify({
        from_token: USDT_ADDRESS,
        to_token: tokenAddress,
        amount: amountWei
      })
    });

    if (!res.ok) {
      throw new Error(`Server returned HTTP network error status code ${res.status}`);
    }

    const data = await res.json();

    if (data.status !== "ok") {
      logToPanel(`❌ Swap Pipeline Processing Error: ${data.message}`);
      return data;
    }

    logToPanel(`📡 Swap verified on server! Prompting MetaMask for contract execution signatures...`);

    // Execute transaction payload against V12 Engine smart contract architecture
    const txHash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [{
        from: userAddr,
        to: data.contract, // Resolves safely to V12_ADDR
        data: data.swap_call,
        gas: "0x5B8D8" // 375,000 gas limit override to protect complex cross-pool path processing
      }]
    });

    logToPanel(`🎉 Automation Swap Dispatched Successfully! Hash: ${txHash}`);

    return {
      status: "ok",
      tx_hash: txHash
    };

  } catch (err) {
    logToPanel(`❌ Swap Engine Interface Error: ${err.message}`);
    return {
      status: "error",
      message: err.message
    };
  }
}

window.runSwap = runSwap;
