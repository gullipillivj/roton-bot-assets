// ✅ RoTon Bot Frontend Controller (V12)
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

logToPanel("main_prod.txt loaded successfully");

// ✅ Parse URL params
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";
window.BASE_URL = window.location.origin;
window.WBNB_TOKEN_ADDRESS = "0xBB4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

// ✅ Connect Wallet
document.getElementById("connectwalletBtn")?.addEventListener("click", async () => {
  if (typeof window.ethereum !== "undefined") {
    const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
    window.USER_ADDR = accounts[0];
    document.getElementById("currentWalletLabel").innerText = window.USER_ADDR;
    logToPanel(`✅ Wallet connected: ${window.USER_ADDR}`);
  } else {
    logToPanel("❌ MetaMask not found.");
  }
});

// ✅ Deposit
document.getElementById("depositBtn")?.addEventListener("click", async () => {
  const token = document.getElementById("tokenSelect").value;
  const amount = parseFloat(document.getElementById("depositAmount").value) * 1e6;
  if (!window.USER_ADDR) return logToPanel("❌ Connect wallet first.");
  const payload = { userid: window.USER_ID, wallet: window.USER_ADDR, token, amount };
  try {
    const res = await fetch(`/prodbot/deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload)
    });
    const data = await res.json();
    logToPanel(`✅ Deposit prepared: ${JSON.stringify(data)}`);
  } catch (err) {
    logToPanel(`❌ Deposit error: ${err.message}`);
  }
});

// ✅ Balance Sync
async function syncBalance() {
  if (!window.USER_ADDR) return;
  const token = document.getElementById("tokenSelect").value;
  try {
    const res = await fetch(`/prodbot/balance`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ user: window.USER_ADDR, token })
    });
    const data = await res.json();
    if (data.status === "ok") {
      const balance = parseFloat(data.balance) / 1e6;
      document.getElementById("liveProfitValue").innerText = `${balance.toFixed(2)} USDT`;
      if (typeof window.updateChart === "function") {
        window.updateChart(balance);
      }
      logToPanel(`💰 Balance synced: ${balance.toFixed(2)} USDT`);
    }
  } catch (err) {
    logToPanel(`❌ Balance sync error: ${err.message}`);
  }
}
setInterval(syncBalance, 10000); // every 10 seconds
