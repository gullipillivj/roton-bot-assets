// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;


//run swap
async function runSwap(userAddr, bnbAmount, tokenAddress) {
  try {
    const res = await fetch(`https://roton.store/prodbot/swap`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        router: "0x10ED43C718714eb63d5aA57B78B54704E256024E",
        tokenIn: window.WBNB_TOKEN_ADDRESS,       // input token (BNB wrapped)
        tokenOut: tokenAddress,                   // output token
        amountIn: Math.floor(bnbAmount * 1e18),   // convert to wei
        amountOutMin: 0,                          // set slippage minOut if needed
        dstChainId: null,
        recipient: userAddr
      })
    });

    const data = await res.json();
    if (data.status === "success") {
      logToPanel(`✅ Swap executed successfully. TxHash=${data.tx_hash}`);
      return data;
    } else {
      logToPanel(`❌ Swap failed: ${data.message}`);
      return data;
    }
  } catch (err) {
    logToPanel(`[SWAP ERROR] ${err.message}`);
    return { status: "error", message: err.message };
  }
}

window.runSwap = runSwap;
