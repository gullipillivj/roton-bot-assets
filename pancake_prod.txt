logToPanel("pancake_prod.txt loaded");

async function runSwap(bnbAmount, targetTokenAddress) {
  try {
    const res = await fetch("/prodbot/swap", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        user_wallet: window.USER_ADDR,
        target_token: targetTokenAddress,
        amount_bnb: bnbAmount,
        wbnb_address: window.WBNB_TOKEN_ADDRESS,
        router: "0x10ED43C718714eb63d5aA57B78B54704E256024E" // PancakeSwap V2 Router Address
      })
    });
    return await res.json();
  } catch (err) {
    logToPanel(`❌ Local server route bridge connection broken: ${err.message}`);
    return { status: "error", message: err.message };
  }
}

window.runSwap = runSwap;
