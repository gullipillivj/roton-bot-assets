// pancake_prod.txt — connector layer (approve/deposit/withdraw/swap via secure backend)
logToPanel("pancake_prod.txt loaded");

function safeParse(text) {
  try { return JSON.parse(text); } catch { return null; }
}
function logResult(action, data) {
  if (data && data.tx_hash) {
    logToPanel(`[RESULT] ${action} — TxHash=${data.tx_hash}`);
  } else {
    logToPanel(`[RESULT] ${action} — Raw=${JSON.stringify(data)}`);
  }
}

// === Secure Backend Calls ===
async function callSecurity(action, amount) {
  try {
    const res = await fetch("/prodbot/security", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        wallet: window.USER_ADDR,
        amount,
        action
      })
    });
    return await res.json();
  } catch (err) {
    logToPanel(`[ERROR] ${action} failed: ${err.message}`);
    return null;
  }
}

// === Wrappers ===
async function runApprove(amount) {
  const parsed = await callSecurity("approve", amount);
  logResult("Approve", parsed);
  return parsed;
}
async function runDeposit(amount) {
  const parsed = await callSecurity("deposit", amount);
  logResult("Deposit", parsed);
  return parsed;
}
async function runWithdraw(amount) {
  const parsed = await callSecurity("withdraw", amount);
  logResult("Withdraw/Payout", parsed);
  return parsed;
}
async function runSwap(amount) {
  const parsed = await callSecurity("swap", amount);
  logResult("Swap", parsed);
  return parsed;
}
async function runPayout(amount) {
  const parsed = await callSecurity("payout", amount);
  logResult("Payout", parsed);
  return parsed;
}

// === Expose globally ===
window.runApprove = runApprove;
window.runDeposit = runDeposit;
window.runWithdraw = runWithdraw;
window.runSwap = runSwap;
window.runPayout = runPayout;
window.logResult = logResult;
