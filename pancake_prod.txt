function logToPanel(msg) {

  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);

window.logToPanel = logToPanel;
async function requestTokenSpendApproval(userAddr, amountWei, tokenIn, spenderContract) {
    try {
        const cleanUser = userAddr.toLowerCase().trim();
        const cleanToken = tokenIn.toLowerCase().trim();
        const cleanSpender = spenderContract.toLowerCase().trim();

        const approveMethodId = "0x095ea7b3";
        const paddedSpender = cleanSpender.replace("0x", "").padStart(64, "0");
        const paddedAmount = BigInt(amountWei).toString(16).padStart(64, "0");
        const compiledData = approveMethodId + paddedSpender + paddedAmount;

        logToPanel(`📢 [APPROVAL PROMPT] Authorizing Token Spend Allowance for: ${cleanToken}`);
        
        const approveTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: cleanUser,
                to: cleanToken, 
                data: compiledData,
                value: "0x0"
            }]
        });
        
        logToPanel(`✅ Approval hash recorded: ${approveTxHash}. Waiting 6s for block state propagation...`);
        await new Promise(resolve => setTimeout(resolve, 6000));
        return true;
    } catch (err) {
        logToPanel(`❌ User rejected token allocation balance approval: ${err.message || err}`);
        throw err;
    }
}
window.requestTokenSpendApproval = requestTokenSpendApproval;
async function runbscSwapUniswap(userAddr, amountWei, fromToken, toToken, targetRouter) {
    try {
        const V12_ADDR = "0xe191B1b7E5c3d7b7B9386F4212259241fC62135B"; 
        const cleanUser = userAddr.toLowerCase().trim();
        const cleanFrom = fromToken.toLowerCase().trim();
        const cleanTo = toToken.toLowerCase().trim();
        const cleanRouter = targetRouter.toLowerCase().trim();
        
        await requestTokenSpendApproval(cleanUser, amountWei, cleanFrom, V12_ADDR);

        logToPanel(`⏳ Compiling swap payload data from backend route...`);
        const res = await fetch("/prodbot/swap/bscuniswap", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                router: cleanRouter,
                tokenIn: cleanFrom,
                tokenOut: cleanTo,
                amountIn: BigInt(amountWei).toString(), 
                amountOutMin: "0"
            })
        });
        
        const backendPayload = await res.json();
        if (!backendPayload || backendPayload.status !== "ok") {
            throw new Error(backendPayload.message || "Failed to compile transaction payloads.");
        }

        logToPanel(`📢 [SWAP PROMPT] Confirming execution parameters with the V12 Trading Engine...`);
        const swapTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: cleanUser,
                to: backendPayload.swap_packet.to_contract.toLowerCase().trim(), 
                data: backendPayload.swap_packet.compiled_data, 
                value: "0x0"
            }]
        });

        logToPanel(`🎉 Trade Segment Dispatched! Hash: ${swapTxHash}`);
        return { status: "ok", tx_hash: swapTxHash };
    } catch (e) {
        logToPanel(`❌ BSC Swap Segment Pipeline Interrupted: ${e.message || e}`);
        return { status: "error", message: e.toString() };
    }
}
window.runbscSwapUniswap = runbscSwapUniswap;
