// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;



// ✅ runSwap (frontend delegates router choice to backend)
// ✅ Synchronized runSwap Execution Framework (USDT -> Selected Asset Target)
async function runSwap(userAddr, usdtAmount, tokenAddress, network = "erc20") {
  try {
    const USDT_ADDRESS =
      "0x55d398326f99059fF775485246999027B3197955";

    const amountWei =
      BigInt(Math.floor(Number(usdtAmount) * 1e18)).toString();

    logToPanel(`⏳ Cycle Swap Request`);
    logToPanel(`Network: ${network}`);
    logToPanel(`Amount: ${usdtAmount}`);

    const res = await fetch(`${window.API_BASE}/prodbot/swap`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify({
        from_token: USDT_ADDRESS,
        to_token: tokenAddress,
        amount: amountWei,
        network: network
      })
    });

    logToPanel("✅ Swap route replied");

    const data = await res.json();

    if (data.status !== "ok") {
      throw new Error(data.message);
    }

    logToPanel("✅ Calldata generated");

    const txHash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [{
        from: userAddr,
        to: data.contract,
        data: data.swap_call,
        gas: "0x5B8D8"
      }]
    });

    logToPanel(`✅ TX Sent ${txHash}`);

    return {
      status: "ok",
      tx_hash: txHash
    };

  } catch (e) {
    logToPanel(`❌ ${e.message}`);

    return {
      status: "error",
      message: e.message
    };
  }
}



window.runSwap = runSwap;
