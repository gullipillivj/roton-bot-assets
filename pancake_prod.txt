// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
} logToPanel;

// ✅ runSwap (frontend delegates router choice to backend)
async function runSwap(userAddr, bnbAmount, tokenAddress) {
  try {
    const res = await fetch(`${API_BASE}/prodbot/swap`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        from_token: window.WBNB_TOKEN_ADDRESS,     // input token (BNB wrapped)
        to_token: tokenAddress,                   // output token
        amount: Math.floor(bnbAmount * 1e18),     // convert to wei
        wallet: userAddr                          // recipient wallet
      })
    });

    const data = await res.json();
    if (data.status === "ok") {
      logToPanel(`✅ Swap prepared: ${JSON.stringify(data)}`);
      // 🔑 Next step: execute transaction with MetaMask
      // window.ethereum.request({
      //   method: "eth_sendTransaction",
      //   params: [ { from: userAddr, to: V12_ADDR, data: data.swap_call } ]
      // });
      return data;
    } else {
      logToPanel(`❌ Swap failed: ${data.message}`);
      return data;
    }
  } catch (err) {
    logToPanel(`[SWAP ERROR] ${err.message}`);
    return { status: "error", message: err.message };
  }
}


window.runSwap = runSwap;
