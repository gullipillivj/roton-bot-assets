// 📊 GLOBAL LOGGING UTILITY
// ==================================================================
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// 🚨 CORE UTILITY: WEB3 RAW HEX APPROVAL GENERATOR
async function requestTokenSpendApproval(userAddr, amountWei, tokenIn, spenderContract) {
    try {
        const approveMethodId = "0x095ea7b3";
        const paddedSpender = spenderContract.replace("0x", "").toLowerCase().padStart(64, "0");
        const paddedAmount = BigInt(amountWei).toString(16).padStart(64, "0");
        const compiledData = approveMethodId + paddedSpender + paddedAmount;

        logToPanel(`📢 [WALLET PROMPT] Authorizing Token Spend Allowance for: ${tokenIn}`);
        
        const approveTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: userAddr,
                to: tokenIn, 
                data: compiledData,
                value: "0x0"
            }]
        });
        
        logToPanel(`✅ Approval hash recorded: ${approveTxHash}. Waiting 5s for block confirmation...`);
        await new Promise(resolve => setTimeout(resolve, 5000));
        return true;
    } catch (err) {
        logToPanel(`❌ User rejected token allocation balance approval: ${err.message || err}`);
        throw err;
    }
}

// 🌟 STAGE 4: LINEAR LIVE CROSS-CHAIN BRIDGE PASS-THROUGH PIPELINE
async function runcrossChainSwap(userAddr, amountWei, fromToken, ccipRouter, destinationChainSelector, recipient) {
    try {
        const ENGINE_ADDRESS = "0xe191B1b7E5c3d7b7B9386F4212259241fC62135B";
        logToPanel(`⏳ Initiating cross-chain pipeline request mapping...`);
        
        const res = await fetch("/prodbot/swap/crosschain", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                ccipRouter: ccipRouter,
                tokenIn: fromToken,
                destinationChainSelector: destinationChainSelector,
                amountIn: amountWei,
                recipient: recipient
            })
        });
        
        const payload = await res.json();
        if (!payload || payload.status !== "ok") {
            throw new Error(payload.message || "Failed to compile cross-chain transaction payloads.");
        }

        logToPanel(`📢 [WALLET PROMPT] Authorizing token spending allowance parameters...`);
        const approveTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: userAddr,
                to: payload.approval_packet.to_contract,
                data: payload.approval_packet.compiled_data,
                value: "0x0"
            }]
        });
        
        logToPanel(`✅ Approval hash recorded: ${approveTxHash}. Waiting 12s for block confirmation...`);
        await new Promise(resolve => setTimeout(resolve, 12000));

        logToPanel(`📢 [WALLET PROMPT] Confirming cross-chain bridge processing via Chainlink CCIP...`);
        const hexFeeValue = "0x" + BigInt(payload.swap_packet.value_wei).toString(16);
        
        const bridgeTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: userAddr,
                to: payload.swap_packet.to_contract,
                data: payload.swap_packet.compiled_data,
                value: hexFeeValue 
            }]
        });
        
        logToPanel(`🎉 Cross-Chain Bridge Transaction Dispatched! Hash: ${bridgeTxHash}`);
        return { status: "ok", tx_hash: bridgeTxHash };

    } catch (e) {
        logToPanel(`❌ Cross-Chain Action Interrupted: ${e.message || e}`);
        return { status: "error", message: e.toString() };
    }
}
window.runcrossChainSwap = runcrossChainSwap;
