// pancake_prod.txt — connector layer (approve/deposit/withdraw/swap via secure backend)
logToPanel("pancake_prod.txt loaded");

function safeParse(text) {
  try { return JSON.parse(text); } catch { return null; }
}
function logResult(action, data) {
  if (data && data.tx_hash) {
    logToPanel(`[RESULT] ${action} — TxHash=${data.tx_hash}`);
  } else {
    logToPanel(`[RESULT] ${action} — Raw=${JSON.stringify(data)}`);
  }
}


// === Secure Backend Calls ===
async function callSecurity(action, payload) {
  try {
    const res = await fetch("/prodbot/security", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        wallet: window.USER_ADDR,
        ...payload,
        action
      })
    });
    return await res.json();
  } catch (err) {
    logToPanel(`[ERROR] ${action} failed: ${err.message}`);
    return null;
  }
}

// === Wrappers ===
// Approve ERC20 token for spending by SolidarityFundV2
async function runApprove(amount, tokenAddress) {
  const parsed = await callSecurity("approve", { amount, tokenAddress });
  logResult("Approve", parsed);
  return parsed;
}

// Deposit ERC20 into SolidarityFundV2
async function runDeposit(amount, tokenAddress) {
  const parsed = await callSecurity("deposit", { amount, tokenAddress });
  logResult("Deposit", parsed);
  return parsed;
}

// Withdraw / Swap to donation token (calls SolidarityFundV2.swapToDonation)
async function runSwap(amount, tokenAddress, amountOutMin = "0") {
  // include gas calculation
  const gas = await window.getGasFees("bsc");
  const parsed = await callSecurity("swapToDonation", {
    amount,
    tokenAddress,
    amountOutMin,
    gasPrice: gas.gasPrice,
    gasCostUSD: gas.usdCost
  });
  logResult("SwapToDonation", parsed);
  return parsed;
}

// Convenience alias for payout
async function runPayout(amount, tokenAddress) {
  return await runSwap(amount, tokenAddress, "0");
}

// === Expose globally ===
window.runApprove = runApprove;
window.runDeposit = runDeposit;
window.runWithdraw = runSwap;   // withdraw is swapToDonation
window.runSwap = runSwap;
window.runPayout = runPayout;
window.logResult = logResult;
