// 📊 GLOBAL LOGGING UTILITY
// ==================================================================
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// 🚨 CORE UTILITY: WEB3 RAW HEX APPROVAL GENERATOR
async function requestTokenSpendApproval(userAddr, amountWei, tokenIn, spenderContract) {
    try {
        const approveMethodId = "0x095ea7b3";
        const paddedSpender = spenderContract.replace("0x", "").toLowerCase().padStart(64, "0");
        const paddedAmount = BigInt(amountWei).toString(16).padStart(64, "0");
        const compiledData = approveMethodId + paddedSpender + paddedAmount;

        logToPanel(`📢 [POPUP 1/2 - APPROVAL] Authorizing Token Spend Allowance for: ${tokenIn}`);
        
        const approveTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: userAddr,
                to: tokenIn, // Sent directly to the ERC20 token contract
                data: compiledData,
                value: "0x0"
            }]
        });
        
        logToPanel(`✅ Approval hash recorded: ${approveTxHash}. Waiting 6s for block propagation...`);
        await new Promise(resolve => setTimeout(resolve, 6000));
        return true;
    } catch (err) {
        logToPanel(`❌ User rejected token allocation balance approval: ${err.message || err}`);
        throw err;
    }
}

// 🌟 STAGE 4: FIXED LIVE BSC SWAP PIPELINE (PROMPTS POPUP 2)
async function runbscSwapUniswap(userAddr, amountWei, fromToken, toToken, targetRouter) {
    try {
        const V12_ADDR = "0xe191B1b7E5c3d7b7B9386F4212259241fC62135B"; // Synchronized global variable parameter address
        
        // Step 1: Prompt the wallet for the ERC20 Token Contract Approval popup
        await requestTokenSpendApproval(userAddr, amountWei, fromToken, V12_ADDR);

        // Step 2: Fetch the compiled execution calldata parameters from the Python server
        logToPanel(`⏳ Compiling swap payload data from backend route...`);
        // Replace the backend payload assembly phase inside your runbscSwapUniswap function:
const res = await fetch("/prodbot/swap/bscuniswap", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
        router: targetRouter,
        tokenIn: fromToken,
        tokenOut: toToken,
        amountIn: BigInt(amountWei).toString(), // Ensure pure numeric string execution vectors
        amountOutMin: "0"
    })
});

        
        const backendPayload = await res.json();
        if (!backendPayload || backendPayload.status !== "ok") {
            throw new Error(backendPayload.message || "Failed to compile transaction payloads.");
        }

        // Step 3: Prompt the wallet for POPUP 2/2 (The On-Chain Swap Contract Execution Call)
        logToPanel(`📢 [POPUP 2/2 - SWAP] Confirming execution parameters with the V12 Trading Engine...`);
        const swapTxHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: userAddr,
                to: backendPayload.swap_packet.to_contract, // V12 Smart Contract address
                data: backendPayload.swap_packet.compiled_data, // executeERC20SwapUniswap payload
                value: "0x0"
            }]
        });

        logToPanel(`🎉 Trade Dispatched to Network! Swap Hash: ${swapTxHash}`);
        return { status: "ok", tx_hash: swapTxHash };
    } catch (e) {
        logToPanel(`❌ BSC Swap Pipeline Interrupted: ${e.message || e}`);
        return { status: "error", message: e.toString() };
    }
}
window.runbscSwapUniswap = runbscSwapUniswap;
