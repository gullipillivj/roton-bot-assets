function ensure18Decimals(amountInput) {
    if (!amountInput) return "0";
    const str = amountInput.toString().trim();
    if (str === "0" || str === "") return "0";
    if (str.length > 12) return str;
    
    if (!str.includes('.')) {
        return str + "000000000000000000";
    }
    
    const parts = str.split('.');
    const whole = parts[0];
    let fraction = parts[1];
    
    if (fraction.length < 18) {
        fraction = fraction.padEnd(18, '0');
    } else if (fraction.length > 18) {
        fraction = fraction.substring(0, 18);
    }
    
    const combined = (whole + fraction).replace(/^0+/, '');
    return combined === "" ? "0" : combined;
}

// 🌟 DYNAMIC TRIPLE DEPOSIT ENGINE WITH SEQUENTIAL MATHEMATICAL DEDUCTIONS
async function runDepositPipeline(userAddr, amountHumanOrWei) {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        
        // 1. Calculate Initial Total Base Amount in 18-Decimal Wei
        const totalInputWei = amountHumanOrWei.toString().length > 12 
            ? BigInt(amountHumanOrWei.toString()) 
            : BigInt(ensure18Decimals(amountHumanOrWei));
            
        window.logToPanel(`[DEPOSIT START] Initial Input Parameter: ${(Number(totalInputWei) / 1e18).toFixed(4)} USDT (${totalInputWei.toString()} Wei)`);
        
        // Fixed Constant Deductions in 18-decimal base representation (0.03 USDT = 30000000000000000 Wei)
        const fixedGasDeductionWei = 30000000000000000n; 
        
        const v12ContractAddress = "0x1854170353ea89672638F9fa5acB477c45523871";
        const usdtTokenAddress = "0x55d398326f99059fF775485246999027B3197955";

        // ----------------------------------------------------------
        // 🌟 STEP 1/3: ERC-20 APPROVE VALUE (DEDUCT FIRST \$0.03)
        // ----------------------------------------------------------
        if (totalInputWei <= fixedGasDeductionWei) {
            throw new Error("Input amount is insufficient to cover the basic gas allowance deduction threshold.");
        }
        
        // Compute Pool Balance after first deduction allocation
        const remainingPoolBalanceWei = totalInputWei - fixedGasDeductionWei;
        window.logToPanel(`[STEP 1] Deducting $0.03 approval cost. Available trading pool balance: ${(Number(remainingPoolBalanceWei) / 1e18).toFixed(4)} USDT`);
        
        window.logToPanel(`[STAGE 1/3] Prompting wallet to authorize token spend allowance...`);
        const cleanSpender = v12ContractAddress.toLowerCase().replace("0x", "").padStart(64, "0");
        const cleanValue = BigInt(totalInputWei).toString(16).padStart(64, "0"); // Approve total entry parameters
        const approveCalldata = "0x095ea7b3" + cleanSpender + cleanValue;
        
        const txHash1 = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: userAddr, to: usdtTokenAddress, data: approveCalldata, value: '0x0' }],
        });
        window.logToPanel(`⏳ Stage 1 complete. Allowance Hash: ${txHash1}. Synchronizing block status...`);
        await new Promise(resolve => setTimeout(resolve, 5000)); 

        // ----------------------------------------------------------
        // 🌟 STEP 2/3: depositCommissionFirst WITH commissionPaid = false (4% OF POOL)
        // ----------------------------------------------------------
        // Calculate exactly 4% (0.04 factor) of the adjusted pool balance safely using BigInt scaling math
        const step2TxAmountWei = (remainingPoolBalanceWei * 4n) / 100n;
        window.logToPanel(`[STEP 2] Computed 4% asset fraction: ${(Number(step2TxAmountWei) / 1e18).toFixed(4)} USDT (${step2TxAmountWei.toString()} Wei)`);
        
        window.logToPanel(`[STAGE 2/3] Requesting signature payload for core deposit (commissionPaid = false)...`);
        const res = await fetch(`${baseApiUrl}/prodbot/deposit/usdt/pipeline`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ amount: step2TxAmountWei.toString(), tradingPhase: false })
        });
        const data = await res.json();
        if (!data || data.status !== "ok") throw new Error(data.message || "Failed API payload parsing.");
        
        const txHash2 = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: userAddr, to: v12ContractAddress, data: data.txdata, value: '0x0' }],
        });
        window.logToPanel(`⏳ Stage 2 complete. Core Deposit Hash: ${txHash2}. Synchronizing block status...`);
        await new Promise(resolve => setTimeout(resolve, 5000));

        // ----------------------------------------------------------
        // 🌟 STEP 3/3: depositCommissionFirst WITH commissionPaid = true (REMAINDER MINUS \$0.03)
        // ----------------------------------------------------------
        // Remaining 96% chunk calculation
        const remainderChunkWei = remainingPoolBalanceWei - step2TxAmountWei;
        
        // Deduct second fixed \$0.03 gas allowance step from the remainder
        if (remainderChunkWei <= fixedGasDeductionWei) {
            throw new Error("Remaining pool capital is depleted past fixed execution boundary limits.");
        }
        const step3TxAmountWei = remainderChunkWei - fixedGasDeductionWei;
        
        window.logToPanel(`[STEP 3] Final net transfer amount (Remainder minus $0.03): ${(Number(step3TxAmountWei) / 1e18).toFixed(4)} USDT (${step3TxAmountWei.toString()} Wei)`);
        window.logToPanel(`[STAGE 3/3] Requesting signature payload for fee tracking (commissionPaid = true)...`);
        
        const secondRes = await fetch(`${baseApiUrl}/prodbot/deposit/usdt/pipeline`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ amount: step3TxAmountWei.toString(), tradingPhase: true })
        });
        const secondData = await secondRes.json();
        if (!secondData || secondData.status !== "ok") throw new Error("Failed to compile final stage data.");
        
        const txHash3 = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: userAddr, to: v12ContractAddress, data: secondData.txdata, value: '0x0' }],
        });
        
        window.logToPanel(`🎉 Cascading Pipeline Fully Settled! Master Final Hash: ${txHash3}`);
        
        // Refresh values inside dashboard layout view instantly
        if (typeof window.syncBalanceFromFund === "function") {
            await window.syncBalanceFromFund();
        }
        return { status: "ok", tx_hash: txHash3 };
        
    } catch (err) {
        window.logToPanel(`❌ Downscaling Pipeline Break: ${err.message}`);
        return { status: "error", message: err.message };
    }
}
window.runDepositPipeline = runDepositPipeline;

async function syncBalanceFromFund() {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        if (!window.USER_ADDR || !window.USER_ID) {
            logToPanel("⚠️ Identification vector missing. Halting balance initialization cycles.");
            return null;
        }

        logToPanel(`🔎 Querying clients.db tokens map tracking for User ID: ${window.USER_ID}...`);
        
        let currentToken = "0x55d398326f99059fF775485246999027B3197955"; // Safe default (USDT)
        let lastToken = "0x0E09FaBB73Bd3Ade0a17ECC321fD13a19e81cE82";    // Safe default (CAKE)
        
        // 🟢 STEP 1: Quick fetch targeting the DB token layout entries
        try {
            const tokenDbRes = await fetch(`${baseApiUrl}/prodbot/tokens/get`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ userid: window.USER_ID })
            });
            const dbRecord = await tokenDbRes.json();
            if (dbRecord && dbRecord.status === "ok") {
                if (dbRecord.currenttoken) currentToken = dbRecord.currenttoken;
                if (dbRecord.lasttoken) lastToken = dbRecord.lasttoken;
            }
        } catch (dbErr) {
            console.error("Database reading breakdown:", dbErr);
        }

        logToPanel(`💡 DB Reference Targets -> [Current: ${currentToken}] | [Last: ${lastToken}]`);

        // Helper shortcut function to clear empty RPC strings
        const queryBalance = async (tokenAddr) => {
            const res = await fetch(baseApiUrl + "/prodbot/balance", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ user: window.USER_ADDR, token: tokenAddr })
            });
            const text = await res.text();
            return text.trim() || "0";
        };

        // 🟢 STEP 2: Fire instantaneous parallel queries for both targets
        logToPanel("⏳ Executing high-speed dual-target balance query checks...");
        const [currentWei, lastWei] = await Promise.all([
            queryBalance(currentToken),
            queryBalance(lastToken)
        ]);

        let finalWeiString = "0";
        let resolvedTokenAddress = currentToken;

        // 🟢 STEP 3: Route assets using instantaneous priority gates
        if (BigInt(currentWei) > 0n) {
            finalWeiString = currentWei;
            resolvedTokenAddress = currentToken;
            logToPanel(`🎉 Capital located inside Current Token! Balance: ${finalWeiString}`);
        } else if (BigInt(lastWei) > 0n) {
            finalWeiString = lastWei;
            resolvedTokenAddress = lastToken;
            logToPanel(`🎉 Capital located inside Last Token! Balance: ${finalWeiString}`);
        } else {
            // 🟢 STEP 4: Comprehensive market sweep only if both values return zero
            logToPanel("⚠️ Both tracks empty. Launching wide contract search matrix...");
            try {
                const sweepRes = await fetch(`${baseApiUrl}/prodbot/active-asset`, {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({ user: window.USER_ADDR })
                });
                const activeState = await sweepRes.json();
                if (activeState && activeState.status === "ok" && BigInt(activeState.confirmed_balance_wei) > 0n) {
                    finalWeiString = activeState.confirmed_balance_wei;
                    resolvedTokenAddress = activeState.confirmed_address;
                    logToPanel(`✅ Wide Sweep Discovered Asset: ${activeState.confirmed_symbol} | Balance: ${finalWeiString}`);
                }
            } catch (sweepErr) {
                console.error("Wide discovery search exception hit:", sweepErr);
            }
        }

        logToPanel("[LEDGER SYNC] Process complete. Final active balance: " + finalWeiString);
        window.SolidarityFundBalance = parseFloat(finalWeiString);

        // 🟢 STEP 5: Apply variables directly to dashboard input forms
        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");
        const usdBox = document.getElementById("usd");

        if (startBox) startBox.value = finalWeiString;
        if (investBox) investBox.value = finalWeiString;
        if (usdBox) {
            usdBox.value = (parseFloat(finalWeiString) / 1e18).toFixed(4);
        }

        return window.SolidarityFundBalance;
    } catch (err) {
        logToPanel("❌ Balance Sync Error: " + err.message);
        return null;
    }
}
window.syncBalanceFromFund = syncBalanceFromFund;
