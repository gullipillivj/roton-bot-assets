// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;
window.logview = logToPanel;

logToPanel("main_prod.txt loaded — minimal mode");

// Define WBNB token address globally (still available if needed)
WBNB_TOKEN_ADDRESS = "0xBB4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

// ✅ Query params parser
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";
window.BASE_URL = "https://roton.store";

logToPanel(`✅ User context loaded: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, WALLET=${window.WALLET_NAME}`);

// ✅ Populate dropdowns
async function populateDropdowns() {
  // Example static lists — replace with balanceOf / checkUserBalance calls
  const walletTokens = [
    { symbol: "BNB", address: WBNB_TOKEN_ADDRESS, balance: "1.25" },
    { symbol: "USDT", address: "0x55d398326f99059fF775485246999027B3197955", balance: "200.00" }
  ];
  const contractTokens = [
    { symbol: "RR", address: "0xba67186fd454D0256a0D3744783538bE08C58feD", balance: "5000.00" },
    { symbol: "BUSD", address: "0xe9e7cea3dedca5984780bafc599bd69add087d56", balance: "100.00" }
  ];

  const walletSelect = document.getElementById("walletTokenSelect");
  const contractSelect = document.getElementById("contractTokenSelect");
  walletSelect.innerHTML = "";
  contractSelect.innerHTML = "";

  walletTokens.forEach(t => {
    const opt = document.createElement("option");
    opt.value = t.address;
    opt.textContent = `${t.symbol} (${t.balance})`;
    walletSelect.appendChild(opt);
  });

  contractTokens.forEach(t => {
    const opt = document.createElement("option");
    opt.value = t.address;
    opt.textContent = `${t.symbol} (${t.balance})`;
    contractSelect.appendChild(opt);
  });

  logToPanel("✅ Token dropdowns populated.");
}

// ✅ ConnectWallet button
document.getElementById("connectwalletBtn")?.addEventListener("click", async () => {
  try {
    if (!window.ethereum) {
      logToPanel("❌ No Ethereum provider found — install MetaMask.");
      return;
    }
    const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
    window.USER_ADDR = accounts[0];
    logToPanel(`✅ Wallet connected: ${window.USER_ADDR}`);
    populateDropdowns(); // populate dropdowns after connection
  } catch (err) {
    logToPanel(`[CONNECT ERROR] ${err.message}`);
  }
});

// ✅ Deposit button handler
document.getElementById("depositBtn")?.addEventListener("click", async () => {
  const token = document.getElementById("walletTokenSelect").value;
  const amt = parseFloat(document.getElementById("depositAmount")?.value || "0");
  if (amt <= 0) {
    return logToPanel("❌ Invalid deposit amount.");
  }

  try {
    logToPanel(`[DEBUG] Sending POST to /prodbot/deposit with token=${token}, amount=${amt}`);

    const resp = await fetch(`${window.BASE_URL}/prodbot/deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        wallet: window.USER_ADDR,
        walletname: window.WALLET_NAME,
        token,
        amount: amt
      })
    });

    logToPanel("[DEBUG] Response received from /prodbot/deposit");
    const data = await resp.json();
    logToPanel(`[DEBUG] Deposit response: ${JSON.stringify(data)}`);

    if (!data.deposit_tx) {
      return logToPanel(`❌ Deposit error: ${data.message || "Deposit failed."}`);
    }

    logToPanel(`✅ Deposit executed successfully — TxHash=${data.deposit_tx.result}`);
  } catch (err) {
    logToPanel(`[DEPOSIT ERROR] ${err.message}`);
  }
});
