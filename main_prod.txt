function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

function getQueryParams() {
  try {
    const url = new URL(window.location.href);
    const params = url.searchParams;
    return {
      userid: params.get("userid") || "unknown",
      wallet: params.get("wallet") || null,
      walletname: params.get("walletname") || "unknown"
    };
  } catch (err) {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid;
window.USER_ADDR = q.wallet;
window.WALLET_NAME = q.walletname;

function initializeLabels() {
  const userLabel = document.getElementById("currentUserLabel");
  const walletLabel = document.getElementById("currentWalletLabel");
  const nameLabel = document.getElementById("currentWalletName");

  if (userLabel) userLabel.innerText = "Current User: " + window.USER_ID;
  if (walletLabel) walletLabel.innerText = "Wallet Address: " + window.USER_ADDR;
  if (nameLabel) nameLabel.innerText = "Wallet Name: " + window.WALLET_NAME;
}

function getSelectedToken() {
  const el = document.getElementById("tokenSelect");
  if (!el) return "USDT";
  const val = el.value.toLowerCase();
  if (val === "0x55d398326f99059ff775485246999027b3197955" || val === "usdt") return "USDT";
  if (val === "bnb") return "BNB";
  if (val === "eth") return "ETH";
  return "USDT";
}

async function syncBalanceFromFund() {
  if (!window.USER_ADDR) return 0;
  try {
    const res = await fetch(`${window.API_BASE}/prodbot/balance`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ user: String(window.USER_ADDR), token: "0x55d398326f99059fF775485246999027B3197955" })
    });
    const data = await res.json();
    if (data.status === "ok") {
      window.SolidarityFundBalance = parseFloat(data.balance) || 0; 
      return window.SolidarityFundBalance;
    }
  } catch (err) {
    console.log(`Balance sync notice: ${err.message}`);
  }
  return 0; 
}
window.syncBalanceFromFund = syncBalanceFromFund;
