// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// ✅ Query params parser
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

// ✅ Populate dropdowns (show all tokens from list)
async function populateWalletTokens() {
    if (!window.USER_ADDR) {
        logToPanel("❌ Wallet address not set.");
        return;
    }

    walletSelect = document.getElementById("walletTokenSelect");
    walletSelect.innerHTML = "";

    apiKey = "YOUR_BSCSCAN_API_KEY"; // replace with your key
    url = `https://api.bscscan.com/api?module=account&action=tokentx&address=${window.USER_ADDR}&apikey=${apiKey}`;

    try {
        res = await fetch(url);
        data = await res.json();

        if (data.status !== "1") {
            logToPanel("❌ Error fetching tokens: " + data.message);
            return;
        }

        seen = {};
        for (i = 0; i < data.result.length; i++) {
            tx = data.result[i];
            if (!seen[tx.contractAddress]) {
                opt = document.createElement("option");
                opt.value = tx.contractAddress;
                opt.textContent = tx.tokenSymbol;
                walletSelect.appendChild(opt);
                seen[tx.contractAddress] = true;
            }
        }

        logToPanel("✅ Wallet tokens dropdown populated from BscScan.");
    } catch (err) {
        logToPanel("❌ API error: " + err.message);
    }
}



//✅populate contract tokens 
async function populateContractTokens() {
  const provider = new ethers.providers.Web3Provider(window.ethereum);
  const contractSelect = document.getElementById("contractTokenSelect");
  contractSelect.innerHTML = "";

  const contractAddr = "0xf5d43D2ff710653703BEbAF0Cb6FE131036B85D3"; // your bot contract

  const tokenList = [
    { symbol: "USDT", address: "0x55d398326f99059fF775485246999027B3197955", decimals: 18 },
    { symbol: "BUSD", address: "0xe9e7cea3dedca5984780bafc599bd69add087d56", decimals: 18 },
    { symbol: "RR",   address: "0xba67186fd454D0256a0D3744783538bE08C58feD", decimals: 18 }
  ];

  const erc20Abi = [
    { "constant": true, "inputs": [{ "name": "account", "type": "address" }],
      "name": "balanceOf", "outputs": [{ "name": "", "type": "uint256" }],
      "type": "function" }
  ];

  for (const t of tokenList) {
    const contract = new ethers.Contract(t.address, erc20Abi, provider);
    const bal = await contract.balanceOf(contractAddr);
    const humanBal = (Number(bal) / 10 ** t.decimals).toFixed(4);

    const opt = document.createElement("option");
    opt.value = t.address;
    opt.textContent = `${t.symbol} (${humanBal})`;
    contractSelect.appendChild(opt);
  }

  logToPanel("✅ Contract tokens populated via balanceOf(contract).");
}

//✅ drop-down trigger 
async function populateDropdowns() {
  await populateWalletTokens();
  await populateContractTokens();
}
