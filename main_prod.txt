// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;
window.logview = logToPanel;

logToPanel("main_prod.txt loaded — minimal mode");

// ✅ Query params parser
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

// ✅ ConnectWallet button
document.getElementById("connectwalletBtn")?.addEventListener("click", async () => {
  try {
    if (!window.ethereum) {
      logToPanel("❌ No Ethereum provider found — install MetaMask.");
      return;
    }
    const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
    window.USER_ADDR = accounts[0];
    logToPanel(`✅ Wallet connected: ${window.USER_ADDR}`);
    populateDropdowns(); // populate dropdowns after connection
  } catch (err) {
    logToPanel(`[CONNECT ERROR] ${err.message}`);
  }
});

// ✅ Deposit button handler (SolidarityV11)
document.getElementById("depositBtn")?.addEventListener("click", async () => {
  const amt = parseFloat(document.getElementById("depositAmount")?.value || "0");
  if (amt <= 0) return logToPanel("❌ Invalid deposit amount.");

  const weiAmount = Math.floor(amt * 1e18);
  const tokenAddr = document.getElementById("walletTokenSelect").value;
  const contractAddr = "0xf5d43D2ff710653703BEbAF0Cb6FE131036B85D3"; // SolidarityV11

  try {
    const resp = await fetch(`${window.BASE_URL}/prodbot/deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        wallet: window.USER_ADDR,
        walletname: window.WALLET_NAME,
        token: tokenAddr,
        amount: weiAmount
      })
    });

    const data = await resp.json();
    logToPanel(`[DEBUG] Deposit response: ${JSON.stringify(data)}`);

    if (data.status !== "ok") {
      return logToPanel(`❌ Deposit error: ${data.message || "Deposit failed."}`);
    }

    async function sendEncodedTx(to, dataHex) {
      const txParams = {
        from: window.USER_ADDR,
        to: to,
        data: dataHex,
        value: "0x0"
      };
      const txHash = await window.ethereum.request({
        method: "eth_sendTransaction",
        params: [txParams],
      });
      logToPanel(`✅ Transaction sent — TxHash=${txHash}`);
    }

    logToPanel(`✅ Deposit prepared successfully`);

    // Step 1: Commission transfer
    await sendEncodedTx(contractAddr, data.commission_call);

    // Step 2: Actual deposit (updates _userTradingBalances)
    await sendEncodedTx(contractAddr, data.deposit_call);

  } catch (err) {
    logToPanel(`[DEPOSIT ERROR] ${err.message}`);
  }
});

// ✅ Withdraw button handler (SolidarityV11)
async function runSwap(userAddr, bnbAmount, tokenAddress) {
  try {
    const res = await fetch(`${window.BASE_URL}/prodbot/swap`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        user_address: userAddr,
        bnb_amount: bnbAmount,
        target_token: tokenAddress,
        wbnb_address: window.WBNB_TOKEN_ADDRESS,
        router: "0x10ED43C718714eb63d5aA57B78B54704E256024E" // PancakeSwap V2 Router
      })
    });

    const data = await res.json();
    if (data.status === "success") {
      logToPanel(`✅ Swap executed successfully. TxHash=${data.tx_hash}`);
      return data;
    } else {
      logToPanel(`❌ Swap failed: ${data.message}`);
      return data;
    }
  } catch (err) {
    logToPanel(`[SWAP ERROR] ${err.message}`);
    return { status: "error", message: err.message };
  }
}
