// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.value += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;   // expose globally
window.logview = logToPanel;      // alias

// ✅ Status label updater
window.updateStatus = function (state, message) {
  const label = document.getElementById("txStatusLabel");
  if (!label) return;

  const states = {
    pending: { color: "#FFD700", icon: "⏳", text: "Processing..." },
    success: { color: "#00FF00", icon: "✅", text: "Completed successfully!" },
    error:   { color: "#FF4C4C", icon: "❌", text: "An error occurred." },
    default: { color: "#00FF00", icon: "", text: "Waiting for input..." }
  };

  const s = states[state] || states.default;
  label.style.color = s.color;
  label.textContent = `Status: ${s.icon} ${message || s.text}`;
};

// ✅ Query params
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    if (window.parent && window.parent.location) {
      const params = new URLSearchParams(window.parent.location.search);
      return {
        userid: params.get("userid"),
        wallet: params.get("wallet"),
        walletname: params.get("walletname")
      };
    }
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";
window.BASE_URL = "https://roton.store";

logToPanel(`✅ User context loaded: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, WALLET=${window.WALLET_NAME}`);

// ✅ Wallet connection helper
async function ensureWalletConnected() {
  if (!window.USER_ADDR) {
    switch (window.WALLET_NAME) {
      case "MetaMask":
        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
        window.USER_ADDR = accounts[0];
        logToPanel("✅ MetaMask connection approved.");
        break;
      case "Binance":
        window.web3 = new Web3(new Web3.providers.HttpProvider("https://bsc-dataseed1.binance.org"));
        logToPanel("✅ Binance Wallet RPC connected.");
        break;
      default:
        window.web3 = new Web3(new Web3.providers.HttpProvider("https://bsc-dataseed.binance.org"));
        logToPanel(`⚠️ Fallback RPC used for wallet: ${window.WALLET_NAME}`);
    }
  }
}

// ✅ Balance sync helper
async function syncBalanceFromFund() {
  try {
    const response = await fetch(`${window.BASE_URL}/prodbot/balances`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ wallet_address: window.USER_ADDR, walletname: window.WALLET_NAME })
    });
    const text = await response.text();
    let data;
    try {
      data = JSON.parse(text);
    } catch {
      logToPanel(`[ERROR] Invalid JSON: ${text.slice(0, 100)}...`);
      return;
    }
    if (data.status === "success") {
      const balanceBNB = parseFloat(data.balance);
      document.getElementById("startBalance").value = balanceBNB;
      logToPanel(`✅ Balance synced successfully: ${balanceBNB} BNB`);
    } else {
      logToPanel(`❌ Sync failed: ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[SYNC ERROR] ${err.message}`);
  }
}

// ✅ Deposit
document.getElementById("depositBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();
  const amt = parseFloat(document.getElementById("depositAmount")?.value || "0");
  if (amt <= 0) return logToPanel("❌ Invalid deposit amount.");
  try {
    updateStatus("pending", `Requesting deposit of ${amt} BNB...`);
    const txHash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [{ from: window.USER_ADDR, to: window.FUND_ADDRESS, value: window.web3.utils.toWei(amt.toString(), "ether") }]
    });
    updateStatus("success", `Deposit submitted. Tx: ${txHash}`);
    await syncBalanceFromFund();
  } catch (err) {
    updateStatus("error", err.message);
  }
});

// ✅ Withdraw
document.getElementById("withdrawBtn")?.addEventListener("click", async () => {

  if (amt <= 0) return logToPanel("❌ Invalid withdrawal amount.");

    const txHash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [{
        from: window.USER_ADDR,
        to: window.FUND_ADDRESS,
        data: fund_contract.methods.withdrawUserCapital(window.web3.utils.toWei(amt.toString(), "ether")).encodeABI()
      }]
    });
    updateStatus("success", `Withdrawal submitted. Tx: ${txHash}`);
    await syncBalanceFromFund();
  } catch (err) {
    updateStatus("error", err.message);
  }
});

// ✅ Transaction submit
document.getElementById("submitTxBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();
  const txId = document.getElementById("transactionId")?.value.trim();
  if (!txId) return logToPanel("❌ Please enter a valid transaction ID.");
  try {
    updateStatus("pending", `Mapping deposit tx id ${txId}...`);
    const response = await fetch(`${window.BASE_URL}/prodbot/deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ tx_id: txId, user_address: window.USER_ADDR, walletname: window.WALLET_NAME })
    });
    const text = await response.text();
    let data;
    try {
      data = JSON.parse(text);
    } catch {
      logToPanel(`[ERROR] Invalid JSON: ${text.slice(0, 100)}...`);
      return;
    }
    if (data.status === "success") {
      updateStatus("success", `Deposit mapped successfully. Tx: ${data.tx_hash}`);
      await syncBalanceFromFund();
    } else {
      updateStatus("error", data.message);
    }
  } catch (err) {
    updateStatus("error", err.message);
  }
});

// ✅ Connect Wallet button click
document.getElementById("connectwalletBtn")?.addEventListener("click", async () => {
  try {
    logToPanel("🔗 ConnectWallet button clicked — initializing wallet connection...");
    await ensureWalletConnected();

    if (window.USER_ADDR) {
      logToPanel(`✅ Wallet connected: ${window.USER_ADDR}`);
      updateStatus("success", "Wallet connected successfully!");
    } else {
      logToPanel("❌ Wallet connection failed or cancelled.");
      updateStatus("error", "Wallet connection failed.");
    }
  } catch (err) {
    logToPanel(`[CONNECT ERROR] ${err.message}`);
    updateStatus("error", err.message);
  }
});
window.logToPanel = logToPanel;
