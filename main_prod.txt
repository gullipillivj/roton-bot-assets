// ✅ Global logger
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;
window.logview = logToPanel;

logToPanel("main_prod.txt loaded — minimal mode");

// ✅ Query params parser
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";
window.BASE_URL = "https://roton.store";

logToPanel(`✅ User context loaded: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, WALLET=${window.WALLET_NAME}`);

// ✅ Populate dropdowns (show all tokens from list)
async function populateDropdowns() {
    if (!window.ethereum || !window.USER_ADDR) {
        logToPanel("❌ Wallet not connected.");
        return;
    }

    const provider = new ethers.providers.Web3Provider(window.ethereum);
    const walletSelect = document.getElementById("walletTokenSelect");
    const contractSelect = document.getElementById("contractTokenSelect");
    walletSelect.innerHTML = "";
    contractSelect.innerHTML = "";

    // Token list (include all you want to expose)
    const tokenList = [
        { symbol: "BNB",  address: "native", decimals: 18 }, // native marker
        { symbol: "ETH",  address: "native", decimals: 18 }, // native marker
        { symbol: "USDT", address: "0x55d398326f99059fF775485246999027B3197955", decimals: 18 },
        { symbol: "BUSD", address: "0xe9e7cea3dedca5984780bafc599bd69add087d56", decimals: 18 },
        { symbol: "RR",   address: "0xba67186fd454D0256a0D3744783538bE08C58feD", decimals: 18 }
    ];

    const erc20Abi = [{
        "constant": true,
        "inputs": [{ "name": "account", "type": "address" }],
        "name": "balanceOf",
        "outputs": [{ "name": "", "type": "uint256" }],
        "type": "function"
    }];

    for (const t of tokenList) {
        try {
            let humanBal = "native";
            if (t.address !== "native") {
                const contract = new ethers.Contract(t.address, erc20Abi, provider);
                const bal = await contract.balanceOf(window.USER_ADDR);
                humanBal = (Number(bal) / 10 ** t.decimals).toFixed(4);
            }

            const opt = document.createElement("option");
            opt.value = t.address;
            opt.textContent = `${t.symbol} (${humanBal})`;
            walletSelect.appendChild(opt);

            const opt2 = document.createElement("option");
            opt2.value = t.address;
            opt2.textContent = `${t.symbol} (${humanBal})`;
            contractSelect.appendChild(opt2);
        } catch (err) {
            logToPanel(`❌ Error fetching ${t.symbol} balance: ${err.message}`);
        }
    }

    // ✅ Send userid + wallet to backend for tracking/logging
    try {
        await fetch(`${window.BASE_URL}/prodbot/balance?userid=${window.USER_ID}&wallet=${window.USER_ADDR}`, {
            method: "POST"
        });
        logToPanel(`✅ User context sent: USER_ID=${window.USER_ID}, WALLET=${window.USER_ADDR}`);
    } catch (err) {
        logToPanel("❌ Error sending user context: " + err.message);
    }

    logToPanel("✅ Token dropdowns populated (all tokens shown).");
}

// ✅ ConnectWallet button
document.getElementById("connectwalletBtn")?.addEventListener("click", async () => {
  try {
    if (!window.ethereum) {
      logToPanel("❌ No Ethereum provider found — install MetaMask.");
      return;
    }
    const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
    window.USER_ADDR = accounts[0];
    logToPanel(`✅ Wallet connected: ${window.USER_ADDR}`);
    populateDropdowns(); // populate dropdowns after connection
  } catch (err) {
    logToPanel(`[CONNECT ERROR] ${err.message}`);
  }
});
