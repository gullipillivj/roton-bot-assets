logToPanel("main_prod.txt loaded — Client Sync Mode");

// Parse login parameters from URL
function getQueryParams() {
  const params = new URLSearchParams(window.location.search);
  return {
    userid: params.get("userid"),
    wallet: params.get("wallet"),
    walletname: params.get("walletname")
  };
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";

logToPanel(`✅ User context loaded: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, WALLET=${window.WALLET_NAME}`);

// Base configuration
window.BASE_URL = "https://roton.store";

// Fetch balance from contract and update clients.db
(async function initBalance() {
  if (!window.USER_ADDR || !window.USER_ID) {
    logToPanel("❌ Wallet or User ID not set — skipping balance fetch.");
    return;
  }

  try {
    // Step 1: Get balance from contract
    const response = await fetch(`${window.BASE_URL}/prodbot/balances`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ wallet_address: window.USER_ADDR })
    });

    const data = await response.json();
    if (data.status === "success") {
      const balanceBNB = parseFloat(data.balance);

      // Step 2: Update clients.db via backend
      await fetch(`${window.BASE_URL}/prodbot/update_client`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ userid: window.USER_ID, new_balance: balanceBNB })
      });

      // Step 3: Update UI textbox
      document.getElementById("startBalance").value = balanceBNB;
      logToPanel(`✅ Balance synced for ${window.USER_ID}: ${balanceBNB} BNB`);
    } else {
      logToPanel(`❌ Backend error: ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[ERROR] ${err.message}`);
  }
})();
