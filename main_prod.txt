function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

window.API_BASE = window.location.origin; 
window.WBNB_TOKEN_ADDRESS = "0xBB4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

function getQueryParams() {
  try {
    const url = new URL(window.location.href);
    const params = url.searchParams;
    return {
      userid: params.get("userid") || "unknown",
      wallet: params.get("wallet") || null,
      walletname: params.get("walletname") || "unknown"
    };
  } catch (err) {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid;
window.USER_ADDR = q.wallet;
window.WALLET_NAME = q.walletname;

function initializeLabels() {
  const userLabel = document.getElementById("currentUserLabel");
  const walletLabel = document.getElementById("currentWalletLabel");
  const nameLabel = document.getElementById("currentWalletName");

  if (userLabel) userLabel.innerText = "Current User: " + window.USER_ID;
  if (walletLabel) walletLabel.innerText = "Wallet Address: " + window.USER_ADDR;
  if (nameLabel) nameLabel.innerText = "Wallet Name: " + window.WALLET_NAME;
}

// 📊 RAW PLAIN-TEXT LEDGER BALANCE SYNCHRONIZER WITH DATABASE TOKEN SLOT MATCHING
async function syncBalanceFromFund() {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        if (!window.USER_ADDR || !window.USER_ID) {
            logToPanel("⚠️ Identification vector missing. Halting balance initialization cycles.");
            return null;
        }

        logToPanel(`🔎 Querying clients.db tokens map tracking for User ID: ${window.USER_ID}...`);
        
        // 🟢 STEP 1: Query database tokens table record mapping for current active session user
        let targetTokenAddr = "0x55d398326f99059fF775485246999027B3197955"; // Strict safe standard default
        try {
            const tokenDbRes = await fetch(`${baseApiUrl}/prodbot/tokens/get`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ userid: window.USER_ID })
            });
            const dbRecord = await tokenDbRes.json();
            if (dbRecord && dbRecord.status === "ok" && dbRecord.currenttoken) {
                targetTokenAddr = dbRecord.currenttoken;
                logToPanel(`✅ Database Active Reference verified: ${targetTokenAddr}`);
            } else {
                logToPanel(`💡 No custom layout found in database. Relying on baseline default asset values.`);
            }
        } catch (dbErr) {
            console.error("Database structural tokens check failed:", dbErr);
        }

        // 🟢 STEP 2: Enforce non-BNB constraint verification rules mapping
        const cleanTokenStr = targetTokenAddr.toLowerCase();
        if (cleanTokenStr === "bnb" || cleanTokenStr === window.WBNB_TOKEN_ADDRESS.toLowerCase() || cleanTokenStr === "0x0000000000000000000000000000000000000000") {
            logToPanel("⚠️ Token target matches gas layers. Diverting to alternate custom non-BNB assets...");
            targetTokenAddr = "0xcF6BB5389c92Bdda8a3747Ddb454cB7a64E86902"; // Redirect straight to Venus layer tracking
        }

        logToPanel(`⏳ Pinging live balance endpoint for wallet: ${window.USER_ADDR} (Asset: ${targetTokenAddr})`);
        const res = await fetch(baseApiUrl + "/prodbot/balance", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                user: window.USER_ADDR,
                token: targetTokenAddr
            })
        });

        const rawWeiText = await res.text();
        let cleanWeiString = rawWeiText.trim() || "0";

        // 🟢 STEP 3: Fallback Verification Check
        if (BigInt(cleanWeiString) <= 0n && targetTokenAddr.toLowerCase() !== "0x55d398326f99059ff775485246999027b3197955") {
            targetTokenAddr = "0x55d398326f99059fF775485246999027B3197955";
            logToPanel(`🔎 Target empty. Checking alternative portfolio layer: USDT...`);
            const fallbackRes = await fetch(baseApiUrl + "/prodbot/balance", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ user: window.USER_ADDR, token: targetTokenAddr })
            });
            const fallbackRawText = await fallbackRes.text();
            cleanWeiString = fallbackRawText.trim() || "0";
        }

        logToPanel("[LEDGER SYNC] Balance loaded successfully: " + cleanWeiString);
        window.SolidarityFundBalance = parseFloat(cleanWeiString);

        // 🟢 STEP 4: Update Frontend Fields (Writing initial balance changes cleanly)
        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");
        const usdBox = document.getElementById("usd");

        // Convert the balance from Uint256 Wei to Decimal Float Formats (Assumes standard 18 decimal scale tokens)
        const parsedBalanceEth = parseFloat(cleanWeiString) / 1e18;

        if (startBox) { 
            startBox.type = "text"; 
            startBox.value = cleanWeiString; // Populates with the base raw integer balance strings
        }
        if (investBox) { 
            investBox.type = "text"; 
            investBox.value = cleanWeiString; // Overwrites target initial balance parameter inputs 
        }
        if (usdBox) { 
            usdBox.value = parsedBalanceEth.toFixed(4); // Formatted value mapping
        }

        return window.SolidarityFundBalance;
    } catch (err) {
        logToPanel("❌ Balance Sync Error: " + err.message);
        return null;
    }
}
window.syncBalanceFromFund = syncBalanceFromFund;


// SECURE SYNCHRONOUS LIFECYCLE CONTROLLER
function initDashboardPipeline() {
  initializeLabels();
  
  if (window.USER_ADDR && window.USER_ADDR !== "{wallet}" && window.USER_ADDR !== "null") {
      logToPanel("🔄 Auto-fetching account metrics on page load...");
      syncBalanceFromFund();
  } else {
      logToPanel("💡 System Ready: Please connect your Web3 Wallet to initialize account state maps.");
  }
}

initDashboardPipeline();

window.activeProvider = window.ethereum;

const connBtn = document.getElementById("connectwalletBtn");
if (connBtn) {
  connBtn.onclick = async function() {
    if (typeof window.ethereum !== "undefined") {
      try {
        logToPanel("⏳ Connecting local extension provider...");
        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
        window.USER_ADDR = String(accounts[0] || accounts).toLowerCase(); 
        window.activeProvider = window.ethereum;

        const label = document.getElementById("currentWalletLabel");
        if (label) label.innerText = window.USER_ADDR;
        logToPanel("✅ Connected: " + window.USER_ADDR);
        await syncBalanceFromFund();

        // Triggers background clock updates instantly upon wallet connection success
        if (typeof window.initLiveBalanceClock === "function") {
            window.initLiveBalanceClock();
        }
        
      } catch (err) {
        logToPanel("❌ Connection error: " + err.message);
      }
    } else {
      logToPanel("❌ No wallet provider injected. Please use a Web3 enabled browser.");
    }
  };
}

function getSelectedToken() {
  const el = document.getElementById("tokenSelect");
  if (!el) return "USDT";
  const val = el.value.toLowerCase();
  if (val === "0x55d398326f99059ff775485246999027b3197955" || val === "usdt") return "USDT";
  if (val === "bnb") return "BNB";
  if (val === "eth") return "ETH";
  return "USDT";
}
