// main_prod.txt — Production Web3 Control & UI Wiring
logToPanel("main_prod.txt loaded (Production Event Wiring)");

// === Contract Initialization (USDT + Fund) ===
window.FUND_ADDRESS = "0x8Af76972C8dA792b9286f6FF869645F57e9F191c";
const USDT_TOKEN_ADDRESS = "0x55d398326f99059fF775485246999027B3197955"; // BSC USDT

const ERC20_ABI = [
  { "constant": true, "inputs": [{ "name": "account", "type": "address" }],
    "name": "balanceOf", "outputs": [{ "name": "", "type": "uint256" }], "type": "function" },
  { "constant": false, "inputs": [{ "name": "spender", "type": "address" },{ "name": "amount", "type": "uint256" }],
    "name": "approve", "outputs": [{ "name": "", "type": "bool" }], "type": "function" },
  { "constant": false, "inputs": [{ "name": "sender", "type": "address" },{ "name": "recipient", "type": "address" },{ "name": "amount", "type": "uint256" }],
    "name": "transferFrom", "outputs": [{ "name": "", "type": "bool" }], "type": "function" }
];

window.USDT_CONTRACT = new web3.eth.Contract(ERC20_ABI, USDT_TOKEN_ADDRESS);
logToPanel("✅ Contract initialization complete — Fund + USDT wired");

// === Global State ===
window.cumulativePnL = 0;
window.isBotRunning = false;

// === Utilities ===
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.value += msg + "\n";
        panel.scrollTop = panel.scrollHeight;
    } else { console.log(msg); }
}
window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) statusLabel.textContent = "STATE: " + text;
    else logToPanel(`STATE: ${text}`);
};

// === Balance Synchronizer (fixed BN parsing) ===
async function syncBalanceFromFund() {
    try {
        if (!window.USDT_CONTRACT || !window.FUND_ADDRESS) {
            logToPanel("[SYNC] Missing contract or fund address; skipping sync");
            return null;
        }
        const raw = await window.USDT_CONTRACT.methods.balanceOf(window.FUND_ADDRESS).call();
        const balanceFloat = Number(web3.utils.toBN(raw).toString()) / 1e6; // ✅ USDT decimals

        logToPanel(`[BALANCE SYNC] On-chain Fund Balance: ${balanceFloat.toFixed(2)} USDT`);

        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");
        if (startBox) startBox.value = balanceFloat.toFixed(2);
        if (investBox) investBox.value = balanceFloat.toFixed(2);

        window.SolidarityFundBalance = balanceFloat;
        return balanceFloat;
    } catch (err) {
        logToPanel(`[SYNC ERROR] ${err.message}`);
        return null;
    }
}

// === UI Event Bindings ===
document.addEventListener("DOMContentLoaded", async () => {
    window.setStatus("READY TO TRADE");
    await syncBalanceFromFund();

    // Start Trading button
    document.getElementById("startBtn")?.addEventListener("click", async () => {
        if (window.isBotRunning) {
            logToPanel("[WARN] Trading loop already running.");
            return;
        }
        const cycleInput = document.getElementById("cycleCount");
        const executionCycles = cycleInput ? parseInt(cycleInput.value) || 10 : 10;
        const launchEngine = window.startSimulation;
        if (typeof launchEngine === "function") {
            window.setStatus("RUNNING");
            try { await launchEngine(executionCycles); }
            catch (err) { logToPanel(`[CRITICAL SIM ERROR] ${err.message}`); }
            finally { window.setStatus("COMPLETED"); }
        } else {
            logToPanel("[ERROR] simulate_prod.txt launch engine not detected.");
        }
    });

    // ✅ Keep contracts redirection for deposits
    document.getElementById("depositBtn")?.addEventListener("click", () => {
        const amt = parseFloat(document.getElementById("depositAmount")?.value || "0") || 0;
        if (!window.isBotRunning) {
            window.open("https://roton.store/contracts", "_blank", "width=800,height=600,resizable=yes");
        }
    });

    // Stop Trading button
    document.getElementById("stopBtn")?.addEventListener("click", () => {
        window.isBotRunning = false;
        window.setStatus("STOPPED");
        logToPanel("[COMMAND] Halt instruction received.");
    });
});

// Expose globals
window.syncBalanceFromFund = syncBalanceFromFund;
