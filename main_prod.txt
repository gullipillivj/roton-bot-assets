let USER_TOKEN_ADDR = null;

// === Helpers ===
function safeParse(text) { try { return JSON.parse(text); } catch { return null; } }
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) panel.value += msg + "\n";
  else console.log(msg);
}
window.setStatus = function(text) {
  const statusLabel = document.getElementById("statusLabel");
  if (statusLabel) statusLabel.textContent = "STATE: " + text;
  else logToPanel(`STATE: ${text}`);
};

function disableAllControlsExcept(exceptId) {
  document.querySelectorAll("button, input").forEach(el => {
    if (el.id !== exceptId) el.disabled = true;
  });
}
function disableAllControls() {
  document.querySelectorAll("button, input").forEach(el => el.disabled = true);
}
function enableAllControls() {
  document.querySelectorAll("button, input").forEach(el => el.disabled = false);
}

// === Fee Helpers ===
function applyDepositFees(amount) {
  const commission = (amount * 2) / 5000;
  const netAmount = amount - commission;
  logToPanel(`Deposit: Gross=${amount.toFixed(6)} | Commission=${commission.toFixed(6)} | FundCredited=${netAmount.toFixed(6)}`);
  return netAmount > 0 ? netAmount : 0;
}

function applyWithdrawFees(amount) {
  const gasFee = 0.001 * amount;
  const dexFee = 0.0025 * amount;
  const netAmount = amount - gasFee - dexFee;
  logToPanel(`Withdraw: Gross=${amount.toFixed(6)} | Gas=${gasFee.toFixed(6)} | DEXFee=${dexFee.toFixed(6)} | UserReceived=${netAmount.toFixed(6)}`);
  return netAmount > 0 ? netAmount : 0;
}

// === Web3 + Contract Initialization (single place) ===
let web3, fund, userToken, tokenDecimals = 18;

async function initWeb3AndContracts() {
  if (typeof window.ethereum === "undefined") {
    logToPanel("MetaMask not detected. Please install MetaMask.");
    return;
  }

  try {
    web3 = new Web3(window.ethereum);
    fund = new web3.eth.Contract(fundAbi, FUND_ADDRESS);

    if (!USER_TOKEN_ADDR) {
      logToPanel("USER_TOKEN_ADDR not set yet; token contract will be initialized after login.");
      return;
    }

    userToken = new web3.eth.Contract(tokenAbi, USER_TOKEN_ADDR);

    // fetch token decimals once and cache
    try {
      const d = await userToken.methods.decimals().call();
      tokenDecimals = parseInt(d, 10);
      logToPanel(`Token decimals detected: ${tokenDecimals}`);
    } catch (err) {
      logToPanel(`Decimals fetch failed, defaulting to ${tokenDecimals}: ${err.message}`);
    }
  } catch (err) {
    logToPanel(`Web3 init error: ${err.message}`);
  }
}

// Convert human amount (e.g., 1.5) to token units (BigNumber string)
function toTokenUnits(amount) {
  if (!web3) throw new Error("web3 not initialized");
  const factor = web3.utils.toBN(10).pow(web3.utils.toBN(tokenDecimals));
  const whole = Math.floor(amount);
  const frac = amount - whole;
  const wholeBN = web3.utils.toBN(whole).mul(factor);
  const fracBN = web3.utils.toBN(Math.round(frac * Math.pow(10, tokenDecimals)));
  return wholeBN.add(fracBN).toString();
}

// Convert token units (string/BN) to human float
function fromTokenUnits(amountStr) {
  try {
    const bn = web3.utils.toBN(amountStr);
    const factor = web3.utils.toBN(10).pow(web3.utils.toBN(tokenDecimals));
    const whole = bn.div(factor).toString();
    const rem = bn.mod(factor).toString().padStart(tokenDecimals, "0");
    // Limit precision to 6 decimals for UI
    const floatVal = parseFloat(`${whole}.${rem}`).toFixed(6);
    return parseFloat(floatVal);
  } catch {
    return 0;
  }
}

// === Clients DB / Login handling ===
async function handleLoginSuccess() {
  try {
    const resp = await fetch("/clients");
    const data = await resp.json();

    // data.clients is expected to be an array
    const clients = data.clients || [];
    const client = clients.find(c => {
      const name = (c.profile && c.profile.name) ? c.profile.name.trim() : "";
      return name === window.USER_ID;
    });

    if (!client) {
      logToPanel(`No client entry found for USER_ID=${window.USER_ID}`);
      return;
    }

    // Set wallet address and token address from clients.db
    window.USER_ADDR = client.wallet && client.wallet.address ? client.wallet.address : window.USER_ADDR;
    USER_TOKEN_ADDR = client.wallet && client.wallet.address ? client.wallet.address : USER_TOKEN_ADDR;

    // Balance from clients.db (may be null)
    const balanceRaw = client.wallet && (client.wallet.balance !== null && client.wallet.balance !== undefined)
      ? parseFloat(client.wallet.balance)
      : 0;

    const balance = isNaN(balanceRaw) ? 0 : balanceRaw;

    // Populate UI start/invest textboxes
    const startEl = document.getElementById("startBalance");
    const investEl = document.getElementById("investBalance");
    if (startEl) startEl.value = balance.toFixed(6);
    if (investEl) investEl.value = balance.toFixed(6);

    window.SolidarityFundBalance = balance;

    // Update current user label
    const userLabel = document.getElementById("currentUserLabel");
    if (userLabel) userLabel.textContent = window.USER_ID;

    logToPanel(`Login success: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, Balance=${balance.toFixed(6)}`);

    // Now initialize web3 contracts (token address is known)
    await initWeb3AndContracts();

    // Sync on-chain fund balance if token contract initialized
    await syncBalanceFromFund();
  } catch (err) {
    logToPanel(`handleLoginSuccess error: ${err.message}`);
  }
}

// === Balance Sync Helper ===
async function syncBalanceFromFund() {
  try {
    if (!userToken || !fund) {
      await initWeb3AndContracts();
      if (!userToken) {
        logToPanel("Token contract not initialized; skipping on-chain balance sync.");
        return;
      }
    }

    const raw = await userToken.methods.balanceOf(FUND_ADDRESS).call();
    const balanceFloat = fromTokenUnits(raw);

    logToPanel(`On-chain SolidarityFundBalance=${balanceFloat.toFixed(6)} USDT`);

    // Update UI
    const startEl = document.getElementById("startBalance");
    const investEl = document.getElementById("investBalance");
    if (startEl) startEl.value = balanceFloat.toFixed(6);
    if (investEl) investEl.value = balanceFloat.toFixed(6);
    window.SolidarityFundBalance = balanceFloat;

    // Mirror to clients.db
    await fetch("/prodbot/update_client", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userid: window.USER_ID, new_balance: balanceFloat })
    });
  } catch (err) {
    logToPanel(`SyncError=${err.message}`);
  }
}

// === Approve / Deposit / Withdraw Handlers ===
// NOTE: This code expects a Node-like environment where require() works for 'ethers' and local security module.
// If running purely in browser, adapt to bundler/imports accordingly.
const { ethers } = require("ethers");
const { decryptKey } = require("./security.txt");

// Use Infura mainnet RPC as requested
const INFURA_MAINNET_RPC = "https://mainnet.infura.io/v3/6dc48b3fc4424773ad710e6042e54b23";

async function runApprove(amount) {
  if (amount <= 0) return null;
  try {
    if (!userToken || !fund) await initWeb3AndContracts();

    // Decrypt private key securely
    const encryptedKey = process.env.USER_PK_ENCRYPTED; // stored in DB or env
    const privateKey = decryptKey(encryptedKey);
    if (!privateKey) throw new Error("Decrypted private key is empty");

    const provider = new ethers.JsonRpcProvider(INFURA_MAINNET_RPC);
    const userWallet = new ethers.Wallet(privateKey, provider);

    const amountUnits = toTokenUnits(amount);
    logToPanel(`[SECURE] Broadcasting approval for ${amount.toFixed(6)} USDT...`);

    const tokenContract = new ethers.Contract(USER_TOKEN_ADDR, tokenAbi, userWallet);
    const tx = await tokenContract.approve(FUND_ADDRESS, amountUnits);
    await tx.wait();

    logToPanel(`ApproveTx=${tx.hash}`);

    // Return wallet for reuse by deposit/withdraw
    return userWallet;
  } catch (err) {
    logToPanel(`ApproveError=${err.message}`);
    throw err;
  }
}

async function runDeposit(amount) {
  if (amount <= 0) return;
  const netAmount = applyDepositFees(amount);

  try {
    if (!fund || !userToken) await initWeb3AndContracts();

    // Approve and get wallet
    const userWallet = await runApprove(netAmount);
    if (!userWallet) return;

    // Deposit
    const amountUnits = toTokenUnits(netAmount);
    logToPanel(`[SECURE] Executing deposit for ${netAmount.toFixed(6)} USDT...`);

    const solidarityContract = new ethers.Contract(FUND_ADDRESS, fundAbi, userWallet);
    const tx = await solidarityContract.depositFunds(USER_TOKEN_ADDR, amountUnits);
    await tx.wait();

    logToPanel(`DepositTx=${tx.hash}`);

    // Purge PK from memory
    try { userWallet.privateKey = null; } catch {}

    // Update local UI balances (human units)
    const currentBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    const newBalance = currentBalance + netAmount;
    document.getElementById("investBalance").value = newBalance.toFixed(6);
    document.getElementById("startBalance").value = newBalance.toFixed(6);
    window.SolidarityFundBalance = newBalance;
    logToPanel(`SolidarityFundBalance=${newBalance.toFixed(6)} USDT`);

    // Mirror to clients.db (must be called after deposit)
    await fetch("/prodbot/update_client", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userid: window.USER_ID, new_balance: newBalance })
    });

    return tx;
  } catch (err) {
    logToPanel(`DepositError=${err.message}`);
    throw err;
  }
}

async function runWithdraw(amount) {
  if (amount <= 0) return;
  const netAmount = applyWithdrawFees(amount);

  try {
    if (!fund) await initWeb3AndContracts();

    // Approve and get wallet
    const userWallet = await runApprove(netAmount);
    if (!userWallet) return;

    const amountUnits = toTokenUnits(netAmount);
    logToPanel(`[SECURE] Executing withdrawal for ${netAmount.toFixed(6)} USDT...`);

    const solidarityContract = new ethers.Contract(FUND_ADDRESS, fundAbi, userWallet);
    const tx = await solidarityContract.executeSolidarityPayout(USER_TOKEN_ADDR, amountUnits);
    await tx.wait();

    logToPanel(`WithdrawTx=${tx.hash}`);

    // Purge PK from memory
    try { userWallet.privateKey = null; } catch {}

    // Update local UI balances
    const currentBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    const newBalance = currentBalance - netAmount;
    document.getElementById("investBalance").value = newBalance.toFixed(6);
    document.getElementById("startBalance").value = newBalance.toFixed(6);
    window.SolidarityFundBalance = newBalance;
    logToPanel(`Updated Local Balance=${newBalance.toFixed(6)} USDT`);

    // Mirror to clients.db (must be called after withdraw)
    await fetch("/prodbot/update_client", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userid: window.USER_ID, new_balance: newBalance })
    });

    return tx;
  } catch (err) {
    logToPanel(`WithdrawError=${err.message}`);
    throw err;
  }
}

// Alias for explicit payout (same as withdraw)
async function runPayout(amount) {
  return runWithdraw(amount);
}

// === UI Glue ===
document.addEventListener("DOMContentLoaded", async () => {
  window.setStatus("READY TO RUN");

  // Ensure USER_ID is present (injected by Flask). If not, try to fetch minimal info.
  if (!window.USER_ID || window.USER_ID === "undefined") {
    logToPanel("No USER_ID found on page. Please ensure Flask injects session userid into the template.");
  }

  // Populate user info from clients.db and initialize contracts
  await handleLoginSuccess();

  // Wire buttons with disable/enable and validations
  document.getElementById("approveBtn")?.addEventListener("click", async () => {
    const amt = parseFloat(document.getElementById("approveAmount")?.value || "0") || 0;
    disableAllControls();
    try { await runApprove(amt); } catch (e) { logToPanel(`ApproveBtnError=${e?.message || e}`); }
    enableAllControls();
  });

  document.getElementById("depositBtn")?.addEventListener("click", async () => {
    const amt = parseFloat(document.getElementById("depositAmount")?.value || "0") || 0;
    disableAllControls();
    try { await runDeposit(amt); } catch (e) { logToPanel(`DepositBtnError=${e?.message || e}`); }
    enableAllControls();
  });

  document.getElementById("withdrawBtn")?.addEventListener("click", async () => {
    const amt = parseFloat(document.getElementById("withdrawAmount")?.value || "0") || 0;
    disableAllControls();
    try { await runWithdraw(amt); } catch (e) { logToPanel(`WithdrawBtnError=${e?.message || e}`); }
    enableAllControls();
  });

  document.getElementById("startBtn")?.addEventListener("click", async () => {
    const startBalance = parseFloat(document.getElementById("startBalance")?.value || "0") || 0;
    const investBalance = parseFloat(document.getElementById("investBalance")?.value || "0") || 0;

    // Validation checks
    if (investBalance <= 0) {
      logToPanel("❌ Cannot start trading: Investment balance is 0.");
      return;
    }
    if (investBalance >= startBalance) {
      logToPanel("❌ Cannot start trading: Investment balance must be less than start balance.");
      return;
    }

    window.cumulativePnL = 0;
    disableAllControlsExcept("stopBtn");

    const launchEngine = window.startSimulation;
    if (typeof launchEngine === "function") {
      window.setStatus("RUNNING");
      try {
        await launchEngine(1); // run only 1 cycle
      } catch (err) {
        logToPanel(`SimulationError=${err.message}`);
      } finally {
        window.setStatus("COMPLETED");
        enableAllControls();
      }
    } else {
      logToPanel("No startSimulation function found on window.");
      enableAllControls();
    }
  });

  document.getElementById("stopBtn")?.addEventListener("click", () => {
    window.isBotRunning = false;
    // Purge any active PK references
    try { window.ACTIVE_PK = null; } catch {}
    window.setStatus("STOPPED");
    enableAllControls();
  });

  document.getElementById("logoutBtn")?.addEventListener("click", () => {
    window.location.href = `${window.BASE_URL}/logout`;
  });
});
