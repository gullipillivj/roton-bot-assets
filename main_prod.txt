// === Fee Helpers ===
function applyDepositFees(amount) {
    const commission = (amount * 2) / 5000;
    const netAmount = amount - commission;
    logToPanel(`Deposit: Gross=${amount.toFixed(2)} | Commission=${commission.toFixed(2)} | FundCredited=${netAmount.toFixed(2)}`);
    return netAmount > 0 ? netAmount : 0;
}

function applyWithdrawFees(amount) {
    const gasFee = 0.001 * amount;
    const pancakeFee = 0.0025 * amount;
    const netAmount = amount - gasFee - pancakeFee;
    logToPanel(`Withdraw: Gross=${amount.toFixed(2)} | Gas=${gasFee.toFixed(2)} | Pancake=${pancakeFee.toFixed(2)} | UserReceived=${netAmount.toFixed(2)}`);
    return netAmount > 0 ? netAmount : 0;
}

// Deposit handler
async function runDeposit(amount) {
  if (amount <= 0) return;
  const netAmount = applyDepositFees(amount);

  try {
    // Request MetaMask account
    const accounts = await ethereum.request({ method: 'eth_requestAccounts' });
    const from = accounts[0];

    // Step 1: Approve token spending
    logToPanel(`[WEB3] Broadcasting approval for ${netAmount.toFixed(2)} USDT...`);
    await userToken.methods.approve(fund.options.address, netAmount).send({ from });

    // Step 2: Execute deposit
    logToPanel(`[WEB3] Executing deposit for ${netAmount.toFixed(2)} USDT...`);
    const tx = await fund.methods.depositFunds(USER_TOKEN_ADDR, netAmount).send({ from });

    logToPanel(`DepositTx=${tx.transactionHash}`);

    // ✅ Update local balance
    const currentBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    const newBalance = currentBalance + netAmount;

    document.getElementById("startBalance").value = newBalance.toFixed(2);

}
