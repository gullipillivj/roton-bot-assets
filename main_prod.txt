function toTokenUnits(amount) {
  if (!web3) throw new Error("Web3 not initialized");
  const activeDecimals = (USER_TOKEN_ADDR.toLowerCase() === "0x55d398326f99059ff775485246999027b3197955") ? 18 : tokenDecimals;
  try {
    const amountStr = parseFloat(amount).toFixed(activeDecimals);
    const parts = amountStr.split(".");
    const whole = parts[0];
    const fraction = parts[1] || "";
    const wholeBN = web3.utils.toBN(whole).mul(web3.utils.toBN(10).pow(web3.utils.toBN(activeDecimals)));
    const fractionBN = web3.utils.toBN(fraction.padEnd(activeDecimals, "0").substring(0, activeDecimals));
    return wholeBN.add(fractionBN).toString();
  } catch (e) {
    return Math.floor(parseFloat(amount) * (10 ** activeDecimals)).toString();
  }
}

function fromTokenUnits(amountStr) {
  try {
    const activeDecimals = (USER_TOKEN_ADDR.toLowerCase() === "0x55d398326f99059ff775485246999027b3197955") ? 18 : tokenDecimals;
    const stringRaw = amountStr.toString();
    const bn = web3.utils.toBN(stringRaw);
    const factor = web3.utils.toBN(10).pow(web3.utils.toBN(activeDecimals));
    const whole = bn.div(factor).toString();
    const rem = bn.mod(factor).toString().padStart(activeDecimals, "0");
    return parseFloat(whole + "." + rem);
  } catch(e) { return 0; }
}

let provider = null;

// Initialize Web3 + Contracts using AppKit provider
async function initWeb3AndContracts() {
  try {
    provider = appKit.getProvider();
    web3 = new Web3(provider);
    logToPanel("🌐 WalletConnect AppKit provider initialized.");

    fund = new web3.eth.Contract(fundAbi, FUND_ADDRESS);
    if (USER_TOKEN_ADDR) {
      userToken = new web3.eth.Contract(tokenAbi, USER_TOKEN_ADDR);
    }
  } catch (err) {
    logToPanel("❌ Contract Config Error: " + err.message);
  }
}

// Universal wallet connection (works for MetaMask, Trust, Coinbase, etc.)
async function connectWallet() {
  try {
    const session = await appKit.connect();
    if (session && session.accounts && session.accounts.length > 0) {
      const activeAccount = session.accounts[0];
      window.USER_ADDR = activeAccount;
      logToPanel("✅ Wallet Connected: " + activeAccount);

      const btn = document.getElementById("connectWalletBtn");
      if (btn) {
        btn.textContent = "Connected: " + activeAccount.substring(0, 6) + "..." + activeAccount.slice(-4);
        btn.style.backgroundColor = "#28a745";
        btn.style.color = "#ffffff";
        btn.disabled = true;
      }

      await initWeb3AndContracts();
      await syncBalanceFromFund();
    }
  } catch (err) {
    logToPanel("❌ WalletConnect Error: " + err.message);
  }
}




async function handleLoginSuccess() {
  try {
    const resp = await fetch("/clients");
    if (resp.ok) {
      const data = await resp.json();
      const clients = data.clients || [];
      const client = clients.find(c => (c.profile && c.profile.name ? c.profile.name.trim() : "") === window.USER_ID);
      if (client && client.wallet) {
        window.USER_ADDR = client.wallet.address || window.USER_ADDR;
        USER_TOKEN_ADDR = client.wallet.token_address || USER_TOKEN_ADDR;
        logToPanel("✅ Session user layout map synced.");
      }
    }
  } catch (err) { logToPanel("Sync notice: Base metadata processed."); }

  await initWeb3AndContracts();
  logToPanel("🚀 Interface ready. Click 'Connect Wallet' to activate blockchain functions.");
}

async function syncBalanceFromFund() {
  try {
    if (!web3 || !userToken) return;
    const accounts = await web3.eth.getAccounts();
    const caller = accounts.length > 0 ? accounts[0] : FUND_ADDRESS;
    const raw = await userToken.methods.balanceOf(FUND_ADDRESS).call({ from: caller });
    const balanceFloat = fromTokenUnits(raw);
    logToPanel("🔄 Synced SolidarityFundBalance=" + balanceFloat.toFixed(6) + " USDT");
    if (document.getElementById("startBalance")) document.getElementById("startBalance").value = balanceFloat.toFixed(6);
    if (document.getElementById("investBalance")) document.getElementById("investBalance").value = balanceFloat.toFixed(6);
    window.SolidarityFundBalance = balanceFloat;
  } catch (err) { logToPanel("SyncError=" + err.message); }
}

async function verifyActiveWalletMatches() {
  if (!window.ethereum) { logToPanel("❌ No Web3 wallet extension found."); return false; }
  if (!web3) { logToPanel("⚠️ Web3 not initialized. Please click 'Connect Wallet' first."); return false; }
  
  const accounts = await web3.eth.getAccounts();
  if (!accounts || accounts.length === 0) { logToPanel("⚠️ Wallet locked. Unlock your MetaMask app connection."); return false; }
  
  const activeAddress = accounts[0].toString().toLowerCase();

  if (!window.USER_ADDR || window.USER_ADDR.trim() === "") {
    window.USER_ADDR = activeAddress;
    logToPanel("🌐 Session Address auto-assigned to active wallet: " + activeAddress);
  }

  const registeredAddress = window.USER_ADDR.toString().toLowerCase();
  if (activeAddress !== registeredAddress) {
    logToPanel("❌ WALLET MISMATCH! Active: " + activeAddress + ", Expected: " + registeredAddress);
    alert("Please switch MetaMask to your registered profile wallet:\n" + window.USER_ADDR);
    return false;
  }
  return true;
}

async function runAction(amount, toAddress, type) {
  try {
    if (!window.ethereum || !web3) throw new Error("Wallet provider missing.");
    const accounts = await web3.eth.getAccounts();
    if (!accounts || accounts.length === 0) throw new Error("No connected accounts found.");
    const userWalletAddress = accounts[0].toString();
    const units = toTokenUnits(amount);
    let receipts = [];

    if (type === "approve") {
      logToPanel("Prompting BEP20 Allowance Approval via MetaMask...");
      const approveReceipt = await userToken.methods.approve(toAddress, units).send({ from: userWalletAddress });
      receipts.push({ tx_hash: approveReceipt.transactionHash.toString(), status: approveReceipt.status });
    } else if (type === "deposit") {
      logToPanel("Prompting Contract depositFunds() via MetaMask...");
      const depositReceipt = await fund.methods.depositFunds(USER_TOKEN_ADDR, units).send({ from: userWalletAddress });
      receipts.push({ tx_hash: depositReceipt.transactionHash.toString(), status: depositReceipt.status });
    } else if (type === "payout") {
      logToPanel("Prompting Contract swapToDonation() via MetaMask...");
      // Maps directly to your Solidity swapToDonation interface function call name
      const payoutReceipt = await fund.methods.swapToDonation(USER_TOKEN_ADDR, units, "0").send({ from: userWalletAddress });
      receipts.push({ tx_hash: payoutReceipt.transactionHash.toString(), status: payoutReceipt.status });
    }
    return { type: type, results: receipts };
  } catch (err) { 
    return { error: err.message }; 
  }
}

async function runDeposit(amount) {
  if (!amount || amount <= 0) return logToPanel("❌ Invalid deposit amount");
  const matches = await verifyActiveWalletMatches();
  if (!matches) return;

  const commission = (amount * 2) / 5000;
  const netAmount = amount - commission;
  
  try {
    logToPanel("Initiating Allowance Approval request...");
    const approveNet = await runAction(netAmount, FUND_ADDRESS, "approve");
    if (!approveNet || approveNet.error) return logToPanel("❌ Wallet Approval Failed: " + (approveNet ? approveNet.error : "Failed"));

    logToPanel("Initiating Deposit request...");
    const depositNet = await runAction(netAmount, FUND_ADDRESS, "deposit");
    if (!depositNet || depositNet.error) return logToPanel("❌ Wallet Deposit Failed: " + (depositNet ? depositNet.error : "Failed"));

    if (depositNet.results && depositNet.results.tx_hash) {
      logToPanel("✅ On-Chain Deposit Complete! Tx Hash: " + depositNet.results.tx_hash);
    }
    await syncBalanceFromFund();
    await broadcastSync("deposit_sync");
  } catch (err) { logToPanel("DepositError=" + err.message); }
}

async function runPayout(amount) {
  if (!amount || amount <= 0) return logToPanel("❌ Invalid payout amount");
  const matches = await verifyActiveWalletMatches();
  if (!matches) return;
  
  try {
    logToPanel("Initiating Swap Transaction request...");
    const payoutResult = await runAction(amount, FUND_ADDRESS, "payout");
    if (!payoutResult || payoutResult.error) return logToPanel("❌ Swap Failed: " + (payoutResult ? payoutResult.error : "Failed"));
    
    if (payoutResult.results && payoutResult.results.tx_hash) {
      logToPanel("✅ On-Chain Swap Complete! Tx Hash: " + payoutResult.results.tx_hash);
    }
    await syncBalanceFromFund();
    await broadcastSync("payout_sync");
  } catch (err) { logToPanel("PayoutError=" + err.message); }
}

async function broadcastSync(actionName) {
  try {
    await fetch("/prodbot/update_client", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ userid: window.USER_ID, wallet: window.USER_ADDR, action: actionName })
    });
  } catch (e) {}
}
