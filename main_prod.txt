// ✅ RoTon Bot Frontend Controller (V12 Production-Ready)
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

logToPanel("main_prod.txt loaded successfully");

// ✅ Core Context Variable Management
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";
const API_BASE = window.location.origin;

window.WBNB_TOKEN_ADDRESS = "0xBB4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

function getSelectedToken() {
  const el = document.getElementById("walletTokenSelect") || document.getElementById("contractTokenSelect") || document.getElementById("tokenSelect");
  return el ? el.value : "USDT"; 
}

// ✅ Balanced Fund Synchronization Endpoint Bridge
async function syncBalanceFromFund() {
  if (!window.USER_ADDR) return 1000; // testing placeholder
  try {
    const res = await fetch(`${API_BASE}/prodbot/balance`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ user: String(window.USER_ADDR), token: "0x55d398326f99059fF775485246999027B3197955" })
    });
    const data = await res.json();
    if (data.status === "ok") {
      window.SolidarityFundBalance = parseFloat(data.balance) || 1000; 
      return window.SolidarityFundBalance;
    }
  } catch (err) {
    console.log(`Balance sync notice: ${err.message}`);
  }
  return 1000; 
}
window.syncBalanceFromFund = syncBalanceFromFund;

// 1. Connect Wallet
const connBtn = document.getElementById("connectwalletBtn");
if (connBtn) {
  connBtn.onclick = async function() {
    if (typeof window.ethereum !== "undefined") {
      try {
        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
        // CRITICAL FIX: Extract clean index string out of array
        window.USER_ADDR = String(accounts[0]).toLowerCase(); 
        
        const label = document.getElementById("currentWalletLabel");
        if (label) label.innerText = window.USER_ADDR;
        logToPanel(`✅ Wallet linked: ${window.USER_ADDR}`);
      } catch (err) {
        logToPanel(`❌ Connection error: ${err.message}`);
      }
    } else {
      logToPanel("❌ MetaMask provider not detected.");
    }
  };
}

// 2. Deposit Execution Routing
// Overwrite the deposit handling chunk in your main_prod.txt file:

const depBtn = document.getElementById("depositBtn");
if (depBtn) {
  depBtn.onclick = async function() {
    const token = getSelectedToken();
    const rawAmt = parseFloat(document.getElementById("depositAmount")?.value || "0");
    
    if (!window.USER_ADDR) {
      return logToPanel("❌ Action Halted: Connect your wallet first.");
    }
    if (rawAmt <= 0) {
      return logToPanel("❌ Action Halted: Specify a positive deposit amount value.");
    }

    // Convert input unit metric up to standard contract wei scale (18 decimals)
    const amount = BigInt(Math.floor(rawAmt * 1e18)).toString();
    logToPanel(`⏳ Communicating with backend server pipeline for ${token} deposit configurations...`);

    try {
      let endpoint = `/prodbot/deposit/usdt/approve`; // Baseline default
      if (token === "BNB") endpoint = `/prodbot/deposit/bnb`;
      if (token === "ETH") endpoint = `/prodbot/deposit/eth`;

      const res = await fetch(`${API_BASE}${endpoint}`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ wallet: String(window.USER_ADDR), amount: amount })
      });
      const data = await res.json();

      if (data.status !== "ok") {
        return logToPanel(`❌ Server side processing error: ${data.message || "Unknown exception"}`);
      }

      logToPanel("🚀 Backend compiled data structure successfully. Directing payload request to MetaMask extension...");

      // 🔑 CRITICAL MISSING LINK: Extract hex data string out of the response mapping
      // If it's a multi-step call, look for the primary structural array element
      const transactionPayloadData = data.approve_call || data.deposit_call || data.bnb_refuel_call;
      
      // Determine target destination contract address
      // V12 Trading Contract is the target for USDT interactions; V13 handles raw gas token reserves
      const contractDestinationAddress = (token === "USDT") 
        ? "0x1854170353ea89672638F9fa5acB477c45523871" 
        : "0xc46E6452c1Cb9641dE339eb8bE48faEb70B58dF5";

      if (!transactionPayloadData) {
        return logToPanel("❌ Payload extraction issue: hex transaction string undefined.");
      }

      // ⚡ TRIGGER METAMASK EXTENSION FOR SIGNATURE POPUP WINDOW
      const txHash = await window.ethereum.request({
        method: "eth_sendTransaction",
        params: [{
          from: String(window.USER_ADDR),
          to: contractDestinationAddress,
          data: transactionPayloadData,
          // If sending raw native BNB directly to contract receive pipeline function:
          value: (token === "BNB") ? "0x" + BigInt(amount).toString(16) : "0x0"
        }]
      });

      logToPanel(`🎉 On-chain transaction broadcasted successfully! Tracking TxHash: ${txHash}`);
      logToPanel("⏳ Awaiting network mining confirmation block cycle updates...");

    } catch (err) {
      logToPanel(`❌ Execution failure or wallet signature rejected: ${err.message}`);
    }
  };
}


// 3. Withdrawal Interface Handler
const wthBtn = document.getElementById("withdrawBtn");
if (wthBtn) {
  wthBtn.onclick = async function() {
    if (!window.USER_ADDR) return logToPanel("❌ Link wallet container first.");
    const rawAmount = parseFloat(document.getElementById("withdrawAmount")?.value || "0");
    if (rawAmount <= 0) return logToPanel("❌ Specify valid asset units.");

    try {
      const res = await fetch(`${API_BASE}/prodbot/withdraw`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ 
          userid: window.USER_ID, 
          wallet: String(window.USER_ADDR), 
          token: getSelectedToken(), 
          amount: rawAmount * 1e18 
        })
      });
      const data = await res.json();
      logToPanel(`📥 Withdrawal sequence initiated: ${JSON.stringify(data)}`);
    } catch (err) {
      logToPanel(`❌ Withdrawal tracking connection lost: ${err.message}`);
    }
  };
}
