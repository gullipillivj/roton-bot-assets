logToPanel("main_prod.txt loaded — Initial Balance Fetch Mode");

document.addEventListener("DOMContentLoaded", async () => {
    try {
        if (typeof window.ethereum === "undefined") {
            document.getElementById("depisitmsg").textContent = "❌ MetaMask not detected.";
            return;
        }

        // Initialize Web3
        window.web3 = new Web3(window.ethereum);
        await window.ethereum.request({ method: "eth_requestAccounts" });
        const accounts = await window.ethereum.request({ method: "eth_accounts" });
        const userWallet = accounts[0];
        window.USER_ADDR = userWallet;

        // Contract ABI (only balance function)
        const MASTER_ABI = [
            {
                "inputs": [{"internalType": "address", "name": "user", "type": "address"}],
                "name": "checkUserBalance",
                "outputs": [{"internalType": "uint256", "name": "", "type": "uint256"}],
                "stateMutability": "view",
                "type": "function"
            }
        ];

        // Contract instance
        const contract = new window.web3.eth.Contract(MASTER_ABI, window.FUND_ADDRESS);

        // Fetch balance
        const balanceWei = await contract.methods.checkUserBalance(userWallet).call();
        const balanceBNB = window.web3.utils.fromWei(balanceWei, "ether");

        // Update UI
        document.getElementById("startBalance").value = balanceBNB;
        document.getElementById("investBalance").value = balanceBNB;
        document.getElementById("depisitmsg").textContent = `✅ Current contract balance: ${balanceBNB} BNB`;

        logToPanel(`✅ Balance fetched successfully for ${userWallet}: ${balanceBNB} BNB`);
    } catch (err) {
        document.getElementById("depisitmsg").textContent = `❌ Error fetching balance: ${err.message}`;
        logToPanel(`[ERROR] ${err.message}`);
    }
});
