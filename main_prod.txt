function ensure18Decimals(amountInput) {
    if (!amountInput) return "0";
    const str = amountInput.toString().trim();
    if (str === "0" || str === "") return "0";
    if (str.length > 12) return str;
    
    if (!str.includes('.')) {
        return str + "000000000000000000";
    }
    
    const parts = str.split('.');
    const whole = parts[0];
    let fraction = parts[1];
    
    if (fraction.length < 18) {
        fraction = fraction.padEnd(18, '0');
    } else if (fraction.length > 18) {
        fraction = fraction.substring(0, 18);
    }
    
    const combined = (whole + fraction).replace(/^0+/, '');
    return combined === "" ? "0" : combined;
}

async function runDepositPipeline(userAddr, amountHumanOrWei) {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        const totalInputWei = amountHumanOrWei.toString().length > 12 
            ? BigInt(amountHumanOrWei.toString()) 
            : BigInt(ensure18Decimals(amountHumanOrWei));
            
        window.logToPanel(`[DEPOSIT START] Initial Input: ${(Number(totalInputWei) / 1e18).toFixed(4)} USDT`);
        
        const fixedGasDeductionWei = 30000000000000000n; // Fixed 0.03 USDT
        const v12ContractAddress = "0x1854170353ea89672638F9fa5acB477c45523871";
        const usdtTokenAddress = "0x55d398326f99059fF775485246999027B3197955";

        if (totalInputWei <= fixedGasDeductionWei) {
            throw new Error("Input amount is insufficient to cover gas allocation thresholds.");
        }
        
        const remainingPoolBalanceWei = totalInputWei - fixedGasDeductionWei;
        window.logToPanel(`[STAGE 1/3] Prompting wallet to authorize token spend allowance...`);
        
        const cleanSpender = v12ContractAddress.toLowerCase().replace("0x", "").padStart(64, "0");
        const cleanValue = BigInt(totalInputWei).toString(16).padStart(64, "0"); 
        const approveCalldata = "0x095ea7b3" + cleanSpender + cleanValue;
        
        const txHash1 = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: userAddr, to: usdtTokenAddress, data: approveCalldata, value: '0x0' }],
        });
        await new Promise(resolve => setTimeout(resolve, 5000)); 

        // 🌟 STEP 2/3: Core deposit pipeline (commissionPaid = false)
        const step2TxAmountWei = (remainingPoolBalanceWei * 4n) / 100n;
        const res = await fetch(`${baseApiUrl}/prodbot/deposit/usdt/pipeline`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ amount: step2TxAmountWei.toString(), tradingPhase: false })
        });
        const data = await res.json();
        if (!data || data.status !== "ok") throw new Error(data.message || "Failed API payload.");
        
        const txHash2 = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: userAddr, to: v12ContractAddress, data: data.txdata, value: '0x0' }],
        });
        await new Promise(resolve => setTimeout(resolve, 5000));

        // 🌟 STEP 3/3: Remaining token allocation tracking (commissionPaid = true)
        const remainderChunkWei = remainingPoolBalanceWei - step2TxAmountWei;
        if (remainderChunkWei <= fixedGasDeductionWei) {
            throw new Error("Pool capital depleted below execution bounds.");
        }
        const step3TxAmountWei = remainderChunkWei - fixedGasDeductionWei;
        
        const secondRes = await fetch(`${baseApiUrl}/prodbot/deposit/usdt/pipeline`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ amount: step3TxAmountWei.toString(), tradingPhase: true })
        });
        const secondData = await secondRes.json();
        
        const txHash3 = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: userAddr, to: v12ContractAddress, data: secondData.txdata, value: '0x0' }],
        });
        
        window.logToPanel(`🎉 Cascading Pipeline Fully Settled! Master Final Hash: ${txHash3}`);
        if (typeof window.syncBalanceFromFund === "function") await window.syncBalanceFromFund();
        return { status: "ok", tx_hash: txHash3 };
        
    } catch (err) {
        window.logToPanel(`❌ Pipeline Break: ${err.message}`);
        return { status: "error", message: err.message };
    }
}
window.runDepositPipeline = runDepositPipeline;

async function syncBalanceFromFund() {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        if (!window.USER_ADDR) {
            logToPanel("⚠️ Wallet address vector missing. Aborting tracking loop.");
            return null;
        }

        logToPanel(`🔎 Inspecting decentralized tuple mappings for: ${window.USER_ADDR}...`);
        
        // Autoritative system constants
        let resolvedTokenAddress = "0x55d398326f99059fF775485246999027B3197955"; // Default fallback (USDT)
        let finalWeiString = "0";

        // 🟢 PHASE 1: Query the user's token arrays via the getUserBalanceList registry endpoint
        try {
            const ledgerRes = await fetch(`${baseApiUrl}/prodbot/ledger/balances`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ user: window.USER_ADDR })
            });
            const ledgerData = await ledgerRes.json();
            
            if (ledgerData && ledgerData.status === "ok" && Array.isArray(ledgerData.tokens) && Array.isArray(ledgerData.balances)) {
                logToPanel(`📊 Scan complete. Found ${ledgerData.tokens.length} potential asset coordinates in contract mapping memory.`);
                
                // Explicit sequential loop execution through available contract storage indexes
                for (let i = 0; i < ledgerData.tokens.length; i++) {
                    const currentTokenInLoop = String(ledgerData.tokens[i]).toLowerCase().trim();
                    const currentBalanceWeiString = String(ledgerData.balances[i]).trim();
                    
                    try {
                        const numericBalanceValue = BigInt(currentBalanceWeiString);
                        if (numericBalanceValue > 0n) {
                            resolvedTokenAddress = ledgerData.tokens[i];
                            finalWeiString = currentBalanceWeiString;
                            logToPanel(`✨ High yield matching coordinate located: [Token: ${resolvedTokenAddress}] with Balance: ${finalWeiString}`);
                            break; // Break execution at the first active balance holding pool index slot
                        }
                    } catch (innerBigIntErr) {
                        console.error("Failed to parse BigInt array element tracking strings:", innerBigIntErr);
                    }
                }
                
                // Secondary recovery mapping sequence if all index array balances are zero
                if (finalWeiString === "0" && ledgerData.tokens.length > 0) {
                    resolvedTokenAddress = ledgerData.tokens[0];
                }
            }
        } catch (ledgerErr) {
            console.error("Ledger router lookup bypassed or dropped:", ledgerErr);
        }

        // 🟢 PHASE 2: Force secondary storage state lookup cross-check boundary validation
        try {
            const balanceCheckRes = await fetch(`${baseApiUrl}/prodbot/balance`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({ user: window.USER_ADDR, token: resolvedTokenAddress })
            });
            const freshBalText = await balanceCheckRes.text();
            if (freshBalText && freshBalText.trim() !== "" && !isNaN(freshBalText.trim())) {
                finalWeiString = freshBalText.trim();
            }
        } catch (balErr) {
            console.error("Direct ledger individual balance cross-check failed:", balErr);
        }

        logToPanel(`[LEDGER SYNC] Process complete. Active Target Address: [${resolvedTokenAddress}] | Live Balance: ${finalWeiString}`);
        window.SolidarityFundBalance = parseFloat(finalWeiString);

        // 🟢 PHASE 3: Write extracted contract values directly to browser control element form boxes
        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");
        const usdBox = document.getElementById("usd");

        if (startBox) startBox.value = finalWeiString;
        if (investBox) investBox.value = finalWeiString;
        
        // 🟢 PHASE 4: Execute token decimals normalization and format market values into USD controls
        if (usdBox) {
            try {
                const marketRes = await fetch(`${baseApiUrl}/prodbot/market-analysis`);
                const marketData = await marketRes.json();
                
                const activeTickerList = [
                    ...(marketData.data?.short_term || marketData.short_term || []),
                    ...(marketData.data?.mid_term || marketData.mid_term || []),
                    ...(marketData.data?.long_term || marketData.long_term || [])
                ];

                const lowerAddress = String(resolvedTokenAddress).toLowerCase().trim();
                const matchedToken = activeTickerList.find(t => t.address && String(t.address).toLowerCase().trim() === lowerAddress);

                if (matchedToken && matchedToken.price) {
                    // Override rule for alternative token types like BabyDoge (9 decimals)
                    let decimals = lowerAddress === "0xc748673057861a797275cd8a068abb95a902e8de" ? 9 : 18;
                    const tokenCount = parseFloat(finalWeiString) / Math.pow(10, decimals);
                    const trueUsdValuation = tokenCount * parseFloat(matchedToken.price);
                    
                    usdBox.value = trueUsdValuation.toFixed(4);
                    logToPanel(`📈 Live Token Ticker Synced: 1 ${matchedToken.symbol} = $${matchedToken.price}. Valuation updated.`);
                } else {
                    // Standard 1:1 conversion ratio fallback for stable asset metrics like standard USDT tokens
                    usdBox.value = (parseFloat(finalWeiString) / 1e18).toFixed(4);
                }
            } catch (apiErr) {
                console.error("Failed to compile pricing data valuation updates:", apiErr);
                usdBox.value = (parseFloat(finalWeiString) / 1e18).toFixed(4);
            }
        }
        return window.SolidarityFundBalance;
    } catch (err) {
        logToPanel("❌ Synchronization breakdown exception: " + err.message);
        return null;
    }
}
window.syncBalanceFromFund = syncBalanceFromFund;

function setupActionListeners() {
  
  // Dynamic Capital Processing Router Pipeline
  document.getElementById("withdrawBtn")?.addEventListener("click", async () => {
    if (!window.USER_ADDR) return logToPanel("❌ Connect wallet first.");
    const inputAmount = document.getElementById("withdrawAmount").value;
    if (!inputAmount || parseFloat(inputAmount) <= 0) return logToPanel("❌ Specify a valid amount.");

    const amountWei = ensure18Decimals(inputAmount);
    logToPanel(`🔄 Packaging withdrawal data parameters for: ${amountWei} Wei...`);

    try {
      const res = await fetch(`${window.API_BASE}/prodbot/withdraw`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ token: "0x55d398326f99059fF775485246999027B3197955", amount: amountWei })
      });
      const data = await res.json();
      if (data && data.status === "ok") {
        const txHash = await window.ethereum.request({
            method: 'eth_sendTransaction',
            params: [{ from: window.USER_ADDR, to: "0x1854170353ea89672638F9fa5acB477c45523871", data: data.withdraw_call, value: '0x0' }],
        });
        logToPanel(`✅ Withdrawal Sent! Hash: ${txHash}`);
      }
    } catch (err) {
      logToPanel(`❌ Withdrawal Action Interrupted: ${err.message}`);
    }
  });
}
