function toTokenUnits(amount) {
  if (!web3) throw new Error("Web3 not initialized");
  const factor = web3.utils.toBN(10).pow(web3.utils.toBN(tokenDecimals));
  const whole = Math.floor(amount);
  const frac = amount - whole;
  const wholeBN = web3.utils.toBN(whole).mul(factor);
  const fracBN = web3.utils.toBN(Math.round(frac * Math.pow(10, tokenDecimals)));
  return wholeBN.add(fracBN).toString();
}
function fromTokenUnits(amountStr) {
  try {
    const bn = web3.utils.toBN(amountStr);
    const factor = web3.utils.toBN(10).pow(web3.utils.toBN(tokenDecimals));
    const whole = bn.div(factor).toString();
    const rem = bn.mod(factor).toString().padStart(tokenDecimals, "0");
    return parseFloat(`${whole}.${rem}`);
  } catch { return 0; }
}

// === Initialize Browser Wallet ===async function initWeb3AndContracts() {
  try {
    // Check if an injected wallet exists
    if (typeof window.ethereum !== 'undefined') {
      web3 = new Web3(window.ethereum);
      // Request account access if needed
      await window.ethereum.request({ method: 'eth_requestAccounts' });
    } else {
      logToPanel("❌ No Web3 Wallet detected. Falling back to Read-Only Mode via public RPC.");
      web3 = new Web3("https://binance.org");
    }

    fund = new web3.eth.Contract(fundAbi, FUND_ADDRESS);
    if (!USER_TOKEN_ADDR) return;
    userToken = new web3.eth.Contract(tokenAbi, USER_TOKEN_ADDR);
    const d = await userToken.methods.decimals().call();
    tokenDecimals = parseInt(d, 10);
    logToPanel(`Token decimals detected: ${tokenDecimals}`);
  } catch (err) { 
    logToPanel(`Contract init tracking error: ${err.message}`); 
  }
}

async function syncBalanceFromFund() {
  try {
    if (!userToken || !fund) await initWeb3AndContracts();
    const raw = await userToken.methods.balanceOf(FUND_ADDRESS).call();
    const balanceFloat = fromTokenUnits(raw);
    logToPanel(`On-chain SolidarityFundBalance=${balanceFloat.toFixed(6)} USDT`);
    if (document.getElementById("startBalance")) document.getElementById("startBalance").value = balanceFloat.toFixed(6);
    if (document.getElementById("investBalance")) document.getElementById("investBalance").value = balanceFloat.toFixed(6);
    window.SolidarityFundBalance = balanceFloat;
  } catch (err) { logToPanel(`SyncError=${err.message}`); }
}
async function verifyActiveWalletMatches() {
  if (!window.ethereum) {
    logToPanel("❌ No wallet extension found. Please open your Web3 wallet app.");
    return false;
  }
  const accounts = await web3.eth.getAccounts();
  if (accounts.length === 0) {
    logToPanel("⚠️ Wallet is locked. Please log into your wallet app first.");
    return false;
  }
  const activeAddress = accounts[0].toLowerCase();
  const registeredAddress = window.USER_ADDR.toLowerCase();
  if (activeAddress !== registeredAddress) {
    logToPanel(`❌ WALLET MISMATCH! Active: ${activeAddress}, Expected: ${registeredAddress}`);
    alert(`Switch to your registered wallet:\n${window.USER_ADDR}`);
    return false;
  }
  return true;
}
// === Updated Transaction Router (Uses Wallet Extension) ===async function runAction(amount, toAddress, type) {
  try {
    if (!window.ethereum) throw new Error("Wallet extension not detected.");
    
    const accounts = await web3.eth.getAccounts();
    const userWalletAddress = accounts[0];
    const units = toTokenUnits(amount);
    let receipts = [];

    if (type === "approve") {
      logToPanel(`Setting BEP20 Allowance Spender via Extension...`);
      const approveTx = userToken.methods.approve(toAddress, units);
      const approveReceipt = await approveTx.send({ from: userWalletAddress });
      receipts.push({ tx_hash: approveReceipt.transactionHash, status: approveReceipt.status });
    } else if (type === "deposit") {
      logToPanel(`Invoking Smart Contract depositFunds() via Extension...`);
      const depositTx = fund.methods.depositFunds(USER_TOKEN_ADDR, units);
      const depositReceipt = await depositTx.send({ from: userWalletAddress });
      receipts.push({ tx_hash: depositReceipt.transactionHash, status: depositReceipt.status });
    } else if (type === "payout") {
      logToPanel(`Invoking Smart Contract executeSolidarityPayout() via Extension...`);
      // Maps to payout function: amountIn, inputToken, amountOutMin (set to 0 for demo slippage allowance)
      const payoutTx = fund.methods.executeSolidarityPayout(units, USER_TOKEN_ADDR, "0");
      const payoutReceipt = await payoutTx.send({ from: userWalletAddress });
      receipts.push({ tx_hash: payoutReceipt.transactionHash, status: payoutReceipt.status });
    }
    return { type, results: receipts };
  } catch (err) { 
    return { error: err.message }; 
  }
}
// === User Core Flow ===async function runDeposit(amount) {
  if (!amount || amount <= 0) return logToPanel("❌ Invalid deposit amount");
  
  const matches = await verifyActiveWalletMatches();
  if (!matches) return;

  const commission = (amount * 2) / 5000;
  const netAmount = amount - commission;
  
  try {
    logToPanel("Initiating Net Amount Wallet Approval request...");
    const approveNet = await runAction(netAmount, FUND_ADDRESS, "approve");
    if (!approveNet || approveNet.error) return logToPanel(`❌ Wallet Approval Failed: ${approveNet?.error}`);

    logToPanel("Initiating Deposit Wallet Transaction request...");
    const depositNet = await runAction(netAmount, FUND_ADDRESS, "deposit");
    if (!depositNet || depositNet.error) return logToPanel(`❌ Wallet Deposit Failed: ${depositNet?.error}`);

    if (depositNet.results && depositNet.results[0] && depositNet.results[0].tx_hash) {
      logToPanel(`✅ On-Chain Deposit Complete! Tx Hash: ${depositNet.results[0].tx_hash}`);
    }

    await syncBalanceFromFund();
    await broadcastSync("deposit_sync");
  } catch (err) { logToPanel(`DepositError=${err.message}`); }
}


async function runPayout(amount) {
if (!amount || amount <= 0) return logToPanel("❌ Invalid payout amount");
const matches = await verifyActiveWalletMatches();
if (!matches) return;
try {
logToPanel("Initiating Payout Transaction request...");
const payoutResult = await runAction(amount, FUND_ADDRESS, "payout");

if (payoutResult.results && payoutResult.results[0] && payoutResult.results[0].tx_hash) {
logToPanel(✅ On-Chain Payout Complete! Tx Hash: ${payoutResult.results[0].tx_hash});
}
await syncBalanceFromFund();
await broadcastSync("payout_sync");
} catch (err) { logToPanel(PayoutError=${err.message}); }
}
