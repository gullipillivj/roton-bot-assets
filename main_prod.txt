// ✅ Ensure logToPanel works inside iframe
window.addEventListener("DOMContentLoaded", () => {
  if (typeof window.parent.logToPanel === "function") {
    window.parent.logToPanel("main_prod.txt loaded — Client Sync Mode");
  } else if (typeof logToPanel === "function") {
    logToPanel("main_prod.txt loaded — Client Sync Mode");
  } else {
    console.log("⚠️ logToPanel not found in parent or local scope.");
  }
});

// ✅ Inline fallback logger — replaces old logger.txt
function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.value += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// Parse login parameters from URL
// ✅ Safe query extraction even inside iframe
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch (e) {
    // fallback: read from parent if iframe
    if (window.parent && window.parent.location) {
      const params = new URLSearchParams(window.parent.location.search);
      return {
        userid: params.get("userid"),
        wallet: params.get("wallet"),
        walletname: params.get("walletname")
      };
    }
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";

logToPanel(`✅ User context loaded: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, WALLET=${window.WALLET_NAME}`);


// Wallet connection helper — uses walletname from URL
async function ensureWalletConnected() {
  if (!window.USER_ADDR) {
    switch (window.WALLET_NAME) {
      case "MetaMask":
        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
        window.USER_ADDR = accounts[0];
        logToPanel("✅ MetaMask connection approved.");
        break;

      case "Binance":
        window.web3 = new Web3(new Web3.providers.HttpProvider("https://bsc-dataseed1.binance.org"));
        logToPanel("✅ Binance Wallet RPC connected.");
        break;

      case "Coinbase":
        window.web3 = new Web3(new Web3.providers.HttpProvider("https://mainnet.infura.io/v3/" + INFURA_KEY));
        logToPanel("✅ Coinbase Wallet RPC connected.");
        break;

      default:
        window.web3 = new Web3(new Web3.providers.HttpProvider("https://bsc-dataseed.binance.org"));
        logToPanel(`⚠️ Fallback RPC used for wallet: ${window.WALLET_NAME}`);
    }
  }
}

// Balance enquiry on page load
(async function initBalance() {
  if (!window.USER_ADDR || !window.USER_ID) {
    logToPanel("❌ Wallet or User ID not set — skipping balance fetch.");
    return;
  }

  try {
    const response = await fetch(`${window.BASE_URL}/prodbot/balances`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ wallet_address: window.USER_ADDR, walletname: window.WALLET_NAME })
    });

    const data = await response.json();
    if (data.status === "success") {
      const balanceBNB = parseFloat(data.balance);

      await fetch(`${window.BASE_URL}/prodbot/update_client`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ userid: window.USER_ID, new_balance: balanceBNB })
      });

      document.getElementById("startBalance").value = balanceBNB;
      logToPanel(`✅ Balance synced for ${window.USER_ID}: ${balanceBNB} BNB`);
    } else {
      logToPanel(`❌ Backend error: ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[ERROR] ${err.message}`);
  }
})();

// 1. Deposit function
document.getElementById("depositBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();
  const amt = parseFloat(document.getElementById("depositAmount")?.value || "0");
  if (amt <= 0) return logToPanel("❌ Invalid deposit amount.");

  try {
    logToPanel(`[RPC] Requesting deposit of ${amt} BNB from ${window.USER_ADDR}...`);

    const txHash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [{
        from: window.USER_ADDR,
        to: window.FUND_ADDRESS,
        value: window.web3.utils.toWei(amt.toString(), "ether")
      }]
    });

    logToPanel(`✅ Deposit submitted. Tx: ${txHash}`);
    await syncBalanceFromFund();
  } catch (err) {
    logToPanel(`[DEPOSIT ERROR] ${err.message}`);
  }
});

// 2. Submit button (tx id mapping)
document.getElementById("txSubmitBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();
  const txId = document.getElementById("txId")?.value.trim();
  if (!txId) return logToPanel("❌ Please enter a valid transaction ID.");

  try {
    logToPanel(`[BACKEND] Mapping deposit tx id ${txId} for ${window.USER_ADDR}...`);

    const response = await fetch(`${window.BASE_URL}/prodbot/deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ tx_id: txId, user_address: window.USER_ADDR, walletname: window.WALLET_NAME })
    });

    const data = await response.json();
    if (data.status === "success") {
      logToPanel(`✅ Deposit mapped successfully. Tx: ${data.tx_hash}`);
      await syncBalanceFromFund();
    } else {
      logToPanel(`❌ Deposit mapping failed: ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[SUBMIT ERROR] ${err.message}`);
  }
});

// 3. Withdraw function

    logToPanel(`[WITHDRAW ERROR] ${err.message}`);
  }
});

// Balance sync helper
async function syncBalanceFromFund() {
  try {
    const response = await fetch(`${window.BASE_URL}/prodbot/balances`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ wallet_address: window.USER_ADDR, walletname: window.WALLET_NAME })
    });
    const data = await response.json();
    if (data.status === "success") {
      const balanceBNB = parseFloat(data.balance);
      document.getElementById("startBalance").value = balanceBNB;
      logToPanel(`✅ Balance synced successfully: ${balanceBNB} BNB`);
    } else {
      logToPanel(`❌ Sync failed: ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[SYNC ERROR] ${err.message}`);
  }
}
