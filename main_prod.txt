    panel.value += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}

// ✅ Status label updater
window.updateStatus = function (state, message) {
  const label = document.getElementById("txStatusLabel");
  if (!label) return;

  const states = {
    pending: { color: "#FFD700", icon: "⏳", text: "Processing..." },
    success: { color: "#00FF00", icon: "✅", text: "Completed successfully!" },
    error:   { color: "#FF4C4C", icon: "❌", text: "An error occurred." },
    default: { color: "#00FF00", icon: "", text: "Waiting for input..." }
  };

  const s = states[state] || states.default;
  label.style.color = s.color;
  label.textContent = `Status: ${s.icon} ${message || s.text}`;
};

// ✅ Query params
function getQueryParams() {
  try {
    const params = new URLSearchParams(window.location.search);
    return {
      userid: params.get("userid"),
      wallet: params.get("wallet"),
      walletname: params.get("walletname")
    };
  } catch {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid || "unknown";
window.USER_ADDR = q.wallet || null;
window.WALLET_NAME = q.walletname || "unknown";
window.BASE_URL = "https://roton.store";

logToPanel(`✅ User context loaded: USER_ID=${window.USER_ID}, USER_ADDR=${window.USER_ADDR}, WALLET=${window.WALLET_NAME}`);

// ✅ Wallet connection helper (ethers.js)
async function ensureWalletConnected() {
  if (typeof window.ethereum === "undefined") {
    logToPanel("❌ No Ethereum provider found — please install MetaMask or another wallet.");
    updateStatus("error", "No wallet provider detected.");
    return;
  }
  const provider = new ethers.providers.Web3Provider(window.ethereum);
  await provider.send("eth_requestAccounts", []);
  const signer = provider.getSigner();
  window.USER_ADDR = await signer.getAddress();
  window.provider = provider;
  window.signer = signer;
  logToPanel(`✅ Wallet connected: ${window.USER_ADDR}`);
}

// ✅ Balance sync helper
async function syncBalanceFromFund() {
  try {
    const response = await fetch(`${window.BASE_URL}/prodbot/balances`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ wallet_address: window.USER_ADDR, walletname: window.WALLET_NAME })
    });
    const text = await response.text();
    let data;
    try { data = JSON.parse(text); } catch {
      logToPanel(`[ERROR] Invalid JSON: ${text.slice(0, 100)}...`);
      return;
    }
    if (data.status === "success") {
      const balanceBNB = parseFloat(data.balance);
      document.getElementById("startBalance").value = balanceBNB;
      logToPanel(`✅ Balance synced successfully: ${balanceBNB} BNB`);
    } else {
      logToPanel(`❌ Sync failed: ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[SYNC ERROR] ${err.message}`);
  }
}

// ✅ Deposit (backend + wallet approve)
document.getElementById("depositBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();

  const amt = parseFloat(document.getElementById("depositAmount")?.value || "0");
  if (amt <= 0) return logToPanel("❌ Invalid deposit amount.");

  try {
    updateStatus("pending", `Preparing deposit of ${amt} BNB...`);

    // 🔹 Step 1: Insert pending record in backend
    const recordResp = await fetch(`${window.BASE_URL}/prodbot/add_pending_deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        txid: "pending-" + Date.now(),
        amount: amt
      })
    });
    const recordData = await recordResp.json();
    if (recordData.status !== "success") {
      updateStatus("error", recordData.message || "Failed to log deposit.");
      return;
    }
    logToPanel(`🗂️ Pending deposit recorded: ${recordData.txid}`);

    // 🔹 Step 2: Approve and send deposit to contract
    const contract = new ethers.Contract(window.FUND_ADDRESS, fundAbi, window.signer);
    const approveTx = await contract.depositBNB({ value: ethers.utils.parseEther(amt.toString()) });
    await approveTx.wait();

    updateStatus("success", `Deposit confirmed on-chain. Tx: ${approveTx.hash}`);
    logToPanel(`✅ Deposit executed successfully — TxHash=${approveTx.hash}`);

    // 🔹 Step 3: Sync balance
    await syncBalanceFromFund();
  } catch (err) {
    updateStatus("error", err.message);
    logToPanel(`[DEPOSIT ERROR] ${err.message}`);
  }
});


// ✅ Withdraw (ethers.js)
document.getElementById("withdrawBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();
  const amt = parseFloat(document.getElementById("withdrawAmount")?.value || "0");
  if (amt <= 0) return logToPanel("❌ Invalid withdrawal amount.");
  try {
    updateStatus("pending", `Requesting withdrawal of ${amt} BNB...`);
    const contract = new ethers.Contract(window.FUND_ADDRESS, fundAbi, window.signer);
    const tx = await contract.withdrawUserCapital(ethers.utils.parseEther(amt.toString()));
    await tx.wait();
    updateStatus("success", `Withdrawal submitted. Tx: ${tx.hash}`);
    await syncBalanceFromFund();
  } catch (err) {
    updateStatus("error", err.message);
  }
});

// ✅ Transaction submit (ethers.js)
document.getElementById("submitTxBtn")?.addEventListener("click", async () => {
  await ensureWalletConnected();

  const txId = document.getElementById("transactionId")?.value.trim();
  if (!txId) return logToPanel("❌ Please enter a valid transaction ID.");

  try {
    updateStatus("pending", `Recording pending deposit for ${txId}...`);

    // Step 1: Insert pending record
    await fetch(`${window.BASE_URL}/prodbot/add_pending_deposit`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userid: window.USER_ID,
        txid: txId,
        amount: 0 // leave 0 until confirmed
      })
    });

    // Step 2: Execute depositBNB on contract
    const contract = new ethers.Contract(window.FUND_ADDRESS, fundAbi, window.signer);
    const depositTx = await contract.depositBNB({ value: ethers.utils.parseEther("0.00304") }); // replace with actual amount
    await depositTx.wait();
    logToPanel(`✅ depositBNB executed — TxHash=${depositTx.hash}`);

    // Step 3: Check balance via backend
    const balResp = await fetch(`${window.BASE_URL}/prodbot/balances`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ wallet_address: window.USER_ADDR })
    });
    const balData = await balResp.json();

    if (balData.status === "success" && balData.balance > 0) {
      // Step 4: Update transaction record to success
      await fetch(`${window.BASE_URL}/prodbot/update_transaction`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          txid: txId,
          status: "success",
          amount: balData.balance
        })
      });

      // Step 5: Update UI
      document.getElementById("startBalance").value = balData.balance;
      updateStatus("success", `Deposit confirmed: ${balData.balance} BNB`);
    } else {
      updateStatus("error", "DepositBNB failed or balance not updated.");
      logToPanel("❌ Balance check failed — transaction remains pending.");
    }
  } catch (err) {
    updateStatus("error", err.message);
    logToPanel(`[SUBMIT ERROR] ${err.message}`);
  }
});
