logToPanel("main_prod.txt loaded (Production Event Wiring)");

// === Global State Synchronization ===
window.cumulativePnL = 0;
window.isBotRunning = false;

// === Global Helper Utilities ===
function safeParse(text) { try { return JSON.parse(text); } catch { return null; } }

function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.value += msg + "\n";
        panel.scrollTop = panel.scrollHeight; // Auto-scroll logs
    } else {
        console.log(msg);
    }
}

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) statusLabel.textContent = "STATE: " + text;
    else logToPanel(`STATE: ${text}`);
};

// === Production Ledger Synchronizer ===
async function syncBalanceFromFund() {
    try {
        if (!window.USDT_CONTRACT || !window.FUND_ADDRESS) {
            logToPanel("[SYNC] Missing contract or fund address; skipping sync");
            return null;
        }

        // Query on-chain USDT balance of the SolidarityFund contract
        const raw = await window.USDT_CONTRACT.methods.balanceOf(window.FUND_ADDRESS).call();
        const balanceFloat = parseFloat(web3.utils.fromWei(raw, "ether"));

        logToPanel(`[BALANCE SYNC] On-chain Fund Balance: ${balanceFloat.toFixed(2)} USDT`);

        if (document.getElementById("startBalance")) {
            document.getElementById("startBalance").value = balanceFloat.toFixed(2);
        }
        if (document.getElementById("investBalance")) {
            document.getElementById("investBalance").value = balanceFloat.toFixed(2);
        }

        window.SolidarityFundBalance = balanceFloat;
        return balanceFloat;
    } catch (err) {
        logToPanel(`[SYNC ERROR] ${err.message}`);
        return null;
    }
}

// === Main UI Event Binding Interface ===
document.addEventListener("DOMContentLoaded", async () => {
    window.setStatus("READY TO TRADE");
    await syncBalanceFromFund();

    // Hook up real Start Button loop execution
    document.getElementById("startBtn")?.addEventListener("click", async () => {
        if (window.isBotRunning) {
            logToPanel("[WARN] Trading execution loop is already actively processing.");
            return;
        }

        // Dynamically parse cycles from the layout form input
        const cycleInput = document.getElementById("cycleCount");
        const executionCycles = cycleInput ? parseInt(cycleInput.value) || 10 : 10;

        const launchEngine = window.startSimulation;
        if (typeof launchEngine === "function") {
            window.setStatus("RUNNING");
            try {
                await launchEngine(executionCycles);
            } catch (err) {
                logToPanel(`[CRITICAL SIMULATION ERROR] ${err.message}`);
            } finally {
                window.setStatus("COMPLETED");
            }
        } else {
            logToPanel("[ERROR] The simulate_prod.txt launch engine core function was not detected.");
        }
    });

    //etherscan tx check 
    document.getElementById("txSubmitBtn")?.addEventListener("click", async () => {
  const txIdInput = document.getElementById("txId");
  const txHash = txIdInput ? txIdInput.value.trim() : "";

  if (!txHash) {
    logToPanel("[ERROR] Please enter a valid Transaction Hash before submitting.");
    return;
  }

  try {
    const response = await fetch("/prodbot/verify_tx", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ tx_hash: txHash })
    });
    const data = await response.json();

    if (data.status === "success") {
      logToPanel("[SUCCESS] Transaction confirmed successfully!");
      document.getElementById("statusLabel").textContent = "STATE: TX_CONFIRMED";
    } else if (data.status === "fail") {
      logToPanel("[ALERT] Transaction execution failed or reverted on-chain.");
      document.getElementById("statusLabel").textContent = "STATE: TX_REVERTED";
    } else if (data.status === "pending") {
      logToPanel("[PENDING] Transaction not mined yet.");
    } else {
      logToPanel(`[ERROR] ${data.message}`);
    }
  } catch (err) {
    logToPanel(`[API ERROR] Unable to reach backend: ${err.message}`);
  }
});



    //deposit button 
    document.getElementById("depositBtn")?.addEventListener("click", () => {
        const amt = parseFloat(document.getElementById("depositAmount")?.value || "0") || 0;
        if (!window.isBotRunning) {
        window.open("https://roton.store/contracts", "_blank", "width=800,height=600,resizable=yes");;
      }
    });


    // Stop execution capture hook
    document.getElementById("stopBtn")?.addEventListener("click", () => {
        window.isBotRunning = false;
        window.setStatus("STOPPED");
        logToPanel("[COMMAND] Immediate halt instruction received. Halting cycle updates...");
    });
});
