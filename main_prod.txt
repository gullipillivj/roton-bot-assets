logToPanel("main_prod.txt loaded (Production Event Wiring)");

if (typeof window.ethereum !== 'undefined') {
    window.web3 = new Web3(window.ethereum);
} else {
    window.web3 = new Web3(new Web3.providers.HttpProvider("https://binance.org"));
    logToPanel("[WARN] No Web3 provider detected; falling back to public BSC endpoint.");
}

// Authoritative Master Contract References
window.FUND_ADDRESS = "0x32f8759F23da6C4847c0DCD2446Af73eB0DE1A8D";
window.BOT_WALLET_ADDRESS = "0x07076BD62abB189ed37aC1f4ca7CD6a155B3F4a0"; // Account 2 (Bot Wallet) Derived from your verified tx records
window.WBNB_TOKEN_ADDRESS = "0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c"; // BSC Mainnet WBNB

const MASTER_POOL_ABI = [
  { "inputs": [{"internalType": "address", "name": "user", "type": "address"}], "name": "checkUserBalance", "outputs": [{"internalType": "uint256", "name": "", "type": "uint256"}], "stateMutability": "view", "type": "function" },
  { "inputs": [], "name": "depositBNB", "outputs": [], "stateMutability": "payable", "type": "function" },
  { "inputs": [{"internalType": "uint256", "name": "amount", "type": "uint256"}], "name": "withdrawUserCapital", "outputs": [], "stateMutability": "nonpayable", "type": "function" }
];

window.MASTER_CONTRACT = new web3.eth.Contract(MASTER_POOL_ABI, window.FUND_ADDRESS);
logToPanel("✅ Contract initialization complete — Master Pool wired");

window.cumulativePnL = 0;
window.isBotRunning = false;

function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.value += msg + "\n";
        panel.scrollTop = panel.scrollHeight;
    } else { console.log(msg); }
}

window.setStatus = function(text) {
    const statusLabel = document.getElementById("statusLabel");
    if (statusLabel) statusLabel.textContent = "STATE: " + text;
    else logToPanel(`STATE: ${text}`);
};

async function syncBalanceFromFund() {
    try {
        if (!window.MASTER_CONTRACT || !window.USER_ADDR) {
            logToPanel("[SYNC] Connection components missing; skipping synchronization check.");
            return null;
        }
        // Authoritative ledger fetch from your custom contract mapping matrix
        const raw = await window.MASTER_CONTRACT.methods.checkUserBalance(window.USER_ADDR).call();
        const balanceFloat = parseFloat(web3.utils.fromWei(raw, "ether"));

        logToPanel(`[BALANCE SYNC] Internal Contract Ledger Balance: ${balanceFloat.toFixed(6)} BNB`);

        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");
        if (startBox) startBox.value = balanceFloat.toFixed(6);
        if (investBox) investBox.value = balanceFloat.toFixed(6);

        window.SolidarityFundBalance = balanceFloat;
        return balanceFloat;
    } catch (err) {
        logToPanel(`[SYNC ERROR] ${err.message}`);
        return null;
    }
}

async function handleLoginSuccess() {
    initSessionFromLabels();
    if (window.USER_ADDR) {
        logToPanel(`[SESSION] User authenticated via wallet address: ${window.USER_ADDR}`);
        await syncBalanceFromFund();
    }
}

function initSessionFromLabels() {
    const currentUser = document.getElementById("currentUserLabel")?.textContent.trim();
    const currentWallet = document.getElementById("currentWalletLabel")?.textContent.trim();
    window.USER_ID = currentUser || window.USER_ID;
    window.USER_ADDR = currentWallet || window.USER_ADDR;
}

document.addEventListener("DOMContentLoaded", async () => {
    window.setStatus("READY TO TRADE");
    await handleLoginSuccess();

    document.getElementById("startBtn")?.addEventListener("click", async () => {
        if (window.isBotRunning) {
            logToPanel("[WARN] Trading engine loop is already active.");
            return;
        }
        const cycleInput = document.getElementById("cycleCount");
        const executionCycles = cycleInput ? parseInt(cycleInput.value) || 10 : 10;
        
        if (typeof window.startSimulation === "function") {
            window.setStatus("RUNNING");
            try { 
                await window.startSimulation(executionCycles); 
            } catch (err) { 
                logToPanel(`[CRITICAL SIM ERROR] ${err.message}`); 
            } finally { 
                window.setStatus("COMPLETED"); 
            }
        } else {
            logToPanel("[ERROR] Loop simulation engine script files not located.");
        }
    });

    document.getElementById("depositBtn")?.addEventListener("click", async () => {
        const amt = parseFloat(document.getElementById("depositAmount")?.value || "0") || 0;
        if (amt <= 0) return logToPanel("❌ Please provide a valid deposit allocation size.");
        
        try {
            logToPanel(`[METAMASK] Requesting authorization to deposit ${amt} BNB into your isolated ledger...`);
            const amtWei = web3.utils.toWei(amt.toString(), "ether");
            
            await window.MASTER_CONTRACT.methods.depositBNB().send({
                from: window.USER_ADDR,
                value: amtWei
            });
            logToPanel("✅ On-Chain Deposit Processed Successfully!");
            await syncBalanceFromFund();
        } catch (e) {
            logToPanel(`[DEPOSIT ERROR] ${e.message}`);
        }
    });

    document.getElementById("withdrawBtn")?.addEventListener("click", async () => {
        const amt = parseFloat(document.getElementById("withdrawAmount")?.value || "0") || 0;
        if (amt <= 0) return logToPanel("❌ Please provide a valid withdrawal payout amount.");
        
        try {
            logToPanel(`[METAMASK] Requesting withdrawal execution for ${amt} BNB...`);
            const amtWei = web3.utils.toWei(amt.toString(), "ether");
            
            await window.MASTER_CONTRACT.methods.withdrawUserCapital(amtWei).send({
                from: window.USER_ADDR
            });
            logToPanel("✅ Payout Extraction Completed successfully!");
            await syncBalanceFromFund();
        } catch (e) {
            logToPanel(`[WITHDRAWAL ERROR] ${e.message}`);
        }
    });

    document.getElementById("stopBtn")?.addEventListener("click", () => {
        window.isBotRunning = false;
        window.setStatus("STOPPED");
        logToPanel("[COMMAND] Immediate halt directive processed.");
    });
});

window.syncBalanceFromFund = syncBalanceFromFund;
window.handleLoginSuccess = handleLoginSuccess;
