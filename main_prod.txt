function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

window.API_BASE = window.location.origin; 
window.WBNB_TOKEN_ADDRESS = "0xBB4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

function getQueryParams() {
  try {
    const url = new URL(window.location.href);
    const params = url.searchParams;
    return {
      userid: params.get("userid") || "unknown",
      wallet: params.get("wallet") || null,
      walletname: params.get("walletname") || "unknown"
    };
  } catch (err) {
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

const q = getQueryParams();
window.USER_ID = q.userid;
window.USER_ADDR = q.wallet;
window.WALLET_NAME = q.walletname;

function initializeLabels() {
  const userLabel = document.getElementById("currentUserLabel");
  const walletLabel = document.getElementById("currentWalletLabel");
  const nameLabel = document.getElementById("currentWalletName");

  if (userLabel) userLabel.innerText = "Current User: " + window.USER_ID;
  if (walletLabel) walletLabel.innerText = "Wallet Address: " + window.USER_ADDR;
  if (nameLabel) nameLabel.innerText = "Wallet Name: " + window.WALLET_NAME;
}

function getSelectedToken() {
  const el = document.getElementById("tokenSelect");
  if (!el) return "USDT";
  const val = el.value.toLowerCase();
  if (val === "0x55d398326f99059ff775485246999027b3197955" || val === "usdt") return "USDT";
  if (val === "bnb") return "BNB";
  if (val === "eth") return "ETH";
  return "USDT";
}

async function syncBalanceFromFund() {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        if (!window.USER_ADDR || window.USER_ADDR === "{wallet}" || window.USER_ADDR === "null") {
            return null;
        }
        
        logToPanel(`⏳ Pinging balance endpoint for wallet: ${window.USER_ADDR}`);
        const res = await fetch(`${baseApiUrl}/prodbot/balance`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                user: window.USER_ADDR,
                token: "0x55d398326f99059fF775485246999027B3197955" // USDT Address
            })
        });

        const rawWeiText = await res.text();
        const cleanWeiString = rawWeiText.trim() || "0";
        logToPanel(`[LEDGER SYNC] Received raw Wei balance string from contract: ${cleanWeiString}`);

        window.SolidarityFundBalance = parseFloat(cleanWeiString);

        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");

        if (startBox) {
            startBox.type = "text";
            startBox.value = cleanWeiString;
        }
        if (investBox) {
            investBox.type = "text";
            investBox.value = cleanWeiString;
        }
        return window.SolidarityFundBalance;
    } catch (err) {
        logToPanel(`❌ Balance Sync Error: ${err.message}`);
        return null;
    }
}
window.syncBalanceFromFund = syncBalanceFromFund;

async function initDashboardPipeline() {
  initializeLabels();
  if (window.USER_ADDR && window.USER_ADDR !== "{wallet}" && window.USER_ADDR !== "null") {
      logToPanel("🔄 Auto-fetching account metrics on page load...");
      await syncBalanceFromFund();
  } else {
      logToPanel("💡 System Ready: Please connect your Web3 Wallet to initialize account state maps.");
  }
}
initDashboardPipeline();
