function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

logToPanel("main_prod.txt loaded successfully");

function getQueryParams() {
  try {
    // Ensure full URL parsing even under nested paths like /prodbot
    const url = new URL(window.location.href);
    const params = url.searchParams;

    const userid = params.get("userid") || "unknown";
    const wallet = params.get("wallet") || null;
    const walletname = params.get("walletname") || "unknown";

    // Log for debugging
    logToPanel(`🔍 URL parsed — userid=${userid}, wallet=${wallet}, walletname=${walletname}`);

    return { userid, wallet, walletname };
  } catch (err) {
    logToPanel(`❌ URL parsing error: ${err.message}`);
    return { userid: "unknown", wallet: null, walletname: "unknown" };
  }
}

// ✅ FIXED: Maps correctly to your specific HTML select element values
function getSelectedToken() {
  const el = document.getElementById("tokenSelect");
  if (!el) return "USDT";
  
  const val = el.value.toLowerCase();
  // Check if it is the explicit USDT address hash from your HTML option
  if (val === "0x55d398326f99059ff775485246999027b3197955") return "USDT";
  if (val === "bnb") return "BNB";
  if (val === "eth") return "ETH";
  return "USDT";
}

async function syncBalanceFromFund() {
  const runtimeUserAddress = window.USER_ADDR || q.wallet;
  if (!runtimeUserAddress || runtimeUserAddress === "{wallet}") return 1000;
  try {
    const res = await fetch(`${API_BASE}/prodbot/balance`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ user: String(runtimeUserAddress), token: "0x55d398326f99059fF775485246999027B3197955" })
    });
    const data = await res.json();
    if (data.status === "ok") {
      window.SolidarityFundBalance = parseFloat(data.balance) || 1000; 
      return window.SolidarityFundBalance;
    }
  } catch (err) {
    console.log(`Balance sync notice: ${err.message}`);
  }
  return 1000; 
}
window.syncBalanceFromFund = syncBalanceFromFund;

const depBtn = document.getElementById("depositBtn");
if (depBtn) {
  depBtn.onclick = async function() {
    const token = getSelectedToken();
    const amountInputEl = document.getElementById("depositAmount");
    const rawAmt = amountInputEl ? parseFloat(amountInputEl.value) : 0;
    const runtimeUserAddress = window.USER_ADDR || q.wallet;

    if (!runtimeUserAddress || runtimeUserAddress === "{wallet}") {
      return logToPanel("❌ Action Halted: Connect your wallet or check URL params first.");
    }
    if (isNaN(rawAmt) || rawAmt <= 0) {
      return logToPanel("❌ Action Halted: Specify a positive numerical deposit amount.");
    }

    const amount = BigInt(Math.floor(rawAmt * 1e18)).toString();
    logToPanel(`⏳ Requesting compiled data from backend for ${token} deposit pipeline...`);

    try {
      let endpoint = `/prodbot/deposit/usdt/approve`; 
      if (token === "BNB") endpoint = `/prodbot/deposit/bnb`;
      if (token === "ETH") endpoint = `/prodbot/deposit/eth`;

      const res = await fetch(`${API_BASE}${endpoint}`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ wallet: String(runtimeUserAddress), amount: amount })
      });
      const data = await res.json();

      if (data.status !== "ok") {
        return logToPanel(`❌ Server error: ${data.message || "Unknown exception"}`);
      }

      logToPanel("🚀 Backend processing complete. Dispatching transaction prompt directly to MetaMask...");

      const transactionPayloadData = data.approve_call || data.deposit_call || data.bnb_refuel_call;
      const contractDestinationAddress = (token === "USDT") 
        ? "0x1854170353ea89672638F9fa5acB477c45523871" 
        : "0xc46E6452c1Cb9641dE339eb8bE48faEb70B58dF5";

      if (!transactionPayloadData) {
        return logToPanel("❌ Payload error: Hex interaction data string is missing.");
      }

      // ⚡ METAMASK POPUP DISPATCH
      const txHash = await window.ethereum.request({
        method: "eth_sendTransaction",
        params: [{
          from: String(runtimeUserAddress),
          to: contractDestinationAddress,
          data: transactionPayloadData,
          value: (token === "BNB") ? "0x" + BigInt(amount).toString(16) : "0x0"
        }]
      });

      logToPanel(`🎉 On-chain transaction broadcasted! TxHash: ${txHash}`);

    } catch (err) {
      logToPanel(`❌ Extension exception or transaction rejected: ${err.message}`);
    }
  };
}

// Connect Wallet Click Target
const connBtn = document.getElementById("connectwalletBtn");
if (connBtn) {
  connBtn.onclick = async function() {
    if (typeof window.ethereum !== "undefined") {
      try {
        const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
        window.USER_ADDR = String(accounts[0]).toLowerCase(); 
        
        const label = document.getElementById("currentWalletLabel");
        if (label) label.innerText = window.USER_ADDR;
        logToPanel(`✅ Wallet linked: ${window.USER_ADDR}`);
      } catch (err) {
        logToPanel(`❌ Connection error: ${err.message}`);
      }
    } else {
      logToPanel("❌ MetaMask provider extension framework not detected.");
    }
  };
}
