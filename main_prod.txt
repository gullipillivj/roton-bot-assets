async function initWeb3AndContracts() {
  try {
    const provider = appKit.getProvider();
    web3 = new Web3(provider);
    logToPanel("🌐 WalletConnect AppKit provider initialized.");

    fund = new web3.eth.Contract(fundAbi, FUND_ADDRESS);
    if (USER_TOKEN_ADDR) {
      userToken = new web3.eth.Contract(tokenAbi, USER_TOKEN_ADDR);
    }
  } catch (err) {
    logToPanel("❌ Contract Config Error: " + err.message);
  }
}

async function connectWallet() {
  logToPanel("🔗 connectWallet() triggered");
  try {
    const session = await appKit.connect();
    if (session && session.accounts.length > 0) {
      const activeAccount = session.accounts[0];
      window.USER_ADDR = activeAccount;
      logToPanel("✅ Wallet Connected: " + activeAccount);

      const btn = document.getElementById("connectWalletBtn");
      if (btn) {
        btn.textContent = "Connected: " + activeAccount.substring(0, 6) + "..." + activeAccount.slice(-4);
        btn.style.backgroundColor = "#28a745";
        btn.style.color = "#ffffff";
        btn.disabled = true;
      }

      await initWeb3AndContracts();
      await syncBalanceFromFund();
    }
  } catch (err) {
    logToPanel("❌ WalletConnect Error: " + err.message);
  }
}

async function verifyActiveWalletMatches() {
  if (!web3) { logToPanel("⚠️ Web3 not initialized. Please click 'Connect Wallet' first."); return false; }
  const accounts = await web3.eth.getAccounts();
  if (!accounts || accounts.length === 0) { logToPanel("⚠️ No connected accounts found."); return false; }

  const activeAddress = accounts[0].toLowerCase();
  if (!window.USER_ADDR || window.USER_ADDR.trim() === "") {
    window.USER_ADDR = activeAddress;
    logToPanel("🌐 Session Address auto-assigned: " + activeAddress);
  }

  const registeredAddress = window.USER_ADDR.toLowerCase();
  if (activeAddress !== registeredAddress) {
    logToPanel("❌ WALLET MISMATCH! Active: " + activeAddress + ", Expected: " + registeredAddress);
    alert("Please switch to your registered wallet:\n" + window.USER_ADDR);
    return false;
  }
  return true;
}

async function runAction(amount, toAddress, type) {
  try {
    const accounts = await web3.eth.getAccounts();
    if (!accounts || accounts.length === 0) throw new Error("No connected accounts found.");
    const userWalletAddress = accounts[0];
    const units = toTokenUnits(amount);
    let receipts = [];

    if (type === "approve") {
      const approveReceipt = await userToken.methods.approve(toAddress, units).send({ from: userWalletAddress });
      receipts.push({ tx_hash: approveReceipt.transactionHash, status: approveReceipt.status });
    } else if (type === "deposit") {
      const depositReceipt = await fund.methods.depositFunds(USER_TOKEN_ADDR, units).send({ from: userWalletAddress });
      receipts.push({ tx_hash: depositReceipt.transactionHash, status: depositReceipt.status });
    } else if (type === "payout") {
      const payoutReceipt = await fund.methods.swapToDonation(USER_TOKEN_ADDR, units, "0").send({ from: userWalletAddress });
      receipts.push({ tx_hash: payoutReceipt.transactionHash, status: payoutReceipt.status });
    }
    return { type, results: receipts };
  } catch (err) { return { error: err.message }; }
}
