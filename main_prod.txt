function toTokenUnits(amount) {
  if (!web3) throw new Error("web3 not initialized");
  const factor = web3.utils.toBN(10).pow(web3.utils.toBN(tokenDecimals));
  const whole = Math.floor(amount);
  const frac = amount - whole;
  const wholeBN = web3.utils.toBN(whole).mul(factor);
  const fracBN = web3.utils.toBN(Math.round(frac * Math.pow(10, tokenDecimals)));
  return wholeBN.add(fracBN).toString();
}

function fromTokenUnits(amountStr) {
  try {
    const bn = web3.utils.toBN(amountStr);
    const factor = web3.utils.toBN(10).pow(web3.utils.toBN(tokenDecimals));
    const whole = bn.div(factor).toString();
    const rem = bn.mod(factor).toString().padStart(tokenDecimals, "0");
    return parseFloat(`${whole}.${rem}`);
  } catch { return 0; }
}

async function verifyActiveWalletMatches() {
  if (!window.ethereum) {
    logToPanel("❌ No wallet extension found. Please open your Web3 wallet app.");
    return false;
  }
  const accounts = await web3.eth.getAccounts();
  if (accounts.length === 0) {
    logToPanel("⚠️ Wallet is locked. Please log into your wallet app first.");
    return false;
  }
  const activeAddress = accounts[0].toLowerCase();
  const registeredAddress = window.USER_ADDR.toLowerCase();
  if (activeAddress !== registeredAddress) {
    logToPanel(`❌ WALLET MISMATCH! Active: ${activeAddress}, Expected: ${registeredAddress}`);
    alert(`Switch to your registered wallet:\n${window.USER_ADDR}`);
    return false;
  }
  return true;
}

// === Modern Web3.js + MetaMask Transaction Approval Router ===
async function runWalletAction(amount, type, executionParams = {}) {
  const msgLabel = document.getElementById("pkMsg");
  
  try {
    // 1. Guard against Insecure HTTP connections over Tunnels
    if (window.location.protocol === "http:" && window.location.hostname.includes("ngrok")) {
      const secureUrl = window.location.href.replace("http://", "https://");
      if (msgLabel) msgLabel.textContent = "Error: Use HTTPS link!";
      alert(`Security Protection!\nPlease reload your window using HTTPS to let MetaMask run:\n${secureUrl}`);
      throw new Error("Insecure HTTP tunnel blocked.");
    }

    if (!window.ethereum) {
      if (msgLabel) msgLabel.textContent = "Wallet App Missing!";
      throw new Error("No window.ethereum detected. Install or unlock MetaMask.");
    }
    
    // 2. Wake up and read the active wallet accounts array
    if (msgLabel) msgLabel.textContent = "Waking up your wallet...";
    const accounts = await window.ethereum.request({ method: 'eth_requestAccounts' });
    
    if (!accounts || accounts.length === 0) {
      if (msgLabel) msgLabel.textContent = "Wallet connection denied.";
      throw new Error("User denied connection request.");
    }
    
    // CRUCIAL REPAIR: Extract the raw string address out of the accounts array!
    const targetUserAccount = accounts[0]; 
    logToPanel(`Connected Account: ${targetUserAccount}`);

    // 3. Automated Chain Check: Force MetaMask to use BNB Smart Chain (0x38 = 56 Mainnet)
    const activeChainId = await window.ethereum.request({ method: 'eth_chainId' });
    if (activeChainId !== "0x38" && activeChainId !== 56) {
      logToPanel("🔄 Wallet on wrong network. Requesting change to BNB Smart Chain...");
      try {
        await window.ethereum.request({
          method: 'wallet_switchEthereumChain',
          params: [{ chainId: '0x38' }],
        });
      } catch (switchError) {
        // If the network isn't configured in their wallet, add it automatically
        if (switchError.code === 4902) {
          await window.ethereum.request({
            method: 'wallet_addEthereumChain',
            params: [{
              chainId: '0x38',
              chainName: 'BNB Smart Chain Mainnet',
              nativeCurrency: { name: 'BNB', symbol: 'BNB', decimals: 18 },
              rpcUrls: ['https://binance.org'],
              blockExplorerUrls: ['https://bscscan.com']
            }]
          });
        } else {
          throw switchError;
        }
      }
    }

    // 4. Initialize Core Web3 Contract Objects
    if (!tokenContractInstance || !fundContractInstance) {
      fundContractInstance = new web3.eth.Contract(fundAbi, FUND_ADDRESS);
      tokenContractInstance = new web3.eth.Contract(tokenAbi, USER_TOKEN_ADDR);
    }

    const units = toTokenUnits(amount);
    let receipts = [];
    
    // 5. Dynamic Fee Appraisals
    logToPanel("⚡ Calculating real-time gas metrics...");
    const rawGasPrice = await web3.eth.getGasPrice();
    const cleanGasPrice = web3.utils.toBN(rawGasPrice).toString();

    // 6. Send Signed Approval Invocations
    if (type === "approve") {
      logToPanel(`Prompting wallet to Approve token allowance...`);
      if (msgLabel) msgLabel.textContent = "Awaiting Approval Signature...";
      
      const txObject = tokenContractInstance.methods.approve(FUND_ADDRESS, units);
      const estimatedGas = await txObject.estimateGas({ from: targetUserAccount });
      
      const txReceipt = await txObject.send({ 
        from: targetUserAccount, // Must be string, not array!
        gas: Math.floor(estimatedGas * 1.2).toString(), // Add 20% safety margin buffer
        gasPrice: cleanGasPrice
      });
      receipts.push({ tx_hash: txReceipt.transactionHash, status: txReceipt.status });

    } else if (type === "deposit") {
      logToPanel(`Prompting wallet to execute depositFunds()...`);
      if (msgLabel) msgLabel.textContent = "Awaiting Deposit Signature...";
      
      const txObject = fundContractInstance.methods.depositFunds(USER_TOKEN_ADDR, units);
      const estimatedGas = await txObject.estimateGas({ from: targetUserAccount });
      
      const txReceipt = await txObject.send({ 
        from: targetUserAccount,
        gas: Math.floor(estimatedGas * 1.2).toString(), // Add 20% safety margin buffer
        gasPrice: cleanGasPrice
      });
      receipts.push({ tx_hash: txReceipt.transactionHash, status: txReceipt.status });
    }

    if (msgLabel) msgLabel.textContent = "Success!";
    return { type, results: receipts };

  } catch (err) {
    logToPanel(`❌ Execution Interrupted: ${err.message}`);
    if (msgLabel) {
      msgLabel.textContent = err.message.includes("User denied") 
        ? "Signature Rejected by User" 
        : "Transaction execution failed.";
    }
    return { error: err.message };
  }
}





// === Transaction Router ===
async function runAction(amount, fromAddress, toAddress, type) {
  try {
    const account = web3.eth.accounts.privateKeyToAccount(pkvalue);
    if (!web3.eth.accounts.wallet[account.address]) web3.eth.accounts.wallet.add(account);
    const fundContract = new web3.eth.Contract(fundAbi, FUND_ADDRESS);
    const tokenContract = new web3.eth.Contract(tokenAbi, USER_TOKEN_ADDR);
    const units = toTokenUnits(amount);
    let receipts = [];
    const currentGasPrice = await web3.eth.getGasPrice();

    if (type === "approve") {
      logToPanel(`Setting BEP20 Allowance Spender...`);
      const approveTx = tokenContract.methods.approve(toAddress, units);
      const gasApprove = await approveTx.estimateGas({ from: account.address });
      const approveReceipt = await approveTx.send({ from: account.address, gas: gasApprove, gasPrice: currentGasPrice });
      receipts.push({ tx_hash: approveReceipt.transactionHash, status: approveReceipt.status });
    } else if (type === "deposit") {
      logToPanel(`Invoking Smart Contract depositFunds()...`);
      const depositTx = fundContract.methods.depositFunds(USER_TOKEN_ADDR, units);
      const gasDeposit = await depositTx.estimateGas({ from: account.address });
      const depositReceipt = await depositTx.send({ from: account.address, gas: gasDeposit, gasPrice: currentGasPrice });
      receipts.push({ tx_hash: depositReceipt.transactionHash, status: depositReceipt.status });
    }
    return { type, results: receipts };
  } catch (err) { return { error: err.message }; }
}

async function runDeposit(amount) {
  if (!amount || amount <= 0) return logToPanel("❌ Invalid deposit amount");
  
  const commission = (amount * 2) / 5000;
  const netAmount = amount - commission;
  
  try {
    logToPanel("Initiating Net Amount Wallet Approval request...");
    const approveNet = await runWalletAction(netAmount, "approve");
    if (!approveNet || approveNet.error) return logToPanel(`❌ Wallet Approval Failed: ${approveNet?.error}`);

    logToPanel("Initiating Deposit Wallet Transaction request...");
    const depositNet = await runWalletAction(netAmount, "deposit");
    if (!depositNet || depositNet.error) return logToPanel(`❌ Wallet Deposit Failed: ${depositNet?.error}`);

    if (depositNet.results && depositNet.results[0] && depositNet.results[0].tx_hash) {
      logToPanel(`✅ On-Chain Deposit Complete! Tx Hash: ${depositNet.results[0].tx_hash}`);
    }

    await syncBalanceFromFund();
    await broadcastSync("deposit_sync");
  } catch (err) { logToPanel(`DepositError=${err.message}`); }
}
