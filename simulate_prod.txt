// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

// Thread state delays helper
const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

// UI Status label updater
function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

async function updateTokenRecord(lastToken, currentToken, networkType) {
    try {

        const resp = await fetch("/prodbot/tokens/update", {
            method: "POST",
            headers: { "Content-Type": "application/json" },

                userid: window.USER_ADDR || window.USER_ID, 
                lasttoken: lastToken, 
                currenttoken: currentToken,
                network: networkType || "bep20"
            })
        });
        
        logToPanel(`✅ [STATE CACHE VERIFIED]\nLAST POOL: ${lastToken}\nCURRENT TARGET: ${currentToken}`);
        return true;
    } catch (err) {
        logToPanel(`⚠️ State sync logging alert: ${err.message}`);
        return false;
    }
}
window.updateTokenRecord = updateTokenRecord;

/**
 * On-chain token state resolver.
 * Hits the updated backend route to read active positions from getUserBalanceList tuple fields.
 */
async function getCurrentTokenState() {
    let state = { token: "0x55d398326f99059fF775485246999027B3197955", network: "bep20" };
    
    try {
        const activeIdentityVector = window.USER_ADDR || window.USER_ID;
        if (!activeIdentityVector) {
            return state;
        }

        const resp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: activeIdentityVector })
        });
        
        const data = await resp.json();
        if (data && data.status === "ok" && data.currenttoken) {
            state.token = data.currenttoken;
            state.network = data.network || "bep20";
        }
    } catch (err) {
        logToPanel(`[ON-CHAIN POSITION LOOKUP FAILED] Defaulting to USDT: ${err.message}`);
    }
    
    return state;
}
window.getCurrentTokenState = getCurrentTokenState;

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`\n--- [START] Cycle ${cycle}/${totalCycles} ---`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);

    // Capture the initial investment baseline value before any transactions execute
    let balanceInputBefore = document.getElementById("investBalance").value.toString().trim();
    if (!balanceInputBefore || balanceInputBefore === "0") {
        logToPanel("⚠️ [CYCLE ABORTED] Balance field is empty or zero.");
        return "STOP_ERROR";
    }

    const srcAsset = await getCurrentTokenState();
    const sourceTokenAddress = String(srcAsset.token).toLowerCase().trim();
    const sourceNetwork = String(srcAsset.network).toLowerCase().trim();
    logToPanel(`[PORTFOLIO ROOT IDENTIFIED] Active position: ${srcAsset.token} (${sourceNetwork.toUpperCase()})`);

    let marketData = [];
    try {
        const filterResp = await fetch("/prodbot/filtertokens");
        const filterJson = await filterResp.json();
        if (filterJson && filterJson.status === "ok" && Array.isArray(filterJson.short_term)) {
            marketData = filterJson.short_term;
        }
    } catch (fErr) {
        logToPanel(`⚠️ [SCANNER DELAY] Master filter table route dropped: ${fErr.message}`);
        return "CONTINUE";
    }

    if (marketData.length === 0) {
        logToPanel("⚠️ [SCANNER COLD] Zero token profiles verified. Skipping cycle...");
        return "SKIP";
    }

    // Discover the highest yield candidate asset profile that you do not currently hold
    let isolatedTarget = null;
    for (let i = 0; i < marketData.length; i++) {
        let currentCandidateAddress = String(marketData[i].address).toLowerCase().trim();
        if (currentCandidateAddress !== sourceTokenAddress) {
            isolatedTarget = marketData[i];
            break;
        }
    }

    if (!isolatedTarget) {
        logToPanel("🔄 All high-momentum assets are already held in portfolio. Skipping...");
        return "SKIP";
    }

    const targetTokenAddress = String(isolatedTarget.address).toLowerCase().trim();
    const targetedWinnerSymbol = String(isolatedTarget.symbol).toUpperCase().trim();
    const winnerNetworkType = String(isolatedTarget.tokentype || "bep20").toLowerCase().trim();

    logToPanel(`🎯 [TARGET CHOSEN] Next candidate: ${targetedWinnerSymbol}`);
    logToPanel(`🔀 [SWAP LIFECYCLE INITIATED]`);
    logToPanel(`   ➡️ FROM: [${sourceTokenAddress}] (${sourceNetwork.toUpperCase()})`);
    logToPanel(`   ➡️ TO:   ${targetedWinnerSymbol} [${targetTokenAddress}] (${winnerNetworkType.toUpperCase()})`);

    try {
        let activeTokenIn = sourceTokenAddress;
        let currentAllocationWei = balanceInputBefore;
        let data = null;

        const WBNB_ADDR = window.WBNB_TOKEN_ADDRESS.toLowerCase().trim();
        const WETH_ADDR = "0xc02aaa39b223fe8d0a0e5c4f27ead9083c756cc2"; 

        // --- STEP 1: CONVERT CUSTOM POSITION TO BASE INTERMEDIARY WRAPPERS ---
        if (sourceNetwork === "erc20" && activeTokenIn !== WETH_ADDR) {
            logToPanel(`⚠️ Custom Ethereum Asset detected. Step 1/2: Transferring into WETH...`);
            data = await window.runethSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, WETH_ADDR, "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D");
            await handleTwoStepTxPipeline(data);
            if (typeof syncBalanceFromFund === "function") await window.syncBalanceFromFund();
            activeTokenIn = WETH_ADDR;
            currentAllocationWei = document.getElementById("investBalance").value.toString().trim();
        } else if (sourceNetwork === "bep20" && activeTokenIn !== WBNB_ADDR) {
            logToPanel(`⚠️ Custom Binance Asset detected. Step 1/2: Transferring into WBNB...`);
            data = await window.runbscSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, WBNB_ADDR, "0x10ED43C718714eb63d5aA57B78B54704E256024E");
            await handleTwoStepTxPipeline(data);
            if (typeof syncBalanceFromFund === "function") await window.syncBalanceFromFund();
            activeTokenIn = WBNB_ADDR;
            currentAllocationWei = document.getElementById("investBalance").value.toString().trim();
        }

        // --- STEP 2: DEPLOY ORDER TRANSFERS AGAINST THE DISCOVERED MOMENTUM TARGET ---
        logToPanel(`⏳ Executing deployment target purchase step: ${targetedWinnerSymbol}...`);
        if (winnerNetworkType === "erc20") {
            data = await window.runethSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, targetTokenAddress, "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D");
        } else {
            data = await window.runbscSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, targetTokenAddress, "0x10ED43C718714eb63d5aA57B78B54704E256024E");
        }

        if (!data || data.status !== "ok") {
            logToPanel(`⚠️ Target pipeline configuration payload invalid. Skipping...`);
            return "CONTINUE";
        }

        logToPanel(`📢 [WALLET PROMPT] Authorizing token spend allocation limits...`);
        await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{ from: window.USER_ADDR, to: data.approval_packet.to_contract, data: data.approval_packet.compiled_data, value: "0x0" }]
        });
        await blockDelay(10000);

        logToPanel(`📢 [WALLET PROMPT] Confirming asset exchange processing...`);
        const txHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{ from: window.USER_ADDR, to: data.swap_packet.to_contract, data: data.swap_packet.compiled_data, value: "0x0" }]
        });
        logToPanel(`Exchange Transaction Dispatched! Hash: ${txHash}`);

        // 🌟 STRICTOR CONTROL GATEWAY: Prevents proceeding until the form input shifts
        let settledOnChain = false;
        logToPanel(`⏳ Locking thread loop. Interrogating on-chain ledger arrays until values change...`);
        
        for (let i = 1; i <= 60; i++) {
            await blockDelay(8000);
            if (typeof syncBalanceFromFund === "function") {
                await window.syncBalanceFromFund();
            }
            let checkCurrentFormValue = document.getElementById("investBalance").value.toString().trim();
            
            // The gate requires the input balance to register a change, and prevents running if it accidentally clears to 0 mid-sync
            if (checkCurrentFormValue !== balanceInputBefore && checkCurrentFormValue !== "0") {
                settledOnChain = true;
                break;
            }
        }

        if (!settledOnChain) {
            logToPanel(`🛑 [TIMEOUT WARNING] Control parameters remained unchanged. Halting pipeline execution...`);
            return "STOP_ERROR";
        }

        // ==================================================================
        // 📈 LIVE CALCULATIONS & CHART UPDATE EXTENSION MATRIX
        // ==================================================================
        const currentUsdValueField = document.getElementById("usd");
        const currentUsdValue = currentUsdValueField ? parseFloat(currentUsdValueField.value) : 0;
        
        // Capture initial entry base price configuration parameters to lock PnL changes
        if (typeof window.initialSessionUsdValue === "undefined" || window.initialSessionUsdValue === 0) {
            // Converts from Wei to base human readable valuation units for first iteration reference
            window.initialSessionUsdValue = parseFloat(balanceInputBefore) / 1e18;
        }

        // Derived Mathematical Session Yield
        const sessionProfitUsd = currentUsdValue - window.initialSessionUsdValue;
        window.cumulativePnL = sessionProfitUsd;

        // 🟢 UPDATE 1: Dynamically update the visual text layout display element container properties
        const liveProfitValueContainer = document.getElementById("liveProfitValue");
        if (liveProfitValueContainer) {
            const formattedProfitString = sessionProfitUsd >= 0 
                ? `+${sessionProfitUsd.toFixed(4)}` 
                : `${sessionProfitUsd.toFixed(4)}`;
            
            // Toggle visual layout text colors (Green for profitable delta, Dark Orange for negative yields)
            liveProfitValueContainer.style.color = sessionProfitUsd >= 0 ? "#00FF00" : "#dc3545";
            liveProfitValueContainer.innerText = `${formattedProfitString} USDT`;
        }

        // 🟢 UPDATE 2: Safely append real-time telemetry datasets directly into your ChartJS context
        if (typeof window.updateChart === "function") {
            window.updateChart(currentUsdValue, window.cumulativePnL);
        } else {
            console.warn("[ORCHESTRATOR ALERT] window.updateChart hook is missing from layout framework context.");
        }

        logToPanel(`🎉 Cycle transaction cleared! Net Valuation: $${currentUsdValue.toFixed(4)} | Session Profit: $${sessionProfitUsd.toFixed(4)}`);

    } catch (err) {
        logToPanel(`❌ Loop Boundary Crash: ${err.message}`);
        return "STOP_ERROR";
    }
    return "CONTINUE";
}

// 🛠️ STANDALONE TWO-STEP PIPELINE TRANSACTION PROMPTER
// ==================================================================
async function handleTwoStepTxPipeline(payload, inputToken, outBridgeToken, targetNet) {
    if (!payload || payload.status !== "ok") throw new Error("Bridge route serialization compilation exception.");
    
    const appHash = await window.ethereum.request({
        method: "eth_sendTransaction",
        params: [{ from: window.USER_ADDR, to: payload.approval_packet.to_contract, data: payload.approval_packet.compiled_data, value: "0x0" }]
    });
    await blockDelay(12000);

    const txHash = await window.ethereum.request({
        method: "eth_sendTransaction",
        params: [{ from: window.USER_ADDR, to: payload.swap_packet.to_contract, data: payload.swap_packet.compiled_data, value: "0x0" }]
    });
    
    logToPanel(`⏳ Bridge Tx Sent: ${txHash}. Waiting for block confirmation...`);
    
    let verified = false;
    const beforeWeiBridge = document.getElementById("investBalance").value.toString().trim();
    for (let index = 1; index <= 45; index++) {
        await blockDelay(10000);
        if (typeof syncBalanceFromFund === "function") await syncBalanceFromFund();
        if (document.getElementById("investBalance").value.toString().trim() !== beforeWeiBridge) { 
            verified = true; 
            break; 
        }
    }
    if (!verified) throw new Error("Intermediate bridge asset routing transaction stalled or failed.");
    
    await updateTokenRecord(inputToken, outBridgeToken, targetNet);
    logToPanel(`✅ Wrapped Base Token Position Ledger successfully logged.`);
    await blockDelay(4000);
}

// 🚀 MASTER SIMULATION CONTROLLER AND INTERFACE GATEWAY
// ==================================================================
async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;
    logToPanel(`🚀 [BOT INITIATED] Processing up to ${totalCycles} execution cycles.`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (signal === "STOP_ERROR") {
            logToPanel("🛑 State mismatch runtime alert. Aborting framework loops.");
            break;
        }
        if (signal === "SKIP") {
            await blockDelay(1000); 
            continue;
        }
        if (i < totalCycles) {
            setStatus("WAITING");
            await blockDelay(10000);
        }
    }
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing complete.");
}
window.startSimulation = startSimulation;
