function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

window.isBotRunning = false;
window.stopRequested = false;

async function updateTokenRecord(lastToken, currentToken) {
    try {
        const userid = window.USER_ID || localStorage.getItem("userid");
        if (!userid) {
            logToPanel("[TOKEN SYNC] Missing userid");
            return false;
        }
        const resp = await fetch("/prodbot/tokens/update", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: userid, lasttoken: lastToken, currenttoken: currentToken })
        });
        const data = await resp.json();
        if (data && data.message) {
            logToPanel(`[DATABASE COMMIT VERIFIED]\nLAST=${lastToken}\nCURRENT=${currentToken}`);
            return true;
        }
        return false;
    } catch (err) {
        logToPanel(`[TOKEN SYNC ERROR] ${err.message}`);
        return false;
    }
}


//Section 2
async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
    } catch (err) {
        console.log("pickRandomCoin missed query, using fallback:", err);
    }
    return "CAKE";
}

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    if (!window.isBotRunning) {
        logToPanel("[INFO] Bot is already completely idle.");
        return;
    }
    window.stopRequested = true;
    setStatus("HALTING...");
    logToPanel("⏹️ [STOP REQUESTED] Graceful exit active. Enforcing stablecoin checks before shutting down...");
}
window.stopBot = stopBot;

//Section 3
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`[INFO] Cycle ${cycle}/${totalCycles}`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);

    const syncedBalance = await syncBalanceFromFund();
    if (syncedBalance === null) {
        logToPanel("[WARN] Balance sync failed. Stopping orchestration loop.");
        return "STOP_ERROR"; 
    }

    let tradingBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    if (tradingBalance <= 0) {
        logToPanel("[WARN] No trading balance available. Stopping loop.");
        return "STOP_ERROR"; 
    }

    window.beforeBalanceUSD = parseFloat(document.getElementById("usd").value) || 0;
    if (cycle === 1) { window.initialSimulationUSD = window.beforeBalanceUSD; }

    const totalBalance = BigInt(document.getElementById("startBalance").value.toString().trim());
    const remainingCycles = Math.max(1, totalCycles - cycle + 1);
    const allocationUSD = (totalBalance / BigInt(remainingCycles)).toString();

    logToPanel(`[ALLOCATION]\nTOTAL=${totalBalance}\nALLOCATED=${allocationUSD}`);

    let currentToken = "0x55d398326f99059fF775485246999027B3197955"; 
    const usdtAddress = "0x55d398326f99059fF775485246999027B3197955";

    try {
        const tokenResp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: window.USER_ID })
        });
        const tokenData = await tokenResp.json();
        if (tokenData && tokenData.currenttoken) { currentToken = tokenData.currenttoken; logToPanel(`[TOKEN LOAD]\n${currentToken}`); }
    } catch (err) { logToPanel(`[TOKEN LOAD ERROR]\n${err.message}`); }

    let selectedCoin = "CAKE";
    try { selectedCoin = await pickRandomCoin(); } catch (pickErr) { selectedCoin = "CAKE"; }

    let coinInfo = null;
    try { coinInfo = await window.evaluateCoin(selectedCoin); } catch (evalErr) { console.error(evalErr); }

    if (!coinInfo || !coinInfo.tokenAddress) {
        logToPanel("⚠️ Dynamic pricing lookup stalled. Injecting stable test asset mapping.");
        coinInfo = { symbol: "CAKE", tokenAddress: "0x0e09fabb73bd3ade0a17ecc321fd13a19e81ce82", change24h: 1.8, lastPrice: 1.95 };
    }

    if (coinInfo.tokenAddress.toLowerCase() === currentToken.toLowerCase()) {
        logToPanel("🔄 Target asset matches current holding token profile. Alternating destination target...");
        if (currentToken.toLowerCase() === usdtAddress.toLowerCase()) {
            coinInfo = { symbol: "CAKE", tokenAddress: "0x0e09fabb73bd3ade0a17ecc321fd13a19e81ce82", change24h: 1.8, lastPrice: 1.95 };
        } else {
            coinInfo = { symbol: "USDT", tokenAddress: usdtAddress, change24h: 0.0, lastPrice: 1.0 };
        }
    }

    logToPanel(`[TARGET]\n${coinInfo.symbol}\nMOVE=${coinInfo.change24h}%\nADDRESS=${coinInfo.tokenAddress}`);
    const beforeBalance = document.getElementById("investBalance").value.toString().trim();

    try {
        logToPanel(`[SWAP REQUEST]\nFROM=${currentToken}\nTO=${coinInfo.tokenAddress}\nAMOUNT=${allocationUSD}`);
        const swapResp = await window.runSwapUniswap(window.USER_ADDR, allocationUSD, currentToken, coinInfo.tokenAddress);

        if (!swapResp || !swapResp.tx_hash) { logToPanel("[SWAP PIPELINE SKIPPED OR REJECTED]"); return "CONTINUE"; }
        logToPanel(`[TX SENT]\n${swapResp.tx_hash}`);


//Section 4
        let balanceChanged = false;
        for (let attempt = 1; attempt <= 45; attempt++) {
            await blockDelay(10000);
            await syncBalanceFromFund();
            const currentBalance = document.getElementById("investBalance").value.toString().trim();
            logToPanel(`[VERIFY ${attempt}/45]\nSTART=${beforeBalance}\nNOW=${currentBalance}`);

            if (currentBalance !== beforeBalance) {
                balanceChanged = true;
                logToPanel(`[BALANCE VERIFIED]\n${beforeBalance}\n➡\n${currentBalance}`);
                break;
            }
        }

        if (!balanceChanged) { logToPanel("❌ [NO BALANCE MOVEMENT DETECTED] Enforcing safe lockdown. Halting loop."); return "STOP_ERROR"; }

        logToPanel("⏳ Synchronizing database token allocation tracking...");
        await updateTokenRecord(currentToken, coinInfo.tokenAddress);
        currentToken = coinInfo.tokenAddress;

        if (cycle === totalCycles || window.stopRequested) {
            if (currentToken.toLowerCase() !== usdtAddress.toLowerCase()) {
                logToPanel(`🛡️ [SETTLEMENT REQUIRED] Returning final cycle portfolio to safe USDT reserves...`);
                const volatileWeiBalance = document.getElementById("investBalance").value.toString().trim();
                logToPanel(`[SHIELD SWAP REQUEST]\nFROM=${currentToken}\nTO=${usdtAddress}\nAMOUNT=${volatileWeiBalance}`);
                const settleResp = await window.runSwapUniswap(window.USER_ADDR, volatileWeiBalance, currentToken, usdtAddress);
                
                if (settleResp && settleResp.tx_hash) {
                    logToPanel(`✅ [SHIELD SETTLED] Capital securely locked in USDT. Hash: ${settleResp.tx_hash}`);
                    await updateTokenRecord(currentToken, usdtAddress);
                    currentToken = usdtAddress;
                } else {
                    logToPanel("⚠️ [SHIELD FAILED] Settlement rejected. Assets remain stranded in alternative slot.");
                    return "STOP_ERROR";
                }
            } else { logToPanel("✅ [SETTLEMENT SKIPPED] Assets already resting cleanly in USDT stable reserves."); }
        } else { logToPanel(`📝 [HOLDING POSITION] Asset portfolio (${coinInfo.symbol}) preserved for upcoming Cycle ${cycle + 1} tracks.`); }

        await syncBalanceFromFund();
        const afterBalanceUSD = parseFloat(document.getElementById("usd").value) || 0;
        const profitDeltaUSD = afterBalanceUSD - window.beforeBalanceUSD;
        window.cumulativePnL = (window.cumulativePnL || 0) + profitDeltaUSD;

        const liveProfitText = document.getElementById("liveProfitValue");
        if (liveProfitText) {
            liveProfitText.innerText = window.cumulativePnL.toFixed(4) + " USDT";
            liveProfitText.style.color = window.cumulativePnL >= 0 ? "#00FF00" : "#FF3B30";
        }
        if (typeof updateChart === "function") { updateChart(afterBalanceUSD, window.cumulativePnL); }

    } catch (err) { logToPanel(`[RUNTIME RUN EXCEPTION]\n${err.message}`); return "STOP_ERROR"; }

    return "CONTINUE";
}

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.stopRequested = false;
    window.cumulativePnL = 0;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    if (window.balanceChart) {
        window.balanceChart.data.labels = [];
        window.balanceChart.data.datasets[0].data = [];
        window.balanceChart.data.datasets[1].data = [];
        window.balanceChart.update();
    }

    logToPanel(`[BOT START] ${totalCycles} cycles`);

    for (let i = 1; i <= totalCycles; i++) {
        if (window.stopRequested) { logToPanel("🛑 [GRACEFUL EXIT CONFIRMED] Execution track safe. Terminating simulation engine."); break; }
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (signal === "STOP_ERROR") { logToPanel("🛑 State check failed or shield swap rejected. Stopping loop to protect records."); break; }
        if (window.stopRequested) { logToPanel("🛑 [GRACEFUL EXIT CONFIRMED] Execution track safe. Terminating simulation engine."); break; }

        if (i < totalCycles) { setStatus("WAITING"); await blockDelay(10000); }
    }
    
    window.isBotRunning = false;
    window.stopRequested = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED]");
}

window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
