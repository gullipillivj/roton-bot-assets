logToPanel("simulate_prod.txt loaded successfully with Dynamic Grounding Engine.");

window.setStatus = function(statusText) {
    console.log(`[STATUS UPDATE]: ${statusText}`);
    logToPanel(`[STATUS]: ${statusText}`);
};

// 🧠 Dynamic verification handler targeting live smart contract state maps
window.discoverActiveOnChainAsset = function(userAddress) {
    const baseApiUrl = window.API_BASE || window.location.origin;
    return fetch(`${baseApiUrl}/prodbot/active-asset`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ user: userAddress })
    })
    .then(res => res.json())
    .catch(err => {
        console.error("Grounding connection lost:", err);
        return { status: "error", confirmed_symbol: "USDT", confirmed_address: "0x55d398326f99059fF775485246999027B3197955", confirmed_balance_wei: "0" };
    });
};

// 🟢 EXCLUSIVE BACKEND ROUTER: Queries custom `/prodbot/balance` endpoint directly
async function queryTokenBalanceDirect(userAddr, tokenAddress) {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        const res = await fetch(baseApiUrl + "/prodbot/balance", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: userAddr, token: tokenAddress })
        });
        const rawWeiText = await res.text();
        return rawWeiText.trim() || "0";
    } catch (err) {
        return "0";
    }
}

window.updateUIBalanceControl = async function(userAddr, tokenAddress) {
    try {
        // Query only the definitive backend asset storage path
        const cleanWeiString = await queryTokenBalanceDirect(userAddr, tokenAddress);

        // Update raw layout storage fields safely if the core engine loop is idle
        if (!window.isBotRunning) {
            const startBox = document.getElementById("startBalance");
            const investBox = document.getElementById("investBalance");
            if (startBox) startBox.value = cleanWeiString;
            if (investBox) investBox.value = cleanWeiString;
        }

        // HUMAN READABLE CONVERSION: Convert raw units and apply to the UI dollar field
        const usdBox = document.getElementById("usd");
        if (usdBox) {
            const floatTokens = parseFloat(cleanWeiString) / 1e18;
            let tokenPriceUSD = 1.0; 

            try {
                // Read from cache to discover valuation properties without printing log lines
                const buckets = typeof window.categorizeCoins === "function" ? await window.categorizeCoins() : {};
                const localPool = [...(buckets.long_term || []), ...(buckets.mid_term || []), ...(buckets.short_term || [])];
                
                const matchedAsset = localPool.find(c => c.tokenAddress && c.tokenAddress.toLowerCase() === tokenAddress.toLowerCase());
                if (matchedAsset && matchedAsset.price) {
                    tokenPriceUSD = parseFloat(matchedAsset.price);
                } else if (tokenAddress.toLowerCase() === "0x55d398326f99059ff775485246999027b3197955") {
                    tokenPriceUSD = 1.0;
                } else {
                    tokenPriceUSD = 3.2; // Custom asset price fallback proxy baseline
                }
            } catch (e) {
                // Silently handle exceptions to prevent filling the log console
            }

            const totalCalculatedUSD = floatTokens * tokenPriceUSD;
            if (!window.isBotRunning) {
                usdBox.value = totalCalculatedUSD.toFixed(2); 
            }
        }

        return cleanWeiString;
    } catch (err) {
        return "0";
    }
};

// LIVE BACKGROUND INTERMISSION TIMING LAYER: Keeps values accurate every 5 seconds without flooding logs
window.initLiveBalanceClock = function() {
    if (window.balanceClockIntervalId) {
        clearInterval(window.balanceClockIntervalId);
    }
    
    window.balanceClockIntervalId = setInterval(async () => {
        if (!window.USER_ADDR || window.isBotRunning) return; 
        try {
            const stateRecord = await window.discoverActiveOnChainAsset(window.USER_ADDR);
            const activeTokenAddr = stateRecord.confirmed_address || "0x55d398326f99059fF775485246999027B3197955";
            await window.updateUIBalanceControl(window.USER_ADDR, activeTokenAddr);
        } catch (error) {
            // Silently absorb runtime interruptions
        }
    }, 5000); 
};

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

// ==================================================================
// 🚀 SECTION 2: GROUNDED ROTATIONAL PROFIT EVALUATOR & LOOP ENGINE
// ==================================================================
async function startSimulation(totalCycles = 10) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    logToPanel(`[LIVE ENGINE INITIALIZED] Guarding runtime loop using direct on-chain contract indices.`);
    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";
    const RR_BNB_ADDR = "0xba67186fd454D0256a0D3744783538bE08C58feD";
    const RR_ETH_ADDR = "0x8A0cAC8E7F6305cE6EeB585005919290F78808f9";
    
    // 🟢 BSC TOKEN CONTRACT CHECKLIST DEFINITIONS FOR BACKEND LOOKUP SCANNING
    const VENUS_XVS_ADDRESS = "0xcF6BB5389c92Bdda8a3747Ddb454cB7a64E86902"; 
    const WBNB_ADDRESS = "0xbb4CdB9CBd36B01bD1cBaEBF2De08d9173bc095c";

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        logToPanel(`\n[INFO] — Cycle ${i}/${totalCycles} starting...`);
        window.setStatus(`RUNNING cycle ${i}/${totalCycles}`);

        logToPanel("🔎 Querying contract state maps via Brownie node link...");
        const stateRecord = await window.discoverActiveOnChainAsset(window.USER_ADDR);
        
        let currentAssetAddress = stateRecord.confirmed_address;
        let currentAssetSymbol = stateRecord.confirmed_symbol;
        let initialFieldText = stateRecord.confirmed_balance_wei;

        logToPanel(`[GROUNDED] Contract Verified State: Holding ${currentAssetSymbol} (${initialFieldText} base units)`);

        // DYNAMIC SYNC GATEWAY: Query backend balance route natively to verify asset limits
        if (BigInt(initialFieldText) <= 0n) {
            logToPanel(`⚠️ Contract storage slot empty for ${currentAssetSymbol}. Scanning user portfolio fields...`);
            initialFieldText = await queryTokenBalanceDirect(window.USER_ADDR, currentAssetAddress);
            
            // 🟢 RUN CLEAN PORTFOLIO DISCOVERY: Scan alternate addresses using backend routes only
            if (BigInt(initialFieldText) <= 0n) {
                const scannerProfiles = [
                    { symbol: "XVS", address: VENUS_XVS_ADDRESS },
                    { symbol: "BNB", address: WBNB_ADDRESS },
                    { symbol: "USDT", address: USDT_ADDRESS }
                ];
                
                for (let profile of scannerProfiles) {
                    if (profile.address.toLowerCase() === currentAssetAddress.toLowerCase()) continue;
                    
                    logToPanel(`🔎 Querying backend balance route for asset: ${profile.symbol}...`);
                    let balanceCheck = await queryTokenBalanceDirect(window.USER_ADDR, profile.address);
                    
                    if (BigInt(balanceCheck) > 0n) {
                        currentAssetAddress = profile.address;
                        currentAssetSymbol = profile.symbol;
                        initialFieldText = balanceCheck;
                        logToPanel(`✅ Capital Found! Switched active baseline token focus to ${currentAssetSymbol}.`);
                        break;
                    }
                }
            }

            // FORCE UI REFRESH: Manually project the discovered balance straight to the text containers
            const startBox = document.getElementById("startBalance");
            const investBox = document.getElementById("investBalance");
            const usdBox = document.getElementById("usd");
            if (startBox) startBox.value = initialFieldText;
            if (investBox) investBox.value = initialFieldText;
            if (usdBox) {
                const floatTokens = parseFloat(initialFieldText) / 1e18;
                usdBox.value = (floatTokens * 1.0).toFixed(2); 
            }

            // Total depletion confirmation gate
            if (BigInt(initialFieldText) <= 0n) {
                logToPanel(`❌ Pipeline Halted: Verified total depletion across all scanned asset profiles.`);
                break;
            }
        }

        let targetSymbol = "USDT";
        let targetAddress = USDT_ADDRESS;
        let targetNetwork = "bep20";

        if (i !== totalCycles) {
            const buckets = await window.categorizeCoins();
            const fullPool = [...(buckets.long_term || []), ...(buckets.mid_term || []), ...(buckets.short_term || [])];

            let highestProfitableCandidate = null;
            const floatBalance = parseFloat(initialFieldText) / 1e18; 
            
            const baselineOverhead = 0.25;
            const estimatedGasOverheadUSD = floatBalance < 5.0 ? 0.0001 : baselineOverhead;

            fullPool.sort((a, b) => (b.change24h || 0) - (a.change24h || 0));

            for (let candidate of fullPool) {
                if (
                    candidate.symbol === "BNB" || candidate.symbol === "ETH" || candidate.symbol === "USDT" ||
                    candidate.tokenAddress.toLowerCase() === RR_BNB_ADDR.toLowerCase() ||
                    candidate.tokenAddress.toLowerCase() === RR_ETH_ADDR.toLowerCase()
                ) {
                    continue; 
                }

                if (candidate.tokenAddress.toLowerCase() === currentAssetAddress.toLowerCase()) {
                    continue;
                }

                const trendAccelerationFactor = candidate.change24h > 0 ? (candidate.change24h * 0.05) : 0.01;
                const projectedGrossProfitUSD = floatBalance * (trendAccelerationFactor / 100);

                if (projectedGrossProfitUSD > estimatedGasOverheadUSD) {
                    highestProfitableCandidate = candidate;
                    logToPanel(`✨ Profit Gate Passed: ${candidate.symbol} yields $${projectedGrossProfitUSD.toFixed(4)}`);
                    break; 
                }
            }

            if (highestProfitableCandidate) {
                targetSymbol = highestProfitableCandidate.symbol;
                targetAddress = highestProfitableCandidate.tokenAddress;
                targetNetwork = highestProfitableCandidate.network || "bep20";
                logToPanel(`🎯 Market Yield Verified: Selecting ${targetSymbol} for rotational execution.`);
            } else {
                if (currentAssetAddress.toLowerCase() !== USDT_ADDRESS.toLowerCase()) {
                    targetSymbol = "USDT";
                    targetAddress = USDT_ADDRESS;
                    logToPanel(`⚖️ Low momentum: Re-routing assets back to USDT stability layer.`);
                } else {
                    if (currentAssetAddress.toLowerCase() === RR_BNB_ADDR.toLowerCase()) {
                        targetSymbol = "RR-ETH";
                        targetAddress = RR_ETH_ADDR;
                    } else {
                        targetSymbol = "RR-BNB";
                        targetAddress = RR_BNB_ADDR;
                    }
                }
            }
        }

        if (targetAddress.toLowerCase() === currentAssetAddress.toLowerCase()) {
            await blockDelay(3000);
            continue;
        }

        logToPanel(`[ROUTE] Preparing transaction pipeline: Swap ${currentAssetSymbol} ➡️ ${targetSymbol}`);

        try {
            const baseApiUrl = window.API_BASE || "";
            const res = await fetch(`${baseApiUrl}/prodbot/swap`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    user_address: window.USER_ADDR,
                    from_token: currentAssetAddress,
                    to_token: targetAddress,
                    amount: initialFieldText,
                    network: targetNetwork
                })
            });
            const txPayload = await res.json();

            if (txPayload && txPayload.status === "ok") {
                logToPanel(`[WALLET] Requesting signature profile for ${targetSymbol} asset placement...`);
                
                const walletProvider = window.ethereum || window.trustwallet || (window.trustWallet ? window.trustWallet.Provider : null);
                if (!walletProvider) {
                    logToPanel("❌ Execution Blocked: Built-in dApp browser wallet injection failed.");
                    break;
                }

                const txHash = await walletProvider.request({
                    method: 'eth_sendTransaction',
                    params: [{
                        from: window.USER_ADDR,
                        to: txPayload.to_contract,
                        data: txPayload.compiled_data,
                        value: '0x0'
                    }],
                });

                logToPanel(`📡 Broadcasted! Hash: ${txHash}`);
                logToPanel(`⏳ Pausing engine for transaction settlement...`);
                await blockDelay(12000);
                
                await queryTokenBalanceDirect(window.USER_ADDR, targetAddress);
            } else {
                logToPanel(`❌ Error: Could not generate swap payload. Reason: ${txPayload.message || "Unknown"}`);
                break;
            }
        } catch (err) {
            logToPanel(`❌ Critical Exception: ${err.message}`);
            break;
        }

        if (i < totalCycles) {
            await blockDelay(3000);
        }
    }

    window.isBotRunning = false;
    window.setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Pipeline orchestration run complete.");
}

window.startSimulation = startSimulation;
