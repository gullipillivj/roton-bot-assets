// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

// Thread state delays helper
const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

/**
 * On-chain token state resolver.
 * Hits the updated backend route to read active positions from getUserBalanceList tuple fields.
 */
async function getCurrentTokenState() {
    let state = { token: "0x55d398326f99059fF775485246999027B3197955", network: "bep20" };
    
    try {
        const activeIdentityVector = window.USER_ADDR || window.USER_ID;
        if (!activeIdentityVector) {
            return state;
        }

        const resp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: activeIdentityVector })
        });
        
        const data = await resp.json();
        if (data && data.status === "ok" && data.currenttoken) {
            state.token = data.currenttoken;
            state.network = data.network || "bep20";
        }
    } catch (err) {
        logToPanel(`[ON-CHAIN POSITION LOOKUP FAILED] Defaulting to USDT: ${err.message}`);
    }
    
    return state;
}
window.getCurrentTokenState = getCurrentTokenState;

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`\n--- [START] Cycle ${cycle}/${totalCycles} ---`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);

    // Grab the live balance string populated straight from your smart contract views
    let balanceInputBefore = document.getElementById("investBalance").value.toString().trim();
    if (!balanceInputBefore || balanceInputBefore === "0") {
        logToPanel("⚠️ [CYCLE ABORTED] Balance field is empty or zero.");
        return "STOP_ERROR";
    }

    const srcAsset = await getCurrentTokenState();
    const sourceTokenAddress = String(srcAsset.token).toLowerCase().trim();
    const sourceNetwork = String(srcAsset.network).toLowerCase().trim();
    logToPanel(`[PORTFOLIO ROOT] Active Position: ${srcAsset.token} (${sourceNetwork.toUpperCase()})`);

    // Fetch market data candidates directly from your coins.txt runtime array engine
    let marketData = [];
    try {
        if (typeof window.getTradeCandidates === "function") {
            marketData = await window.getTradeCandidates();
        } else {
            const filterResp = await fetch("/prodbot/filtertokens");
            const filterJson = await filterResp.json();
            if (filterJson && filterJson.status === "ok") {
                marketData = filterJson.short_term || filterJson;
            }
        }
    } catch (fErr) {
        logToPanel(`⚠️ [SCANNER DELAY] Unable to pull candidate tokens array: ${fErr.message}`);
        return "CONTINUE";
    }

    if (!Array.isArray(marketData) || marketData.length === 0) {
        logToPanel("⚠️ [SCANNER COLD] Zero validated tokens compiled. Skipping cycle...");
        return "SKIP";
    }

    logToPanel(`🔬 Inspecting ${marketData.length} trending candidates sorted by momentum...`);

    let executionSuccessful = false;
    let currentCandidateIndex = 0;

    // Loop through the token trend array elements sequentially
    while (currentCandidateIndex < marketData.length) {
        let candidate = marketData[currentCandidateIndex];
        
        let rawAddress = candidate.address_or_geckoid || candidate.address || "n/a";
        let targetedWinnerSymbol = String(candidate.token_symbol || candidate.symbol || candidate.name || "UNKNWN").toUpperCase().trim();
        let candidateMomentum = parseFloat(candidate.change_24h_percent || candidate["%profit"] || 0);
        let winnerNetworkType = String(candidate.tokentype || "bep20").toLowerCase().trim();

        // Safe Clean: Strip internet scrape prefixes safely (e.g. "bsc:0x...", "ethereum:0x...")
        if (String(rawAddress).includes(":")) {
            const structuralParts = String(rawAddress).split(":");
            rawAddress = structuralParts[1];
        }

        let candidateAddress = String(rawAddress).toLowerCase().trim();

        // Guard A: Skip empty slots or instances trying to buy the asset we already hold
        if (!candidateAddress || candidateAddress === "n/a" || candidateAddress === sourceTokenAddress) {
            currentCandidateIndex++;
            continue;
        }

        // Guard B: Instantly drop non-EVM address slots (like raw Solana entries)
        if (!candidateAddress.startsWith("0x") || candidateAddress.length !== 42) {
            currentCandidateIndex++;
            continue;
        }

        logToPanel(`🔍 Testing Candidate [${currentCandidateIndex + 1}/${marketData.length}]: ${targetedWinnerSymbol} (+${candidateMomentum.toFixed(2)}%)`);

        try {
            // --- PURE PRE-FLIGHT SIMULATION CHECK ---
            const simRes = await fetch("/prodbot/swap/simulateuniswap", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    router: "0x10ED43C718714eb63d5aA57B78B54704E256024E",
                    tokenIn: sourceTokenAddress,
                    tokenOut: candidateAddress,
                    amountIn: balanceInputBefore,
                    amountOutMin: "0"
                })
            });

            const simData = await simRes.json();
            
            // Validate simulation data fields safely BEFORE calling wallet actions
            if (!simData || simData.status !== "ok" || !simData.expected_output_wei || BigInt(simData.expected_output_wei) <= 0n) {
                logToPanel(`⚠️ [SIMULATION REJECTED] ${targetedWinnerSymbol} skipped: Shallow liquidity depths or execution revert.`);
                currentCandidateIndex++;
                continue; // Immediately jumps down to evaluate the next token item in your array
            }

            // Target token passes the backend simulation check -> Proceed to Section 2 for execution
            executionSuccessful = await executeVerifiedSwapPipeline(
                simData, candidateAddress, targetedWinnerSymbol, winnerNetworkType, sourceNetwork, sourceTokenAddress, balanceInputBefore
            );

            if (executionSuccessful) {
                break; // Break the candidate index tracking loop cleanly upon successful trade completion
            } else {
                currentCandidateIndex++;
            }

        } catch (err) {
            logToPanel(`⚠️ [PIPELINE EXCEPTION] Token ${targetedWinnerSymbol} execution dropped: ${err.message || err}`);
            currentCandidateIndex++;
        }
    }

    if (!executionSuccessful) {
        logToPanel("🔄 [CYCLE SKIPPED] Sequenced analysis complete. No verified swappable targets found in current token batch.");
        return "SKIP";
    }

    return "CONTINUE";
}
window.simulateCycle = simulateCycle;

async function executeVerifiedSwapPipeline(simData, candidateAddress, targetedWinnerSymbol, winnerNetworkType, sourceNetwork, sourceTokenAddress, balanceInputBefore) {
    logToPanel(`🔥 [PRE-FLIGHT PASSED] Expected: ${simData.expected_output_wei} Wei. Initiating live Web3 execution chain for: ${targetedWinnerSymbol}`);

    let activeTokenIn = sourceTokenAddress;
    let currentAllocationWei = balanceInputBefore;
    let data = null;

    const WBNB_ADDR = window.WBNB_TOKEN_ADDRESS ? window.WBNB_TOKEN_ADDRESS.toLowerCase().trim() : "0xbb4cdb9cbd36b01bd1cbaebf2de08d9173bc095c";
    const WETH_ADDR = "0xc02aaa39b223fe8d0a0e5c4f27ead9083c756cc2"; 

    // --- STEP 1: BRIDGING INTERMEDIARY CHECK ---
    try {
        if (sourceNetwork === "erc20" && activeTokenIn !== WETH_ADDR) {
            logToPanel(`⚠️ Custom Ethereum Asset. Selling into WETH bridge...`);
            data = await window.runethSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, WETH_ADDR, "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D");
            await handleTwoStepTxPipeline(data, activeTokenIn, WETH_ADDR, "erc20");
            if (typeof window.syncBalanceFromFund === "function") await window.syncBalanceFromFund();
            activeTokenIn = WETH_ADDR;
            currentAllocationWei = document.getElementById("investBalance").value.toString().trim();
        } else if (sourceNetwork === "bep20" && activeTokenIn !== WBNB_ADDR) {
            logToPanel(`⚠️ Custom Binance Asset. Selling into WBNB bridge...`);
            data = await window.runbscSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, WBNB_ADDR, "0x10ED43C718714eb63d5aA57B78B54704E256024E");
            await handleTwoStepTxPipeline(data, activeTokenIn, WBNB_ADDR, "bep20");
            if (typeof window.syncBalanceFromFund === "function") await window.syncBalanceFromFund();
            activeTokenIn = WBNB_ADDR;
            currentAllocationWei = document.getElementById("investBalance").value.toString().trim();
        }
    } catch (bridgeErr) {
        logToPanel(`⚠️ [BRIDGE ROUTE CRASH] Pipeline error during route parsing: ${bridgeErr.message}`);
        return false; 
    }

    // --- STEP 2: DYNAMIC ROUTER SWAP EXECUTION ---
    try {
        if (winnerNetworkType === "erc20") {
            data = await window.runethSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, candidateAddress, "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D");
        } else {
            data = await window.runbscSwapUniswap(window.USER_ADDR, currentAllocationWei, activeTokenIn, candidateAddress, "0x10ED43C718714eb63d5aA57B78B54704E256024E");
        }

        if (!data || data.status !== "ok" || !data.approval_packet || !data.swap_packet) {
            logToPanel(`⚠️ [ROUTER PAYLOAD ERROR] Serialization dropped for ${targetedWinnerSymbol}.`);
            return false;
        }

        // Prompt wallet token permissions step safely with explicit rejection catch blocks
        logToPanel(`📢 [WALLET PROMPT] Authorizing token spend allocation...`);
        await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{ from: window.USER_ADDR, to: data.approval_packet.to_contract, data: data.approval_packet.compiled_data, value: "0x0" }]
        });
        await blockDelay(10000);

        // Dispatch execution transaction payload parameters
        logToPanel(`📢 [WALLET PROMPT] Confirming asset exchange processing...`);
        const txHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{ from: window.USER_ADDR, to: data.swap_packet.to_contract, data: data.swap_packet.compiled_data, value: "0x0" }]
        });
        logToPanel(`✅ Exchange Transaction Dispatched! Hash: ${txHash}`);

        // Polling cycle logic checking on-chain data transitions
        let settledOnChain = false;
        logToPanel(`⏳ Awaiting block state confirmation rules...`);
        
        for (let i = 1; i <= 60; i++) {
            await blockDelay(8000);
            if (typeof window.syncBalanceFromFund === "function") {
                await window.syncBalanceFromFund();
            }
            let checkCurrentFormValue = document.getElementById("investBalance").value.toString().trim();
            if (checkCurrentFormValue !== balanceInputBefore && checkCurrentFormValue !== "0") {
                settledOnChain = true;
                break;
            }
        }

        if (!settledOnChain) {
            logToPanel(`🛑 [TIMEOUT] On-chain indices remained static for ${targetedWinnerSymbol}.`);
            return false;
        }

        // Sync storage tracking maps on-chain
        await updateTokenRecord(activeTokenIn, candidateAddress, winnerNetworkType);
        return true;

    } catch (txRuntimeErr) {
        logToPanel(`❌ [USER REJECTION OR EVM EXCEPTION] Swap pipeline interrupted: ${txRuntimeErr.message || txRuntimeErr}`);
        return false;
    }
}

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;
    logToPanel(`🚀 [BOT INITIATED] Processing up to ${totalCycles} execution cycles.`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (signal === "STOP_ERROR") {
            logToPanel("🛑 State mismatch runtime alert. Aborting framework loops.");
            break;
        }
        if (signal === "SKIP") {
            await blockDelay(1000); 
            continue;
        }
        if (i < totalCycles) {
            setStatus("WAITING");
            await blockDelay(10000);
        }
    }
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing complete.");
}
window.startSimulation = startSimulation;
