// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Asset scan matrix paused by user.");
}
window.stopBot = stopBot;

// 🔍 ZERO-GAS BROWNIE SIMULATOR CORE (CALLS CALCULATE_ON_CHAIN_SLIPPAGE)
// ==================================================================
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`🚀 [ROUTING DIAGNOSTIC] Accessing read-only backend slippage route...`);

    const PANCAKE_ROUTER = "0x10ED43C718714eb63d5aA57B78B54704E256024E";
    const BASE_TOKEN_IN = "0x55d398326f99059fF775485246999027B3197955"; // USDT Base Contract

    for (let i = 1; i <= totalCycles; i++) {
        if (window.isBotRunning === false) {
            logToPanel("🛑 Loop halted by request.");
            break;
        }

        logToPanel(`\n--- [START] Simulation Cycle ${i}/${totalCycles} ---`);
        setStatus(`RUNNING ${i}/${totalCycles}`);

        // Read layout balance field to model standard price impact calculations
        let balanceInputStr = document.getElementById("investBalance") ? document.getElementById("investBalance").value.toString().trim() : "10";
        if (!balanceInputStr || balanceInputStr === "0") balanceInputStr = "10";
        
        // Pad out simulated amount to standard 18 decimal places (e.g. 10 -> 10000000000000000000 Wei)
        let simulatedAmountInWei = balanceInputStr.includes("0000") ? balanceInputStr : balanceInputStr + "000000000000000000";

        let marketData = [];
        try {
            if (typeof window.getTradeCandidates === "function") {
                marketData = await window.getTradeCandidates();
            } else {
                logToPanel("⚠️ window.getTradeCandidates registry access function is missing.");
                continue;
            }
        } catch (fErr) {
            logToPanel(`❌ Local Array Ingestion Error: ${fErr.message}`);
            continue;
        }

        if (!Array.isArray(marketData) || marketData.length === 0) {
            logToPanel("❌ Filtered token list data is completely empty.");
            continue;
        }

        logToPanel(`🎯 Dispatching ${marketData.length} tokens to read-only validator endpoint...`);

        for (let index = 0; index < marketData.length; index++) {
            let candidate = marketData[index];
            let symbolLabel = candidate.symbol || "UNKNOWN";
            let targetAddress = candidate.address || "MISSING";
            let tokenNetwork = (candidate.tokentype || "bep20").toLowerCase();

            if (targetAddress === "MISSING" || !targetAddress.startsWith("0x")) {
                continue;
            }

            // Firewall: Restrict token assets strictly to EVM BSC mapping criteria
            if (tokenNetwork !== "bep20" && tokenNetwork !== "bsc") {
                logToPanel(`   🔹 [SKIP] ${symbolLabel} (${tokenNetwork.toUpperCase()} asset bypassed on PancakeSwap router)`);
                continue;
            }

            try {
                // Call the pure view simulation route on your Flask server
                let response = await fetch("/prodbot/swap/simulateuniswap", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        router: PANCAKE_ROUTER,
                        tokenIn: BASE_TOKEN_IN,
                        tokenOut: targetAddress,
                        amountIn: simulatedAmountInWei,
                        amountOutMin: 0
                    })
                });

                let resData = await response.json();

                if (resData.status === "ok") {
                    let expectedOutput = resData.expected_output_wei || "0";
                    let slippageCost = resData.slippage_cost_pct || "0.0";

                    // If getAmountsOut returned 0, no pool pair exists or liquidity is dead
                    if (expectedOutput === "0" || expectedOutput === 0) {
                        logToPanel(`   ❌ [NOT SWAPPABLE] ${symbolLabel} (${targetAddress.slice(0, 8)}...): Liquidity validation check returned 0 or dead pair routing paths.`);
                    } else {
                        logToPanel(`   🟩 [SWAPPABLE] ${symbolLabel} | Slippage Rate: ${slippageCost}% | Expected Out: ${expectedOutput} Wei`);

                        // ==================================================================
                        // 📍 PLACEMENT LOCATION: GAS LOOKUP, DIRECT TRANSACTING & AUTO-SYNC
                        // ==================================================================
                        try {
                            logToPanel(`   ⛽ Fetching active network metrics via window.ethereum...`);
                            const hexGasPrice = await window.ethereum.request({ method: "eth_gasPrice" });
                            const currentGasPriceWei = BigInt(hexGasPrice);
                            
                            // Establish token network properties for gas check criteria defaults
                            let estimatedGasLimit = 150000n;
                            if (tokenNetwork === "erc20" || tokenNetwork === "ethereum") {
                                estimatedGasLimit = 450000n;
                            }
                            
                            const transactionGasCostWei = currentGasPriceWei * estimatedGasLimit;
                            logToPanel(`   ⛽ [GAS METRICS] Dynamic path gas fee cost: ${transactionGasCostWei.toString()} Wei`);

                            let transactionOutcome = null;

                            // Route directly to your pancake_prod.txt dispatch pipelines based on asset token network type
                            if (tokenNetwork === "erc20" || tokenNetwork === "ethereum") {
                                logToPanel(`   ⚡ [ETH PIPELINE] Forwarding token payload to ERC20 swap pipeline...`);
                                if (typeof window.runethSwapUniswap === "function") {
                                    transactionOutcome = await window.runethSwapUniswap(
                                        window.USER_ADDR, simulatedAmountInWei, BASE_TOKEN_IN, targetAddress, PANCAKE_ROUTER
                                    );
                                } else {
                                    logToPanel("   ❌ window.runethSwapUniswap execution route handle is missing.");
                                }
                            } else {
                                logToPanel(`   ⚡ [BSC PIPELINE] Forwarding token payload to BEP20 swap pipeline...`);
                                if (typeof window.runbscSwapUniswap === "function") {
                                    transactionOutcome = await window.runbscSwapUniswap(
                                        window.USER_ADDR, simulatedAmountInWei, BASE_TOKEN_IN, targetAddress, PANCAKE_ROUTER
                                    );
                                } else {
                                    logToPanel("   ❌ window.runbscSwapUniswap execution route handle is missing.");
                                }
                            }

                            // If broadcast succeeded, await settlement via syncBalanceFromFund checks then refresh dashboard values
                            if (transactionOutcome && transactionOutcome.status === "ok") {
                                logToPanel(`   🚀 [BROADCAST SUCCESS] Transaction written to blockchain indices. Awaiting ledger update...`);
                                
                                let startBox = document.getElementById("startBalance");
                                let previousWeiBalance = startBox ? startBox.value.toString().trim() : "0";
                                let maxRetries = 20;
                                
                                for (let attempt = 1; attempt <= maxRetries; attempt++) {
                                    if (window.isBotRunning === false) break;
                                    
                                    if (typeof window.syncBalanceFromFund === "function") {
                                        let currentRawWei = await window.syncBalanceFromFund();
                                        if (BigInt(currentRawWei || "0") !== BigInt(previousWeiBalance)) {
                                            logToPanel(`   🎉 [SETTLEMENT CONFIRMED] Ledger shift detected! Fresh Balance: ${currentRawWei} Wei.`);
                                            
                                            // Trigger visualization updates on chart layout elements
                                            let usdValBox = document.getElementById("usd");
                                            let liveUsdValue = usdValBox ? parseFloat(usdValBox.value) || 0 : 0;
                                            if (typeof window.updateChart === "function") {
                                                window.updateChart(liveUsdValue, liveUsdValue - (parseFloat(window.InitialBaseCapital) || 0));
}
break;
}
}
logToPanel(      🔹 Check [${attempt}/${maxRetries}]: State unchanged. Waiting 6s...);
await blockDelay(6000);
}
} else {
logToPanel(   ❌ [ROUTING REJECTION] Live execution rejected payload: ${transactionOutcome ? transactionOutcome.message : "User denied signature or RPC failure."});
}
} catch (walletErr) {
logToPanel(   ❌ [WALLET ERROR] Live execution block failed to complete: ${walletErr.message || walletErr});
}
// ==================================================================
}
} else {
logToPanel(   ❌ [BACKEND REJECTION] ${symbolLabel}: ${resData.message});
}
} catch (err) {
logToPanel(   ❌ [ROUTE TIMEOUT] Failed calling simulation path for ${symbolLabel}: ${err.message});
}
// Short 150ms structural delay between checks to avoid spamming the local RPC node
await blockDelay(150);
}
logToPanel(✅ Cycle ${i}/${totalCycles} analysis finalized.);
setStatus("WAITING");
await blockDelay(3000);
}
logToPanel(\n=== 📊 [FINISHED] Pre-flight structural scan operations complete. ==);
return "STOP_ERROR";
}
window.simulateCycle = simulateCycle;
// Master controller dashboard trigger logic mapping
async function startSimulation(totalCycles) {
if (window.isBotRunning) return;
window.isBotRunning = true;
logToPanel(🚀 [LAUNCHING] Read-only background asset validation sequence engaged...);
await simulateCycle(1, 10, 0, 0);
window.isBotRunning = false;
setStatus("STOPPED");
logToPanel("[BOT FINISHED] Run parameters finalized.");
}
window.startSimulation = startSimulation;
