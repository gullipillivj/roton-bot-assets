// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Asset scan matrix paused by user.");
}
window.stopBot = stopBot;
// Direct provider account hook replaces fragile URL strings entirely
async function getLiveWalletAddress() {
    if (window.ethereum) {
        try {
            const accounts = await window.ethereum.request({ method: 'eth_accounts' });
            if (accounts && accounts.length > 0) {
                return accounts.toLowerCase().trim();
            }
        } catch (err) {
            console.error("Error reading accounts from window.ethereum:", err);
        }
    }
    return window.USER_ADDR ? window.USER_ADDR.toLowerCase().trim() : null;
}

// Helper to dynamically locate which token currently holds a balance inside the contract vault
async function getActiveLedgerAsset(userAddress) {
    try {
        logToPanel(`🔎 Querying ledger data for address: ${userAddress}...`);
        const res = await fetch("/prodbot/ledger/balances", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: userAddress })
        });
        const ledgerData = await res.json();

        if (ledgerData && ledgerData.status === "ok" && Array.isArray(ledgerData.tokens)) {
            for (let i = 0; i < ledgerData.tokens.length; i++) {
                const currentBalWei = BigInt(ledgerData.balances[i] || "0");
                if (currentBalWei > 0n) {
                    logToPanel(`🎯 Active Capital Found: ${ledgerData.tokens[i]} | Balance: ${currentBalWei.toString()} Wei`);
                    return {
                        address: ledgerData.tokens[i].toLowerCase().trim(),
                        balanceWei: currentBalWei.toString()
                    };
                }
            }
            if (ledgerData.tokens.length > 0) {
                return { address: ledgerData.tokens.toLowerCase().trim(), balanceWei: "0" };
            }
        }
    } catch (err) {
        logToPanel(`⚠️ Failed checking multi-asset ledger array: ${err.message}`);
    }
    return { address: "0x55d398326f99059ff775485246999027b3197955", balanceWei: "0" };
}
