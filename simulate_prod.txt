logToPanel("simulate_prod.txt (Container 2) loaded successfully with RR-Circulation Engine.");

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

async function startSimulation(totalCycles = 10) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    logToPanel(`[ENGINE INITIALIZED] Executing conditional profit analysis for ${totalCycles} iterations.`);
    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";
    
    // Define the custom ecosystem tracking parameters explicitly
    const RR_BNB_ADDR = "0xba67186fd454D0256a0D3744783538bE08C58feD";
    const RR_ETH_ADDR = "0x8A0cAC8E7F6305cE6EeB585005919290F78808f9";

    window.setStatus("SYNCHRONIZING");
    let currentAssetAddress = USDT_ADDRESS;
    let currentAssetSymbol = "USDT";

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        logToPanel(`\n[INFO] — Cycle ${i}/${totalCycles} starting...`);
        window.setStatus(`RUNNING cycle ${i}/${totalCycles}`);

        let initialFieldText = await window.updateUIBalanceControl(window.USER_ADDR, currentAssetAddress);
        logToPanel(`[STATUS] Active Internal Balance: ${initialFieldText} base units`);

        if (BigInt(initialFieldText) <= 0n) {
            logToPanel(`❌ Pipeline Halted: Display balance textbox evaluates to 0.`);
            break;
        }

        let targetSymbol = "USDT";
        let targetAddress = USDT_ADDRESS;
        let targetNetwork = "bep20";
        let executeMarketSwap = false;

        // Final loop cycle forcefully returns all active token allocations back to USDT safety stability structures
        if (i !== totalCycles) {
            // 📊 1. FETCH FULL 30-COIN MOMENTUM ARRAY MATRIX
            const buckets = await window.categorizeCoins();
            const fullPool = [...buckets.long_term, ...buckets.mid_term, ...buckets.short_term];

            let highestProfitableCandidate = null;
            const estimatedGasOverheadUSD = 0.25; // Base execution friction cost on BSC mainnet

            // 💸 2. EVALUATE CONDITIONAL MARKET PROFITABILITY CEILINGS
            for (let candidate of fullPool) {
                if (candidate.symbol === "BNB" || candidate.symbol === "ETH" || candidate.symbol === "USDT") {
                    continue; // Skip gas markers
                }

                const trendAccelerationFactor = candidate.change24h > 0 ? (candidate.change24h * 0.05) : 0.5;
                const floatBalance = parseFloat(initialFieldText) / 1e18; 
                const projectedGrossProfitUSD = floatBalance * (trendAccelerationFactor / 100);

                // Check if candidate net gains completely scale clear past overhead costs
                if (projectedGrossProfitUSD > estimatedGasOverheadUSD) {
                    highestProfitableCandidate = candidate;
                    break; 
                }
            }

            // 🔄 3. ROUTING ARBITRATION SELECTION GATEWAY
            if (highestProfitableCandidate) {
                // Scenario A: A market token passed the financial tests cleanly
                targetSymbol = highestProfitableCandidate.symbol;
                targetAddress = highestProfitableCandidate.tokenAddress;
                targetNetwork = highestProfitableCandidate.network;
                executeMarketSwap = true;
                logToPanel(`🎯 Market Yield Verified: Selecting ${targetSymbol} to execute arbitrage swap.`);
            } else {
                // Scenario B: No market coin crossed profitability boundaries. Divert routing parameters to volume generation!
                executeMarketSwap = true; // Still initialize transaction routines, but use RR contract allocations instead
                
                // Toggle between your asset pairs dynamically based on the loop cycle steps to generate volume back and forth
                if (currentAssetAddress.toLowerCase() === RR_BNB_ADDR.toLowerCase()) {
                    targetSymbol = "RR-ETH";
                    targetAddress = RR_ETH_ADDR;
                    logToPanel(`🔁 [ECOSYSTEM INJECTION] Market yields low. Routing volume from RR-BNB ➡️ RR-ETH for circulation.`);
                } else {
                    targetSymbol = "RR-BNB";
                    targetAddress = RR_BNB_ADDR;
                    logToPanel(`🔁 [ECOSYSTEM INJECTION] Market yields low. Routing volume ➡️ ${targetSymbol} for circulation.`);
                }
            }
        }

        // Safety gate checks to prevent duplicate node executions
        if (targetAddress.toLowerCase() === currentAssetAddress.toLowerCase()) {
            logToPanel(`[INFO] Asset sequence already aligned to target. Skipping loop step.`);
            await blockDelay(3000);
            continue;
        }

        logToPanel(`[ROUTE] Preparing transaction pipeline: Swap ${currentAssetSymbol} ➡️ ${targetSymbol}`);

        try {
            const baseApiUrl = window.API_BASE || "";
            const res = await fetch(`${baseApiUrl}/prodbot/swap`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    user_address: window.USER_ADDR,
                    from_token: currentAssetAddress,
                    to_token: targetAddress,
                    amount: initialFieldText,
                    network: targetNetwork
                })
            });
            const txPayload = await res.json();

            if (txPayload && txPayload.status === "ok") {
                logToPanel(`[METAMASK] Requesting signature profile for ${targetSymbol} asset placement...`);
                const txHash = await window.ethereum.request({
                    method: 'eth_sendTransaction',
                    params: [{
                        from: window.USER_ADDR,
                        to: txPayload.to_contract,
                        data: txPayload.compiled_data,
                        value: '0x0'
                    }],
                });

                logToPanel(`📡 Broadcasted! Hash: ${txHash}`);
                logToPanel(`⏳ Pausing engine for transaction settlement...`);

                // Give 12 seconds for network transactions to fully settle on-chain
                await blockDelay(12000);
                
                await window.updateUIBalanceControl(window.USER_ADDR, targetAddress);
                currentAssetAddress = targetAddress;
                currentAssetSymbol = targetSymbol;
                logToPanel(`✅ Cycle step complete. Advancing loop pipeline records.`);

            } else {
                logToPanel(`❌ Error: Could not generate swap payload. Reason: ${txPayload.message || "Unknown"}`);
                break;
            }
        } catch (err) {
            logToPanel(`❌ Critical Exception: ${err.message}`);
            break;
        }

        if (i < totalCycles) {
            await blockDelay(3000);
        }
    }

    window.isBotRunning = false;
    window.setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Pipeline orchestration run complete.");
}
window.startSimulation = startSimulation;
