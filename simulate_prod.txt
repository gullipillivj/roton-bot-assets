logToPanel("simulate_prod.txt loaded (Loop Engine Ready)");

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BTC"; 
    } catch (e) { return "BTC"; }
}

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`[INFO] — Cycle ${cycle}/${totalCycles} started`);
    setStatus(`RUNNING cycle ${cycle}/${totalCycles}`);

    // Refresh balance
    const syncedBalance = await syncBalanceFromFund();
    if (syncedBalance === null) {
        logToPanel("[WARN] Could not sync on-chain balance; skipping cycle");
        return "CONTINUE";
    }

    let tradingBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    const startBalance = parseFloat(document.getElementById("startBalance").value) || 0;

    if (tradingBalance <= 0) {
        logToPanel("[INFO] Trading balance is zero; skipping cycle");
        return "SKIP";
    }

    const allocationUSD = tradingBalance / Math.max(1, totalCycles - (cycle - 1));

    // Pick coin and fetch dynamic address
    const coinBase = await pickRandomCoin();
    const targetSymbol = `${coinBase}USDT`;
    
    // Fixed: get address mapping directly
    const tokenAddress = await window.getTokenAddress(targetSymbol);
    if (!tokenAddress) {
        logToPanel(`[WARN] Token address for ${coinBase} not found; skipping cycle`);
        return "CONTINUE";
    }

    const stats = await window.getBinanceStats(targetSymbol);
    if (!stats || !stats.lastPrice) {
        logToPanel("[WARN] Market stats unavailable; skipping cycle");
        return "CONTINUE";
    }

    // Fee calculation
    let feeCalc;
    try {
        feeCalc = await window.calculateTradeCost(targetSymbol, allocationUSD, "bsc");
    } catch (feeErr) {
        logToPanel(`[WARN] Fee check failed: ${feeErr.message} — skipping trade`);
        return "CONTINUE";
    }

    const totalFees = feeCalc.totalCost || 0;
    
    // Execute Swap safely
    const beforeBalance = window.SolidarityFundBalance || tradingBalance;
    try {
        // Fallback check to prevent calling missing arguments
        const userKey = typeof pkvalue !== 'undefined' ? pkvalue : ""; 
        
        // Execute swap wrapper route
        const swapResp = await window.runSwap(window.USER_ADDR, allocationUSD, userKey, tokenAddress);
        if (swapResp && swapResp.tx_hash) {
            logToPanel(`🎉 Swap executed — TxHash=${swapResp.tx_hash}`);
        }

        await syncBalanceFromFund();
        const afterBalance = window.SolidarityFundBalance || beforeBalance;
        const profitDelta = afterBalance - beforeBalance;

        window.cumulativePnL = (window.cumulativePnL || 0) + profitDelta;
        updateChart(afterBalance, window.cumulativePnL);

    } catch (err) {
        logToPanel(`[ERROR] Swap execution failed: ${err.message}`);
        return "CONTINUE";
    }

    const totalChangePct = (window.cumulativePnL / tradingBalance) * 100;
    if (totalChangePct >= profitTargetPct) return "STOP_PROFIT";
    if (totalChangePct <= -stopLossPct) return "STOP_LOSS";

    return "CONTINUE";
}
window.simulateCycle = simulateCycle;


async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    logToPanel(`[BOT INITIALIZATION] Booting automation pipeline script engine for ${totalCycles} trade iterations.`);

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        const stateSignal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (stateSignal === "STOP_PROFIT" || stateSignal === "STOP_LOSS" || stateSignal === "SKIP") break;

        if (i < totalCycles) {
            logToPanel("[COOLDOWN] Cooling down execution pipeline contexts before scanning next asset target...");
            for (let countdown = 10; countdown > 0; countdown--) {
                if (!window.isBotRunning) break;
                if (countdown % 5 === 0) setStatus(`WAITING (${countdown}s)`);
                await blockDelay(1000);
            }
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Sequence pipeline processing completed successfully.");
}

window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
