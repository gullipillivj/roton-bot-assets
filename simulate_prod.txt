// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Trading halted by user.");
}
window.stopBot = stopBot;

async function updateTokenRecord(lastToken, currentToken, networkType = "bep20") {
    try {
        const userid = window.USER_ID || localStorage.getItem("userid");
        if (!userid) return false;
        const resp = await fetch("/prodbot/tokens/update", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ 
                userid: userid, 
                lasttoken: lastToken, 
                currenttoken: currentToken,
                network: networkType
            })
        });
        const data = await resp.json();
        return !!(data && data.message);
    } catch (err) {
        return false;
    }
}

async function getCurrentTokenState() {
    let state = { token: "0x55d398326f99059fF775485246999027B3197955", network: "bep20" };
    try {
        const resp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: window.USER_ID })
        });
        const data = await resp.json();
        if (data && data.currenttoken) {
            state.token = data.currenttoken;
            state.network = data.network || "bep20";
        }
    } catch (err) {}
    return state;
}

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`\n=========================================`);
    logToPanel(`🔄 [START] Engine Cycle ${cycle}/${totalCycles}`);
    logToPanel(`=========================================`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);

    let tradingBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    if (tradingBalance <= 0) return "STOP_ERROR"; 

    const allocationWei = document.getElementById("startBalance").value.toString().trim();
    const srcAsset = await getCurrentTokenState();

    // 1 & 2. Pull down filtered market assets array from coins.txt context
    let candidates = [];
    if (typeof window.getTradeCandidates === "function") {
        candidates = await window.getTradeCandidates();
    }

    if (!candidates || candidates.length === 0) {
        logToPanel("⚠️ [CYCLE ALERT] Candidate token array registry is completely empty.");
        return "SKIP";
    }

    // Dynamic router resolution parameters mapping
    let targetRouter = "0x10ED43C718714eb63d5aA57B78B54704E256024E"; // Pancake V2
    if (["erc20", "eth"].includes(srcAsset.network.toLowerCase())) {
        targetRouter = "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D"; // Uniswap V2 ETH
    }

    // Force strict 10-token bounding limits for the batch cycle sweep
    let tokensToScan = Math.min(10, candidates.length);
    let anySwapTriggered = false;

    // -------------------------------------------------------------
    // 🌟 SUB-LOOP: SEQUENTIALLY SCAN TOKENS 1 TO 10
    // -------------------------------------------------------------
    for (let t = 0; t < tokensToScan; t++) {
        if (!window.isBotRunning) break;

        let token = candidates[t];
        let targetAddress = token.address || token.address_or_geckoid;

        logToPanel(`\n🔍 [TOKEN ${t + 1}/${tokensToScan}] Testing ${token.symbol} (${targetAddress})...`);

        // STAGE 3: Pass token to the zero-overhead backend simulation pipe
        try {
            const simResult = await window.runsimulateSwapUniswap(window.USER_ADDR, allocationWei, srcAsset.token, targetAddress, targetRouter);

            // If simulation isn't fully successful, skip this specific token instantly
            if (!simResult || simResult.status !== "ok") {
                logToPanel(`❌ [SIM REVERT] ${token.symbol} failed validation. Skipping...`);
                continue; 
            }

            logToPanel(`🎯 [SIM SUCCESS] Path confirmed viable for ${token.symbol}!`);
            logToPanel(`🚀 [STAGE 4] Launching pass-through payload compile to fire popup...`);

            // STAGE 4: Invoke linear pass-through routes to generate bytecode data maps
            let liveSwapResult = null;
            if (["erc20", "eth"].includes(srcAsset.network.toLowerCase())) {
                liveSwapResult = await window.runethSwapUniswap(window.USER_ADDR, allocationWei, srcAsset.token, targetAddress, targetRouter);
            } else {
                liveSwapResult = await window.runbscSwapUniswap(window.USER_ADDR, allocationWei, srcAsset.token, targetAddress, targetRouter);
            }

            if (liveSwapResult && liveSwapResult.status === "ok") {
                logToPanel("[INFO] Instantly pushing bytecode array directly to extension browser proxy...");
                
                // 🌟 DIRECT UNMUTATED POPUP CALL TRIGGER
                const txHash = await window.ethereum.request({
                    method: "eth_sendTransaction",
                    params: [{
                        from: window.USER_ADDR,
                        to: liveSwapResult.to_contract,
                        data: liveSwapResult.compiled_data,
                        value: "0x0"
                    }]
                });

                logToPanel(`✅ Swap Executed! Tracking Hash: ${txHash}`);
                anySwapTriggered = true;

                // Lock thread execution while the transaction hits on-chain block states
                await blockDelay(15000);
                if (typeof syncBalanceFromFund === "function") await syncBalanceFromFund();
                await updateTokenRecord(srcAsset.token, targetAddress, srcAsset.network);
                
                // Exit sub-loop early since a profitable swap has been successfully committed
                break; 
            }

        } catch (err) {
            logToPanel(`⚠️ Error parsing token iteration step: ${err.message}`);
            continue;
        }
    }

    // If none of the 10 tokens pass simulation checks, skip the entire cycle layout cleanly
    if (!anySwapTriggered) {
        logToPanel(`\n⚠️ [CYCLE FINISHED] All ${tokensToScan} tokens failed simulation matrix constraints.`);
        logToPanel(`⏭️ Skipping Cycle ${cycle} completely to secure wallet assets.`);
        return "SKIP";
    }

    return "CONTINUE";
}

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    logToPanel(`🚀 [BOT INITIATED] Processing up to ${totalCycles} execution cycles.`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        
        if (signal === "STOP_ERROR") {
            logToPanel("🛑 Runtime parameter breakdown. Aborting master framework loops.");
            break;
        }
        
        if (i < totalCycles) {
            setStatus("WAITING");
            // Standard cooling interval delay between master cycle runs
            await blockDelay(10000);
        }
    }
    
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing complete.");
}

window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
