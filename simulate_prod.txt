function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) { statusLabel.textContent = msg; } 
    else { logToPanel(`[STATUS] ${msg}`); }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    window.CURRENT_WORKER_ID++;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Cross-chain batch sequence forcefully aborted.");
}
window.stopBot = stopBot;

async function getLiveWalletAddress() {
    if (window.ethereum) {
        try {
            const accounts = await window.ethereum.request({ method: 'eth_accounts' });
            if (accounts && accounts.length > 0) {
                return accounts[0].toLowerCase().trim();
            }
        } catch (err) {
            console.error("Error reading accounts from window.ethereum:", err);
        }
    }
    return window.USER_ADDR ? window.USER_ADDR.toLowerCase().trim() : null;
}

async function getActiveLedgerAsset(userAddress) {
    try {
        logToPanel(`🔎 Querying ledger data for address: ${userAddress}...`);
        const res = await fetch("/prodbot/ledger/balances", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: userAddress })
        });
        const ledgerData = await res.json();

        if (ledgerData && ledgerData.status === "ok" && Array.isArray(ledgerData.tokens)) {
            for (let i = 0; i < ledgerData.tokens.length; i++) {
                const currentBalWei = BigInt(ledgerData.balances[i] || "0");
                if (currentBalWei > 0n) {
                    logToPanel(`🎯 Active Capital Found: ${ledgerData.tokens[i]} | Balance: ${currentBalWei.toString()} Wei`);
                    return {
                        address: ledgerData.tokens[i].toLowerCase().trim(),
                        balanceWei: currentBalWei.toString()
                    };
                }
            }
            if (ledgerData.tokens.length > 0) {
                return { address: ledgerData.tokens[0].toLowerCase().trim(), balanceWei: "0" };
            }
        }
    } catch (err) {
        logToPanel(`⚠️ Failed checking multi-asset ledger array: ${err.message}`);
    }
    return { address: "0x55d398326f99059ff775485246999027b3197955", balanceWei: "0" };
}

}async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct, designatedWorkerId) {
    const PANCAKE_ROUTER = "0x10ED43C718714eb63d5aA57B78B54704E256024E";
    const UNISWAP_ROUTER = "0x7a250d5630b4cf539739df2c5dacb4c659f2488d";
    const STABLE_USDT_BSC = "0x55d398326ff99059ff775485246999027b3197955";

    logToPanel("🚀 Starting Cross-Network Virtual Optimization Loop...");
    
    const currentActiveWallet = await getLiveWalletAddress();
    if (!currentActiveWallet) {
        logToPanel("❌ [ABORTED] No wallet connected.");
        return "STOP_ERROR";
    }

    const initialAssetSnapshot = await getActiveLedgerAsset(currentActiveWallet);
    let currentVirtualToken = initialAssetSnapshot.address.toLowerCase().trim();
    
    let balanceInputStr = document.getElementById("investBalance") ? document.getElementById("investBalance").value.toString().trim() : "0";
    let virtualAmountInWei = BigInt(balanceInputStr === "0" || !balanceInputStr ? initialAssetSnapshot.balanceWei : balanceInputStr);
    
    const REAL_STARTING_TOKEN = currentVirtualToken;
    let customBridgesTouched = false;
    let finalTargetNetwork = "bep20"; // Tracks if final trade should run on BSC or ETH

    // 🟢 PHASE 1: Run 10 loops in memory with dynamic router assignment
    for (let i = 1; i <= totalCycles; i++) {
        if (window.isBotRunning === false || window.CURRENT_WORKER_ID !== designatedWorkerId) {
            logToPanel("🛑 [THREAD MUTEX] Loop processing stopped.");
            break;
        }

        logToPanel(`\n--- [VIRTUAL CYCLE] Hop ${i}/${totalCycles} ---`);
        setStatus(`SIMULATING ${i}/${totalCycles}`);
        logToPanel(`📥 Current Virtual Token: ${currentVirtualToken} | Vol: ${virtualAmountInWei.toString()} Wei`);

        const universalPillars = getUniversalPillarsMesh();
        const customBridges = getCustomBridgesMesh();
        let marketData = [];

        // Dynamic routing rules based on cycle indices
        if (i === 3 || i === 7 || !customBridgesTouched) {
            logToPanel("🛡️ Enforcing custom bridge routing protocol condition rules...");
            marketData = customBridges;
            customBridgesTouched = true;
        } else if (i === totalCycles) {
            logToPanel("🔒 Capital Preservation: Heading to stable pool destination...");
            marketData = [{ symbol: "USDT-BSC", address: STABLE_USDT_BSC, tokentype: "bep20" }];
        } else {
            try {
                if (typeof window.getTradeCandidates === "function") {
                    marketData = await window.getTradeCandidates();
                }
            } catch (fErr) {}
            marketData = marketData.concat(universalPillars);
        }

        let bestHopCandidate = null;
        let highestSimulatedOutput = 0n;

        for (let index = 0; index < marketData.length; index++) {
            let candidate = marketData[index];
            let targetAddress = candidate.address || "MISSING";
            if (targetAddress === "MISSING" || targetAddress.toLowerCase().trim() === currentVirtualToken) continue;

            let tokenNetwork = (candidate.tokentype || "bep20").toLowerCase();
            
            // 🟢 FIXED: Swap router address dynamically on every hop based on target token network type
            let activeRouterForSimulation = PANCAKE_ROUTER;
            if (tokenNetwork === "erc20" || tokenNetwork === "ethereum") {
                activeRouterForSimulation = UNISWAP_ROUTER;
            }

            try {
                let response = await fetch("/prodbot/swap/simulateuniswap", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        router: activeRouterForSimulation,
                        tokenIn: currentVirtualToken,
                        tokenOut: targetAddress,
                        amountIn: virtualAmountInWei.toString(),
                        amountOutMin: 0
                    })
                });
                let resData = await response.json();
                if (resData.status === "ok") {
                    let outWei = BigInt(resData.expected_output_wei || "0");
                    if (outWei > highestSimulatedOutput) {
                        highestSimulatedOutput = outWei;
                        bestHopCandidate = candidate;
                    }
                }
            } catch (err) {}
        }

        if (bestHopCandidate && highestSimulatedOutput > 0n) {
            logToPanel(`📈 Virtual Step Match: -> ${bestHopCandidate.symbol} [${(bestHopCandidate.tokentype || "bep20").toUpperCase()}]`);
            currentVirtualToken = bestHopCandidate.address.toLowerCase().trim();
            virtualAmountInWei = highestSimulatedOutput;
            finalTargetNetwork = (bestHopCandidate.tokentype || "bep20").toLowerCase();
        } else {
            logToPanel("ℹ️ Stagnant variance detected. Shifting to fallback stable pool anchor.");
            currentVirtualToken = STABLE_USDT_BSC;
            finalTargetNetwork = "bep20";
        }

        logToPanel("⏱️ Simulating chain validation weights... Awaiting propagation lapse delay...");
        await blockDelay(3000);
    }

    // 🟢 PHASE 2: Execute exactly ONE matching cross-network transaction at loop settlement
    if (window.isBotRunning && window.CURRENT_WORKER_ID !== designatedWorkerId) {
        logToPanel(`\n🔥 [VIRTUAL MATRIX COMPLETE] Dispatching ONLY ONE actual on-chain transaction request!`);
        logToPanel(`🗺️ Ultimate Path: ${REAL_STARTING_TOKEN} -> ${currentVirtualToken}`);

        if (REAL_STARTING_TOKEN === currentVirtualToken) {
            logToPanel("⚠️ Trade optimization identical to starting asset. Real-world gas preservation triggered.");
        } else {
            setStatus("EXECUTING FINAL TRADE");
            let txResult = null;

            // 🟢 FIXED: Selects the appropriate frontend script function based on final network results
            if (finalTargetNetwork === "erc20" || finalTargetNetwork === "ethereum") {
                logToPanel("⛓️ Network Target: Ethereum Mainnet. Directing request to Uniswap Pipeline...");
                txResult = await window.runethSwapUniswap(currentActiveWallet, balanceInputStr, REAL_STARTING_TOKEN, currentVirtualToken, UNISWAP_ROUTER);
            } else {
                logToPanel("⛓️ Network Target: Binance Smart Chain. Directing request to PancakeSwap Pipeline...");
                txResult = await window.runbscSwapUniswap(currentActiveWallet, balanceInputStr, REAL_STARTING_TOKEN, currentVirtualToken, PANCAKE_ROUTER);
            }

            if (txResult && txResult.status === "ok") {
                logToPanel("🚀 [FINAL TRADE DISPATCH SUCCESS] Syncing fund ledger balances on-chain...");
                await blockDelay(8000);
                if (typeof window.syncBalanceFromFund === "function") { await window.syncBalanceFromFund(); }
            } else {
                logToPanel("❌ On-chain settlement transaction was rejected or encountered a network error.");
            }
        }
    }

    logToPanel("\n=== 📊 [FINISHED] 10 multi-router hops processed with single settlement transaction finalized. ===");
    return "STOP_ERROR";
}
window.simulateCycle = simulateCycle;

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    logToPanel("🚀 [LAUNCHING] Dynamic multi-router cross-network loop active...");
    await simulateCycle(1, totalCycles || 10, 0, 0, window.CURRENT_WORKER_ID);
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Run parameters finalized.");
}
window.startSimulation = startSimulation;
