// ==================================================================
// CONTAINER 1: DOM STATE SYNCER & LAYOUT CONTROLLER 
// ==================================================================
logToPanel("simulate_prod.txt (Container 1) loaded successfully.");

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) statusLabel.textContent = msg;
    else logToPanel(`[STATUS] ${msg}`);
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Trading halted by user.");
}
window.stopBot = stopBot;

async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BNB"; 
    } catch (e) { return "BNB"; }
}

// 📊 SYNCHRONIZATION PIPE: CONVERTS BOXES TO TEXT CHANNELS AND ASSIGNS VALUE
async function updateUIBalanceControl(user, token) {
    try {
        const baseApiUrl = window.API_BASE || "";
        const res = await fetch(`${baseApiUrl}/prodbot/balance`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: user, token: token })
        });
        const rawWeiText = await res.text();
        const cleanWeiString = rawWeiText.trim() || "0";

        // Target your explicit HTML control targets
        const startInput = document.getElementById("startBalance");
        const investInput = document.getElementById("investBalance");
        
        // Change type properties to 'text' to avoid floating-point truncations
        if (startInput) {
            startInput.type = "text";
            startInput.value = cleanWeiString;
        }
        if (investInput) {
            investInput.type = "text";
            investInput.value = cleanWeiString;
        }
        return cleanWeiString;
    } catch (err) {
        return "0";
    }
}
window.updateUIBalanceControl = updateUIBalanceControl;

logToPanel("simulate_prod.txt (Container 2) loaded successfully.");

// 🔎 PURE TEXTBOX OBSERVATION POLLING BLOCK
async function pollForTextboxValueChange(userAddr, tokenAddress, initialTextboxValue, timeoutSeconds) {
    const pollIntervalMs = 4000;
    const maximumAttempts = (timeoutSeconds * 1000) / pollIntervalMs;
    const startInputBox = document.getElementById("startBalance");

    for (let attempt = 0; attempt < maximumAttempts; attempt++) {
        if (!window.isBotRunning) return false;
        await blockDelay(pollIntervalMs);
        
        // Re-ping the server to update the layout box value natively
        await window.updateUIBalanceControl(userAddr, tokenAddress);
        
        // Directly extract what is currently written inside the input form field element
        let currentWrittenValue = startInputBox ? startInputBox.value : "0";
        
        if (String(currentWrittenValue) !== String(initialTextboxValue)) {
            logToPanel(`📈 Box change captured! Old text: ${initialTextboxValue} -> New text: ${currentWrittenValue}`);
            return true;
        }
        if ((attempt + 1) % 3 === 0) {
            setStatus(`POLLING (${timeoutSeconds - Math.floor((attempt * pollIntervalMs)/1000)}s left)`);
        }
    }
    return false;
}

// ✅ MAIN AUTOMATED SWAP SIMULATION SCHEDULER
async function startSimulation(totalCycles = 10) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    logToPanel(`[ENGINE INITIALIZED] Executing layout tracking array routines for ${totalCycles} iterations.`);
    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";
    const startInputBox = document.getElementById("startBalance");

    // Initialize layout parameters cleanly before parsing cycle chains
    setStatus("SYNCHRONIZING");
    let currentAssetAddress = USDT_ADDRESS;
    let currentAssetSymbol = "USDT";

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        logToPanel(`[INFO] — Cycle ${i}/${totalCycles} started`);
        setStatus(`RUNNING cycle ${i}/${totalCycles}`);

        // Update the form input tracking fields directly from the blockchain state
        let initialFieldText = await window.updateUIBalanceControl(window.USER_ADDR, currentAssetAddress);
        logToPanel(`[STATUS] Observed Field Balance: ${initialFieldText} base units for ${currentAssetSymbol}`);

        if (BigInt(initialFieldText) <= 0n) {
            logToPanel(`❌ Pipeline Halted: Display balance textbox evaluates to 0.`);
            break;
        }

        let targetSymbol = (i === totalCycles) ? "USDT" : "";
        let targetAddress = (i === totalCycles) ? USDT_ADDRESS : "";
        let targetNetwork = "bep20";

        if (i !== totalCycles) {
            const generatedCoin = await pickRandomCoin();
            targetSymbol = `${generatedCoin}`;
            const evalMetrics = await window.evaluateCoin(`${generatedCoin}USDT`);
            targetAddress = evalMetrics?.tokenAddress || "0xeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeee";
            targetNetwork = evalMetrics?.network || "bep20";
        }

        if (targetAddress.toLowerCase() === currentAssetAddress.toLowerCase()) {
            logToPanel(`[INFO] Duplicate asset address detected. Skipping loop step.`);
            continue;
        }

        logToPanel(`[ROUTE] Preparing transaction pipeline: Swap ${currentAssetSymbol} ➡️ ${targetSymbol}`);

        try {
            const baseApiUrl = window.API_BASE || "";
            const res = await fetch(`${baseApiUrl}/prodbot/swap`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    user_address: window.USER_ADDR,
                    from_token: currentAssetAddress,
                    to_token: targetAddress,
                    amount: initialFieldText, // Passes the text box value straight into transaction data variables
                    network: targetNetwork
                })
            });
            const txPayload = await res.json();

            if (txPayload && txPayload.status === "ok") {
                logToPanel(`[METAMASK] Requesting secure signature profile...`);
                const txHash = await window.ethereum.request({
                    method: 'eth_sendTransaction',
                    params: [{
                        from: window.USER_ADDR,
                        to: txPayload.to_contract,
                        data: txPayload.compiled_data,
                        value: '0x0'
                    }],
                });

                logToPanel(`📡 Broadcasted! Hash: ${txHash}`);
                logToPanel(`⏳ Monitoring "startBalance" textbox for changes...`);

                // Freeze loop here. It unfreezes only when the numbers inside the box shift
                const layoutAltered = await pollForTextboxValueChange(window.USER_ADDR, currentAssetAddress, initialFieldText, 120);

                if (layoutAltered) {
                    logToPanel(`✅ Display value text transformation confirmed!`);
                    currentAssetAddress = targetAddress;
                    currentAssetSymbol = targetSymbol;
                } else {
                    logToPanel(`❌ Loop Error: Textbox update timed out. Halting bot.`);
                    break;
                }
            } else {
                logToPanel(`❌ Error: Could not generate swap payload.`);
                break;
            }
        } catch (err) {
            logToPanel(`❌ Critical Exception: ${err.message}`);
            break;
        }

        if (i < totalCycles) {
            await blockDelay(3000);
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Pipeline run complete.");
}
window.startSimulation = startSimulation;
