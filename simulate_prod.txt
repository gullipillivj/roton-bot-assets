logToPanel("simulate_prod.txt (Section 1) loaded successfully.");

function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) statusLabel.textContent = msg;
    else logToPanel(`[STATUS] ${msg}`);
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Trading halted by user.");
}
window.stopBot = stopBot;

async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BNB"; 
    } catch (e) { return "BNB"; }
}

// 📊 BALANCES SYNC: FORCES EXACT UINT256 STRINGS DIRECTLY TO YOUR UI CONTROLS
// Open Section 1 of simulate_prod.txt and update fetchLiveContractBalance to read text():
async function fetchLiveContractBalance(user, token) {
    try {
        const baseApiUrl = window.API_BASE || "";
        const res = await fetch(`${baseApiUrl}/prodbot/balance`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: user, token: token })
        });
        
        // ✅ CHANGED: Read direct response content as a text string instead of .json()
        const rawWeiString = await res.text();
        const verifiedNumberString = rawWeiString.trim() || "0";

        // Push values smoothly down to input structures without numeric padding conflicts
        const investInput = document.getElementById("investBalance");
        const startInput = document.getElementById("startBalance");
        
        if (investInput) {
            investInput.type = "text"; 
            investInput.value = verifiedNumberString;
        }
        if (startInput) {
            startInput.type = "text";
            startInput.value = verifiedNumberString;
        }
        return verifiedNumberString;
    } catch (err) {
        return "0";
    }
}
window.fetchLiveContractBalance = fetchLiveContractBalance;

async function pollForSpentTokenBalanceChange(userAddr, spentTokenAddress, initialValue, maxTimeoutSeconds) {
    const checkIntervalMs = 4000;
    const maxAttempts = (maxTimeoutSeconds * 1000) / checkIntervalMs;
    
    for (let attempt = 0; attempt < maxAttempts; attempt++) {
        if (!window.isBotRunning) return false;
        await blockDelay(checkIntervalMs);
        
        let currentLiveBalance = await window.fetchLiveContractBalance(userAddr, spentTokenAddress);
        
        if (BigInt(currentLiveBalance) !== BigInt(initialValue)) {
            logToPanel(`📈 Balance change found! Old: ${initialValue} -> New: ${currentLiveBalance}`);
            return true;
        }
        if ((attempt + 1) % 3 === 0) {
            setStatus(`POLLING (${maxTimeoutSeconds - Math.floor((attempt * checkIntervalMs)/1000)}s left)`);
        }
    }
    return false;
}

// 🔍 STEP 1 CONDITIONAL PORTFOLIO GATEKEEPER
async function autoSynchronizeActivePortfolioState(userAddr) {
    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";
    try {
        logToPanel("🔎 Step 1: Checking user internal trading balance for USDT...");
        let usdtBalanceWei = await window.fetchLiveContractBalance(userAddr, USDT_ADDRESS);
        
        if (BigInt(usdtBalanceWei) > 0n) {
            logToPanel(`✅ [DIRECT ENTRY] Active USDT capital discovered: ${usdtBalanceWei} Wei. Bypassing rescue scanner.`);
            return { address: USDT_ADDRESS, symbol: "USDT" };
        }
        
        logToPanel("⚠️ USDT balance is 0. Triggering Backend Rescue Scan engine...");
        const baseApiUrl = window.API_BASE || "";
        const res = await fetch(`${baseApiUrl}/prodbot/rescue-scan`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: userAddr })
        });
        
        const scanResult = await res.json();
        if (scanResult && (scanResult.status === "found" || scanResult.status === "clean")) {
            logToPanel(`🔍 [SYNC SUCCESS] Recovered active asset: ${scanResult.symbol} (${scanResult.token_address})`);
            return { address: scanResult.token_address, symbol: scanResult.symbol };
        }
        throw new Error("Scanner optimization error.");
    } catch (err) {
        logToPanel(`⚠️ Synchronization Notice: ${err.message}. Defaulting to main tracking profiles.`);
        return { address: USDT_ADDRESS, symbol: "USDT" };
    }
}

// ✅ MAIN REFACTORED MASTER CYCLING EXECUTION SYSTEM
async function startSimulation(totalCycles = 10) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    logToPanel(`[BOT INITIALIZATION] Booting Dynamic State Tracker Engine for ${totalCycles} iterations.`);
    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";

    setStatus("SYNCHRONIZING");
    const activePortfolio = await autoSynchronizeActivePortfolioState(window.USER_ADDR);
    let currentlyHeldTokenAddress = activePortfolio.address;
    let currentlyHeldTokenSymbol = activePortfolio.symbol;

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        logToPanel(`[INFO] — Cycle ${i}/${totalCycles} started`);
        setStatus(`RUNNING cycle ${i}/${totalCycles}`);

        let spentTokenInitialBalance = await window.fetchLiveContractBalance(window.USER_ADDR, currentlyHeldTokenAddress);
        logToPanel(`[STATUS] Active Target: ${currentlyHeldTokenSymbol} (Balance: ${spentTokenInitialBalance} Wei)`);

        if (BigInt(spentTokenInitialBalance) <= 0n) {
            logToPanel(`❌ Pipeline Halted: Balance mapping evaluates to 0.`);
            break;
        }

        let targetTokenSymbol = (i === totalCycles) ? "USDT" : "";
        let targetTokenAddress = (i === totalCycles) ? USDT_ADDRESS : "";
        let networkParam = "bep20";

        if (i !== totalCycles) {
            const coinBase = await pickRandomCoin();
            targetTokenSymbol = `${coinBase}`;
            const coinInfo = await window.evaluateCoin(`${coinBase}USDT`);
            targetTokenAddress = coinInfo?.tokenAddress || "0xeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeee";
            networkParam = coinInfo?.network || "bep20";
        }

        if (targetTokenAddress.toLowerCase() === currentlyHeldTokenAddress.toLowerCase()) {
            logToPanel(`[INFO] Selected identical asset variant. Auto-skipping to avoid redundant logs.`);
            continue;
        }

        logToPanel(`[INFO] Processing Route: Swap ${currentlyHeldTokenSymbol} ➡️ ${targetTokenSymbol}`);

        try {
            const baseApiUrl = window.API_BASE || "";
            const res = await fetch(`${baseApiUrl}/prodbot/swap`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    user_address: window.USER_ADDR,
                    from_token: currentlyHeldTokenAddress,
                    to_token: targetTokenAddress,
                    amount: spentTokenInitialBalance,
                    network: networkParam
                })
            });
            const backendData = await res.json();

            if (backendData && backendData.status === "ok") {
                logToPanel(`[INFO] Prompting secure wallet signature...`);
                const txHash = await window.ethereum.request({
                    method: 'eth_sendTransaction',
                    params: [{
                        from: window.USER_ADDR,
                        to: backendData.to_contract,
                        data: backendData.compiled_data,
                        value: '0x0'
                    }],
                });

                logToPanel(`📡 Transaction broadcasted! Hash identity: ${txHash}`);
                const txCleared = await pollForSpentTokenBalanceChange(window.USER_ADDR, currentlyHeldTokenAddress, spentTokenInitialBalance, 120);

                if (txCleared) {
                    logToPanel(`✅ State modification cleared by blockchain miners!`);
                    currentlyHeldTokenAddress = targetTokenAddress;
                    currentlyHeldTokenSymbol = targetTokenSymbol;
                } else {
                    logToPanel(`❌ Execution Timeout: State failed to alter. Halted pipeline.`);
                    break;
                }
            } else {
                logToPanel(`❌ Backend Error: Compiled calldata generation failed.`);
                break;
            }
        } catch (err) {
            logToPanel(`❌ Pipeline Exception Encountered: ${err.message}`);
            break;
        }

        if (i < totalCycles) {
            logToPanel("[COOLDOWN] Waiting 3s for next loop iteration step...");
            await blockDelay(3000);
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Sequence pipeline processing completed successfully.");
}
window.startSimulation = startSimulation;
