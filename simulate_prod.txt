// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Asset scan matrix paused by user.");
}
window.stopBot = stopBot;

// 🔍 ZERO-GAS BROWNIE SIMULATOR CORE WITH POST-CYCLE BATCH MATCHING
// ==================================================================
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`🚀 [ROUTING DIAGNOSTIC] Accessing read-only backend slippage route...`);

    const PANCAKE_ROUTER = "0x10ED43C718714eb63d5aA57B78B54704E256024E";
    const BASE_TOKEN_IN = "0x55d398326f99059fF775485246999027B3197955"; // USDT Base Contract

    for (let i = 1; i <= totalCycles; i++) {
        if (window.isBotRunning === false) {
            logToPanel("🛑 Loop halted by request.");
            break;
        }

        logToPanel(`\n--- [START] Simulation Cycle ${i}/${totalCycles} ---`);
        setStatus(`RUNNING ${i}/${totalCycles}`);

        // Read dynamic UI field containing the true 18-digit Wei string from dashboard ledger metrics
        let balanceInputStr = document.getElementById("investBalance") ? document.getElementById("investBalance").value.toString().trim() : "10";
        if (!balanceInputStr || balanceInputStr === "0") balanceInputStr = "10";
        
        let simulatedAmountInWei = balanceInputStr;

        logToPanel(`💰 Pure Transaction Principal Weight: ${simulatedAmountInWei} Wei`);

        let marketData = [];
        try {
            if (typeof window.getTradeCandidates === "function") {
                marketData = await window.getTradeCandidates();
            } else {
                logToPanel("⚠️ window.getTradeCandidates registry access function is missing.");
                continue;
            }
        } catch (fErr) {
            logToPanel(`❌ Local Array Ingestion Error: ${fErr.message}`);
            continue;
        }

        // 🟢 CUSTOM TOKEN INJECTION MATRIX: Merging your specified custom assets explicitly
        const customTokens = [
            { symbol: "RR-BNB", address: "0xba67186fd454D0256a0D3744783538bE08C58feD", tokentype: "bep20" },
            { symbol: "RR-ETH", address: "0x8A0cAC8E7F6305cE6EeB585005919290F78808f9", tokentype: "erc20" },
            { symbol: "WRR-ETH", address: "0xf3414e85dEa5cf3Fab62F5A967F98Fd032d5BE2A", tokentype: "erc20" },
            { symbol: "WRR-WBNB", address: "0xBf9d926BBF7b9408D1182c45fCf64f7f27909837", tokentype: "bep20" }
        ];

        // Combine system assets with your custom trading matrix tokens seamlessly
        marketData = marketData.concat(customTokens);

        if (!Array.isArray(marketData) || marketData.length === 0) {
            logToPanel("❌ Filtered token list data is completely empty.");
            continue;
        }

        logToPanel(`🎯 Dispatching ${marketData.length} tokens (${customTokens.length} custom injected) to validator endpoint...`);

        // Dedicated array matrix to cache evaluation metrics safely away from loop collisions
        let cycleCandidatesArray = [];
        for (let index = 0; index < marketData.length; index++) {
            let candidate = marketData[index];
            let symbolLabel = candidate.symbol || "UNKNOWN";
            let targetAddress = candidate.address || "MISSING";
            let tokenNetwork = (candidate.tokentype || "bep20").toLowerCase();

            if (targetAddress === "MISSING" || !targetAddress.startsWith("0x")) {
                continue;
            }

            // Route dynamically through PancakeSwap parameters or Ethereum router constraints based on definition metrics
            let activeRouterForSimulation = PANCAKE_ROUTER;
            if (tokenNetwork === "erc20" || tokenNetwork === "ethereum") {
                // If your backend simulation uses a distinct ETH mainnet router, swap out this string constraint
                activeRouterForSimulation = "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D"; 
            }

            try {
                let response = await fetch("/prodbot/swap/simulateuniswap", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        router: activeRouterForSimulation,
                        tokenIn: BASE_TOKEN_IN,
                        tokenOut: targetAddress,
                        amountIn: simulatedAmountInWei,
                        amountOutMin: 0
                    })
                });

                let resData = await response.json();

                if (resData.status === "ok") {
                    let expectedOutput = resData.expected_output_wei || "0";
                    let slippageCost = resData.slippage_cost_pct || "0.0";

                    if (expectedOutput === "0" || expectedOutput === 0) {
                        logToPanel(`   ❌ [NOT SWAPPABLE] ${symbolLabel} (${targetAddress.slice(0, 8)}...): Liquidity validation check returned 0 or dead pool routing paths.`);
                    } else {
                        logToPanel(`   🟩 [SWAPPABLE] ${symbolLabel} | Slippage Rate: ${slippageCost}% | Expected Out: ${expectedOutput} Wei`);
                        
                        // Push token metrics into data array buffer for post-cycle evaluation
                        cycleCandidatesArray.push({
                            symbol: symbolLabel,
                            address: targetAddress,
                            network: tokenNetwork,
                            expectedOutWei: BigInt(expectedOutput),
                            slippage: parseFloat(slippageCost)
                        });
                    }
                } else {
                    logToPanel(`   ❌ [BACKEND REJECTION] ${symbolLabel}: ${resData.message}`);
                }
            } catch (err) {
                logToPanel(`   ❌ [ROUTE TIMEOUT] Failed calling simulation path for ${symbolLabel}: ${err.message}`);
            }

            await blockDelay(150);
        }
        // ==================================================================
        // ⚖️ POST-CYCLE EVALUATION MATRIX: CHOOSE AND RUN EXACTLY ONE SWAP
        // ==================================================================
        logToPanel(`\n📊 [CYCLE AUDIT] Evaluating ${cycleCandidatesArray.length} swappable pool candidates...`);

        if (cycleCandidatesArray.length === 0) {
            logToPanel("   💤 No executable candidates cleared parameter validation mappings for this loop.");
        } else {
            try {
                if (!window.USER_ADDR) {
                    logToPanel("   ❌ [ABORTED] No wallet session address connected.");
                } else {
                    const userBalanceHex = await window.ethereum.request({ method: "eth_getBalance", params: [window.USER_ADDR, "latest"] });
                    const walletGasBalanceWei = BigInt(userBalanceHex);
                    const networkGasPriceHex = await window.ethereum.request({ method: "eth_gasPrice" });
                    const liveGasPriceWei = BigInt(networkGasPriceHex);

                    let targetSelection = null;
                    let highestExpectedOutput = 0n;

                    for (let c = 0; c < cycleCandidatesArray.length; c++) {
                        let currentAsset = cycleCandidatesArray[c];
                        let standardGasLimit = currentAsset.network === "erc20" || currentAsset.network === "ethereum" ? 450000n : 150000n;
                        let computedGasFeeWei = liveGasPriceWei * standardGasLimit;

                        // Verify network gas token eligibility constraints 
                        if (walletGasBalanceWei < computedGasFeeWei) continue;

                        if (currentAsset.expectedOutWei > highestExpectedOutput) {
                            highestExpectedOutput = currentAsset.expectedOutWei;
                            targetSelection = currentAsset;
                        }
                    }

                    if (!targetSelection) {
                        logToPanel("   ⚠️ [SKIPPED] All assets bypassed due to wallet gas restrictions.");
                    } else {
                        logToPanel(`\n🔥 [OPTIMIZER SELECTION] Dispatching parameters for: ${targetSelection.symbol} (${targetSelection.address})`);
                        
                        let txResult = null;
                        if (targetSelection.network === "erc20" || targetSelection.network === "ethereum") {
                            logToPanel("   ⚡ Forwarding payload parameters to ERC20 mainnet pipeline...");
                            txResult = await window.runethSwapUniswap(
                                window.USER_ADDR, simulatedAmountInWei, BASE_TOKEN_IN, targetSelection.address, PANCAKE_ROUTER
                            );
                        } else {
                            logToPanel("   ⚡ Forwarding payload parameters to BEP20 production pipeline...");
                            txResult = await window.runbscSwapUniswap(
                                window.USER_ADDR, simulatedAmountInWei, BASE_TOKEN_IN, targetSelection.address, PANCAKE_ROUTER
                            );
                        }

                        if (txResult && txResult.status === "ok") {
                            logToPanel("   🚀 [SWAP DISPATCH SUCCESS] Token approval and route execution initialized. Locking loop for ledger checks... ");
                            
                            let startBox = document.getElementById("startBalance");
                            let baseWeiState = startBox ? startBox.value.toString().trim() : "0";
                            let transactionConfirmed = false;
                            
                            // 30-step sequential wait balance validator loop
                            for (let checkNum = 1; checkNum <= 30; checkNum++) {
                                if (window.isBotRunning === false) break;
                                
                                if (typeof window.syncBalanceFromFund === "function") {
                                    let currentWeiVal = await window.syncBalanceFromFund();
                                    
                                    if (BigInt(currentWeiVal || "0") !== BigInt(baseWeiState)) {
                                        logToPanel(`   🎉 [SETTLEMENT CONFIRMED] Block updated! Shift: ${baseWeiState} -> ${currentWeiVal} Wei.`);
                                        
                                        let usdBox = document.getElementById("usd");
                                        let floatingUsd = usdBox ? parseFloat(usdBox.value) || 0 : 0;
                                        if (typeof window.updateChart === "function") {
                                            window.updateChart(floatingUsd, floatingUsd - (parseFloat(window.InitialBaseCapital) || 0));
                                        }
                                        transactionConfirmed = true;
                                        break; 
                                    }
                                }
                                logToPanel(`      ⏳ Polling update [${checkNum}/30]: Ledger state stable. Awaiting network validation (6s)...`);
                                await blockDelay(6000);
                            }

                            if (!transactionConfirmed) {
                                logToPanel("   ⚠️ [POLLING EXCEEDED] No contract state balance alterations caught inside the execution window.");
                            }

                            // ⏱️ FORCE CRITICAL 2-MINUTE COOLING LOCK WITH LOG PANEL FEEDBACK COUNTDOWN
                            logToPanel("\n⏱️ [SAFETY COOLING LOCK] Enforcing a mandatory 120s pause to protect account state parameters...");
                            for (let countdown = 120; countdown > 0; countdown -= 5) {
                                if (window.isBotRunning === false) break;
                                setStatus(`COOLING (${countdown}s)`);
                                logToPanel(`   ⏳ Network Sync Cooling: ${countdown} seconds remaining before next cycle...`);
                                await blockDelay(5000);
                            }
                            
                        } else {
                            logToPanel(`   ❌ [ROUTING REJECTION] Execution router dropped payload: ${txResult ? txResult.message : "User cancelled signature or RPC fault."}`);
                        }
                    }
                }
            } catch (walletErr) {
                logToPanel(`   ❌ [CRITICAL FAULT] Wallet state calculation error: ${walletErr.message}`);
            }
        }

        logToPanel(`✅ Cycle ${i}/${totalCycles} analysis finalized.`);
        setStatus("WAITING");
        await blockDelay(4000); 
    }

    logToPanel(`\n=== 📊 [FINISHED] Pre-flight structural scan operations complete. ==`);
    return "STOP_ERROR";
}
window.simulateCycle = simulateCycle;

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    logToPanel("🚀 [LAUNCHING] Read-only background asset validation sequence engaged...");
    await simulateCycle(1, 10, 0, 0);
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Run parameters finalized.");
}
window.startSimulation = startSimulation;
