function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

async function updateTokenRecord(lastToken, currentToken) {
    try {
        const userid = window.USER_ID || localStorage.getItem("userid");
        if (!userid) {
            logToPanel("[TOKEN SYNC] Missing userid");
            return;
        }
        const resp = await fetch("/prodbot/tokens/update", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: userid, lasttoken: lastToken, currenttoken: currentToken })
        });
        const data = await resp.json();
        logToPanel(`[TOKEN SYNC] ${data.message}`);
    } catch (err) {
        logToPanel(`[TOKEN SYNC ERROR] ${err.message}`);
    }
}

async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
    } catch (err) {
        console.log(err);
    }
    return "CAKE";
}

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Trading halted by user.");
}
window.stopBot = stopBot;

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`[INFO] Cycle ${cycle}/${totalCycles}`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);

    const syncedBalance = await syncBalanceFromFund();
    if (syncedBalance === null) {
        logToPanel("[WARN] Balance sync failed. Skipping loop cycle.");
        return "CONTINUE"; 
    }

    let tradingBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    if (tradingBalance <= 0) {
        logToPanel("[WARN] No trading balance available.");
        return "CONTINUE"; 
    }

    // Capture baseline metrics before executing transaction routines
    window.beforeBalanceUSD = parseFloat(document.getElementById("usd").value) || 0;
    if (cycle === 1) {
        window.initialSimulationUSD = window.beforeBalanceUSD;
    }

    const totalBalance = BigInt(document.getElementById("startBalance").value.toString().trim());
    const remainingCycles = Math.max(1, totalCycles - cycle + 1);
    const allocationUSD = (totalBalance / BigInt(remainingCycles)).toString();

    logToPanel(`[ALLOCATION]\nTOTAL=${totalBalance}\nALLOCATED=${allocationUSD}`);

    let currentToken = "0x55d398326f99059fF775485246999027B3197955";
    try {
        const tokenResp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: window.USER_ID })
        });
        const tokenData = await tokenResp.json();
        if (tokenData && tokenData.currenttoken) {
            currentToken = tokenData.currenttoken;
            logToPanel(`[TOKEN LOAD]\n${currentToken}`);
        }
    } catch (err) {
        logToPanel(`[TOKEN LOAD ERROR]\n${err.message}`);
    }

    const selectedCoin = await pickRandomCoin();
    const coinInfo = await window.evaluateCoin(selectedCoin);

    if (!coinInfo || !coinInfo.tokenAddress) {
        logToPanel("[WARN] Invalid or unprofiled target token. Proceeding...");
        return "CONTINUE"; 
    }

    logToPanel(`[TARGET]\n${coinInfo.symbol}\nMOVE=${coinInfo.change24h}%\nADDRESS=${coinInfo.tokenAddress}`);
    const beforeBalance = document.getElementById("investBalance").value.toString().trim();

    try {
        logToPanel(`[SWAP REQUEST]\nFROM=${currentToken}\nTO=${coinInfo.tokenAddress}\nAMOUNT=${allocationUSD}`);
        const swapResp = await window.runSwapUniswap(
            window.USER_ADDR,
            allocationUSD,
            currentToken,
            coinInfo.tokenAddress
        );

        if (!swapResp || !swapResp.tx_hash) {
            logToPanel("[SWAP PIPELINE SKIPPED OR REJECTED]");
            return "CONTINUE"; 
        }

        logToPanel(`[TX SENT]\n${swapResp.tx_hash}`);

        let balanceChanged = false;
        for (let attempt = 1; attempt <= 30; attempt++) {
            await blockDelay(10000);
            await syncBalanceFromFund();
            const currentBalance = document.getElementById("investBalance").value.toString().trim();
            logToPanel(`[VERIFY ${attempt}/30]\nSTART=${beforeBalance}\nNOW=${currentBalance}`);

            if (currentBalance !== beforeBalance) {
                balanceChanged = true;
                logToPanel(`[BALANCE VERIFIED]\n${beforeBalance}\n➡\n${currentBalance}`);
                break;
            }
        }

        if (!balanceChanged) {
            logToPanel("[NO BALANCE MOVEMENT DETECTED — PASSING TO NEXT LOOP]");
            return "CONTINUE"; 
        }

        await updateTokenRecord(currentToken, coinInfo.tokenAddress);
        logToPanel(`[TOKEN UPDATED]\nLAST=${currentToken}\nCURRENT=${coinInfo.tokenAddress}`);

        await syncBalanceFromFund();
        
        const afterBalanceUSD = parseFloat(document.getElementById("usd").value) || 0;
        const profitDeltaUSD = afterBalanceUSD - window.beforeBalanceUSD;
        window.cumulativePnL = (window.cumulativePnL || 0) + profitDeltaUSD;

        const liveProfitText = document.getElementById("liveProfitValue");
        if (liveProfitText) {
            liveProfitText.innerText = window.cumulativePnL.toFixed(4) + " USDT";
            liveProfitText.style.color = window.cumulativePnL >= 0 ? "#00FF00" : "#FF3B30";
        }

        if (typeof updateChart === "function") {
            updateChart(afterBalanceUSD, window.cumulativePnL);
        }

    } catch (err) {
        logToPanel(`[RUNTIME RUN EXCEPTION]\n${err.message}`);
        return "CONTINUE"; 
    }

    // Stripped of early exits to force all 10 continuous cycles to compile
    return "CONTINUE";
}

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    logToPanel(`[BOT START] ${totalCycles} cycles`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (i < totalCycles) {
            setStatus("WAITING");
            await blockDelay(10000);
        }
    }
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED]");
}
window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
