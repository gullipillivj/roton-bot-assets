// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Compounding execution matrix paused by user.");
}
window.stopBot = stopBot;

// 🔄 DYNAMIC COMPOUNDING ENGINE: PER-CYCLE NET-PROFIT MAXIMIZER
// ==================================================================
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`🚀 [COMPOUNDING CORE ENGAGED] Scanning up to ${totalCycles} cycles sequentially...`);

    const PANCAKE_ROUTER = "0x10ED43C718714eb63d5aA57B78B54704E256024E";
    const BASE_TOKEN_IN = "0x55d398326f99059fF775485246999027B3197955"; // USDT Base

    // Read initial capital configuration from layout screen input box on startup
    let balanceInputStr = document.getElementById("investBalance") ? document.getElementById("investBalance").value.toString().trim() : "100";
    let activePrincipal = parseFloat(balanceInputStr) || 100.0;
    
    let userProfitTarget = parseFloat(document.getElementById("profitTarget") ? document.getElementById("profitTarget").value : "1.0") || 1.0;

    // Outer orchestration loop processing cycles dynamically step-by-step
    for (let i = 1; i <= totalCycles; i++) {
        if (window.isBotRunning === false) {
            logToPanel("🛑 Compounding execution loop halted by user.");
            break;
        }

        logToPanel(`\n--- [START] Compounding Cycle ${i}/${totalCycles} ---`);
        setStatus(`RUNNING ${i}/${totalCycles}`);
        logToPanel(`💰 Starting Cycle Principal: ${activePrincipal.toFixed(4)} USDT`);

        // Convert current compounded floating principal into a secure 18-decimal Wei string layout
        let simulatedAmountInWei = BigInt(Math.floor(activePrincipal * 10000)) * BigInt(10**14);

        // 1. Fetch fresh data array vectors straight out of your coins.txt registry function PER CYCLE
        let marketData = [];
        try {
            if (typeof window.getTradeCandidates === "function") {
                marketData = await window.getTradeCandidates();
            } else {
                logToPanel("⚠️ window.getTradeCandidates function is missing from current scope layout.");
                continue;
            }
        } catch (fErr) {
            logToPanel(`❌ Dynamic Data Array Retrieval Interrupted: ${fErr.message}`);
            continue;
        }

        if (!Array.isArray(marketData) || marketData.length === 0) {
            logToPanel("❌ Ingestion tracking market data array is completely empty for this cycle.");
            continue;
        }

        let selectedWinner = null;
        let highestNetProfitAbsolute = -999999;

        logToPanel(`🎯 Evaluating liquidity depth profiles for ${marketData.length} active candidates...`);

        // 2. Inner scanner loop auditing tokens against live backend simulation metrics
        for (let index = 0; index < marketData.length; index++) {
            let candidate = marketData[index];
            let symbolLabel = candidate.symbol || "UNKNOWN";
            let targetAddress = candidate.address || "MISSING";
            let tokenNetwork = (candidate.tokentype || "bep20").toLowerCase();
            let momentum24h = parseFloat(candidate.change_24h_percent || candidate["%profit"] || "0.0");

            if (targetAddress === "MISSING" || !targetAddress.startsWith("0x")) continue;
            if (tokenNetwork !== "bep20" && tokenNetwork !== "bsc") continue;

            // Model network transaction friction parameters based on the asset routing type
            // Cross-chain actions include Chainlink CCIP oracle fees; standard BEP20 swaps carry base router gas costs only
            let frictionOverheadGasUsdt = 1.50; // Standard baseline BEP20 swap gas cost
            if (symbolLabel.toUpperCase().includes("CROSS") || symbolLabel.toUpperCase().includes("BRIDGE")) {
                frictionOverheadGasUsdt = 7.00; // Multi-chain validation fee allocation
            } else if (symbolLabel.toUpperCase() === "VENUSFLUX") {
                frictionOverheadGasUsdt = 2.00; // Highly optimized liquidity pair overhead cost
            }

            try {
                let response = await fetch("/prodbot/swap/simulateuniswap", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        router: PANCAKE_ROUTER,
                        tokenIn: BASE_TOKEN_IN,
                        tokenOut: targetAddress,
                        amountIn: simulatedAmountInWei.toString(),
                        amountOutMin: 0
                    })
                });

                let resData = await response.json();

                if (resData.status === "ok") {
                    let expectedOutputWei = resData.expected_output_wei || "0";
                    let slippagePct = parseFloat(resData.slippage_cost_pct || "0.0");

                    if (expectedOutputWei === "0" || expectedOutputWei === 0) continue;

                    // Deduct liquidity pool price impact costs from gross value
                    let poolEfficiencyFactor = (100.0 - slippagePct) / 100.0;
                    
                    // Model gross balance returns using 24h market momentum velocity adjustments
                    let grossReturnUsdt = activePrincipal * (1.0 + (momentum24h / 100.0)) * poolEfficiencyFactor;
                    
                    // Deduct the initial principal and the specific operational asset fee block
                    let absoluteNetProfitUsdt = grossReturnUsdt - activePrincipal - frictionOverheadGasUsdt;
                    let calculatedNetProfitPct = (absoluteNetProfitUsdt / activePrincipal) * 100.0;

                    logToPanel(`   🔹 ${symbolLabel} -> Gross: $${grossReturnUsdt.toFixed(2)} | Friction: $${frictionOverheadGasUsdt.toFixed(2)} | Net: +$${absoluteNetProfitUsdt.toFixed(2)} (${calculatedNetProfitPct.toFixed(2)}%)`);

                    // CRITICAL LOGIC SELECTOR: Track the token yielding the highest absolute net balance growth
                    if (absoluteNetProfitUsdt > highestNetProfitAbsolute) {
                        highestNetProfitAbsolute = absoluteNetProfitUsdt;
                        selectedWinner = {
                            symbol: symbolLabel,
                            address: targetAddress,
                            netUsdtGrowth: absoluteNetProfitUsdt,
                            netPctGrowth: calculatedNetProfitPct,
                            slippage: slippagePct,
                            friction: frictionOverheadGasUsdt
                        };
                    }
                }
            } catch (err) {
                // Background exception tracking bypassed silently to preserve loop runtime stability
            }

            await blockDelay(150);
        }

        // ==================================================================
        // 🚀 DYNAMIC CAPITAL RE-INJECTION & COMPOUNDING CHECK
        // ==================================================================
        if (selectedWinner && selectedWinner.netUsdtGrowth > 0 && selectedWinner.netPctGrowth >= userProfitTarget) {
            logToPanel(`\n🔥 [OPTIMIZATION SUCCESSFUL - WINNER SELECTED] 🔥`);
            logToPanel(`🏅 Token Asset: ${selectedWinner.symbol}`);
            logToPanel(`🛡️ Overhead Spent: $${selectedWinner.friction.toFixed(2)} (Slippage: ${selectedWinner.slippage}%)`);
            logToPanel(`📈 Realized Net Balance Growth: +$${selectedWinner.netUsdtGrowth.toFixed(4)} USDT`);
            
            // 🟢 DYNAMIC RE-INJECTION: Add the net earnings back into your principal for the next cycle loop
            activePrincipal = activePrincipal + selectedWinner.netUsdtGrowth;
            logToPanel(`🌱 Principal grew to: ${activePrincipal.toFixed(4)} USDT`);
            
            // Optional: Update your screen text input box live so you can watch your balance build up visually
            const inputField = document.getElementById("investBalance");
            if (inputField) inputField.value = activePrincipal.toFixed(2);
            
        } else {
            logToPanel(`\n💤 [CYCLE PASSED] No candidates provided a healthy net-yield profile above ${userProfitTarget}%.`);
            logToPanel(`🔒 Principal safe. Re-injecting unmodified balance of ${activePrincipal.toFixed(4)} USDT into next cycle verification checks.`);
        }

        logToPanel(`✅ Compounding Cycle ${i}/${totalCycles} analysis finalized.`);
        setStatus("WAITING");
        await blockDelay(4000);
    }

    logToPanel(`\n=== 📊 [FINISHED] Completed all requested data compounding scan cycles successfully. Final Balance: ${activePrincipal.toFixed(4)} USDT ===`);
    return "STOP_ERROR";
}
window.simulateCycle = simulateCycle;

    logToPanel(`🚀 [LAUNCHING] Compounding loop master controller initialization active...`);

    window.isBotRunning = false;
    setStatus("STOPPED");
