function logToPanel(msg) {
    const panel = document.getElementById("logPanel");

    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}

window.logToPanel = logToPanel;

const blockDelay = ms =>
    new Promise(resolve => setTimeout(resolve, ms));

// ==========================================
// TOKEN DB SYNC
// ==========================================
async function updateTokenRecord(
    lastToken,
    currentToken
) {
    try {
        const userid =
            window.USER_ID ||
            localStorage.getItem("userid");

        if (!userid) {
            logToPanel(
                "[TOKEN SYNC] Missing userid"
            );
            return;
        }

        const resp = await fetch(
            "/prodbot/tokens/update",
            {
                method: "POST",
                headers: {
                    "Content-Type":
                        "application/json"
                },
                body: JSON.stringify({
                    userid: userid,
                    lasttoken: lastToken,
                    currenttoken: currentToken
                })
            }
        );

        const data = await resp.json();

        logToPanel(
            `[TOKEN SYNC] ${data.message}`
        );
    } catch (err) {
        logToPanel(
            `[TOKEN SYNC ERROR] ${err.message}`
        );
   
}
}


async function pickRandomCoin() {

    try {

        if (typeof window.getBinancePairs === "function") {

            const pairs =
                await window.getBinancePairs();

            if (pairs && pairs.length > 0) {

                return pairs[
                    Math.floor(
                        Math.random() * pairs.length
                    )
                ].replace("USDT", "");
            }
        }

    } catch (err) {
        console.log(err);
    }

    return "CAKE";
}

function setStatus(msg) {

    const statusLabel =
        document.querySelector(".status-label");

    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}

window.setStatus = setStatus;

function stopBot() {

    window.isBotRunning = false;

    setStatus("STOPPED");

    logToPanel(
        "[BOT STOPPED] Trading halted by user."
    );
}

window.stopBot = stopBot;

// FALLBACK SWAP
// =====================================================

async function swapRR(amount) {

    try {

        const resp =
            await window.runSwapUniswap(

                window.USER_ADDR,

                amount,

                "0xba67186fd454D0256a0D3744783538bE08C58feD",

                "0x8A0cAC8E7F6305cE6EeB585005919290F78808f9"
            );

        if (resp && resp.tx_hash) {

            logToPanel(
                `[RR_FALLBACK] Swap Executed - ${resp.tx_hash}`
            );
        }

    } catch (err) {

        logToPanel(
            `[RR_FALLBACK ERROR] ${err.message}`
        );
    }

    return "CONTINUE";
}

// =====================================================
// SINGLE TRADING CYCLE
// =====================================================
async function simulateCycle(
    cycle,
    totalCycles,
    profitTargetPct,
    stopLossPct
) {
    logToPanel(
        `[INFO] Cycle ${cycle}/${totalCycles}`
    );

    setStatus(
        `RUNNING ${cycle}/${totalCycles}`
    );

    const syncedBalance =
        await syncBalanceFromFund();

    if (syncedBalance === null) {
        logToPanel(
            "[WARN] Balance sync failed."
        );
        return await swapRR(0);
    }

    let tradingBalance =
        parseFloat(
            document.getElementById(
                "investBalance"
            ).value
        ) || 0;

    if (tradingBalance <= 0) {
        logToPanel(
            "[WARN] No trading balance."
        );
        return await swapRR(0);
    }

    const allocationUSD =
        tradingBalance /
        Math.max(
            1,
            totalCycles - (cycle - 1)
        );

    if (cycle === 6) {
        logToPanel(
            "[RR SUPPORT] Dedicated RR circulation cycle."
        );
        return await swapRR(allocationUSD);
    }

    // =====================================
    // LOAD CURRENT TOKEN FROM DB
    // =====================================

    let currentToken =
        "0x55d398326f99059fF775485246999027B3197955";

    try {

        const tokenResp = await fetch(
            "/prodbot/tokens/get",
            {
                method: "POST",
                headers: {
                    "Content-Type":
                        "application/json"
                },
                body: JSON.stringify({
                    userid:
                        window.USER_ID ||
                        localStorage.getItem(
                            "userid"
                        )
                })
            }
        );

        const tokenData =
            await tokenResp.json();

        if (
            tokenData &&
            tokenData.currenttoken
        ) {
            currentToken =
                tokenData.currenttoken;

            logToPanel(
                `[TOKEN LOAD] Current Token = ${currentToken}`
            );
        }

    } catch (err) {

        logToPanel(
            `[TOKEN LOAD ERROR] ${err.message}`
        );

    }

    const selectedCoin =
        await pickRandomCoin();

    const targetSymbol =
        `${selectedCoin}USDT`;

    const coinInfo =
        await window.evaluateCoin(
            targetSymbol
        );

    if (
        !coinInfo ||
        !coinInfo.tokenAddress
    ) {
        logToPanel(
            `[WARN] Could not resolve ${targetSymbol}`
        );

        return await swapRR(
            allocationUSD
        );
    }

    logToPanel(
        `[TARGET] ${coinInfo.symbol}`
    );

    const beforeBalance =
        window.SolidarityFundBalance ||
        tradingBalance;

    try {

        const swapResp =
            await window.runSwapUniswap(
                window.USER_ADDR,
                allocationUSD,
                currentToken,
                coinInfo.tokenAddress
            );

        if (
            swapResp &&
            swapResp.tx_hash
        ) {

            logToPanel(
                `🎉 Swap Success: ${swapResp.tx_hash}`
            );

            // =====================================
            // UPDATE TOKEN TABLE
            // =====================================

            await updateTokenRecord(
                currentToken,
                coinInfo.tokenAddress
            );

            logToPanel(
                `[TOKEN UPDATED]
LAST=${currentToken}
CURRENT=${coinInfo.tokenAddress}`
            );

        } else {

            logToPanel(
                "[WARN] Swap failed."
            );

            return await swapRR(
                allocationUSD
            );
        }

        await syncBalanceFromFund();

        const afterBalance =
            window.SolidarityFundBalance ||
            beforeBalance;

        const profitDelta =
            afterBalance -
            beforeBalance;

        window.cumulativePnL =
            (window.cumulativePnL || 0) +
            profitDelta;

        if (
            typeof updateChart ===
            "function"
        ) {
            updateChart(
                afterBalance,
                window.cumulativePnL
            );
        }

    } catch (err) {

        logToPanel(
            `[ERROR] ${err.message}`
        );

        return await swapRR(
            allocationUSD
        );
    }

    const totalChangePct =
        (
            window.cumulativePnL /
            tradingBalance
        ) * 100;

    if (
        totalChangePct >=
        profitTargetPct
    ) {
        logToPanel(
            "[TARGET REACHED]"
        );

        return "STOP_PROFIT";
    }

    if (
        totalChangePct <=
        -stopLossPct
    ) {
        logToPanel(
            "[STOP LOSS HIT]"
        );

        return "STOP_LOSS";
    }

    return "CONTINUE";
}

// =====================================================
// BOT START
// =====================================================

async function startSimulation(
    totalCycles
) {

    if (window.isBotRunning)
        return;

    window.isBotRunning = true;

    window.cumulativePnL = 0;

    const pTarget =
        parseFloat(
            document.getElementById(
                "profitTarget"
            ).value
        ) || 1.0;

    const sLoss =
        parseFloat(
            document.getElementById(
                "stopLoss"
            ).value
        ) || 0.5;

    logToPanel(
        `[BOT START] ${totalCycles} cycles`
    );

    for (
        let i = 1;
        i <= totalCycles;
        i++
    ) {

        if (!window.isBotRunning)
            break;

        const signal =
            await simulateCycle(
                i,
                totalCycles,
                pTarget,
                sLoss
            );

        if (
            signal === "STOP_PROFIT" ||
            signal === "STOP_LOSS"
        ) {
            break;
        }

        if (i < totalCycles) {

            setStatus(
                "WAITING"
            );

            await blockDelay(
                10000
            );
        }
    }

    window.isBotRunning = false;

    setStatus("STOPPED");

    logToPanel(
        "[BOT FINISHED]"
    );
}

window.startSimulation =
    startSimulation;

window.simulateCycle =
    simulateCycle;
