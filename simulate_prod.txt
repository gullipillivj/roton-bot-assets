const blockDelay = ms => new Promise(res => setTimeout(res, ms));

// === Dynamic Live Coin Pair Picker ===
async function pickRandomCoin() {
    try {
        // Safe check for Binance pairs payload list hook
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BTC"; 
    } catch (e) {
        return "BTC";
    }
}

// === Automated Transaction Logic Core ===
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Cycle ${cycle}/${totalCycles} started`);
    setStatus(`RUNNING cycle ${cycle}/${totalCycles}`);

    // Refresh authoritative balance
    const syncedBalance = await syncBalanceFromFund();
    if (syncedBalance === null) {
        logToPanel("[WARN] Could not sync on-chain balance; skipping cycle");
        return "CONTINUE";
    }

    let tradingBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    const startBalance = parseFloat(document.getElementById("startBalance").value) || 0;

    if (tradingBalance < startBalance) {
        tradingBalance = startBalance;
        const investEl = document.getElementById("investBalance");
        if (investEl) investEl.value = tradingBalance.toFixed(SYNC_DECIMALS);
        logToPanel(`[INFO] investBalance reset to on-chain balance`);
    }

    if (tradingBalance <= 0) {
        logToPanel("[INFO] Trading balance is zero or invalid; skipping cycle");
        return "SKIP";
    }

    const allocationUSD = tradingBalance / Math.max(1, totalCycles - (cycle - 1));

    // Pick coin and fetch stats
    const coinBase = await pickRandomCoin();
    const stats = await window.getBinanceStats(`${coinBase}USDT`);
    if (!stats || !stats.lastPrice) {
        logToPanel("[WARN] Market stats unavailable; skipping cycle");
        return "CONTINUE";
    }

    // Estimate potential profit
    const currentPrice = stats.lastPrice;
    const newPrice = currentPrice * (1 + (stats.priceChangePercent || 0) / 100);
    const potentialPnLUSD = (newPrice - currentPrice) * (allocationUSD / currentPrice);

    // Fee calculation
    let feeCalc;
    try {
        feeCalc = await window.calculateTradeCost(`${coinBase}USDT`, allocationUSD, "bsc");
    } catch (feeErr) {
        logToPanel(`[WARN] Fee check failed: ${feeErr.message} — skipping trade`);
        return "CONTINUE";
    }

    const totalFees = feeCalc.totalCost || 0;

    // Hard-coded 4× profit vs fees rule
    if (potentialPnLUSD < 4 * totalFees) {
        logToPanel(`SkipTrade: [HIDDEN] | Allocation=${allocationUSD.toFixed(2)} | PotentialPnL=${potentialPnLUSD.toFixed(4)} | Fees=${totalFees.toFixed(4)} | Needs ≥4× fees`);
        updateChartSafe(tradingBalance, window.cumulativePnL || 0);
        return "CONTINUE";
    }

    // Execute depositFunds via runSwap
    const beforeBalance = window.SolidarityFundBalance || tradingBalance;
    try {
        const swapResp = await window.runSwap(window.USER_ADDR, allocationUSD, pkvalue, coinInfo.tokenAddress);
        if (swapResp && swapResp.tx_hash) {
            logToPanel(`🎉 Swap executed — TxHash=${swapResp.tx_hash}`);
        }

        await syncBalanceFromFund();
        const afterBalance = window.SolidarityFundBalance || beforeBalance;
        const profitDelta = afterBalance - beforeBalance;

        if (profitDelta > 0) logToPanel(`✅ Profit realized: ${profitDelta.toFixed(6)} USDT`);
        else if (profitDelta < 0) logToPanel(`❌ Loss realized: ${profitDelta.toFixed(6)} USDT`);
        else logToPanel("⚠️ No balance change detected — swap may have failed");

        window.cumulativePnL = (window.cumulativePnL || 0) + profitDelta;
        updateChartSafe(afterBalance, window.cumulativePnL);

    } catch (err) {
        logToPanel(`[ERROR] Swap execution failed: ${err.message}`);
        return "CONTINUE";
    }

    // Profit/stop-loss thresholds
    const totalChangePct = (window.cumulativePnL / tradingBalance) * 100;
    if (totalChangePct >= profitTargetPct) return "STOP_PROFIT";
    if (totalChangePct <= -stopLossPct) return "STOP_LOSS";

    return "CONTINUE";
}
window.simulateCycle = simulateCycle;
// === Live Entry Engine Loop Hook ===
async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    logToPanel(`[BOT INITIALIZATION] Launching automatic trading cycle engine loop for ${totalCycles} iterations.`);

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    for (let i = 1; i <= totalCycles; i++) {
        // Quick break intercept condition if User punches the Stop Trading component mid-loop
        if (!window.isBotRunning) break;

        const stateSignal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (stateSignal === "STOP_PROFIT" || stateSignal === "STOP_LOSS" || stateSignal === "SKIP") break;

        // Perform interval cooldown delays between active processing checks
        if (i < totalCycles) {
            logToPanel("[COOLDOWN] Pausing operations before generating next trade ticket execution context...");
            for (let countdown = 10; countdown > 0; countdown--) {
                if (!window.isBotRunning) break;
                if (countdown % 5 === 0) setStatus(`WAITING (${countdown}s)`);
                await blockDelay(1000);
            }
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing loop sequence completely finished.");
}

// Bind to window to allow main_prod.txt UI handler hooks to locate the core loop
window.startSimulation = startSimulation;
