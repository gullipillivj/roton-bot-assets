};

// 🧠 Dynamic verification handler targeting live smart contract state maps
window.discoverActiveOnChainAsset = function(userAddress) {
    const baseApiUrl = window.API_BASE || window.location.origin;
    return fetch(`${baseApiUrl}/prodbot/active-asset`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ user: userAddress })
    })
    .then(res => res.json())
    .catch(err => {
        console.error("Grounding connection lost:", err);
        return { 
            status: "error", 
            confirmed_symbol: "USDT", 
            confirmed_address: "0x55d398326f99059fF775485246999027B3197955", 
            confirmed_balance_wei: "0" 
        };
    });
};

// 🟢 DIRECT BALANCING AGENT: Queries the absolute ledger balance cleanly
async function queryTokenBalanceDirect(userAddr, tokenAddress) {
    try {
        const baseApiUrl = window.API_BASE || window.location.origin;
        const res = await fetch(baseApiUrl + "/prodbot/balance", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: userAddr, token: tokenAddress })
        });
        const rawWeiText = await res.text();
        return rawWeiText.trim() || "0";
    } catch (err) {
        return "0";
    }
}
window.queryTokenBalanceDirect = queryTokenBalanceDirect;

window.updateUIBalanceControl = async function(userAddr, tokenAddress) {
    try {
        const cleanWeiString = await queryTokenBalanceDirect(userAddr, tokenAddress);

        if (BigInt(cleanWeiString) > 0n || !window.isBotRunning) {
            const startBox = document.getElementById("startBalance");
            const investBox = document.getElementById("investBalance");
            const usdBox = document.getElementById("usd");
            
            if (startBox) startBox.value = cleanWeiString;
            if (investBox) investBox.value = cleanWeiString;
            
            if (usdBox) {
                const floatTokens = parseFloat(cleanWeiString) / 1e18;
                usdBox.value = floatTokens.toFixed(4); 
            }
        }
        return cleanWeiString;
    } catch (err) {
        return "0";
    }
};

// LIVE BACKGROUND SYNC CLOCK: Maintains absolute ledger state safely without UI noise
window.initLiveBalanceClock = function() {
    if (window.balanceClockIntervalId) {
        clearInterval(window.balanceClockIntervalId);
    }
    
    window.balanceClockIntervalId = setInterval(async () => {
        if (!window.USER_ADDR || window.isBotRunning) return; 
        try {
            const stateRecord = await window.discoverActiveOnChainAsset(window.USER_ADDR);
            const activeTokenAddr = stateRecord.confirmed_address || "0x55d398326f99059fF775485246999027B3197955";
            await window.updateUIBalanceControl(window.USER_ADDR, activeTokenAddr);
        } catch (error) {
            // Silently handled
        }
    }, 5000); 
};

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

// 🚀 SECTION 2: GROUNDED ROTATIONAL PROFIT EVALUATOR & LOOP ENGINE (PART A)

async function startSimulation(totalCycles = 10) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    logToPanel(`[LIVE ENGINE INITIALIZED] Launching loop routines via velocity scoring.`);
    const baseApiUrl = window.API_BASE || window.location.origin;

    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";
    const CAKE_ADDRESS = "0x0E09FaBB73Bd3Ade0a17ECC321fD13a19e81cE82";

    // Authority dynamic token references tracking maps
    let currentAssetAddress = USDT_ADDRESS;
    let currentAssetSymbol = "USDT";
    let runningBalanceWeiLock = "0";

    // 🟢 DUAL-QUERY INITIAL RECOVERY HOOKS
    try {
        const tokenDbRes = await fetch(`${baseApiUrl}/prodbot/tokens/get`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: window.USER_ID })
        });
        const dbRecord = await tokenDbRes.json();
        let checkTokens = [USDT_ADDRESS, CAKE_ADDRESS];
        if (dbRecord && dbRecord.status === "ok") {
            if (dbRecord.currenttoken) checkTokens.unshift(dbRecord.currenttoken);
            if (dbRecord.lasttoken) checkTokens.push(dbRecord.lasttoken);
        }
        for (let targetAddr of checkTokens) {
            let balCheck = await queryTokenBalanceDirect(window.USER_ADDR, targetAddr);
            if (BigInt(balCheck) > 0n) {
                runningBalanceWeiLock = balCheck;
                currentAssetAddress = targetAddr;
                currentAssetSymbol = (targetAddr.toLowerCase() === USDT_ADDRESS.toLowerCase()) ? "USDT" : "CAKE";
                break;
            }
        }
    } catch (initErr) {
        console.error("Pipeline discovery failed:", initErr);
    }

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        logToPanel(`\n[INFO] — Cycle ${i}/${totalCycles} starting...`);
        window.setStatus(`RUNNING cycle ${i}/${totalCycles}`);

        // Pulling absolute values using current checked parameters dynamically
        let fetchedBalance = await queryTokenBalanceDirect(window.USER_ADDR, currentAssetAddress);
        if (BigInt(fetchedBalance) > 0n) {
            runningBalanceWeiLock = fetchedBalance;
            logToPanel(`[LEDGER RECORD] Active capital confirmed: ${runningBalanceWeiLock} Wei units inside ${currentAssetSymbol}.`);
        } else {
            logToPanel(`⚠️ Current token balance returned 0. Retaining balance vector: ${runningBalanceWeiLock}`);
        }

        let initialFieldText = runningBalanceWeiLock;
        const startBox = document.getElementById("startBalance");
        const investBox = document.getElementById("investBalance");
        const usdBox = document.getElementById("usd");
        if (startBox) startBox.value = initialFieldText;
        if (investBox) investBox.value = initialFieldText;
        if (usdBox) usdBox.value = (parseFloat(initialFieldText) / 1e18).toFixed(4);

        if (BigInt(initialFieldText) <= 0n) {
            logToPanel("❌ Pipeline Halted: Verified total depletion across all asset channels.");
            break;
        }

        let targetSymbol = "USDT";
        let targetAddress = USDT_ADDRESS;

        if (i !== totalCycles) {
            if (currentAssetAddress.toLowerCase() !== USDT_ADDRESS.toLowerCase()) {
                targetSymbol = "USDT";
                targetAddress = USDT_ADDRESS;
                logToPanel(`⚖️ Rotational Check: Re-routing assets back to USDT stability layer.`);
            } else {
                logToPanel(`⚖️ Velocity Engine Active: Filtering horizon vectors for maximum speed...`);
                try {
                    const dynamicMarket = await categorizeCoins();
                    let combinedPool = [];
                    if (dynamicMarket && Array.isArray(dynamicMarket.short_term)) combinedPool = combinedPool.concat(dynamicMarket.short_term);
                    if (dynamicMarket && Array.isArray(dynamicMarket.mid_term)) combinedPool = combinedPool.concat(dynamicMarket.mid_term);
                    if (dynamicMarket && Array.isArray(dynamicMarket.long_term)) combinedPool = combinedPool.concat(dynamicMarket.long_term);

                    let bestCandidate = null;
                    let peakVelocityRecorded = -999999;
                    let uniqueTracker = new Set();

                    for (let token of combinedPool) {
                        if (!token || !token.tokenAddress || uniqueTracker.has(token.tokenAddress.toLowerCase())) continue;
                        uniqueTracker.add(token.tokenAddress.toLowerCase());
                        let currentSpeed = parseFloat(token.change24h || 0.0);
                        if (currentSpeed > peakVelocityRecorded) {
                            peakVelocityRecorded = currentSpeed;
                            bestCandidate = token;
                        }
                    }

                    if (bestCandidate && peakVelocityRecorded > 0.05) {
                        targetSymbol = bestCandidate.symbol;
                        targetAddress = bestCandidate.tokenAddress;
                        logToPanel(`🔥 [MOMENTUM CHOICE] Selected ${targetSymbol} -> Climbing at +${bestCandidate.change24h}% velocity.`);
                    } else {
                        targetSymbol = "CAKE";
                        targetAddress = CAKE_ADDRESS;
                    }
                } catch (err) {
                    targetSymbol = "CAKE";
                    targetAddress = CAKE_ADDRESS;
                }
            }
        }

        // 🟢 ZERO-GAS OVERLAP PROTECTION CHECK
        if (targetAddress.toLowerCase() === currentAssetAddress.toLowerCase()) {
            logToPanel(`🛡️ [GAS PROTECTION] Asset sequence already aligned to target (${targetSymbol}). Skipping transaction execution to prevent fee burn.`);
            await blockDelay(5000);
            continue;
        }

        // ✅ FIX: Part B now returns the updated state values back to the orchestrator layer
        let executionState = await window.executeActualContractSwapBlock(baseApiUrl, initialFieldText, currentAssetAddress, targetAddress, currentAssetSymbol, targetSymbol);
        
        if (!executionState || !executionState.success) {
            break;
        }

        // ✅ FIX: Update the memory track pointers using the state variables returned from Part B
        currentAssetAddress = executionState.updatedAddress;
        currentAssetSymbol = executionState.updatedSymbol;

        if (i < totalCycles) await blockDelay(3000);
    }

    window.isBotRunning = false;
    window.setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Pipeline orchestration run complete.");
}
window.startSimulation = startSimulation;

// 🚀 SECTION 2: GROUNDED ROTATIONAL PROFIT EVALUATOR & LOOP ENGINE (PART B)
// ==================================================================
async function executeActualContractSwapBlock(baseApiUrl, initialFieldText, currentAssetAddress, targetAddress, currentAssetSymbol, targetSymbol) {
    logToPanel(`💾 Synchronizing tokens mapping index inside clients.db...`);
    try {
        await fetch(`${baseApiUrl}/prodbot/tokens/update`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
                userid: window.USER_ID,
                currenttoken: targetAddress,       
                lasttoken: currentAssetAddress     
            })
        });
        logToPanel(`✅ Database updated: [Last: ${currentAssetSymbol}] ➡️ [Current: ${targetSymbol}]`);
    } catch (dbUpdateErr) {
        console.error("Database connection failed:", dbUpdateErr);
    }

    logToPanel(`[ROUTE] Preparing transaction pipeline: Swap ${currentAssetSymbol} ➡️ ${targetSymbol}`);

    try {
        const swapResult = await window.runSwap(window.USER_ADDR, initialFieldText, currentAssetAddress, targetAddress);

        if (swapResult && swapResult.status === "ok") {
            logToPanel(`⏳ Signature confirmed. Starting settlement monitoring loop on-chain...`);
            let postSwapBalanceConfirmed = 0n;
            let pollAttempts = 0;
            
            while (postSwapBalanceConfirmed === 0n && pollAttempts < 40) {
                if (!window.isBotRunning) return { success: false };
                pollAttempts++;
                logToPanel(`🔍 Checking target balance via node array mapping (Attempt ${pollAttempts}/40)...`);
                
                let liveCheckWei = await queryTokenBalanceDirect(window.USER_ADDR, targetAddress);
                postSwapBalanceConfirmed = BigInt(liveCheckWei);
                
                if (postSwapBalanceConfirmed > 0n) {
                    logToPanel(`🎉 On-Chain Non-Zero balance discovered! Confirmed: ${liveCheckWei} Wei inside ${targetSymbol}.`);
                    await window.updateUIBalanceControl(window.USER_ADDR, targetAddress);
                    
                    // ✅ FIX: Package and return the brand new tracking pointers back to the main thread loop
                    return {
                        success: true,
                        updatedAddress: targetAddress,
                        updatedSymbol: targetSymbol
                    };
                }
                await blockDelay(5000); 
            }
            if (postSwapBalanceConfirmed === 0n) {
                logToPanel(`⚠️ Settlement check reached timeout thresholds. Forcing process break.`);
                return { success: false };
            }
        } else {
            const failReason = (swapResult && swapResult.message) ? swapResult.message : "Signature Denied / Nonce collision";
            logToPanel(`❌ [ORCHESTRATION BREAK] Loop halted: ${failReason}`);
            return { success: false };
        }
    } catch (err) {
        logToPanel(`❌ Critical Loop Exception: ${err.message}`);
        return { success: false };
    }
    return { success: false };
}
window.executeActualContractSwapBlock = executeActualContractSwapBlock;
