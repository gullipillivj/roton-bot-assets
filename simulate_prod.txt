// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

// Thread state delays helper
const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

// UI Status label updater
function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

// Emergency Halt Button Hook
function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Tracking execution cycle paused by user.");
}
window.stopBot = stopBot;

// ==================================================================
// 🔍 SINGLE-FUNCTION PURE INVENTORY INSPECTOR (NO ROUTERS / NO DB ROUTES)
// ==================================================================
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`\n=== 📊 [INVENTORY SCAN] Cycle ${cycle}/${totalCycles} ===`);
    setStatus(`SCANNING ${cycle}/${totalCycles}`);

    // Read the current live balance parameter straight from the form element container properties
    let balanceInputBefore = document.getElementById("investBalance") ? document.getElementById("investBalance").value.toString().trim() : "0";
    
    // FIXED: Pulled live coordinates straight from the synchronized memory variable populated by syncBalanceFromFund()
    // Defaulting directly to the token address or standard USDT coordinate without triggering broken database endpoints
    let currentTokenAddress = window.USER_ADDR || "0x55d398326f99059fF775485246999027B3197955";
    logToPanel(`💳 Account Address Vector: [${currentTokenAddress}] | Parameter Balance: ${balanceInputBefore} Wei`);

    // Ingest the token trend data array from your coins.txt registry function
    let marketData = [];
    try {
        if (typeof window.getTradeCandidates === "function") {
            marketData = await window.getTradeCandidates();
        } else {
            logToPanel("⚠️ window.getTradeCandidates function is missing from current visibility layout context.");
            return "SKIP";
        }
    } catch (fErr) {
        logToPanel(`❌ Data Array Retrieval Interrupted: ${fErr.message}`);
        return "CONTINUE";
    }

    if (!Array.isArray(marketData) || marketData.length === 0) {
        logToPanel("❌ Ingestion tracking market data array is completely empty.");
        return "SKIP";
    }

    logToPanel(`📋 Printing ${marketData.length} records discovered inside coins.txt array matrix:`);

    // Parse and view properties sequentially inside a pure single execution loop block
    for (let index = 0; index < marketData.length; index++) {
        let candidate = marketData[index];
        
        let symbolLabel = candidate.symbol || candidate.token_symbol || candidate.name || "UNKNWN";
        let targetAddress = candidate.address || candidate.address_or_geckoid || "MISSING";
        let profitVelocity = candidate.change_24h_percent || candidate["%profit"] || 0;
        let networkFramework = candidate.tokentype || "bep20";
        let marketPrice = candidate.price || candidate.live_price_usd || 0;

        logToPanel(`🔹 [Token ${index + 1}/${marketData.length}]: ${String(symbolLabel).toUpperCase()}
   ▫️ Address Hex  : ${String(targetAddress).toLowerCase().trim()}
   ▫️ Network Type : ${networkFramework}
   ▫️ 24h Momentum : ${profitVelocity}%
   ▫️ Market Price : $${marketPrice}
   ───────────────────────────────────────────────────────`);
    }

    logToPanel(`=== 📊 [SCAN SUCCESSFUL] Cycle loop finished. Wallet providers bypassed. ===`);
    return "SKIP"; // Instructs master loop manager to skip transaction executions cleanly
}
window.simulateCycle = simulateCycle;

// ==================================================================
// 🚀 MASTER SIMULATION CONTROLLER AND INTERFACE GATEWAY
// ==================================================================
async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    logToPanel(`🚀 [DEBUG BOT STARTED] Initiated data array inspection for up to ${totalCycles} loops.`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        
        // Triggers the pure isolated data parsing layout function defined above
        const signal = await simulateCycle(i, totalCycles, 0, 0);
        
        if (signal === "STOP_ERROR") {
            logToPanel("🛑 Debug run interrupted execution check.");
            break;
        }
        
        // Small interval loop separation buffer delay parameter
        await blockDelay(1000);
    }
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing complete.");
}
window.startSimulation = startSimulation;
