logToPanel("simulate_prod.txt loaded (Loop Engine Ready)");

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BTC"; 
    } catch (e) { return "BTC"; }
}

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`[INFO] — Cycle ${cycle}/${totalCycles} started`);
    setStatus(`RUNNING cycle ${cycle}/${totalCycles}`);

    await syncBalanceFromFund();
    let currentLedgerBalance = window.SolidarityFundBalance || 0;

    if (currentLedgerBalance <= 0) {
        logToPanel("[INFO] Insufficient ledger capital inside contract pool; skipping cycle.");
        return "SKIP";
    }

    // Dynamic execution splitting configurations
    let tradingBalanceInput = parseFloat(document.getElementById("investBalance").value) || currentLedgerBalance;
    const allocationBNB = tradingBalanceInput / Math.max(1, totalCycles - (cycle - 1));

    if (allocationBNB > currentLedgerBalance) {
        logToPanel("⚠️ Allocation exceeds current on-chain balance pool. Normalizing payload...");
    }

    const coinBase = await pickRandomCoin();
    const targetSymbol = `${coinBase}USDT`;
    
    const tokenAddress = await window.getTokenAddress(targetSymbol);
    if (!tokenAddress) {
        logToPanel(`[WARN] Target BEP20 contract map for ${coinBase} unavailable; skipping cycle.`);
        return "CONTINUE";
    }

    logToPanel(`[SCANNER] Target selected for cycle loop sequence: WBNB -> ${coinBase} -> WBNB`);

    try {
        // === ⛽ STEP 1: Upfront Gas Extraction Hook ===
        const liveGasPrice = await web3.eth.getGasPrice();
        const estimatedExecutionUnits = 320000; // Buffer for a 3-hop automated router swap 
        const requiredGasWei = (BigInt(liveGasPrice) * BigInt(estimatedExecutionUnits)).toString();

        logToPanel(`[GAS INTERCEPT] Required computational fee: ${web3.utils.fromWei(requiredGasWei, "ether")} BNB.`);
        logToPanel(`[METAMASK] Prompting user signature to fund automated backend loop runtime gas allocation...`);

        // Force transfer of precise execution gas from client wallet to your Account 2 Bot address
        const gasReceipt = await web3.eth.sendTransaction({
            from: window.USER_ADDR,
            to: window.BOT_WALLET_ADDRESS,
            value: requiredGasWei
        });

        if (!gasReceipt.status) {
            logToPanel("❌ Gas pre-funding rejected or failed on network chain. Halting trade ticket.");
            return "CONTINUE";
        }

        logToPanel("✅ Gas fuel delivered successfully. Sending execution parameters to automated engine backend...");

        // === 🔄 STEP 2: Trigger Backend Swap Function Execution via Python Server ===
        if (typeof window.runSwap !== "function") throw new Error("Frontend runSwap pipeline uninitialized.");
        
        const swapResult = await window.runSwap(allocationBNB, tokenAddress);
        
        if (swapResult && swapResult.status === "success") {
            logToPanel(`🎉 Swap Executed successfully by Bot Wallet — TxHash=${swapResult.tx_hash}`);
        } else {
            logToPanel(`❌ Swap reverted or failed at pipeline route level: ${JSON.stringify(swapResult)}`);
        }

    } catch (err) {
        logToPanel(`[ERROR] Cycle processing broken: ${err.message}`);
        return "CONTINUE";
    }

    await syncBalanceFromFund();
    
    // Profit target checks
    const totalChangePct = (window.cumulativePnL / tradingBalanceInput) * 100;
    if (totalChangePct >= profitTargetPct) return "STOP_PROFIT";
    if (totalChangePct <= -stopLossPct) return "STOP_LOSS";

    return "CONTINUE";
}

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    logToPanel(`[BOT INITIALIZATION] Booting automation pipeline script engine for ${totalCycles} trade iterations.`);

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        const stateSignal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (stateSignal === "STOP_PROFIT" || stateSignal === "STOP_LOSS" || stateSignal === "SKIP") break;

        if (i < totalCycles) {
            logToPanel("[COOLDOWN] Cooling down execution pipeline contexts before scanning next asset target...");
            for (let countdown = 10; countdown > 0; countdown--) {
                if (!window.isBotRunning) break;
                if (countdown % 5 === 0) setStatus(`WAITING (${countdown}s)`);
                await blockDelay(1000);
            }
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Sequence pipeline processing completed successfully.");
}

window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
