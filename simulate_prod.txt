function logToPanel(msg) {
  const panel = document.getElementById("logPanel");
  if (panel) {
    panel.innerText += "\n" + msg;
    panel.scrollTop = panel.scrollHeight;
  } else {
    console.log(msg);
  }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

async function pickRandomCoin() {
    try {
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BNB"; 
    } catch (e) { return "BNB"; }
}

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) statusLabel.textContent = msg;
    else logToPanel(`[STATUS] ${msg}`);
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Trading halted by user.");
}
window.stopBot = stopBot;

async function fetchLiveContractBalance(user, token) {
    try {
        const baseApiUrl = window.API_BASE || "";
        const res = await fetch(`${baseApiUrl}/prodbot/balance`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: user, token: token })
        });
        const data = await res.json();
        return data.balance || "0";
    } catch (err) {
        return "0";
    }
}

async function pollForSpentTokenBalanceChange(userAddr, spentTokenAddress, initialValue, maxTimeoutSeconds) {
    const checkIntervalMs = 4000;
    const maxAttempts = (maxTimeoutSeconds * 1000) / checkIntervalMs;
    
    for (let attempt = 0; attempt < maxAttempts; attempt++) {
        if (!window.isBotRunning) return false;
        await blockDelay(checkIntervalMs);
        
        let currentLiveBalance = await fetchLiveContractBalance(userAddr, spentTokenAddress);
        if (BigInt(currentLiveBalance) !== BigInt(initialValue)) {
            return true;
        }
        if ((attempt + 1) % 3 === 0) {
            setStatus(`POLLING (${maxTimeoutSeconds - Math.floor((attempt * checkIntervalMs)/1000)}s left)`);
        }
    }
    return false;
}

// ✅ Main Refactored Master Asynchronous State Trading Execution Engine
async function startSimulation(totalCycles = 10) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    logToPanel(`[BOT INITIALIZATION] Booting Dynamic State Tracker Engine for ${totalCycles} iterations.`);

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    const USDT_ADDRESS = "0x55d398326f99059fF775485246999027B3197955";

    // 💡 STEP 1: RUN THE AUTO-RECOVERY HARVESTER BEFORE THE SWAP LOOPS START
    setStatus("SYNCHRONIZING");
    const activePortfolio = await autoSynchronizeActivePortfolioState(window.USER_ADDR);
    
    // Dynamic tracking variables update to point exactly to where your money is right now
    let currentlyHeldTokenAddress = activePortfolio.address;
    let currentlyHeldTokenSymbol = activePortfolio.symbol;

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;

        logToPanel(`[INFO] — Cycle ${i}/${totalCycles} started`);
        setStatus(`RUNNING cycle ${i}/${totalCycles}`);

        // Query the live balance of the dynamic asset we currently hold
        let spentTokenInitialBalance = await fetchLiveContractBalance(window.USER_ADDR, currentlyHeldTokenAddress);
        logToPanel(`[STATUS] Active Portfolio Target: ${currentlyHeldTokenSymbol} (Balance: ${spentTokenInitialBalance} Wei)`);

        if (BigInt(spentTokenInitialBalance) <= 0n) {
            logToPanel(`❌ Pipeline Halted: Balance mapping for ${currentlyHeldTokenSymbol} evaluates to 0.`);
            break;
        }

        let targetTokenSymbol = "";
        let targetTokenAddress = "";
        let networkParam = "bep20";

        if (i === totalCycles) {
            // 💡 THE 10TH CYCLE FINAL EXIT RULE: Always clear custom assets back into USDT
            targetTokenSymbol = "USDT";
            targetTokenAddress = USDT_ADDRESS;
            logToPanel(`[EXIT STRATEGY] Cycle 10 reached. Liquidating portfolio positions back into USDT.`);
        } else {
            // Cycles 1-9: Dynamic random generation from asset text pools
            const coinBase = await pickRandomCoin();
            targetTokenSymbol = `${coinBase}`;
            const targetSymbolPair = `${coinBase}USDT`;
            
            const coinInfo = await window.evaluateCoin(targetSymbolPair);
            targetTokenAddress = coinInfo?.tokenAddress || "0xeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeeee";
            networkParam = coinInfo?.network || "bep20";
        }

        // Avoid identical redundant circular routings
        if (targetTokenAddress.toLowerCase() === currentlyHeldTokenAddress.toLowerCase()) {
            logToPanel(`[INFO] Selected identical asset variant ${targetTokenSymbol}. Auto-skipping to avoid empty loop logs.`);
            continue;
        }

        logToPanel(`[INFO] Processing Route: Swap ${currentlyHeldTokenSymbol} ➡️ ${targetTokenSymbol}`);

        try {
            const baseApiUrl = window.API_BASE || "";
            const res = await fetch(`${baseApiUrl}/prodbot/swap`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    user_address: window.USER_ADDR,
                    from_token: currentlyHeldTokenAddress,
                    to_token: targetTokenAddress,
                    amount: spentTokenInitialBalance,
                    network: networkParam
                })
            });

            const backendData = await res.json();

            if (backendData && backendData.status === "ok") {
                logToPanel(`[INFO] Prompting secure browser signature from your active Web3 wallet layout...`);

                const txHash = await window.ethereum.request({
                    method: 'eth_sendTransaction',
                    params: [{
                        from: window.USER_ADDR,
                        to: backendData.to_contract,
                        data: backendData.compiled_data,
                        value: '0x0'
                    }],
                });

                logToPanel(`📡 Transaction broadcasted to miners! Hash identity: ${txHash}`);
                logToPanel(`⏳ Freezing timer loop. Polling contract mapping database for state modification...`);

                // Hold code iteration step open until on-chain storage changes status
                const txCleared = await pollForSpentTokenBalanceChange(
                    window.USER_ADDR, 
                    currentlyHeldTokenAddress, 
                    spentTokenInitialBalance, 
                    120
                );

                if (txCleared) {
                    logToPanel(`✅ State modification cleared by blockchain miners!`);
                    
                    // Shift state tracking variables over to reflect the swap outcome asset
                    currentlyHeldTokenAddress = targetTokenAddress;
                    currentlyHeldTokenSymbol = targetTokenSymbol;
                } else {
                    logToPanel(`❌ Execution Timeout: Block confirmation state failed to alter. Halted pipeline.`);
                    break;
                }

            } else {
                logToPanel(`❌ Backend Error: Compiled calldata string compilation failed or skipped.`);
                break;
            }

        } catch (err) {
            logToPanel(`❌ Pipeline Exception Encountered: ${err.message}`);
            break;
        }

        if (i < totalCycles) {
            logToPanel("[COOLDOWN] Restoring operational loops frameworks for next iteration scan...");
            await blockDelay(3000);
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Sequence pipeline processing completed successfully.");
}

window.startSimulation = startSimulation;
