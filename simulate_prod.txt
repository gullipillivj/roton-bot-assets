function logToPanel(msg) {
    const panel = document.getElementById("logPanel");

    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}

window.logToPanel = logToPanel;

const blockDelay = ms =>
    new Promise(resolve => setTimeout(resolve, ms));

// ==========================================
// TOKEN DB SYNC
// ==========================================
async function updateTokenRecord(
    lastToken,
    currentToken
) {
    try {
        const userid =
            window.USER_ID ||
            localStorage.getItem("userid");

        if (!userid) {
            logToPanel(
                "[TOKEN SYNC] Missing userid"
            );
            return;
        }

        const resp = await fetch(
            "/prodbot/tokens/update",
            {
                method: "POST",
                headers: {
                    "Content-Type":
                        "application/json"
                },
                body: JSON.stringify({
                    userid: userid,
                    lasttoken: lastToken,
                    currenttoken: currentToken
                })
            }
        );

        const data = await resp.json();

        logToPanel(
            `[TOKEN SYNC] ${data.message}`
        );
    } catch (err) {
        logToPanel(
            `[TOKEN SYNC ERROR] ${err.message}`
        );
   
}
}


async function pickRandomCoin() {

    try {

        if (typeof window.getBinancePairs === "function") {

            const pairs =
                await window.getBinancePairs();

            if (pairs && pairs.length > 0) {

                return pairs[
                    Math.floor(
                        Math.random() * pairs.length
                    )
                ].replace("USDT", "");
            }
        }

    } catch (err) {
        console.log(err);
    }

    return "CAKE";
}

function setStatus(msg) {

    const statusLabel =
        document.querySelector(".status-label");

    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}

window.setStatus = setStatus;

function stopBot() {

    window.isBotRunning = false;

    setStatus("STOPPED");

    logToPanel(
        "[BOT STOPPED] Trading halted by user."
    );
}

window.stopBot = stopBot;

async function swapRR(amount) {

    try {

        const resp =
            await window.runSwapUniswap(

                window.USER_ADDR,

                amount,

                "0xba67186fd454D0256a0D3744783538bE08C58feD",

                "0x8A0cAC8E7F6305cE6EeB585005919290F78808f9"
            );

        if (resp && resp.tx_hash) {

            logToPanel(
                `[RR_FALLBACK] Swap Executed - ${resp.tx_hash}`
            );
        }

    } catch (err) {

        logToPanel(
            `[RR_FALLBACK ERROR] ${err.message}`
        );
    }

    return "CONTINUE";
}

// SINGLE TRADING CYCLE
// =====================================================
// =====================================================
// SINGLE TRADING CYCLE (FULLY CORRECTED FOR DYNAMIC API)
// =====================================================
// =====================================================================
// SINGLE TRADING CYCLE (FULLY STRIPPED OF RR FALLBACK CALLS)
// =====================================================================
async function simulateCycle(
    cycle,
    totalCycles,
    profitTargetPct,
    stopLossPct
) {

    logToPanel(
        `[INFO] Cycle ${cycle}/${totalCycles}`
    );

    setStatus(
        `RUNNING ${cycle}/${totalCycles}`
    );

    const syncedBalance =
        await syncBalanceFromFund();

    if (syncedBalance === null) {
        logToPanel(
            "[WARN] Balance sync failed. Skipping loop cycle."
        );
        return "CONTINUE"; // 🌟 RR REMOVED
    }

    let tradingBalance =
        parseFloat(
            document.getElementById(
                "investBalance"
            ).value
        ) || 0;

    if (tradingBalance <= 0) {
        logToPanel(
            "[WARN] No trading balance available."
        );
        return "CONTINUE"; // 🌟 RR REMOVED
    }

    // ==========================================
    // PER-CYCLE ALLOCATION
    // ==========================================

    const totalBalance =
        BigInt(
            document
                .getElementById(
                    "startBalance"
                )
                .value
                .toString()
                .trim()
        );

    const remainingCycles =
        Math.max(
            1,
            totalCycles - cycle + 1
        );

    const allocationUSD =
        (
            totalBalance /
            BigInt(remainingCycles)
        ).toString();

    logToPanel(
        `[ALLOCATION]
TOTAL=${totalBalance}
ALLOCATED=${allocationUSD}`
    );

    // ==========================================
    // LOAD CURRENT TOKEN
    // ==========================================

    let currentToken =
        "0x55d398326f99059fF775485246999027B3197955";

    try {

        const tokenResp =
            await fetch(
                "/prodbot/tokens/get",
                {
                    method: "POST",
                    headers: {
                        "Content-Type":
                            "application/json"
                },
                    body: JSON.stringify({
                        userid:
                            window.USER_ID
                    })
                }
            );

        const tokenData =
            await tokenResp.json();

        if (
            tokenData &&
            tokenData.currenttoken
        ) {
            currentToken =
                tokenData.currenttoken;

            logToPanel(
                `[TOKEN LOAD]
${currentToken}`
            );
        }

    } catch (err) {
        logToPanel(
            `[TOKEN LOAD ERROR]
${err.message}`
        );
    }

    // ==========================================
    // PICK COIN
    // ==========================================

    const selectedCoin =
        await pickRandomCoin();

    const coinInfo =
        await window.evaluateCoin(
            selectedCoin
        );

    if (
        !coinInfo ||
        !coinInfo.tokenAddress
    ) {
        logToPanel(
            "[WARN] Invalid or unprofiled target token. Proceeding..."
        );
        return "CONTINUE"; // 🌟 RR REMOVED
    }

    // ==========================================
    // COST-AWARE FILTER
    // ==========================================

    if (
        Number(
            coinInfo.change24h || 0
        ) < 3
    ) {
        logToPanel(
            `[SKIP]
${coinInfo.symbol}
MOVE=${coinInfo.change24h}%`
        );
        return "CONTINUE";
    }

    logToPanel(
        `[TARGET]
${coinInfo.symbol}
MOVE=${coinInfo.change24h}%
ADDRESS=${coinInfo.tokenAddress}`
    );

    const beforeBalance =
        document.getElementById(
            "investBalance"
        ).value
        .toString()
        .trim();

    try {

        logToPanel(
            `[SWAP REQUEST]
FROM=${currentToken}
TO=${coinInfo.tokenAddress}
AMOUNT=${allocationUSD}`
        );

        const swapResp =
            await window.runSwapUniswap(
                window.USER_ADDR,
                allocationUSD,
                currentToken,
                coinInfo.tokenAddress
            );

        if (
            !swapResp ||
            !swapResp.tx_hash
        ) {
            logToPanel(
                "[SWAP PIPELINE SKIPPED OR REJECTED]"
            );
            return "CONTINUE"; // 🌟 RR REMOVED
        }

        logToPanel(
            `[TX SENT]
${swapResp.tx_hash}`
        );

        // ==========================================
        // WAIT FOR ACTUAL BALANCE CHANGE
        // ==========================================

        let balanceChanged = false;

        for (
            let attempt = 1;
            attempt <= 30;
            attempt++
        ) {

            await blockDelay(
                10000
            );

            await syncBalanceFromFund();

            const currentBalance =
                document
                    .getElementById(
                        "investBalance"
                    )
                    .value
                    .toString()
                    .trim();

            logToPanel(
                `[VERIFY ${attempt}/30]
START=${beforeBalance}
NOW=${currentBalance}`
            );

            if (
                currentBalance !==
                beforeBalance
            ) {
                balanceChanged =
                    true;

                logToPanel(
                    `[BALANCE VERIFIED]
${beforeBalance}
➡
${currentBalance}`
                );
                break;
            }
        }

        // ==========================================
        // NO MOVEMENT TRACKING
        // ==========================================

        if (!balanceChanged) {
            logToPanel(
                "[NO BALANCE MOVEMENT DETECTED — PASSING TO NEXT LOOP]"
            );
            return "CONTINUE"; // 🌟 RR REMOVED
        }

        // ==========================================
        // UPDATE TOKENS TABLE
        // ==========================================

        await updateTokenRecord(
            currentToken,
            coinInfo.tokenAddress
        );

        logToPanel(
            `[TOKEN UPDATED]
LAST=${currentToken}
CURRENT=${coinInfo.tokenAddress}`
        );

        await syncBalanceFromFund();

        const afterBalance =
            parseFloat(
                document
                    .getElementById(
                        "investBalance"
                    )
                    .value
            ) || 0;

        const beforeFloat =
            parseFloat(
                beforeBalance
            ) || 0;

        const profitDelta =
            afterBalance -
            beforeFloat;

        window.cumulativePnL =
            (
                window.cumulativePnL ||
                0
            ) +
            profitDelta;

        if (
            typeof updateChart ===
            "function"
        ) {
            updateChart(
                afterBalance,
                window.cumulativePnL
            );
        }

    } catch (err) {
        logToPanel(
            `[RUNTIME RUN EXCEPTION]
${err.message}`
        );
        return "CONTINUE"; // 🌟 RR REMOVED
    }

    const totalChangePct =
        (
            window.cumulativePnL /
            Math.max(
                tradingBalance,
                1
            )
        ) * 100;

    if (
        totalChangePct >=
        profitTargetPct
    ) {
        logToPanel(
            "[TARGET REACHED]"
        );
        return "STOP_PROFIT";
    }

    if (
        totalChangePct <=
        -stopLossPct
    ) {
        logToPanel(
            "[STOP LOSS HIT]"
        );
        return "STOP_LOSS";
    }

    return "CONTINUE";
}





// =====================================================
// BOT START
// =====================================================

async function startSimulation(
    totalCycles
) {

    if (window.isBotRunning)
        return;

    window.isBotRunning = true;

    window.cumulativePnL = 0;

    const pTarget =
        parseFloat(
            document.getElementById(
                "profitTarget"
            ).value
        ) || 1.0;

    const sLoss =
        parseFloat(
            document.getElementById(
                "stopLoss"
            ).value
        ) || 0.5;

    logToPanel(
        `[BOT START] ${totalCycles} cycles`
    );

    for (
        let i = 1;
        i <= totalCycles;
        i++
    ) {

        if (!window.isBotRunning)
            break;

        const signal =
            await simulateCycle(
                i,
                totalCycles,
                pTarget,
                sLoss
            );

        if (
            signal === "STOP_PROFIT" ||
            signal === "STOP_LOSS"
        ) {
            break;
        }

        if (i < totalCycles) {

            setStatus(
                "WAITING"
            );

            await blockDelay(
                10000
            );
        }
    }

    window.isBotRunning = false;

    setStatus("STOPPED");

    logToPanel(
        "[BOT FINISHED]"
    );
}

window.startSimulation =
    startSimulation;

window.simulateCycle =
    simulateCycle;
