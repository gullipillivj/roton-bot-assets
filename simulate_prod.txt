// simulate_prod.txt — Live Runtime Simulation Cycle Engine Loop
logToPanel("simulate_prod.txt loaded (Loop Engine Ready)");

const blockDelay = ms => new Promise(res => setTimeout(res, ms));

// === Dynamic Live Coin Pair Picker ===
async function pickRandomCoin() {
    try {
        // Safe check for Binance pairs payload list hook
        if (typeof window.getBinancePairs === "function") {
            const pairs = await window.getBinancePairs();
            if (pairs && pairs.length > 0) {
                return pairs[Math.floor(Math.random() * pairs.length)].replace("USDT", "");
            }
        }
        return "BTC"; 
    } catch (e) {
        return "BTC";
    }
}

// === Automated Transaction Logic Core ===
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`[CYCLE START] ${new Date().toLocaleTimeString()} — Processing Loop ${cycle}/${totalCycles}`);
    setStatus(`RUNNING CYCLE ${cycle}/${totalCycles}`);

    const tradingBalance = parseFloat(document.getElementById("investBalance").value) || 0;
    if (tradingBalance <= 0) {
        logToPanel("[SKIP] Terminating loop: Available investment balance is 0 USDT.");
        return "SKIP";
    }

    // Allocate funds step calculation across remaining iteration tracks
    const allocationUSD = tradingBalance / Math.max(1, totalCycles - (cycle - 1));
    const targetAsset = await pickRandomCoin();
    
    logToPanel(`[ANALYSIS] Target asset: ${targetAsset}/USDT | Allocation: ${allocationUSD.toFixed(2)} USDT`);

    // Simulated market calculation based on live tracking parameters
    let variance = (Math.random() * 2 - 0.95); // Yields simulated change percent shifts
    let netAfterFees = (allocationUSD * variance) / 100;

    if (netAfterFees <= 0) {
        logToPanel(`[SKIP TRADE] Negative yield index for ${targetAsset}. Bypassing execution step to safeguard capital.`);
        return "CONTINUE";
    }

    // Increment global earnings metrics and report results
    window.cumulativePnL = (window.cumulativePnL || 0) + netAfterFees;
    logToPanel(`[SUCCESS] Orders matched for ${targetAsset}. Realized PnL: +${netAfterFees.toFixed(4)} USDT`);

    // Sync Live Dashboard Metric text updates
    const liveProfitSpan = document.getElementById("liveProfitValue");
    if (liveProfitSpan) {
        liveProfitSpan.textContent = `${window.cumulativePnL.toFixed(2)} USDT`;
        liveProfitSpan.style.color = window.cumulativePnL >= 0 ? "#28a745" : "#dc3545";
    }

    // Dynamic Chart rendering node pipeline processing update
    if (window.balanceChart) {
        const timeLabel = new Date().toLocaleTimeString();
        window.balanceChart.data.labels.push(timeLabel);
        // Current investment value pipeline tracing
        window.balanceChart.data.datasets[0].data.push(tradingBalance + window.cumulativePnL);
        // Net profit progression line tracing
        if (window.balanceChart.data.datasets[1]) {
            window.balanceChart.data.datasets[1].data.push(window.cumulativePnL);
        }
        window.balanceChart.update();
    }

    // Evaluate Risk Limit Threshold Boundaries
    const totalChangePct = (window.cumulativePnL / tradingBalance) * 100;
    if (totalChangePct >= profitTargetPct) {
        logToPanel(`[TRIGGER] Profit Target milestone of ${profitTargetPct}% achieved! Closing cycle loop safely.`);
        return "STOP_PROFIT";
    }
    if (totalChangePct <= -stopLossPct) {
        logToPanel(`[TRIGGER] Stop Loss threshold bound at -${stopLossPct}% breached. Liquidating assets to protect wallet.`);
        return "STOP_LOSS";
    }

    return "CONTINUE";
}

// === Live Entry Engine Loop Hook ===
async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    logToPanel(`[BOT INITIALIZATION] Launching automatic trading cycle engine loop for ${totalCycles} iterations.`);

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    for (let i = 1; i <= totalCycles; i++) {
        // Quick break intercept condition if User punches the Stop Trading component mid-loop
        if (!window.isBotRunning) break;

        const stateSignal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (stateSignal === "STOP_PROFIT" || stateSignal === "STOP_LOSS" || stateSignal === "SKIP") break;

        // Perform interval cooldown delays between active processing checks
        if (i < totalCycles) {
            logToPanel("[COOLDOWN] Pausing operations before generating next trade ticket execution context...");
            for (let countdown = 10; countdown > 0; countdown--) {
                if (!window.isBotRunning) break;
                if (countdown % 5 === 0) setStatus(`WAITING (${countdown}s)`);
                await blockDelay(1000);
            }
        }
    }

    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing loop sequence completely finished.");
}

// Bind to window to allow main_prod.txt UI handler hooks to locate the core loop
window.startSimulation = startSimulation;
