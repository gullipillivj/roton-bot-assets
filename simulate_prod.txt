/**
 * On-chain token state resolver.
 * Hits the updated backend route to read active positions from getUserBalanceList tuple fields.
 */
async function getCurrentTokenState() {
    let state = { token: "0x55d398326f99059fF775485246999027B3197955", network: "bep20" };
    
    try {
        const activeIdentityVector = window.USER_ADDR || window.USER_ID;
        if (!activeIdentityVector) {
            return state;
        }

        const resp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: activeIdentityVector })
        });
        
        const data = await resp.json();
        if (data && data.status === "ok" && data.currenttoken) {
            state.token = data.currenttoken;
            state.network = data.network || "bep20";
        }
    } catch (err) {
        logToPanel(`[ON-CHAIN POSITION LOOKUP FAILED] Defaulting to USDT: ${err.message}`);
    }
    
    return state;
}
window.getCurrentTokenState = getCurrentTokenState;

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`\n--- [START] Cycle ${cycle}/${totalCycles} ---`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);

    let balanceInputBefore = document.getElementById("investBalance").value.toString().trim();
    if (!balanceInputBefore || balanceInputBefore === "0") {
        logToPanel("⚠️ [CYCLE ABORTED] Visual input field 'investBalance' is empty or zero.");
        return "STOP_ERROR";
    }

    const srcAsset = await getCurrentTokenState();
    const sourceTokenAddress = String(srcAsset.token).toLowerCase().trim();
    logToPanel(`💳 [PORTFOLIO BALANCE STATE] Current Active Asset Hex: [${sourceTokenAddress}]`);

    let marketData = [];
    try {
        if (typeof window.getTradeCandidates === "function") {
            marketData = await window.getTradeCandidates();
        }
    } catch (fErr) {
        logToPanel(`⚠️ [SCANNER DELAY] Failed to invoke window.getTradeCandidates: ${fErr.message}`);
        return "CONTINUE";
    }

    if (!Array.isArray(marketData) || marketData.length === 0) {
        logToPanel("⚠️ [SCANNER COLD] Zero validated tokens compiled by data layer registry. Skipping cycle...");
        return "SKIP";
    }

    // DIAGNOSTIC 1: Print the complete cleaned array data inside the layout text panel
    logToPanel(`📊 [INGESTION REPORT] Discovered ${marketData.length} clean market targets inside registry data vector.`);
    let targetSymbolsList = marketData.map(t => t.symbol || t.name || "UNKNWN");
    logToPanel(`📋 [SCANNER TRACKING ARRAY]: ${JSON.stringify(targetSymbolsList)}`);

    let executionSuccessful = false;
    let currentCandidateIndex = 0;

    while (currentCandidateIndex < marketData.length) {
        let candidate = marketData[currentCandidateIndex];
        let candidateAddress = String(candidate.address).toLowerCase().trim();
        let targetedWinnerSymbol = String(candidate.symbol || "UNKNWN").toUpperCase().trim();
        let winnerNetworkType = String(candidate.tokentype || "bep20").toLowerCase().trim();

        if (!candidateAddress || candidateAddress === "n/a" || candidateAddress === sourceTokenAddress) {
            currentCandidateIndex++;
            continue;
        }

        // DIAGNOSTIC 2: Log explicit route parameters to track exactly what inputs hit your simulation endpoint
        logToPanel(`\n🔍 [PRE-FLIGHT CHECK ENGINE] Testing Path Configuration:`);
        logToPanel(`   ➡️ Input Token In  (You Hold) : [${sourceTokenAddress}]`);
        logToPanel(`   ➡️ Output Token Out (Candidate): [${candidateAddress}] (${targetedWinnerSymbol})`);
        logToPanel(`   ➡️ Expected Allocation Size   : ${balanceInputBefore} Wei`);

        try {
            const simRes = await fetch("/prodbot/swap/simulateuniswap", {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    router: "0x10ED43C718714eb63d5aA57B78B54704E256024E",
                    tokenIn: sourceTokenAddress,
                    tokenOut: candidateAddress,
                    amountIn: balanceInputBefore,
                    amountOutMin: "0"
                })
            });

            const contentType = simRes.headers.get("content-type");
            if (!contentType || !contentType.includes("application/json")) {
                const rawCrashHTML = await simRes.text();
                logToPanel(`❌ [SIMULATION CRASHED] Backend returned HTML error code payload block instead of JSON.`);
                console.error("HTML Crash Log:", rawCrashHTML);
                currentCandidateIndex++;
                continue;
            }

            const simData = await simRes.json();
            
            // DIAGNOSTIC 3: Print the exact structural JSON object returned by the server simulation endpoint
            logToPanel(`📦 [RAW BACKEND SIMULATION RESPONSE]:\n${JSON.stringify(simData, None, 2)}`);
            
            // THE FINANCIAL FIREWALL: Instantly drop the candidate if backend validation checks return errors
            if (!simData || simData.status !== "ok" || !simData.expected_output_wei || BigInt(simData.expected_output_wei) <= 0n) {
                logToPanel(`⚠️ [ROUTE REJECTED] Token ${targetedWinnerSymbol} failed simulation verification. Moving to next target...`);
                currentCandidateIndex++;
                continue; 
            }

            let calculatedOutput = BigInt(simData.expected_output_wei);
            let baselineInput = BigInt(balanceInputBefore);
            
            if (calculatedOutput <= baselineInput) {
                logToPanel(`⏭️ [ROUTE REJECTED] Token ${targetedWinnerSymbol} skipped: Simulation indicates negative returns or high slippage impact.`);
                currentCandidateIndex++;
                continue;
            }

            // Verification Complete! Direct swap execution will only run if parameters are 100% clean
            logToPanel(`🎯 [ROUTE VERIFIED CLEAN] Target verified successfully. Dispatched straight to contract routers...`);
            executionSuccessful = await executeDirectOnChainSwap(
                candidateAddress, targetedWinnerSymbol, winnerNetworkType, sourceTokenAddress, balanceInputBefore
            );

            if (executionSuccessful) {
                break; 
            } else {
                currentCandidateIndex++;
            }

        } catch (err) {
            logToPanel(`⚠️ [SIMULATION ENGINE EXCEPTION] Simulation failed for ${targetedWinnerSymbol}: ${err.message || err}`);
            currentCandidateIndex++;
        }
    }

    if (!executionSuccessful) {
        logToPanel("🔄 [CYCLE SKIPPED] Processing complete. No liquid or profitable swappable paths were cleared in this batch.");
        return "SKIP";
    }

    return "CONTINUE";
}
window.simulateCycle = simulateCycle;

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;
    window.cumulativePnL = 0;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;
    logToPanel(`🚀 [BOT INITIATED] Processing up to ${totalCycles} execution cycles.`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        if (signal === "STOP_ERROR") {
            logToPanel("🛑 State mismatch runtime alert. Aborting framework loops.");
            break;
        }
        if (signal === "SKIP") {
            await blockDelay(1000); 
            continue;
        }
        if (i < totalCycles) {
            setStatus("WAITING");
            await blockDelay(10000);
        }
    }
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Processing complete.");
}
window.startSimulation = startSimulation;
