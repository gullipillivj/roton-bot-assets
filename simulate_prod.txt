// Global log dashboard adapter
function logToPanel(msg) {
    const panel = document.getElementById("logPanel");
    if (panel) {
        panel.innerText += "\n" + msg;
        panel.scrollTop = panel.scrollHeight;
    } else {
        console.log(msg);
    }
}
window.logToPanel = logToPanel;

const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

function stopBot() {
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT STOPPED] Asset scan matrix paused by user.");
}
window.stopBot = stopBot;

// Direct provider account hook replaces fragile URL strings entirely
async function getLiveWalletAddress() {
    if (window.ethereum) {
        try {
            const accounts = await window.ethereum.request({ method: 'eth_accounts' });
            if (accounts && accounts.length > 0) {
                return accounts[0].toLowerCase().trim();
            }
        } catch (err) {
            console.error("Error reading accounts from window.ethereum:", err);
        }
    }
    return window.USER_ADDR ? window.USER_ADDR.toLowerCase().trim() : null;
}
// Helper to dynamically locate which token currently holds a balance inside the contract vault
async function getActiveLedgerAsset(userAddress) {
    try {
        logToPanel(`🔎 Querying ledger data for address: ${userAddress}...`);
        const res = await fetch("/prodbot/ledger/balances", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ user: userAddress })
        });
        const ledgerData = await res.json();

        if (ledgerData && ledgerData.status === "ok" && Array.isArray(ledgerData.tokens)) {
            for (let i = 0; i < ledgerData.tokens.length; i++) {
                const currentBalWei = BigInt(ledgerData.balances[i] || "0");
                if (currentBalWei > 0n) {
                    logToPanel(`🎯 Active Capital Found: ${ledgerData.tokens[i]} | Balance: ${currentBalWei.toString()} Wei`);
                    return {
                        address: ledgerData.tokens[i].toLowerCase().trim(),
                        balanceWei: currentBalWei.toString()
                    };
                }
            }
            if (ledgerData.tokens.length > 0) {
                return { address: ledgerData.tokens[0].toLowerCase().trim(), balanceWei: "0" };
            }
        }
    } catch (err) {
        logToPanel(`⚠️ Failed checking multi-asset ledger array: ${err.message}`);
    }
    return { address: "0x55d398326f99059ff775485246999027b3197955", balanceWei: "0" };
}

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    const PANCAKE_ROUTER = "0x10ED43C718714eb63d5aA57B78B54704E256024E";

    for (let i = 1; i <= totalCycles; i++) {
        if (window.isBotRunning === false) {
            logToPanel("🛑 Loop halted by request.");
            break;
        }

        logToPanel(`\n--- [START] Simulation Cycle ${i}/${totalCycles} ---`);
        setStatus(`RUNNING ${i}/${totalCycles}`);

        const currentActiveWallet = await getLiveWalletAddress();
        if (!currentActiveWallet) {
            logToPanel("❌ [ABORTED] No wallet session address connected.");
            break;
        }

        const activeAsset = await getActiveLedgerAsset(currentActiveWallet);
        const DYNAMIC_TOKEN_IN = activeAsset.address.toLowerCase().trim();
        
        let balanceInputStr = document.getElementById("investBalance") ? document.getElementById("investBalance").value.toString().trim() : "0";
        if (balanceInputStr === "0" || !balanceInputStr) {
            balanceInputStr = activeAsset.balanceWei !== "0" ? activeAsset.balanceWei : "10";
        }
        
        let simulatedAmountInWei = balanceInputStr;
        logToPanel(`📥 Current Routing Base Asset: ${DYNAMIC_TOKEN_IN}`);
        logToPanel(`💰 Pure Transaction Principal Weight: ${simulatedAmountInWei} Wei`);

        const universalPillars = getUniversalPillarsMesh();
        const customBridges = getCustomBridgesMesh();

        // 🛡️ Pre-Simulation Shield: Check target matrix alignment before testing paths
        const holdingPillarToken = universalPillars.some(t => t.address.toLowerCase().trim() === DYNAMIC_TOKEN_IN);
        const holdingBridgeToken = customBridges.some(t => t.address.toLowerCase().trim() === DYNAMIC_TOKEN_IN);

        let marketData = [];
        if (!holdingPillarToken && !holdingBridgeToken) {
            // 🚨 EXOTIC DETECTED: Immediately intercept loop and restrict to strong pillar exit paths only
            logToPanel("⚠️ [EXOTIC RETREAT INTERVENTION] Token held is outside safety matrix. Bypassing scan list to force exit routes straight into heavy pillars (WBNB/USDT/USDC/WETH).");
            marketData = universalPillars;
        } else {
            // Safe baseline: We hold a core anchor asset. Ingest full scanner candidate paths.
            try {
                if (typeof window.getTradeCandidates === "function") {
                    marketData = await window.getTradeCandidates();
                } else {
                    logToPanel("⚠️ window.getTradeCandidates function is missing.");
                    continue;
                }
            } catch (fErr) {
                logToPanel(`❌ Local Array Ingestion Error: ${fErr.message}`);
                continue;
            }
            marketData = marketData.concat(customBridges).concat(universalPillars);
        }

        if (!Array.isArray(marketData) || marketData.length === 0) {
            logToPanel("❌ Filtered token list data is completely empty.");
            continue;
        }

        logToPanel(`🎯 Dispatching ${marketData.length} tokens to validator endpoint...`);
        let cycleCandidatesArray = [];

        for (let index = 0; index < marketData.length; index++) {
            let candidate = marketData[index];
            let symbolLabel = candidate.symbol || "UNKNOWN";
            let targetAddress = candidate.address || "MISSING";
            let tokenNetwork = (candidate.tokentype || "bep20").toLowerCase();

            if (targetAddress === "MISSING" || !targetAddress.startsWith("0x")) continue;
            if (targetAddress.toLowerCase().trim() === DYNAMIC_TOKEN_IN) continue;

            let activeRouterForSimulation = PANCAKE_ROUTER;
            if (tokenNetwork === "erc20" || tokenNetwork === "ethereum") {
                activeRouterForSimulation = "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D"; 
            }

            try {
                let response = await fetch("/prodbot/swap/simulateuniswap", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify({
                        router: activeRouterForSimulation,
                        tokenIn: DYNAMIC_TOKEN_IN,
                        tokenOut: targetAddress,
                        amountIn: simulatedAmountInWei,
                        amountOutMin: 0
                    })
                });

                let resData = await response.json();
                if (resData.status === "ok") {
                    let expectedOutput = resData.expected_output_wei || "0";
                    let slippageCost = resData.slippage_cost_pct || "0.0";

                    if (expectedOutput !== "0" && expectedOutput !== 0) {
                        logToPanel(`   🟩 [SWAPPABLE] ${symbolLabel} | Slippage Rate: ${slippageCost}% | Expected Out: ${expectedOutput} Wei`);
                        
                        cycleCandidatesArray.push({
                            symbol: symbolLabel,
                            address: targetAddress,
                            network: tokenNetwork,
                            expectedOutWei: BigInt(expectedOutput),
                            slippage: parseFloat(slippageCost)
                        });
                    }
                }
            } catch (err) {}
            await blockDelay(100);
        }

        logToPanel(`\n📊 [CYCLE AUDIT] Evaluating ${cycleCandidatesArray.length} swappable pool candidates...`);

        if (cycleCandidatesArray.length === 0) {
            logToPanel("   💤 No executable candidates cleared parameter validation mappings.");
        } else {
            try {
                const userBalanceHex = await window.ethereum.request({ method: "eth_getBalance", params: [currentActiveWallet, "latest"] });
                const walletGasBalanceWei = BigInt(userBalanceHex);
                const networkGasPriceHex = await window.ethereum.request({ method: "eth_gasPrice" });
                const liveGasPriceWei = BigInt(networkGasPriceHex);

                let targetSelection = null;
                let highestExpectedOutput = 0n;

                for (let c = 0; c < cycleCandidatesArray.length; c++) {
                    let currentAsset = cycleCandidatesArray[c];
                    let standardGasLimit = currentAsset.network === "erc20" || currentAsset.network === "ethereum" ? 450000n : 150000n;
                    let computedGasFeeWei = liveGasPriceWei * standardGasLimit;

                    if (walletGasBalanceWei < computedGasFeeWei) continue;

                    if (currentAsset.expectedOutWei > highestExpectedOutput) {
                        highestExpectedOutput = currentAsset.expectedOutWei;
                        targetSelection = currentAsset;
                    }
                }

                if (!targetSelection) {
                    logToPanel("   ⚠️ [SKIPPED] All assets bypassed due to gas restrictions.");
                } else {
                    logToPanel(`\n🔥 [OPTIMIZER SELECTION] Dispatching parameters for: ${targetSelection.symbol}`);
                    let txResult = null;
                    if (targetSelection.network === "erc20" || targetSelection.network === "ethereum") {
                        txResult = await window.runethSwapUniswap(
                            currentActiveWallet, simulatedAmountInWei, DYNAMIC_TOKEN_IN, targetSelection.address, PANCAKE_ROUTER
                        );
                    } else {
                        txResult = await window.runbscSwapUniswap(
                            currentActiveWallet, simulatedAmountInWei, DYNAMIC_TOKEN_IN, targetSelection.address, PANCAKE_ROUTER
                        );
                    }
                    if (txResult && txResult.status === "ok") {
                        logToPanel("   🚀 [SWAP DISPATCH SUCCESS] Awaiting confirmation loop...");
                        
                        let transactionConfirmed = false;
                        for (let checkNum = 1; checkNum <= 15; checkNum++) {
                            if (window.isBotRunning === false) break;
                            
                            let verifiedState = await getActiveLedgerAsset(currentActiveWallet);
                            if (verifiedState.address.toLowerCase().trim() !== DYNAMIC_TOKEN_IN) {
                                logToPanel("   🎉 [SETTLEMENT CONFIRMED] Asset state changed successfully!");
                                transactionConfirmed = true;
                                break;
                            }
                            await blockDelay(6000);
                        }

                        if (!transactionConfirmed) {
                            logToPanel(`❌ [REVERT DETECTED] On-chain state failed to update. Halting execution loop to save gas.`);
                            window.isBotRunning = false;
                            setStatus("STOPPED");
                            break;
                        }

                        logToPanel("\n⏱️ [SAFETY COOLING LOCK] Enforcing a mandatory 15s pause...");
                        await blockDelay(15000);
                    } else {
                        logToPanel(`   ❌ [ROUTING REJECTION] Router dropped payload.`);
                    }
                }
            } catch (walletErr) {
                logToPanel(`   ❌ [CRITICAL FAULT] Wallet state loop exception: ${walletErr.message}`);
            }
        }
        logToPanel(`✅ Cycle ${i}/${totalCycles} analysis finalized.`);
        setStatus("WAITING");
        await blockDelay(4000);
    }
    logToPanel("\n=== 📊 [FINISHED] Scan operations complete. ==");
    return "STOP_ERROR";
}
window.simulateCycle = simulateCycle;

async function startSimulation(totalCycles) {
if (window.isBotRunning) return;
window.isBotRunning = true;
logToPanel("🚀 [LAUNCHING] Read-only background asset validation sequence engaged...");
await simulateCycle(1, 10, 0, 0);
window.isBotRunning = false;
setStatus("STOPPED");
logToPanel("[BOT FINISHED] Run parameters finalized.");
}
window.startSimulation = startSimulation;
