const blockDelay = ms => new Promise(resolve => setTimeout(resolve, ms));

function setStatus(msg) {
    const statusLabel = document.querySelector(".status-label");
    if (statusLabel) {
        statusLabel.textContent = msg;
    } else {
        logToPanel(`[STATUS] ${msg}`);
    }
}
window.setStatus = setStatus;

// Database record synchronization hook
async function updateTokenRecord(lastToken, currentToken, networkType = "bep20") {
    try {
        const userid = window.USER_ID || localStorage.getItem("userid");
        if (!userid) return false;
        const resp = await fetch("/prodbot/tokens/update", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ 
                userid: userid, 
                lasttoken: lastToken, 
                currenttoken: currentToken,
                network: networkType
            })
        });
        return resp.ok;
    } catch (err) {
        return false;
    }
}

// State container fetching hook
async function getCurrentTokenState() {
    let state = { token: "0x55d398326f99059fF775485246999027B3197955", network: "bep20" };
    try {
        const resp = await fetch("/prodbot/tokens/get", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ userid: window.USER_ID })
        });
        const data = await resp.json();
        if (data && data.currenttoken) {
            state.token = data.currenttoken;
            state.network = data.network || "bep20";
        }
    } catch (err) {
        logToPanel(`[STATE LOAD ERROR] Fallback engaged: ${err.message}`);
    }
    return state;
}

async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
    logToPanel(`\n--- [START] Cycle ${cycle}/${totalCycles} ---`);
    setStatus(`RUNNING ${cycle}/${totalCycles}`);
    const base = window.API_BASE || "";

    const totalBalance = BigInt(document.getElementById("startBalance").value.toString().trim());
    const allocationWei = totalBalance.toString();

    // 1. Fetch live contextual state mapping
    const srcAsset = await getCurrentTokenState();
    
    // 2. Query production-ready target variables from backend database cache
    logToPanel("⏳ Fetching qualified candidate collection from server filter cache...");
    let tokenCandidates = [];
    try {
        const tokenRes = await fetch("/prodbot/filtertokens");
        const tokenData = await tokenRes.json();
        if (tokenData && tokenData.short_term) {
            tokenCandidates = tokenData.short_term;
        }
    } catch (err) {
        logToPanel("⚠️ Failed to load target list from filtered engine cache.");
        return "CONTINUE";
    }

    if (tokenCandidates.length === 0) {
        logToPanel("⚠️ Filter registry empty. Skipping execution loop iteration.");
        return "SKIP";
    }

    // Determine parameter router mapping criteria based on token properties
    let activeRouter = "0x10ED43C718714eb63d5aA57B78B54704E256024E"; // BSC default
    let networkEndpoint = `${base}/prodbot/swap/bscuniswap`;

    if (["erc20", "eth"].includes(srcAsset.network.toLowerCase())) {
        activeRouter = "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D"; // ETH default
        networkEndpoint = `${base}/prodbot/swap/ethuniswap`;
    }

    let winningToken = null;
    let highestYield = -99999.0;
    let successfulBytecodePayload = null;
    let successfulProjectedWei = "0";

    // 3. FRONTEND ITERATION MATRIX: Sweeps assets and alters parameters to evaluate returns
    logToPanel(`🔬 Sweeping ${tokenCandidates.length} tokens using wallet node simulation tests...`);
    for (let candidate of tokenCandidates) {
        if (!candidate.address || candidate.address.toLowerCase() === srcAsset.token.toLowerCase()) {
            continue;
        }

        try {
            // Alter parameters and dispatch transaction call signature to simulation compiler
            const simRes = await fetch(`${base}/prodbot/swap/simulateuniswap`, {
                method: "POST",
                headers: { "Content-Type": "application/json" },
                body: JSON.stringify({
                    router: activeRouter,
                    tokenIn: srcAsset.token,
                    tokenOut: candidate.address,
                    amountIn: allocationWei,
                    amountOutMin: "0" // Rigid baseline parameter
                })
            });

            const simData = await simRes.json();

            // Check response payload conditions against strict profitability filters
            if (simData && simData.status === "ok") {
                const expectedOutWei = BigInt(simData.expected_output_wei || "0");
                if (expectedOutWei <= 0n) continue;

                const momentumChange = parseFloat(candidate["%profit"] || 0.0);

                // Isolate the highest momentum token that clears node simulation rules
                if (momentumChange > highestYield) {
                    highestYield = momentumChange;
                    winningToken = candidate;
                    successfulBytecodePayload = simData.compiled_data;
                    successfulProjectedWei = simData.expected_output_wei;
                }
            }
        } catch (simErr) {
            continue; // Safely skip problematic pairs
        }
    }

    if (!winningToken || !successfulBytecodePayload) {
        logToPanel("⚠️ [CYCLE ALERT] No active candidate tokens passed structural profit targets.");
        logToPanel(`⏭️ Skipping Cycle ${cycle} cleanly to protect working balance reserves.`);
        return "SKIP";
    }

    logToPanel(`🏆 Target Selected: ${winningToken.symbol} | Yield: ${highestYield}%`);
    logToPanel(`📈 Projected Output Returns: ${successfulProjectedWei} Wei units.`);

    // 4. Send exact compiled rigid bytecode payload straight to the user's extension wallet
    try {
        logToPanel("[INFO] Requesting secure transaction signature execution from extension wallet...");
        const txHash = await window.ethereum.request({
            method: "eth_sendTransaction",
            params: [{
                from: window.USER_ADDR,
                to: V12_ADDR, // Rigid destination wrapper contract address variable
                data: successfulBytecodePayload,
                value: "0x0"
            }]
        });

        logToPanel(`✅ Broadcast Succeeded! Transaction Hash: ${txHash}`);

        // 5. Track block confirmation lifecycle states
        const initialWeiBox = document.getElementById("investBalance").value.toString().trim();
        let confirmed = false;
        for (let check = 1; check <= 45; check++) {
            await blockDelay(10000);
            if (typeof window.syncBalanceFromFund === "function") await window.syncBalanceFromFund();
            if (document.getElementById("investBalance").value.toString().trim() !== initialWeiBox) {
                confirmed = true;
                break;
            }
        }

        if (!confirmed) {
            logToPanel("🛑 Transaction validation surpassed allowed timeout constraints.");
            return "STOP_ERROR";
        }

        // 6. Settle tracking parameters inside database records
        await updateTokenRecord(srcAsset.token, winningToken.address, srcAsset.network);

    } catch (broadcastErr) {
        logToPanel(`❌ Wallet execution rejected or crashed: ${broadcastErr.message}`);
        return "STOP_ERROR";
    }

    return "CONTINUE";
}

async function startSimulation(totalCycles) {
    if (window.isBotRunning) return;
    window.isBotRunning = true;

    const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
    const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

    logToPanel(`🚀 [ORCHESTRATOR ACTIVE] Processing up to ${totalCycles} parameter-altering cycles.`);

    for (let i = 1; i <= totalCycles; i++) {
        if (!window.isBotRunning) break;
        
        const signal = await simulateCycle(i, totalCycles, pTarget, sLoss);
        
        if (signal === "STOP_ERROR") {
            logToPanel("🛑 State runtime constraint mismatched. Terminating framework loops.");
            break;
        }
        if (signal === "SKIP") {
            await blockDelay(1000);
            continue;
        }
        if (i < totalCycles) {
            setStatus("WAITING");
            await blockDelay(10000);
        }
    }
    
    window.isBotRunning = false;
    setStatus("STOPPED");
    logToPanel("[BOT FINISHED] Orchestration processing loop completed.");
}

window.startSimulation = startSimulation;
window.simulateCycle = simulateCycle;
