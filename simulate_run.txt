// === Fixed Coin Picker (Relying directly on window memory strings) ===
async function pickRandomCoin() {
  try {
    const pairs = await window.getBinancePairs();
    if (!pairs || pairs.length === 0) return "BTC";
    const idx = Math.floor(Math.random() * pairs.length);
    // Strip USDT to provide structural base token context
    return pairs[idx].replace("USDT", "");
  } catch (e) {
    logToPanel("[ERROR] pickRandomCoin failed: " + e.message);
    return "BTC";
  }
}
window.pickRandomCoin = pickRandomCoin;

// === Custom Non-blocking Delay Helper ===
const blockDelay = ms => new Promise(res => setTimeout(res, ms));

// === Active Simulation Logic Cycle ===
async function simulateCycle(cycle, totalCycles, profitTargetPct, stopLossPct) {
  logToPanel(`[INFO] ${new Date().toLocaleTimeString()} — Cycle ${cycle}/${totalCycles} started`);
  setStatus(`RUNNING cycle ${cycle}/${totalCycles}`);

  const investBal = parseFloat(document.getElementById("investBalance").value) || 0;
  if (investBal <= 0) {
    logToPanel(`[WARN] Cycle ${cycle}: Investment pool depleted. Skipping.`);
    return "SKIP";
  }

  // Linear allocation slicing per test execution cycle block
  const allocationUSD = investBal / totalCycles;
  logToPanel(`[INFO] Cycle ${cycle}: Allocating ${allocationUSD.toFixed(2)} USDT to trade path`);

  const coinBase = await pickRandomCoin();
  const fullSymbol = `${coinBase}USDT`;
  logToPanel(`[DEBUG] Selected target validation pair: ${fullSymbol}`);

  // Fetch real market movements from window context to calculate real P&L
  const stats = await window.getBinanceStats(fullSymbol);
  if (!stats) {
    logToPanel(`[WARN] Skipping cycle ${cycle}: No price feed telemetry for ${fullSymbol}`);
    return "SKIP";
  }

  const currentPrice = stats.lastPrice;
  const standardDayChangePct = stats.priceChangePercent; // e.g., +2.5 or -1.2
  
  // Realized math output: Balance factor scaled against live index changes
  const cyclePnLUSD = allocationUSD * (standardDayChangePct / 100);
  cumulativePnL += cyclePnLUSD;

  logToPanel(`[INFO] Trade execution index: ${coinBase} entry asset at ${currentPrice} USDT`);
  
  try {
    // 1. Log transaction approval structure matching the ERC20 ABI specifications
    logToPanel(`[WEB3] Broadcasting approval transaction for ${coinBase}...`);
    // Mock or transmit JSON RPC call via route proxy maps
    const txApprove = await window.approveToken(allocationUSD, USER_ADDR, USER_PK);
    if (txApprove?.tx_hash) logToPanel(`[SUCCESS] Approve TX: ${txApprove.tx_hash}`);

    // 2. Log exact swap execution layout using verified Brownie types
    logToPanel(`[WEB3] Routing swap payload via PancakeRouter proxy address...`);
    const txSwap = await window.swapToTarget(allocationUSD, USER_ADDR, USER_PK);
    if (txSwap?.tx_hash) logToPanel(`[SUCCESS] Swap V2 Router TX: ${txSwap.tx_hash}`);

  } catch (err) {
    logToPanel(`[ERROR] Contract transaction trace failure: ${err.message}`);
  }

  // Update real-time metric UI tags dynamically
  const liveProfitSpan = document.getElementById("liveProfitValue");
  if (liveProfitSpan) {
    liveProfitSpan.textContent = `${cumulativePnL >= 0 ? '+' : ''}${cumulativePnL.toFixed(2)} USDT`;
    liveProfitSpan.style.color = cumulativePnL >= 0 ? "#00FF00" : "#FF3333";
  }

  // Cumulative graph output balancing layout constraints
  const updatedRunningBalance = startingBalanceAllocated + cumulativePnL;
  window.updateChart(updatedRunningBalance, cumulativePnL);

   // Check Stop-Loss / Take-Profit criteria threshold checks
  const totalChangePct = (cumulativePnL / investBal) * 100;
  
  if (totalChangePct >= profitTargetPct) {
    logToPanel(`[TAKE PROFIT] Target of ${profitTargetPct}% matched! Triggering profit target exit.`);
    return "STOP_PROFIT";
  }
  
  if (totalChangePct <= -stopLossPct) {
    // Log asset protection warning but allow simulation to proceed to the next automated cycle
    logToPanel(`[ALERT-STOP LOSS] Limit of -${stopLossPct}% breached (${totalChangePct.toFixed(2)}%). Simulation continues for block verification.`);
    return "CONTINUE";
  }

  logToPanel(`[INFO] Cycle ${cycle} completed. P&L Delta: ${cyclePnLUSD.toFixed(2)} USDT`);
  return "CONTINUE";

}

// === Modified Simulation Engine Management Wrapper ===
async function startSimulation(totalCycles) {
  if (window.isBotRunning) {
    logToPanel("[WARN] Bot runner already thread active.");
    return;
  }

  logToPanel(`[START] Initializing target system pipeline for ${totalCycles} blocks`);
  window.isBotRunning = true;
  cumulativePnL = 0;
  
  startingBalanceAllocated = parseFloat(document.getElementById("startBalance").value) || 1000;
  const pTarget = parseFloat(document.getElementById("profitTarget").value) || 1.0;
  const sLoss = parseFloat(document.getElementById("stopLoss").value) || 0.5;

  // Clear previous metric layout entries prior to rendering fresh data sets
  if (window.balanceChart) {
    window.balanceChart.data.labels = [];
    window.balanceChart.data.datasets[0].data = [];
    if (window.balanceChart.data.datasets[1]) {
        window.balanceChart.data.datasets[1].data = [];
    }
    window.balanceChart.update();
  }

  for (let i = 1; i <= totalCycles; i++) {
    if (!window.isBotRunning) {
      logToPanel("[MANUAL] Simulation loop intercepted by stop sequence.");
      break;
    }

    const stateSignal = await simulateCycle(i, totalCycles, pTarget, sLoss);
    
    // Check if loop was targeted to stop explicitly by Take Profit 
    if (stateSignal === "STOP_PROFIT") {
      logToPanel("[INFO] Terminating simulation due to Profit Target completion.");
      break;
    }

    // Skip the loop interval pause on the very final cycle execution block
    if (i < totalCycles) {
      logToPanel(`[WAIT] Pacing delay active — waiting 1 minute before executing Cycle ${i + 1}...`);
      
      // 60-second non-blocking countdown delay (60,000 milliseconds)
      for (let countdown = 60; countdown > 0; countdown--) {
        if (!window.isBotRunning) break;
        if (countdown % 15 === 0) { // Log status update every 15 seconds to keep panel clean
          setStatus(`WAITING (${countdown}s remaining)`);
        }
        await blockDelay(1000); 
      }
    }
  }

  window.isBotRunning = false;
  setStatus("STOPPED");
  logToPanel("[FINISH] 10-cycle simulated crypto run sequence complete.");
}
window.startSimulation = startSimulation;
