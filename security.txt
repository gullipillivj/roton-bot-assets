const ENCRYPTION_KEY = window.BOT_MASTER_KEY || "YOUR32CHARMASTERKEY"; // must be 32 chars
const IV_LENGTH = 16; // AES block size

function encryptKey(privateKey) {
  const iv = CryptoJS.lib.WordArray.random(IV_LENGTH);
  const encrypted = CryptoJS.AES.encrypt(privateKey, CryptoJS.enc.Utf8.parse(ENCRYPTION_KEY), {
    iv: iv,
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
  });
  return iv.toString(CryptoJS.enc.Hex) + ":" + encrypted.ciphertext.toString(CryptoJS.enc.Hex);
}

function decryptKey(encryptedText) {
  const parts = encryptedText.split(":");
  const iv = CryptoJS.enc.Hex.parse(parts.shift());
  const encryptedBuffer = CryptoJS.enc.Hex.parse(parts.join(":"));
  const decrypted = CryptoJS.AES.decrypt({ ciphertext: encryptedBuffer }, CryptoJS.enc.Utf8.parse(ENCRYPTION_KEY), {
    iv: iv,
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
  });
  return decrypted.toString(CryptoJS.enc.Utf8);
}

// Expose globally
window.encryptKey = encryptKey;
window.decryptKey = decryptKey;
